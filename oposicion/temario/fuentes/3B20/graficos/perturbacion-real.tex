% 3.B.20: una perturbación real negativa con tipo de cambio fijo (a) y
% flexible (b), en Mundell-Fleming con movilidad perfecta del capital (nota
% del tema antiguo sobre la flotación de 1973; mismas rectas y colores que
% mf-fijo-perfecta del 3.A.38, con la IS hacia la izquierda).
%   BP: i = i* = 2,5;  IS: i = 2,5 - (Y - 4,5);  IS': i = 2,5 - (Y - 3)
%   LM: i = 2,5 + 0,8 (Y - 4,5);  LM': i = 2,5 + 0,8 (Y - 3)
%   A = (4,5; 2,5); B = IS' ∩ LM = (3,667; 1,833); C (fijo) = (3; 2,5)
% Capas: ejes, BP, IS y LM, A (1-2) → IS' y B (3) → fijo: LM' y C;
% flexible: la depreciación devuelve la IS (4).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.75cm,y=0.75cm,
    declare function={is(\y)=2.5-(\y-4.5); isn(\y)=2.5-(\y-3);
      lm(\y)=2.5+0.8*(\y-4.5); lmn(\y)=2.5+0.8*(\y-3);},
    explica/.style={etiqueta,text=tceerojo,font=\footnotesize,align=center},
    titulo/.style={font=\small\bfseries,text=tceemorado}]
  \foreach \dx/\tit in {0/{a) Tipo de cambio fijo}, 9.4/{b) Tipo de cambio flexible}}{
    \begin{scope}[xshift=\dx*0.75cm]
      % 1: ejes, BP, IS, LM y A
      \capa{1}{\ejes{$Y$}{$i$}{7.6}{5.0}}
      \capa{1}{\node[titulo] at (3.8,5.6) {\tit};}
      \capa{1}{\draw[curva3] (0,2.5) node[left,text=black] {$i^*$} -- (7.2,2.5) node[etiqueta,right,text=tceeverde] {BP};}
      \capa{1}{\draw[curva2] (2.3,{is(2.3)}) -- (6.6,{is(6.6)}) node[etiqueta,right,text=tceeazul] {IS};}
      \capa{1}{\draw[curva1] (2.2,{lm(2.2)}) -- (6.3,{lm(6.3)}) node[etiqueta,right,text=tceemorado] {LM};}
      \capa{2}{\draw[guia] (4.5,2.5) -- (4.5,0) node[below,text=black] {$Y_0$};}
      \capa{2}{\node[punto,label={[etiqueta]45:$A$}] at (4.5,2.5) {};}
      % 3: perturbación real negativa: IS'
      \capa{3}{\draw[curva2,desplazada] (0.8,{isn(0.8)}) -- (5.1,{isn(5.1)}) node[etiqueta,below,text=tceeazul] {IS$'$};}
      \capa{3}{\node[punto,label={[etiqueta]-90:$B$}] at (3.667,1.833) {};}
    \end{scope}}
  % 4a: tipo fijo: sale capital, el banco central vende divisas, LM' y C
  \capa{4}{\draw[curva1,desplazada] (0.7,{lmn(0.7)}) -- (4.8,{lmn(4.8)}) node[etiqueta,above,text=tceemorado] {LM$'$};}
  \capa{4}{\draw[flecha] (2.6,0.8) -- (1.5,0.8);}
  \capa{4}{\draw[guia,tceerojo] (3,2.5) -- (3,0) node[below] {$Y_1$};}
  \capa{4}{\node[punto,fill=tceerojo,label={[etiqueta,text=tceerojo]135:$C$}] at (3,2.5) {};}
  \capa{4}{\node[explica,anchor=north] at (3.6,-0.7) {sale capital: el banco central\\vende divisas y cae el dinero};}
  % 4b: tipo flexible: la moneda se deprecia y la IS' vuelve a la IS
  \begin{scope}[xshift=7.05cm]
    \capa{4}{\draw[flecha] (4.5,1.2) -- (5.6,1.2);}
    \capa{4}{\node[explica,anchor=north] at (3.6,-0.7) {sale capital: la moneda se deprecia\\y suben las exportaciones netas};}
    \capa{4}{\node[etiqueta,text=tceerojo] at (5.3,3.3) {$C=A$};}
  \end{scope}
\end{tikzpicture}
\end{document}
