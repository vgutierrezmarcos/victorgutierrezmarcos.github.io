% El cártel en el plano (q_1, q_2): curva de colusión e incentivo a desviarse.
% Rehace la Imagen 2 («Colusión») del tema original. Misma base y escala que
% cournot-stackelberg.tex (en la pizarra es su última capa).
% Escala: cantidades en unidades de (a-c)/b; q^M = 1/2, q^c = 1.
%   R_1(q_2) = (1 - q_2)/2, R_2(q_1) = (1 - q_1)/2; Cournot C = (1/3, 1/3)
%   Isobeneficio de la empresa 1 de nivel k: q_2 = iso(k, q_1) = 1 - q_1 - k/q_1;
%   el de la 2, q_1 = iso(k, q_2).
%   Lente: los isobeneficios de nivel 1/9 por C se cortan en C y en (1/6, 1/6).
%   Curva de colusión: q_1 + q_2 = 1/2. En (a, 1/2 - a) el isobeneficio de la 1
%   tiene nivel a/2 y pendiente (1/2 - a)/a, igual que el de la 2 (nivel
%   (1/2 - a)/2): tangencias reales. Pares dibujados: a = 0,36 y a = 0,14.
%   Cártel K = (1/4, 1/4), B_i = 1/8. Desviación: R_1(1/4) = 3/8.
% Marco: x = y = 7 cm; el mismo que cournot-stackelberg.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}

\begin{tikzpicture}[tcee,x=7cm,y=7cm,
    declare function={iso(\k,\q)=1-\q-\k/\q;}]
  \useasboundingbox (-0.2,-0.17) rectangle (1.34,1.2);
  % 1: ejes, R_1, R_2 y C, de referencia (trazo fino)
  \capa{1}{\ejes{$q_1$}{$q_2$}{1.18}{1.15}}
  \capa{1}{\draw[curva1,thick] (0.5,0) -- (0,1) node[etiqueta,above right] {$R_1(q_2)$};}
  \capa{1}{\draw[curva2,thick] (0,0.5) -- (1,0) node[etiqueta,above right] {$R_2(q_1)$};}
  \capa{1}{\node[etiqueta,below] at (0.5,0) {$q^M$};}
  \capa{1}{\node[etiqueta,left] at (0,0.5) {$q^M$};}
  \capa{1}{\node[etiqueta,below] at (1,0) {$q^c$};}
  \capa{1}{\node[etiqueta,left] at (0,1) {$q^c$};}
  \capa{1}{\node[punto,label={[etiqueta]above right:$C$}] (C) at ({1/3},{1/3}) {};}
  % 2: isobeneficios de las 2 empresas por C y la lente
  \capa{2}{\draw[curva1,thin] plot[domain=0.15:0.62,samples=60] (\x,{iso(1/9,\x)}) node[etiqueta,right,text=tceemorado] {$B_1^C$};}
  \capa{2}{\draw[curva2,thin] plot[domain=0.15:0.62,samples=60] ({iso(1/9,\x)},\x) node[etiqueta,right,text=tceeazul] {$B_2^C$};}
  \capa{2}{\fill[area,fill=tceeazul] plot[domain=1/6:1/3,samples=40] (\x,{iso(1/9,\x)}) -- plot[domain=1/3:1/6,samples=40] ({iso(1/9,\x)},\x) -- cycle;}
  \capa{2}{\draw[thin,tceegris] (0.312,0.318) -- (0.64,0.42) node[etiqueta,right,text=black,align=left] {colusión imperfecta\\ (dentro de la lente)};}
  % 3: pares de isobeneficios tangentes (en (0,36; 0,14) y en (0,14; 0,36))
  \capa{3}{\draw[curva1,thick] plot[domain=0.25:0.5,samples=40] (\x,{iso(0.18,\x)});}
  \capa{3}{\draw[curva2,thick] plot[domain=0.09:0.24,samples=40] ({iso(0.07,\x)},\x);}
  \capa{3}{\draw[curva1,thick] plot[domain=0.09:0.24,samples=40] (\x,{iso(0.07,\x)});}
  \capa{3}{\draw[curva2,thick] plot[domain=0.25:0.5,samples=40] ({iso(0.18,\x)},\x);}
  % 4: curva de colusión (de contrato), de (q^M, 0) a (0, q^M)
  \capa{4}{\draw[curva3] (0.5,0) -- (0,0.5);}
  \capa{4}{\draw[thin,tceeverde,->] (0.6,0.07) node[etiqueta,right,text=tceeverde,align=left] {curva de\\ colusión} -- (0.42,0.08);}
  % 5: el cártel, con reparto a partes iguales
  \capa{5}{\node[punto,label={[etiqueta]below left:$K$}] (K) at (0.25,0.25) {};}
  % 6: desviación: la empresa 1 produce por encima de su cuota, hasta R_1
  \capa{6}{\draw[flecha] (K) -- (0.375,0.25) node[etiqueta,midway,above=0pt,font=\footnotesize,text=tceerojo] {desviación};}
\end{tikzpicture}
