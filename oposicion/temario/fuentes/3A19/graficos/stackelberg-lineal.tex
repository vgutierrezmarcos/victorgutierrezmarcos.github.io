% Stackelberg con demanda lineal y los números del anexo (anexo:stackelberg-lineal):
% p = 13 - Q, coste marginal c = 1 para las 2 (a - c = 12, b = 1).
% Parte del gráfico de Stackelberg (stackelberg.tex), con valores.
%   R_2(q_1) = 6 - q_1/2 (la restricción de la líder); R_1 y C = (4, 4) en gris
%   Isobeneficio de la empresa 1 de nivel k: q_1 (12 - q_1 - q_2) = k
%     -> q_2 = iso(k, q_1) = 12 - q_1 - k/q_1, máximo en q_1 = sqrt(k)
%     k = 16: por C (Cournot);  k = 18: tangente a R_2 en S = (6, 3), porque
%     su pendiente, -1 + 18/q_1^2, vale -1/2 en q_1 = 6
%   Stackelberg S = (6, 3): Q = 9, p = 4, B_1 = 18, B_2 = 9
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.6cm,y=0.6cm,
    declare function={iso(\k,\q)=12-\q-\k/\q;}]
  \useasboundingbox (-1.6,-1.5) rectangle (18.6,14.2);
  % 1: ejes
  \capa{1}{\ejes{$q_1$}{$q_2$}{14.2}{13.8}}
  % 2: lo que queda de Cournot: R_1, C y el isobeneficio de la 1 por C, en gris
  \capa{2}{\draw[desplazada,thick,tceegris] (6,0) -- (0,12) node[etiqueta,above right,text=tceegris] {$R_1(q_2)=6-q_2/2$};}
  \capa{2}{\draw[curva1,thin,tceegris] plot[domain=1.85:9,samples=60] (\x,{iso(16,\x)});}
  \capa{2}{\draw[thin,tceegris] (5.6,{iso(16,5.6)}) -- (8.2,5.3) node[etiqueta,right,text=tceegris] {$B_1^C=16$ (por $C$)};}
  \capa{2}{\node[punto,fill=tceegris,label={[etiqueta,text=tceegris]above right:$C$}] (C) at (4,4) {};}
  % 3: la restricción de la líder: la función de reacción de la seguidora
  \capa{3}{\draw[curva2] (0,6) -- (12,0);}
  \capa{3}{\node[etiqueta,anchor=south west,text=tceeazul] at (11.9,0.35) {$q_2=R_2(q_1)=6-q_1/2$};}
  \capa{3}{\node[etiqueta,left] at (0,6) {$6$};}
  \capa{3}{\node[etiqueta,below] at (12,0) {$12$};}
  % 4: el isobeneficio más bajo de la líder que toca R_2: tangencia en S
  \capa{4}{\draw[curva1] plot[domain=2.2:10.4,samples=60] (\x,{iso(18,\x)});}
  \capa{4}{\node[etiqueta,anchor=west,text=tceemorado] at (2.35,1.25) {$B_1^S=18$};}
  % 5: el equilibrio de Stackelberg
  \capa{5}{\node[punto,label={[etiqueta,fill=white,inner sep=1pt]below left:$S=(6;\,3)$}] (S) at (6,3) {};}
  \capa{5}{\proyeccion{(S)}{$6$}{$3$}}
  \capa{5}{\node[etiqueta,text=tceemorado,anchor=west,align=left,font=\footnotesize] at (7.5,9) {la líder elige, sobre $R_2$,\\ el punto de su isobeneficio\\ más bajo (más beneficio):\\ tangencia en $S$};}
\end{tikzpicture}
