% Colusión con estrategias de gatillo: valor descontado del beneficio de cada
% periodo, coludiendo siempre o desviándose hoy. Rehace la Imagen 13 del tema
% original: las 2 curvas ya están descontadas (beta^t B), de modo que se
% comparan directamente las áreas: la ganancia de desviarse hasta que las
% demás reaccionan (rojo) frente a la pérdida del castigo después (azul).
% Valores: B^D = 4, B^C = 2,8, B^N = 0,9; beta = 0,82 por periodo; reacción en
% tau = 2 (con tau = 1, la reacción es al periodo siguiente).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.75cm,y=1cm,
    declare function={des(\B,\t)=\B*0.82^\t;}]
  \def\BD{4}\def\BC{2.8}\def\BN{0.9}\def\treac{2}\def\T{14}
  \useasboundingbox (-1.3,-0.8) rectangle (17.2,4.9);
  % 1: ejes (tiempo y beneficio descontado)
  \capa{1}{\ejes{$t$}{$\beta^t B$}{\T.6}{4.6}}
  % 2: coludir siempre: beta^t B^C
  \capa{2}{\draw[curva1] plot[domain=0:\T,samples=60] (\x,{des(\BC,\x)});}
  \capa{2}{\node[etiqueta,text=tceemorado,anchor=south west] at (10.5,{des(\BC,10.5)}) {coludir};}
  \capa{2}{\node[left] at (0,\BC) {$B^{\mathrm C}$};}
  % 3: desviarse: beta^t B^D hasta que reaccionan, y la ganancia (rojo)
  \capa{3}{\fill[area,fill=tceerojo,fill opacity=0.45] plot[domain=0:\treac,samples=30] (\x,{des(\BD,\x)}) -- plot[domain=\treac:0,samples=30] (\x,{des(\BC,\x)}) -- cycle;}
  \capa{3}{\draw[curva4] plot[domain=0:\treac,samples=30] (\x,{des(\BD,\x)});}
  \capa{3}{\node[left] at (0,\BD) {$B^{\mathrm D}$};}
  % 4: el castigo: beta^t B^N desde tau, y la pérdida (azul)
  \capa{4}{\draw[curva4] (\treac,{des(\BD,\treac)}) -- (\treac,{des(\BN,\treac)});}
  \capa{4}{\fill[area,fill=tceeazul,fill opacity=0.35] plot[domain=\treac:\T,samples=50] (\x,{des(\BC,\x)}) -- plot[domain=\T:\treac,samples=50] (\x,{des(\BN,\x)}) -- cycle;}
  \capa{4}{\draw[curva4] plot[domain=\treac:\T,samples=50] (\x,{des(\BN,\x)});}
  \capa{4}{\node[etiqueta,text=tceedorado,anchor=south west] at (0.9,{des(\BD,0.9)}) {desviarse};}
  \capa{4}{\node[left] at (0,\BN) {$B^{\mathrm N}$};}
  \capa{4}{\draw[guia] (0,\BN) -- (\treac,{des(\BN,\treac)});}
  \capa{4}{\draw (\treac,0) -- (\treac,-0.08) node[below] {$\tau$};}
  \capa{4}{\node[etiqueta,below,font=\footnotesize,align=center] at (\treac,-0.35) {reacción};}
  % 5: rótulos y la condición
  \capa{5}{\node[etiqueta,text=tceerojo,anchor=west,align=left] at (3.2,3.1) {ganancia de desviarse\\ hasta que reaccionan};}
  \capa{5}{\node[etiqueta,text=tceeazul,anchor=west,align=left] at (6.2,1.75) {pérdida por el castigo\\ desde entonces};}
  \capa{5}{\node[draw=tceemorado,thick,rounded corners=2pt,inner sep=4pt,fill=white] at (10.8,3.9) {la colusión aguanta si \textcolor{tceerojo}{rojo} $\leq$ \textcolor{tceeazul}{azul}};}
\end{tikzpicture}
\end{document}
