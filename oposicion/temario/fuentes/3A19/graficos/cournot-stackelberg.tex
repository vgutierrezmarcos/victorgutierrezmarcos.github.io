% Cournot y Stackelberg en el plano (q_1, q_2): funciones de reacción e
% isobeneficios. Rehace las Imágenes 1 y 3 y el gráfico de la nota 33 del tema
% original (a partir de Pindyck y Rubinfeld).
% Escala: demanda p = a - bQ, coste marginal c igual para las 2; las cantidades
% van en unidades de (a-c)/b y el beneficio en unidades de (a-c)^2/b.
%   q^M = 1/2 (monopolio), q^c = 1 (competencia perfecta)
%   R_1(q_2) = (1 - q_2)/2: de (1/2, 0) a (0, 1);  R_2(q_1) = (1 - q_1)/2
%   Isobeneficio de la empresa 1 de nivel k: B_1 = q_1 (1 - q_1 - q_2) = k
%     -> q_2 = iso(k, q_1) = 1 - q_1 - k/q_1, máximo en q_1 = sqrt(k), sobre R_1
%     k = 1/4: el punto de monopolio (1/2, 0); k = 1/9: Cournot; k = 1/8: líder
%     de Stackelberg; k = 0,04: uno más alto (menos beneficio), máximo en (0,2; 0,6)
%   Empresa 2: el simétrico, q_1 = iso(k, q_2)
%   Cournot C = (1/3, 1/3), B_i = 1/9
%   Lente: los isobeneficios de nivel 1/9 de las 2 se cortan en C y en (1/6, 1/6)
%   Stackelberg S = (1/2, 1/4): iso(1/8, q_1) tiene pendiente -1 + 1/(8 q_1^2) = -1/2
%     en q_1 = 1/2, la de R_2: tangencia real
% Para el gráfico «cartel» (bloque III): la curva de contrato de las 2 empresas
% con costes iguales es el segmento q_1 + q_2 = q^M = 1/2 (los isobeneficios
% iso(k,.) de las 2 son tangentes en él); el cártel simétrico es (1/4, 1/4),
% con B_i = 1/8; el isobeneficio de la 1 por (1/4, 1/4) es iso(1/8, q_1), el
% mismo que el de la líder de Stackelberg, y el de la 2, su simétrico.
% Marco: x = y = 7 cm; \useasboundingbox (-0.2,-0.17) rectangle (1.34,1.2).
% Autor: Víctor Gutiérrez Marcos

\begin{tikzpicture}[tcee,x=7cm,y=7cm,
    declare function={iso(\k,\q)=1-\q-\k/\q;}]
  \useasboundingbox (-0.2,-0.17) rectangle (1.34,1.2);
  % 1: ejes
  \capa{1}{\ejes{$q_1$}{$q_2$}{1.18}{1.15}}
  % 2: función de reacción de la empresa 1, q_1 = R_1(q_2), de (q^M,0) a (0,q^c)
  \capa{2}{\draw[curva1] (0.5,0) -- (0,1) node[etiqueta,above right] {$R_1(q_2)$};}
  \capa{2}{\node[punto] (M1) at (0.5,0) {};}
  \capa{2}{\node[punto] (c1) at (0,1) {};}
  \capa{2}{\node[etiqueta,below] at (0.5,0) {$q^M$};}
  \capa{2}{\node[etiqueta,left] at (0,1) {$q^c$};}
  % 3: isobeneficios de la empresa 1, con el máximo sobre R_1; más abajo, más beneficio
  \capa{3}{\draw[curva1,thick] plot[domain=0.1:0.5,samples=60] (\x,{iso(0.04,\x)});}
  \capa{3}{\draw[curva1,thick] plot[domain=0.15:0.62,samples=60] (\x,{iso(1/9,\x)}) node[etiqueta,right,text=tceemorado] {$B_1^C$};}
  \capa{3}{\node[etiqueta,above right,text=tceemorado] at (0.5,0) {$B_1^M$};}
  \capa{3}{\draw[flecha,tceemorado] (0.45,0.46) -- (0.45,0.31) node[etiqueta,right,midway,text=tceemorado] {más $B_1$};}
  % 4: función de reacción de la empresa 2, q_2 = R_2(q_1), de (0,q^M) a (q^c,0)
  \capa{4}{\draw[curva2] (0,0.5) -- (1,0) node[etiqueta,above right] {$R_2(q_1)$};}
  \capa{4}{\node[etiqueta,left] at (0,0.5) {$q^M$};}
  \capa{4}{\node[etiqueta,below] at (1,0) {$q^c$};}
  % 5: equilibrio de Cournot
  \capa{5}{\node[punto,label={[etiqueta]above right:$C$}] (C) at ({1/3},{1/3}) {};}
  \capa{5}{\proyeccion{(C)}{$q^C$}{$q^C$}}
  % 6: isobeneficio de la empresa 2 por C y la lente en que las 2 ganan más
  \capa{6}{\draw[curva2,thick] plot[domain=0.15:0.62,samples=60] ({iso(1/9,\x)},\x) node[etiqueta,right,text=tceeazul] {$B_2^C$};}
  \capa{6}{\fill[area,fill=tceeazul] plot[domain=1/6:1/3,samples=40] (\x,{iso(1/9,\x)}) -- plot[domain=1/3:1/6,samples=40] ({iso(1/9,\x)},\x) -- cycle;}
  % 7: rectas de la cantidad total de monopolio y de competencia perfecta
  \capa{7}{\draw[desplazada,thick,tceegris] (0,0.5) -- (0.5,0);}
  \capa{7}{\node[etiqueta,font=\footnotesize,text=tceegris,fill=white,inner sep=0.5pt,rotate=-45] at (0.103,0.338) {$q_1+q_2=q^M$};}
  \capa{7}{\draw[desplazada,thick,tceegris] (0,1) -- (1,0) node[etiqueta,text=tceegris,pos=0.72,sloped,above] {$q_1+q_2=q^c$};}
  % 8: Stackelberg: el isobeneficio más bajo de la líder que toca R_2, tangente en S
  \capa{8}{\draw[curva4] plot[domain=0.21:0.8,samples=60] (\x,{iso(1/8,\x)}) node[etiqueta,right,text=tceedorado] {$B_1^S$};}
  \capa{8}{\node[punto,label={[etiqueta]above right:$S$}] (S) at (0.5,0.25) {};}
  \capa{8}{\proyeccion{(S)}{}{$q_2^S$}}
  \capa{8}{\node[etiqueta,below=11pt] at (0.5,0) {$q_1^S$};}
\end{tikzpicture}
