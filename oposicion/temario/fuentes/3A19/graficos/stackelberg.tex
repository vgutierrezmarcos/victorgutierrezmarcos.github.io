% Stackelberg en el plano (q_1, q_2): la líder elige sobre R_2 el punto de su
% isobeneficio más bajo (tangencia, S). Rehace la Imagen 3 del tema original.
% Parte del gráfico de Cournot (cournot.tex), que se deja en la pizarra:
% R_1 y C en gris como referencia, R_2 y los isobeneficios de la 1 por C y por S.
% Escala: la de cournot.tex; S = (1/2, 1/4), B_1^S = 1/8 > B_1^C = 1/9.

\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=7cm,y=7cm,
    declare function={iso(\k,\q)=1-\q-\k/\q;}]
  \useasboundingbox (-0.2,-0.17) rectangle (1.34,1.2);
  % 1: ejes
  \capa{1}{\ejes{$q_1$}{$q_2$}{1.18}{1.15}}
  % 2: lo que queda de Cournot: R_1 y C, como referencia
  \capa{2}{\draw[desplazada,thick,tceegris] (0.5,0) -- (0,1) node[etiqueta,above right,text=tceegris] {$R_1(q_2)$};}
  \capa{2}{\node[punto,fill=tceegris,label={[etiqueta,text=tceegris]above right:$C$}] (C) at ({1/3},{1/3}) {};}
  \capa{2}{\draw[curva1,thin,tceegris] plot[domain=0.15:0.62,samples=60] (\x,{iso(1/9,\x)}) node[etiqueta,right,text=tceegris] {$B_1^C$};}
  % 3: la restricción de la líder: la función de reacción de la seguidora
  \capa{3}{\draw[curva2] (0,0.5) -- (1,0) node[etiqueta,above right] {$q_2=R_2(q_1)$};}
  \capa{3}{\node[etiqueta,left] at (0,0.5) {$q^M$};}
  \capa{3}{\node[etiqueta,below] at (1,0) {$q^c$};}
  % 4: el isobeneficio más bajo de la líder que toca R_2: tangencia en S
  \capa{4}{\draw[curva1] plot[domain=0.21:0.8,samples=60] (\x,{iso(1/8,\x)}) node[etiqueta,right,text=tceemorado] {$B_1^S$};}
  \capa{4}{\draw[flecha,tceemorado] (0.62,0.36) -- (0.62,0.24) node[etiqueta,right,midway,text=tceemorado] {más $B_1$};}
  \capa{5}{\node[punto,label={[etiqueta]above right:$S$}] (S) at (0.5,0.25) {};}
  \capa{5}{\proyeccion{(S)}{$q_1^S=q^M$}{$q_2^S$}}
\end{tikzpicture}
