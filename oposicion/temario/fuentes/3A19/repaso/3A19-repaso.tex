\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: «la avaricia rompe el saco».
  \begin{itemize}
  \item \autor{Smith} (1776): el mismo oficio rara vez se reúne sin conspirar contra el público.
  \item 2016: la CE multa con 2.926 M€ a 4 fabricantes de camiones (1997-2011); MAN lo denuncia y se libra (1.200 M€).
  \item \emph{Si les conviene ponerse de acuerdo, ¿por qué compiten?}
  \end{itemize}
\item \textbf{Relevancia}:
  \begin{itemize}
  \item \emph{empírica}: telecos, energía, banca, aviación, plataformas; margen sobre CMg en EE.\,UU. 21\,\% (1980) $\to$ 61\,\% (2016);
  \item \emph{teórica}: laboratorio de la teoría de juegos (\autor{Cournot} anticipa a \autor{Nash}); base de \vertema{3.B.8} \vertema{3.B.6} \vertema{3.B.27};
  \item \emph{de política}: $p>\mathrm{CMg}$, falla el 1TFEB \vertema{3.A.22} $\Rightarrow$ defensa de la competencia \vertema{3.A.20} \vertema{3.B.39}.
  \end{itemize}
\item \textbf{Contextualización}:
  \begin{itemize}
  \item 2 estructuras polares (\vertema{3.A.16} \vertema{3.A.17}) y 2 intermedias (\vertema{3.A.18} y oligopolio).
  \item \concepto{Oligopolio}: pocos vendedores, cada uno capaz de mover el precio $\Rightarrow$ \concepto{interdependencia estratégica}: teoría de juegos aplicada \vertema{3.A.14}.
  \item Historia: \autor{Cournot}, \autor{Bertrand}, \autor{Edgeworth}, \autor{Stackelberg} $\to$ \autor{von Neumann}-\autor{Morgenstern}, \autor{Nash}, \autor{Selten}, \autor{Harsanyi} $\to$ \autor{Stigler}, \autor{Friedman}, \autor{Tirole}.
  \item «Cooperativo», 2 sentidos:
    \begin{itemize}
    \item \emph{por el resultado}: \concepto{colusorio} (cerca del beneficio de monopolio);
    \item \emph{por el instrumento}: acuerdos vinculantes.
    \item \concepto{Colusión tácita}: resultado cooperativo de un juego \emph{no} cooperativo; \concepto{cártel}: el caso cooperativo.
    \end{itemize}
  \end{itemize}
\item \textbf{Problemática}: ¿cuánto se alejan del resultado competitivo, según variable y orden? ¿Se coopera sin contratos? ¿Qué sostiene y qué rompe un cártel? ¿Cómo se reparte lo ganado?
\item \textbf{Estructura}: de la competencia a la cooperación: \hyperref[rep:I]{I}, una sola vez; \hyperref[rep:II]{II}, repetida; \hyperref[rep:III]{III}, cártel.
\end{itemize}

\bloque[11]{I. No cooperativas: competencia en una sola vez\label{rep:I}}
\begin{ideaclave}
Con un solo encuentro el precio queda entre el competitivo y el de monopolio, según la variable (cantidades o precios) y el orden (a la vez o por turnos); en todos los casos las empresas ganarían más produciendo menos juntas.
\end{ideaclave}
\graficorepaso[0.85\linewidth]{escala-precios}{Dónde queda el precio de cada modelo (duopolio lineal, costes iguales)}

\sub{I.1. Caracterización del oligopolio\label{rep:I.1}}
\begin{itemize}
\item \emph{Supuestos} \vertema{3.A.16}:
  \begin{itemize}
  \item se mantienen racionalidad (+ \emph{conocimiento común}), información completa y producto homogéneo (el diferenciado, \vertema{3.A.18});
  \item \emph{se levanta el 4}: pocos vendedores (2 por sencillez), que se saben interdependientes; compradores muchos y precio-aceptantes;
  \item casi siempre \emph{el 5}: barreras \vertema{3.A.15} estratégicas, legales o tecnológicas (\concepto{oligopolio natural}).
  \end{itemize}
\item \emph{Juego}: jugadores = empresas; estrategias = $q$ o $p$; pagos = beneficios.
  \begin{itemize}
  \item \concepto{Función de reacción} $R_i(\cdot)$: mejor respuesta.
  \item \concepto{Equilibrio de \autor{Nash}} (EN): cada una da la mejor respuesta a las demás.
  \item \concepto{ENPS}: además, creíble fuera del camino de equilibrio (inducción hacia atrás).
  \end{itemize}
\item \autor{Bulow}-\autor{Geanakoplos}-\autor{Klemperer} (1985): \concepto{sustitutos estratégicos} ($\partial^2B_i/\partial x_i\partial x_j<0$, reacción decreciente: cantidades) y \concepto{complementos} ($>0$, creciente: precios).
  \begin{itemize}
  \item 2 variables $\times$ 2 órdenes = 4 modelos: \autor{Cournot}, \autor{Stackelberg}, \autor{Bertrand}, liderazgo en precios.
  \end{itemize}
\end{itemize}

\sub{I.2. \autor{Cournot} (1838): cantidades a la vez\label{rep:I.2}}
\subsub{a) Idea y b) modelo}
\begin{itemize}
\item \emph{Idea}: 2 manantiales; cada uno elige $q_i$ dada la del otro; un subastador vacía el mercado.
\item \emph{Supuestos}: $n$ empresas, $Q=\sum q_i$; $p(Q)$ decreciente, $C_i(q_i)$ (lineal: $p=a-bQ$, $C_i=c_iq_i$); \concepto{conjetura de \autor{Nash}} (variación conjetural nula).
\item \emph{Desarrollo}: el IMg solo recoge la bajada de precio sobre las unidades \emph{propias} $\Rightarrow$ cada una produce más de lo que conviene al conjunto:
\begin{formula}
\centering $\optim{\max}{q_i}{B_i=p(q_i+Q_{-i})\,q_i-C_i(q_i)}{q_i\geq0}$\\[3pt]
$\Rightarrow\ \underbrace{p(Q)+p'(Q)\,q_i}_{\mathrm{IMg}_i}=\mathrm{CMg}_i(q_i)$
\end{formula}
  \begin{itemize}
  \item Lineal, duopolio: $R_i(q_j)=\frac{a-c_i}{2b}-\frac12q_j$ (pendiente $-1/2$, corta el eje en $q^M$).
  \end{itemize}
\item \emph{Equilibrio}, $n$ iguales, lineal:
\begin{formula}
\centering $q^C=\dfrac{a-c}{(n+1)b}$,\quad $Q^C=\dfrac{n}{n+1}\cdot\dfrac{a-c}{b}$,\quad $p^C=\dfrac{a+nc}{n+1}$
\end{formula}
  \begin{itemize}
  \item duopolio: $q^C=(a-c)/3b$, $p^C=(a+2c)/3$, $B_i=(a-c)^2/9b$;
  \end{itemize}
\end{itemize}
\graficorepaso[0.45\linewidth]{cournot}{\autor{Cournot} (C): reacciones e isobeneficios (en U invertida); en C se cortan}
\begin{itemize}
\item \emph{Implicaciones}:
  \begin{enumerate}
  \item EN, no en estrategias dominantes.
  \item \emph{Poder proporcional a la cuota} ($s_i=q_i/Q$; menor CMg $\Rightarrow$ más cuota y margen):
\funcionclave{\mathcal{L}_i\equiv\frac{p-\mathrm{CMg}_i}{p}=\frac{s_i}{|\varepsilon_{Q,p}|}}
  \item \emph{Concentración} (\autor{Cowling}-\autor{Waterson}, 1976): $\sum_is_i\mathcal{L}_i=\dfrac{\mathrm{IHH}}{10.000\,|\varepsilon_{Q,p}|}$, $\mathrm{IHH}=\sum(100s_i)^2$.
  \item \emph{Bienestar}: $Q^M<Q^C<Q^c$; ineficiencia \emph{asignativa} (PIE) y \emph{productiva} (CMg distintos).
  \item \emph{Ineficiente para las empresas}: «lente» entre isobeneficios, \emph{dilema del prisionero} \vertema{3.A.14}.
  \item $n\to\infty$: $p^C\to c$, $\mathcal{L}=1/(n|\varepsilon|)\to0$ («concurrencia indefinida»): une monopolio y competencia.
  \end{enumerate}
\end{itemize}
\subsub{c) Extensiones}
\begin{itemize}
\item \concepto{Variaciones conjeturales} (\autor{Bowley}, 1924), $v_i=dQ_{-i}/dq_i$:
  \begin{itemize}
  \item $\mathcal{L}_i=s_i(1+v_i)/|\varepsilon_{Q,p}|$: $v=0$ \autor{Cournot}, $v=-1$ competencia, $v=n-1$ monopolio;
  \item crítica: en un juego simultáneo nadie reacciona (\autor{Tirole}).
  \end{itemize}
\item \emph{Costes privados}: \concepto{equilibrio bayesiano de \autor{Nash}}: la de coste alto produce \emph{más}, la de bajo \emph{menos} \vertema{3.A.13}.
\end{itemize}
\subsub{d) Evidencia y e) valoración}
\begin{itemize}
\item \autor{Brander}-\autor{Zhang} (1990), aerolíneas: más cerca de \autor{Cournot}.
\item Razonable; críticas: no siempre a la vez y son las empresas quienes fijan el precio.
\end{itemize}

\sub{I.3. \autor{Stackelberg} (1934): cantidades por turnos\label{rep:I.3}}
\subsub{a) Idea y b) modelo}
\begin{itemize}
\item \emph{Supuestos}: la \concepto{líder} elige $q_1$ primero, irreversible; la \concepto{seguidora} la observa y elige $q_2$.
\item \emph{Desarrollo}: seguidora, $q_2=R_2(q_1)$ (responde a lo hecho, no a lo esperado); la líder maximiza con esa restricción:
\begin{formula}
\centering $\optim{\max}{q_1,q_2}{B_1=p(q_1+q_2)\,q_1-C_1(q_1)}{q_2=R_2(q_1)\\ q_1\geq0}$\\[3pt]
$p+p'(Q)\,q_1\big[1+R_2'(q_1)\big]=\mathrm{CMg}_1$
\end{formula}
  \begin{itemize}
  \item $R_2'<0$: conjetura negativa \emph{y correcta} $\Rightarrow$ produce más.
  \item Lineal: $q_1^S=(a-c)/2b$ (= $q^M$), $q_2^S=(a-c)/4b$, $p^S=(a+3c)/4$.
  \item $B_1=(a-c)^2/8b$, $B_2=(a-c)^2/16b$ (\autor{Cournot}: $/9b$).
  \end{itemize}
\end{itemize}
\graficorepaso[0.45\linewidth]{stackelberg}{\autor{Stackelberg} (S): la líder elige sobre $R_2$ su isobeneficio más bajo}
\begin{itemize}
\item \emph{Implicaciones}:
  \begin{itemize}
  \item ENPS: la amenaza de la seguidora de producir $q^C$ no es creíble;
  \item \emph{ventaja del primero de compromiso, no de información} (inversión irreversible: \autor{Dixit} \vertema{3.A.15});
  \item $Q^S=3(a-c)/4b>Q^C$: más competitivo que \autor{Cournot}.
  \end{itemize}
\end{itemize}
\subsub{c) Valoración}
\begin{itemize}
\item Explica la ventaja de las establecidas, pero el orden es exógeno.
\end{itemize}

\sub{I.4. \autor{Bertrand} (1883): precios a la vez\label{rep:I.4}}
\subsub{a) Idea y b) modelo}
\begin{itemize}
\item \emph{Idea}: reseña de \autor{Cournot}: las empresas eligen \emph{precios}.
\item \emph{Supuestos}: 2 empresas, homogéneo, mismo CMg $c$ constante, sin límite de capacidad, una vez; compran a la más barata (empate: mitad):
\begin{formula}
\centering $x_i(p_i,p_j)=\begin{cases}x(p_i) & p_i<p_j\\ \tfrac12x(p_i) & p_i=p_j\\ 0 & p_i>p_j\end{cases}$
\end{formula}
\item \emph{Desarrollo}: sin CPO; mejor respuesta: $p^M$ si $p_j>p^M$; un céntimo menos si $c<p_j\leq p^M$; no vender si $p_j\leq c$.
  \begin{itemize}
  \item Único EN: $p_1^B=p_2^B=c$; reacciones crecientes.
  \end{itemize}
\end{itemize}
\graficorepaso[0.4\linewidth]{bertrand-reaccion}{\autor{Bertrand}: reacciones en precios; EN en $(c,c)$}
\begin{itemize}
\item \emph{Implicaciones}: \concepto{paradoja de \autor{Bertrand}}: con 2 empresas, $p=c$ y $B=0$, como con infinitas; eficiente, pero contradice los datos.
  \begin{itemize}
  \item Descansa en 5 supuestos: CMg iguales, homogéneo, sin límite de capacidad, estático, sin cooperación.
  \end{itemize}
\end{itemize}
\subsub{c) Extensiones: las salidas a la paradoja}
\begin{enumerate}
\item \emph{$c_1<c_2$}: la 1 cobra un céntimo bajo $c_2$ (o $p^M(c_1)$) y se queda todo el mercado.
\item \emph{CMg crecientes}: sin EN en puras; si hay que servir toda la demanda, equilibrios con $p\geq p^c$ (\autor{Dastidar}, 1995).
\item \emph{Capacidad} (\autor{Edgeworth}, 1897): $k<x(c)\leq2k$.
  \begin{itemize}
  \item \concepto{Racionamiento eficiente} $\Rightarrow$ la cara tiene \concepto{demanda residual} $x(p)-k$.
  \item $p=c$ no es EN; la que sube cobra $\overline{p}=\arg\max(p-c)[x(p)-k]$.
  \item Rebajas sucesivas hasta $\underline{p}$: $(\underline{p}-c)k=(\overline{p}-c)[x(\overline{p})-k]$, y vuelta a $\overline{p}$: \concepto{ciclos de \autor{Edgeworth}}.
  \item Sin EN en puras (sí en mixtas); con $k$ pequeña, EN en $p(2k)$. Gasolina en Perth.
  \end{itemize}
\end{enumerate}
\graficorepaso[0.65\linewidth]{edgeworth}{Capacidad limitada: demanda residual y ciclo de \autor{Edgeworth}}
\begin{enumerate}\setcounter{enumi}{3}
\item \emph{\autor{Kreps}-\autor{Scheinkman}} (1983): 1.º capacidad, 2.º precios (racionamiento eficiente) $\Rightarrow$ ENPS = \autor{Cournot}: elegir capacidad es elegir cantidad.
\item \emph{Diferenciado}: $p>c$ y $p^B<p^C$ (\autor{Singh}-\autor{Vives}, 1984) \vertema{3.A.18}.
\item Repetición (\hyperref[rep:II]{II}) y cártel (\hyperref[rep:III]{III}).
\end{enumerate}
\subsub{d) Valoración: ¿\autor{Cournot} o \autor{Bertrand}?}
\begin{itemize}
\item Mercados distintos: \autor{Cournot}, capacidad fija y cara de cambiar (acero, cemento, aviación); \autor{Bertrand}, capacidad sobrante o por encargo.
\end{itemize}

\sub{I.5. Liderazgo en precios\label{rep:I.5}}
\subsub{a) Idea y b) modelo}
\begin{itemize}
\item \emph{Idea}: con 2 iguales y homogéneo, mover primero no sirve; sentido con \concepto{empresa dominante} (grande, costes bajos) y \concepto{franja competitiva} precio-aceptante (\autor{Forchheimer}, 1908).
\item \emph{Desarrollo}: demanda residual $x_R(p)=x(p)-q_f(p)$, quebrada ($=x$ bajo $p_0$; nula desde $p_1$).
  \begin{itemize}
  \item La dominante es monopolista de ella \vertema{3.A.17}: $\mathrm{IMg}_R(q_L)=\mathrm{CMg}_L(q_L)$ y fija $p^*$.
  \item \autor{Landes}-\autor{Posner} (1981):
\begin{formula}
\centering $\mathcal{L}_L=\dfrac{s_L}{|\varepsilon_{Q,p}|+(1-s_L)\,\varepsilon_f}$
\end{formula}
  \item El poder depende de lo que puede crecer la franja, no solo de la cuota.
  \end{itemize}
\end{itemize}
\graficorepaso[0.45\linewidth]{liderazgo-precios}{Empresa dominante y franja competitiva}
\subsub{c) Valoración}
\begin{itemize}
\item $p$ más bajo cuanto más elástica la franja; a LP se erosiona la cuota; puede ser coordinación.
\item \emph{Comparación} ($p=a-bQ$, 2 empresas, CMg $c$):
  \begin{itemize}
  \item producción (en $(a-c)/b$): \autor{Bertrand} $1$ $>$ \autor{Stackelberg} $3/4$ $>$ \autor{Cournot} $2/3$ $>$ monopolio o cártel $1/2$;
  \item PIE (en $(a-c)^2/b$): $0$, $1/32$, $1/18$, $1/8$;
  \end{itemize}
\end{itemize}

\bloque[7]{II. Competencia repetida: colusión tácita\label{rep:II}}
\begin{ideaclave}
Sin un final conocido y con empresas pacientes, la amenaza de una guerra de precios basta para sostener precios altos sin ningún acuerdo: un resultado cooperativo con un equilibrio no cooperativo, posible pero no seguro.
\end{ideaclave}

\sub{II.1. Horizonte finito\label{rep:II.1}}
\begin{itemize}
\item \emph{Supuestos}: \autor{Bertrand} repetido $T$ periodos con final conocido (\concepto{juego repetido}); descuento $\beta\equiv1/(1+\rho)$.
\item \emph{Desarrollo}: inducción hacia atrás $\Rightarrow$ único ENPS, $p=c$ siempre.
  \begin{itemize}
  \item No saca de la paradoja (con varios EN en la etapa, sí puede).
  \end{itemize}
\end{itemize}

\sub{II.2. Horizonte infinito\label{rep:II.2}}
\subsub{a) Idea y b) modelo}
\begin{itemize}
\item \emph{Idea}: sin último periodo, el futuro disciplina el presente (\autor{Friedman}, 1971).
\item \emph{Estrategias}:
  \begin{itemize}
  \item \emph{incondicionada} ($p=c$ siempre): ENPS, no salva la paradoja;
  \item condicionada, \concepto{gatillo} (\emph{trigger}): precio de colusión entre $c$ y $p^M$ mientras todas cumplan; si alguna se desvía, $p=c$ para siempre.
  \end{itemize}
\item \emph{Condición}: $B^{\mathrm C}/(1-\beta)\geq B^{\mathrm D}+\beta B^{\mathrm N}/(1-\beta)$, es decir:
\funcionclave{\beta\;\geq\;\frac{B^{\mathrm D}-B^{\mathrm C}}{B^{\mathrm D}-B^{\mathrm N}}}
  \begin{itemize}
  \item lo que se gana hoy desviándose, entre lo que se pierde cada periodo después.
  \end{itemize}
\item \emph{\autor{Bertrand}}: $B^{\mathrm C}=B^M/n$, $B^{\mathrm D}\approx B^M$, $B^{\mathrm N}=0$ $\Rightarrow$ $\beta\geq1-1/n$ (duopolio: $1/2$).
\item \emph{\autor{Cournot}} lineal: $B^{\mathrm C}=(a-c)^2/8b$, $B^{\mathrm D}=9(a-c)^2/64b$, $B^{\mathrm N}=(a-c)^2/9b$ $\Rightarrow$ $\beta\geq9/17\approx0{,}53$ (castigo más suave).
\item Castigo creíble (EN en cada subjuego) $\Rightarrow$ el gatillo es ENPS.
  \begin{itemize}
  \item \emph{Retraso} $\tau$: $\beta^{\tau}\geq(B^{\mathrm D}-B^{\mathrm C})/(B^{\mathrm D}-B^{\mathrm N})$.
  \item Más fácil cuanto $\uparrow B^{\mathrm C}$, $\uparrow\beta$, $\downarrow B^{\mathrm D}$, $\downarrow B^{\mathrm N}$, $\downarrow\tau$.
  \end{itemize}
\end{itemize}
\graficorepaso[0.7\linewidth]{colusion-tiempo}{Beneficio descontado: ganancia de desviarse frente a pérdida por el castigo}
\begin{itemize}
\item \emph{Implicaciones}: \concepto{teorema de tradición oral} (\emph{folk theorem}).
  \begin{itemize}
  \item Con $\beta$ alto, cualquier $p\in[c,p^M]$ es ENPS: infinitos (también $p=c$); la colusión es \emph{posible}, no segura.
  \item \emph{Tácita}: sin comunicación, difícil de perseguir.
  \end{itemize}
\item \autor{Axelrod} (1984): ganó \emph{tit for tat}, pero solo es ENPS para $\beta$ muy concretos.
\end{itemize}
\subsub{c) Extensiones}
\begin{itemize}
\item \autor{Green}-\autor{Porter} (1984): solo se ve el precio y la demanda es incierta.
  \begin{itemize}
  \item Precio umbral; \emph{guerras de precios en equilibrio} tras caídas de la demanda (\autor{Stigler}, 1964: rebajas secretas).
  \end{itemize}
\item \autor{Rotemberg}-\autor{Saloner} (1986): demanda observada que fluctúa.
  \begin{itemize}
  \item Guerras en los \emph{auges}: márgenes anticíclicos.
  \end{itemize}
\item \emph{Castigos óptimos} (\autor{Abreu}): «palo y zanahoria»; la \emph{renegociación} debilita la amenaza.
\end{itemize}
\subsub{d) Evidencia empírica}
\begin{itemize}
\item Cártel ferroviario JEC, 1880-1886 (\autor{Porter}, 1983).
\item Algoritmos de precios que coluden solos (\autor{Calvano} et al., 2020).
\end{itemize}
\subsub{e) Valoración: qué facilita y qué dificulta la colusión}
\begin{itemize}
\item Factores (\autor{Ivaldi} et al., 2003):
  \begin{itemize}
  \item (+) barreras, frecuencia, transparencia, demanda creciente, contacto multimercado, prácticas facilitadoras;
  \item ($-$) nº de empresas, asimetrías, innovación, compradores grandes, renegociación;
  \item (±) diferenciación, capacidad excedente.
  \end{itemize}
\item La colusión \emph{puede} sostenerse, pero el modelo no dice qué precio ni cómo repartir $\Rightarrow$ cártel.
\end{itemize}

\bloque[8,5]{III. Soluciones cooperativas: el cártel y el reparto\label{rep:III}}
\begin{ideaclave}
El cártel maximiza el beneficio conjunto como un monopolista multiplanta, pero el acuerdo no es un equilibrio de \autor{Nash}; lo ganado se reparte según lo que cada una obtendría compitiendo, y la entrada y la ley lo limitan.
\end{ideaclave}

\sub{III.1. Colusión explícita: el cártel\label{rep:III.1}}
\subsub{a) Idea}
\begin{itemize}
\item \concepto{Colusión}: ejercicio conjunto del poder de mercado.
  \begin{itemize}
  \item \emph{explícita} (\concepto{cártel}: precios, cuotas, clientes o territorios; prohibido \vertema{3.B.39}) o \emph{tácita}.
  \end{itemize}
\item \autor{Stigler} (1950): beneficio conjunto máximo actuando como un monopolista.
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item \emph{Desarrollo}: monopolista con 2 plantas \vertema{3.A.17}: \emph{cuánto} (IMg = suma horizontal de CMg) y \emph{cómo repartir} (igualar CMg):
\begin{formula}
\centering $\optim{\max}{q_1,q_2}{p(q_1+q_2)(q_1+q_2)-C_1(q_1)-C_2(q_2)}{q_1,q_2\geq0}$\\[3pt]
$\Rightarrow\ \mathrm{IMg}(Q)=\mathrm{CMg}_1(q_1)=\mathrm{CMg}_2(q_2)$
\end{formula}
  \begin{itemize}
  \item CMg constantes distintos: solo produce la eficiente $\Rightarrow$ \emph{pagos laterales}.
  \end{itemize}
\end{itemize}
\begin{itemize}
\item En $(q_1,q_2)$: \concepto{curva de colusión} (tangencias de isobeneficios) de $(q^M,0)$ a $(0,q^M)$; C, fuera de ella.
\end{itemize}
\graficorepaso[0.45\linewidth]{cartel}{Curva de colusión e incentivo a desviarse}
\begin{itemize}
\item \emph{Implicaciones}:
  \begin{itemize}
  \item \concepto{colusión perfecta} ($p^M$ con reparto de coste mínimo) e \emph{imperfecta} (cualquier punto de la «lente»);
  \item \emph{no es EN}: en la cuota,
\begin{formula}
\centering $\dfrac{\partial B_1}{\partial q_1}\Big|_{\text{cártel}}=p+p'(Q)\,q_1-\mathrm{CMg}_1=-p'(Q)\,q_2>0$
\end{formula}
  \item \concepto{dilema del prisionero}: $p=13-Q$, $c=1$: cuota 3 y 3 $\to$ (18; 18); \autor{Cournot} 4 y 4 $\to$ (16; 16); uno se desvía $\to$ (20; 15);
  \item se sostiene por repetición o por un tercero (contrato imposible: es ilegal); los más duraderos, entre Estados (OPEP).
  \end{itemize}
\end{itemize}
\subsub{c) Extensiones: cárteles parciales y su estabilidad}
\begin{itemize}
\item \concepto{Paradoja de las fusiones} (\autor{Salant}-\autor{Switzer}-\autor{Reynolds}, 1983):
  \begin{itemize}
  \item con \autor{Cournot} lineal no compensa salvo $\approx$80\,\% del mercado (3 iguales: 2 juntas $1/9<2/16$);
  \item el que se queda fuera gana más ($1/9>1/12$).
  \end{itemize}
\item Estabilidad \emph{interna} y \emph{externa}: cártel = dominante con franja; se forma con pocas (\autor{Selten}, 1973).
\end{itemize}

\sub{III.2. Reparto: \autor{Nash}, núcleo y \autor{Shapley}\label{rep:III.2}}
\subsub{a) Idea y b) modelo}
\begin{itemize}
\item \emph{Supuestos}: acuerdos vinculantes y \concepto{utilidad transferible}; \concepto{punto de desacuerdo} = beneficio de \autor{Cournot}, $B_i^0$.
\item \emph{2 empresas}: \concepto{solución de negociación de \autor{Nash}} (1950), $\gamma$ = poder de la 1:
\begin{formula}
\centering $\optim{\max}{B_1,B_2}{(B_1-B_1^0)^{\gamma}(B_2-B_2^0)^{1-\gamma}}{B_1+B_2=B^M}$\\[3pt]
$B_1=B_1^0+\gamma\,(B^M-B_1^0-B_2^0)$
\end{formula}
  \begin{itemize}
  \item Ej.: $p=13-Q$, $c_1=1$, $c_2=4$: \autor{Cournot} $B^0=(25;4)$; cártel solo con la 1: $B^M=36$, excedente 7.
  \item $\gamma=1/2$: la 1 se queda 28,5 y paga 7,5 a la 2 por no producir.
  \end{itemize}
\item \emph{3 empresas} ($p=4-Q$, $c=0$): $v(\{i\})=1$, $v(\{i,j\})=16/9$, $v(N)=4$.
  \begin{itemize}
  \item \concepto{Núcleo} (\autor{Gillies}, 1959): repartos que ninguna coalición mejora: $x_i\geq1$, $\sum x_i=4$ (no vacío).
  \item \concepto{Valor de \autor{Shapley}} (1953): aportación marginal media: $1$, $7/9$, $20/9$ $\Rightarrow$ $\phi_i=4/3$, en el núcleo.
  \end{itemize}
\end{itemize}
\graficorepaso[0.85\linewidth]{reparto-cartel}{a) Negociación de \autor{Nash}; b) núcleo y valor de \autor{Shapley} con 3}
\begin{itemize}
\item \emph{Implicaciones}:
  \begin{itemize}
  \item reparto según opciones externas y poder; la ineficiente puede cobrar por no producir;
  \item los cárteles parciales no compensan ($16/9<2$);
  \item \emph{estabilidad cooperativa $\neq$ no cooperativa}: fuera ganaría $16/9>4/3$: el núcleo no basta, hace falta poder hacerlo cumplir.
  \end{itemize}
\end{itemize}
\subsub{c) Valoración}
\begin{itemize}
\item Describe mejor la cooperación legal (empresas conjuntas, I+D) que los cárteles (cuotas históricas).
\end{itemize}

\sub{III.3. Límites: entrada y defensa de la competencia\label{rep:III.3}}
\subsub{a) La amenaza de entrada}
\begin{itemize}
\item Barreras legales, de factores, técnicas o estratégicas \vertema{3.A.15}.
\item \concepto{Precio límite} (\autor{Bain}, 1956; \autor{Modigliani}, 1958): solo disuade si es creíble (ENPS): compromiso de capacidad (\autor{Dixit}, 1980).
\item \concepto{Mercados impugnables} (\autor{Baumol}-\autor{Panzar}-\autor{Willig}, 1982):
  \begin{itemize}
  \item sin costes hundidos: \emph{hit and run} obliga a $p=\mathrm{CMe}$ aun con una empresa.
  \end{itemize}
\end{itemize}
\subsub{b) La defensa de la competencia}
\begin{itemize}
\item \emph{Prohibición}: art.~101 TFUE; art.~1 Ley 15/2007; muy grave, hasta el 10\,\% del volumen de negocios (art.~63).
\item \emph{Prueba}: el paralelismo solo prueba concertación sin otra explicación plausible: la tácita pura no es ilegal.
\item \emph{Clemencia} (arts.~65-66 Ley 15/2007):
  \begin{itemize}
  \item exención a la 1.ª que aporte pruebas; reducciones 30-50\,\%, 20-30\,\%, hasta 20\,\%;
  \item en el gatillo: $\downarrow B^{\mathrm C}$ (multa) y $\uparrow B^{\mathrm D}$ (exención) $\Rightarrow$ $\uparrow$ umbral;
  \item carrera por confesar (\autor{Motta}-\autor{Polo}, 2003) \vertema{4.A.15}.
  \end{itemize}
\end{itemize}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}:
  \begin{itemize}
  \item una vez: entre competencia y monopolio según variable y orden, $\mathcal{L}_i=s_i/|\varepsilon|$;
  \item indefinidamente: gatillo, colusión tácita si pocas y pacientes;
  \item acuerdo: monopolista multiplanta, reparto según opciones externas, pero no es equilibrio.
  \end{itemize}
\item \textbf{Relevancia}: estructura de los mercados que más importan; guía la defensa de la competencia (cuota e IHH, concentraciones, cárteles).
\item \textbf{Extensiones}: \vertema{3.A.18} \vertema{3.A.14} \vertema{3.A.15} \vertema{3.A.20} \vertema{3.B.39} \vertema{4.A.15} \vertema{3.B.8} \vertema{3.B.6}; algoritmos.
\item \textbf{Opinión}: no hay un modelo único; la política de competencia no puede limitarse a contar empresas: qué hace estable la colusión en cada mercado e instituciones (clemencia) que la desestabilicen.
\item \textbf{Idea final}: compiten porque a cada una le conviene romper el acuerdo.
  \begin{itemize}
  \item La avaricia rompe el saco si desviarse hoy gana más que el castigo de después.
  \item La clemencia usa esa avaricia: al primero que confiesa, como MAN, le sale a cuenta.
  \end{itemize}
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Qué pasa con más de 2 empresas?} \autor{Cournot}: $p\to c$; \autor{Bertrand}: basta con 2; colusión más difícil, $\beta\geq1-1/n$.
\item \textbf{¿\autor{Cournot} o \autor{Bertrand}?} Según qué está fijo: con capacidades dadas, \autor{Cournot} (\autor{Kreps}-\autor{Scheinkman}).
\item \textbf{¿Dónde se produce más?} \autor{Bertrand} $>$ \autor{Stackelberg} $>$ \autor{Cournot} $>$ monopolio.
\item \textbf{¿Finito o infinito?} Final conocido: la inducción hacia atrás deshace la colusión; sin final, el gatillo la sostiene.
\item \textbf{¿Qué es la colusión perfecta?} Monopolio con reparto de coste mínimo: $\mathrm{IMg}(Q)=\mathrm{CMg}_1=\mathrm{CMg}_2$.
\item \textbf{Monopolio natural liberalizado: ¿estable?} Pocas, transparencia, barreras: colusión tácita $\Rightarrow$ vigilar \vertema{3.A.17}.
\end{enumerate}
\begin{trampa}
\begin{itemize}
\item Colusión $\neq$ juego cooperativo: la de \autor{Friedman} es un EN no cooperativo.
\item $\mathcal{L}_i=s_i/|\varepsilon_{Q,p}|$: elasticidad de \emph{mercado}.
\item \autor{Stackelberg}: ventaja de \emph{compromiso}, no de información.
\item \emph{Tit for tat} no es en general ENPS.
\item \autor{Green}-\autor{Porter}: guerras tras \emph{caídas}; \autor{Rotemberg}-\autor{Saloner}: en los \emph{auges}.
\end{itemize}
\end{trampa}
