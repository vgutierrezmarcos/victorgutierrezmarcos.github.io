\bloque[2]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: pensiones revalorizadas con el IPC (\dato{+2,7\,\% en 2026}{LGSS}): ¿basta para quedar igual? Importa cuánto cuesta \emph{volver a estar igual de bien} $\Rightarrow$ \concepto{función de gasto}.
\item \textbf{Relevancia}: teórica (\autor{Slutsky} en dos líneas, integrabilidad); empírica (elasticidades para IVA, IPC, pobreza); normativa (VC y VE: coste-beneficio, exceso de gravamen).
\item \textbf{Contexto}: dualidad (\autor{Hotelling} 1932 $\to$ \autor{McKenzie} 1957, \autor{Diewert}); integrabilidad (\autor{Antonelli} 1886 $\to$ \autor{Hurwicz}-\autor{Uzawa} 1971); bienestar (\autor{Dupuit} 1844 $\to$ \autor{Hicks}, \autor{Willig}); sistemas (\autor{Stone} 1954 $\to$ AIDS 1980).
\item \textbf{Problemática}: ¿qué se gana al «dar la vuelta» al problema? ¿Cómo se estiman las demandas? ¿Qué demandas son racionales? ¿Cómo medir el bienestar en euros?
\item \textbf{Estructura}: I. dualidad; II. sistemas de demanda; III. aplicaciones (integrabilidad, \autor{Slutsky}, bienestar).
\end{itemize}

\bloque[10]{I. Análisis teórico de la dualidad}
\sub{I.1. Ingredientes comunes}
\begin{itemize}
\item Mismos ingredientes, papeles cambiados: $S\subseteq\mathbb{R}^n_+$; preferencias (completas, reflexivas, transitivas, continuas $\Rightarrow$ $U$ continua, \autor{Debreu}; insaciab.\ local; convexidad estricta; diferenciab.); precios $\vect{p}\gg\vect{0}$. Estático, info.\ perfecta, precio-aceptante.
\item \concepto{Primal}: objetivo $U$, restricción $\vect{p}\cdot\vect{x}\leq\overline{W}$. \concepto{Dual}: objetivo $\vect{p}\cdot\vect{h}$, restricción $U(\vect{h})\geq\overline{U}$.
\end{itemize}

\sub{I.2. Primal y dual}
\begin{formula}
PMU: $\max_{\vect{x}} U(\vect{x})$ s.a.\ $\vect{p}\cdot\vect{x}\leq\overline{W}$\ \ $\Rightarrow\dfrac{U_i}{p_i}=\lambda$\\
PMG: $\min_{\vect{h}}\vect{p}\cdot\vect{h}$ s.a.\ $U(\vect{h})\geq\overline{U}$\ \ $\Rightarrow\dfrac{p_i}{U_i}=\mu$\\
Ambos: $RMS_{12}=U_1/U_2=p_1/p_2$;\ \ $\mu^*=1/\lambda^*$
\end{formula}
\begin{itemize}
\item $\lambda$ = UMg de la renta $=\partial V/\partial\overline{W}$; $\mu$ = coste marginal de la utilidad $=\partial E/\partial\overline{U}$.
\item \concepto{Marshalliana} $\vect{x}(\vect{p},\overline{W})$: existe, única, continua (\autor{Berge}), homog.\ grado 0 en $(\vect{p},\overline{W})$, ley de \autor{Walras}. \concepto{FIU} $V(\vect{p},\overline{W})$: homog.\ 0, estr.\ creciente en $\overline{W}$, no creciente en $\vect{p}$, \textbf{cuasiconvexa}; \autor{Roy} (1947).
\item \concepto{Hicksiana} $\vect{h}(\vect{p},\overline{U})$: homog.\ 0 en $\vect{p}$, sin exceso de utilidad, \concepto{ley de la demanda compensada} $\partial h_i/\partial p_i\leq0$ (no hay \autor{Giffen}). \textbf{No observable} (depende de $\overline{U}$).
\item \concepto{Función de gasto} $E(\vect{p},\overline{U})=\vect{p}\cdot\vect{h}$: homog.\ \textbf{grado 1} en $\vect{p}$, estr.\ creciente en $\overline{U}$, no decreciente en $\vect{p}$, \textbf{cóncava} en $\vect{p}$ (sustituir abarata respecto a la cesta fija). Mide la utilidad en \emph{euros}.
\end{itemize}
\begin{formula}
\autor{Roy}: $x_i=-\dfrac{\partial V/\partial p_i}{\partial V/\partial\overline{W}}$\qquad \autor{Shephard}: $h_i=\dfrac{\partial E}{\partial p_i}$\\
\autor{Slutsky}: $S_{ij}=\dfrac{\partial^2E}{\partial p_i\partial p_j}=\dfrac{\partial h_i}{\partial p_j}$
\end{formula}
\begin{itemize}
\item $S$: \textbf{simétrica} (\autor{Schwarz}), \textbf{semidefinida negativa} ($E$ cóncava), $S\vect{p}=\vect{0}$ (\autor{Euler}).
\end{itemize}
\graficorepaso[0.7\linewidth]{concavidad-fg}{$E$ cóncava: siempre bajo su plano tangente (pendiente $h_i$, \autor{Shephard})}

\sub{I.3. Relaciones de dualidad}
\begin{itemize}
\item \concepto{Teorema básico} (MWG 3.E.1): con $U$ continua, insaciab.\ local y $\vect{p}\gg\vect{0}$, la cesta del PMU resuelve el PMG con $\overline{U}=U^*$, y viceversa con $\overline{W}=E^*$.
\end{itemize}
\graficorepaso[0.72\linewidth]{primal-dual}{Misma cesta: $\vect{x}^*=\vect{h}^*$ con $\overline{W}=E^*$, $\overline{U}=U^*$}
\begin{formula}
\raggedright Demandas: $\vect{x}(\vect{p},\overline{W})\equiv\vect{h}\big(\vect{p},V(\vect{p},\overline{W})\big)$\\ \phantom{Demandas: }$\vect{h}(\vect{p},\overline{U})\equiv\vect{x}\big(\vect{p},E(\vect{p},\overline{U})\big)$\\
Valor: $E\big(\vect{p},V(\vect{p},\overline{W})\big)\equiv\overline{W}$\\ \phantom{Valor: }$V\big(\vect{p},E(\vect{p},\overline{U})\big)\equiv\overline{U}$
\end{formula}
\begin{itemize}
\item $V$ y $E$ \textbf{inversas} en su 2.º argumento. Basta resolver \emph{un} problema para las cuatro funciones.
\item \emph{Cobb-Douglas}: $x_1=\alpha\overline{W}/p_1$;\\ $E=\overline{U}(p_1/\alpha)^{\alpha}(p_2/(1-\alpha))^{1-\alpha}$; $h_1=\alpha E/p_1$.
\end{itemize}
\begin{ideaclave}
Primal y dual son el mismo problema visto desde dos lados; $E$ mide la utilidad en dinero, sus derivadas son las hicksianas y su concavidad da la ley de la demanda compensada.
\end{ideaclave}

\bloque[5]{II. Sistemas completos de demanda}
\sub{II.1. Idea}
\begin{itemize}
\item Explican a la vez el gasto en \emph{todos} los bienes. Fines: \textbf{medir} elasticidades, \textbf{contrastar} la teoría, \textbf{valorar} el bienestar.
\item Dualidad: se especifica $E$ o $V$ y se deriva (\autor{Shephard}/\autor{Roy}) $\Rightarrow$ coherente con la maximización \emph{por construcción}.
\item \concepto{Requisitos}: coherencia teórica, flexibilidad, agregación, sencillez (lineal).
\end{itemize}

\sub{II.2. Pasos}
\begin{itemize}
\item (1) forma funcional; (2) sistema completo; (3) datos; (4) econometría; (5) estimar y contrastar.
\item \concepto{Cobb-Douglas}: $s_i=\alpha_i$ constantes; $\varepsilon_W=1$, $\varepsilon_{ii}=-1$, cruzadas 0; no recoge \autor{Engel}.
\item \concepto{LES} (\autor{Stone}-\autor{Geary}, 1954): $U=\sum\alpha_i\ln(x_i-\gamma_i)$; $p_ix_i=p_i\gamma_i+\alpha_i(\overline{W}-\sum p_j\gamma_j)$: subsistencia + renta supernumeraria. \autor{Engel} rectas sin pasar por el origen. Límite: \emph{aditivo} (sin inferiores ni complementarios netos).
\item \concepto{Rotterdam} (\autor{Barten}, \autor{Theil}): \autor{Slutsky} en diferencias log.; restricciones lineales; sin $E$ explícita.
\item \concepto{Translog} (\autor{Christensen}-\autor{Jorgenson}-\autor{Lau}, 1975): $\ln V$ de 2.º orden, flexible; no lineal, no agrega.
\item \concepto{AIDS} (\autor{Deaton}-\autor{Muellbauer}, 1980): $E$ \textbf{PIGLOG}, $\ln E=(1-\overline{U})\ln a(\vect{p})+\overline{U}\ln b(\vect{p})$.
\end{itemize}
\begin{formula}
\raggedright $s_i=\alpha_i+\sum_j\gamma_{ij}\ln p_j+\beta_i\ln(\overline{W}/P)$\\
Agregación: $\sum_i\alpha_i=1$, $\sum_i\gamma_{ij}=\sum_i\beta_i=0$\\ Homog.: $\sum_j\gamma_{ij}=0$;\ simetría: $\gamma_{ij}=\gamma_{ji}$;\ $\varepsilon_{x_i,\overline{W}}=1+\dfrac{\beta_i}{s_i}$
\end{formula}
\begin{itemize}
\item $\beta_i>0$ lujo, $\beta_i<0$ necesario (\autor{Working}-\autor{Leser}). Agregación exacta; \emph{LA-AIDS} con índice de \autor{Stone}, MCO. Negatividad no garantizada (\emph{casi} ideal). \concepto{QUAIDS} (\autor{Banks}-\autor{Blundell}-\autor{Lewbel}, 1997): término cuadrático, rango 3.
\item \concepto{Sistema completo}: $\sum s_i=1$; agrupación por \concepto{separabilidad débil} y presupuestación en dos etapas (\autor{Strotz}, \autor{Gorman}).
\item \concepto{Datos}: EPF del INE (\dato{24.000 hogares}{anual desde 2006}); precios: años y regiones, o valores unitarios (\autor{Deaton}). \concepto{Econometría}: covarianzas singulares $\Rightarrow$ se quita una ecuación (MV: invariante, \autor{Barten}); gasto endógeno; ceros; demografía.
\item \concepto{Resultados}: con datos agregados se rechaza la \textbf{homogeneidad} (\autor{Barten}, \autor{Deaton} 1974, AIDS); \autor{Deaton}: agregación, dinámica $\Rightarrow$ microdatos. España: AIDS con EPF para reformas del IVA (\autor{Sanz}-\autor{Romero}, 2012).
\end{itemize}

\sub{II.3. Valoración}
\begin{itemize}
\item \textbf{Aportes}: teoría $\to$ instrumento de medida; restricciones contrastables. \textbf{Límites}: flexibilidad vs.\ parsimonia ($n(n+1)/2$ parámetros); agregación (SMD); marco estático.
\end{itemize}

\bloque[12]{III. Aplicaciones de la dualidad}
\sub{III.1. Integrabilidad}
\begin{itemize}
\item ¿Cuándo procede $\vect{x}(\vect{p},\overline{W})$ de maximizar una $U$? \autor{Antonelli} (1886); \autor{Hurwicz}-\autor{Uzawa} (1971).
\item \concepto{Teorema}: $\vect{x}$ $C^1$ es demanda de una $U$ creciente y cuasicóncava $\Leftrightarrow$ (1) ley de \autor{Walras}, (2) $S$ \textbf{simétrica}, (3) $S$ \textbf{semidef.\ negativa} (+ homogeneidad, necesaria). Necesarias y suficientes.
\end{itemize}
\begin{formula}
\raggedright $S_{ij}=\dfrac{\partial x_i}{\partial p_j}+x_j\dfrac{\partial x_i}{\partial\overline{W}}$\ \ (observable)\\ EDP: $\nabla_{\vect{p}}E(\vect{p},\overline{U})=\vect{x}\big(\vect{p},E(\vect{p},\overline{U})\big)$
\end{formula}
\begin{itemize}
\item Demostración: EDP con solución $\Leftrightarrow$ simetría (\concepto{\autor{Frobenius}}); negatividad $\Rightarrow$ $E$ cóncava; $U(\vect{x}^0)=\min V(\vect{p},\overline{W})$ s.a.\ $\vect{p}\cdot\vect{x}^0\leq\overline{W}$.
\item Con 2 bienes, simetría automática. Preferencia revelada: ADPR $\Rightarrow$ $S$ semidef.\ neg., no simétrica; AFPR (\autor{Houthakker}) sí; \autor{Afriat} (datos finitos).
\item Relevancia: cierra el círculo; contraste sin hallar $U$; bienestar exacto (\autor{Hausman}, 1981).
\end{itemize}

\sub{III.2. Ecuación de \autor{Slutsky}}
\begin{itemize}
\item Derivar $h_i(\vect{p},\overline{U})\equiv x_i\big(\vect{p},E(\vect{p},\overline{U})\big)$ respecto a $p_j$ + \autor{Shephard} ($\partial E/\partial p_j=x_j$):
\end{itemize}
\begin{formula}
\centering $\underbrace{\dfrac{\partial x_i}{\partial p_j}}_{ET}=\underbrace{\dfrac{\partial h_i}{\partial p_j}}_{ES}\ \underbrace{-\,x_j\dfrac{\partial x_i}{\partial\overline{W}}}_{ER}$\\[1pt] Elasticidades: $\varepsilon_{x_i,p_j}=\varepsilon^h_{i,p_j}-s_j\,\varepsilon_{x_i,\overline{W}}$
\end{formula}
\begin{itemize}
\item ES inobservable; ET y ER observables $\Rightarrow$ $S$ calculable. Compensación de \autor{Hicks} (utilidad) $\approx$ de \autor{Slutsky} (poder adquisitivo) en el margen.
\item \textbf{Propio}: ES $\leq0$; \textcolor{tceeverde}{\textbf{normal}} $\Rightarrow$ \textcolor{tceeazul}{\textbf{ordinario}}; \textcolor{tceedorado}{\textbf{inferior}}: ordinario si domina ES, \textcolor{tceerojo}{\textbf{\autor{Giffen}}} si domina ER ($s_i$ alto). \autor{Veblen} fuera.
\item \textbf{Cruzado}: neto (signo de $S_{ij}$) \textbf{simétrico} (\autor{Hicks}-\autor{Allen}, 1934); bruto no. \textcolor{tceeverde}{Sustitutivos netos} + $i$ inferior $\Rightarrow$ \textcolor{tceeverde}{sustitutivos brutos}; \textcolor{tceedorado}{complementarios netos} + $i$ normal $\Rightarrow$ \textcolor{tceedorado}{complementarios brutos}.
\item $S\vect{p}=\vect{0}\Rightarrow$ todo bien tiene un \concepto{sustitutivo neto}; con 2 bienes, sustitutivos netos.
\item Bien normal: marshalliana más plana que la hicksiana.
\end{itemize}

\sub{III.3. Bienestar}
\begin{itemize}
\item Utilidad ordinal: $\overline{U}^1-\overline{U}^0$ no significa nada $\Rightarrow$ medida en euros.
\item \concepto{$\Delta$EC} (\autor{Dupuit}, \autor{Marshall}): $\int_{p_1^1}^{p_1^0}x_1\,dp_1$, observable pero mezcla ES y ER; exacta con ER nulo (cuasilineal); con varios precios depende del camino.
\item \concepto{Métrica monetaria} (\autor{McKenzie}, \autor{Samuelson}): $m(\bar{\vect{p}};\vect{p},\overline{W})\equiv E\big(\bar{\vect{p}},V(\vect{p},\overline{W})\big)$: transformación creciente de $V$ medida en euros; VC y VE son su variación con $\bar{\vect{p}}=\vect{p}^1$ y $\bar{\vect{p}}=\vect{p}^0$.
\end{itemize}
\begin{formula}
$\mathrm{VC}=\overline{W}-E(\vect{p}^1,\overline{U}^0)=\int h_1(\cdot,\overline{U}^0)\,dp_1$\\
$\mathrm{VE}=E(\vect{p}^0,\overline{U}^1)-\overline{W}=\int h_1(\cdot,\overline{U}^1)\,dp_1$
\end{formula}
\begin{itemize}
\item \concepto{VC}: precios \emph{finales}, utilidad \emph{inicial}; si baja $p$, máximo que pagaría. \concepto{VE}: precios \emph{iniciales}, utilidad \emph{final}; mínimo que aceptaría por renunciar.
\item Bien \textcolor{tceeverde}{normal}, bajada de precio: $L\leq\mathrm{VC}\leq\Delta\mathrm{EC}\leq\mathrm{VE}\leq P$; inferior: se invierte; ER nulo: VC $=\Delta$EC $=$ VE.
\end{itemize}
\graficorepaso[0.78\linewidth]{bienestar-vc-ve}{VC, $\Delta$EC y VE (y $L$, $P$) ante una bajada de $p_1$ de un bien normal}
\begin{itemize}
\item \concepto{\autor{Willig}} (1976): $\mathrm{VC}\approx\Delta\mathrm{EC}-\dfrac{\varepsilon_{x_1,\overline{W}}(\Delta\mathrm{EC})^2}{2\overline{W}}$; error $\approx|\varepsilon|\,|\Delta\mathrm{EC}|/2\overline{W}$ (5\,\% de la renta y $\varepsilon=1$: 2,5\,\%). \autor{Hausman} (1981): falla en la pérdida irrecuperable.
\item \textbf{¿Cuál?} Ordenar políticas: \textbf{VE} (base fija $\vect{p}^0$). Compensar (\autor{Kaldor}-\autor{Hicks}): \textbf{VC}. Exceso de gravamen: $\mathrm{EG}=-\mathrm{VE}-T>0$.
\item \concepto{Info.\ incompleta}: $L=(\vect{p}^0-\vect{p}^1)\cdot\vect{x}^0\leq\mathrm{VC}$, $\mathrm{VE}\leq P=(\vect{p}^0-\vect{p}^1)\cdot\vect{x}^1$: \emph{sesgo de sustitución}.
\end{itemize}
\begin{formula}
\autor{Konüs}: $I_K=\dfrac{E(\vect{p}^1,\overline{U})}{E(\vect{p}^0,\overline{U})}$;\ \ $I_L=\dfrac{\vect{p}^1\cdot\vect{x}^0}{\vect{p}^0\cdot\vect{x}^0}\geq I_K(\overline{U}^0)$,\ \ $I_P=\dfrac{\vect{p}^1\cdot\vect{x}^1}{\vect{p}^0\cdot\vect{x}^1}\leq I_K(\overline{U}^1)$
\end{formula}
\graficorepaso[\linewidth]{indice-konus-laspeyres}{\autor{Laspeyres} sobrestima: sesgo de sustitución}
\begin{itemize}
\item \autor{Laspeyres}: aprox.\ de 1.er orden de $E$, cota superior (concavidad). Solo con homotéticas $I_P\leq I_K\leq I_L$. \concepto{Superlativos} (\autor{Diewert}, 1976): \autor{Fisher} $\sqrt{I_LI_P}$, \autor{Törnqvist} (translog).
\item IPC: \autor{Laspeyres} encadenado (base 2025); deflactor del PIB: \autor{Paasche} encadenado.
\end{itemize}
\begin{ideaclave}
$E$ traduce la utilidad a euros: VC y VE exactas; $\Delta$EC las aproxima (\autor{Willig}); \autor{Laspeyres} y \autor{Paasche} las acotan con sesgo de sustitución.
\end{ideaclave}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Por qué $\mu^*=1/\lambda^*$?} Derivando $E(\vect{p},V(\vect{p},\overline{W}))=\overline{W}$ respecto a $\overline{W}$: $\frac{\partial E}{\partial\overline{U}}\frac{\partial V}{\partial\overline{W}}=1$.
\item \textbf{¿Por qué la hicksiana no admite \autor{Giffen}?} No tiene efecto renta y $E$ es cóncava: $\partial h_i/\partial p_i\leq0$.
\item \textbf{¿Por qué $S$ es simétrica?} Son segundas derivadas de $E$ (\autor{Schwarz}); las brutas no, porque el ER cruzado difiere.
\item \textbf{¿Qué condiciones da la integrabilidad?} Ley de \autor{Walras}, simetría y negatividad de $S$; con 2 bienes la simetría es automática.
\item \textbf{¿Por qué es «casi ideal» el AIDS?} Flexible, coherente, agrega exactamente y es lineal (LA-AIDS), pero la negatividad no está garantizada y las curvas de \autor{Engel} son restrictivas.
\item \textbf{¿VC o VE?} VE para ordenar políticas (base fija); VC para compensar a los perjudicados.
\item \textbf{¿Cuándo vale el excedente marshalliano?} Si el ER es pequeño (\autor{Willig}): bien que pesa poco en la renta.
\item \textbf{¿Por qué el IPC sobrestima el coste de la vida?} \autor{Laspeyres} fija la cesta e ignora la sustitución; la concavidad de $E$ lo hace cota superior.
\item \textbf{¿Cómo se demuestra la ecuación de \autor{Slutsky} por dualidad?} Derivando $h_i(\vect{p},\overline{U})\equiv x_i(\vect{p},E(\vect{p},\overline{U}))$ respecto a $p_j$ y usando \autor{Shephard}.
\item \textbf{¿Cómo se mide el exceso de gravamen?} Con la VE: $\mathrm{EG}=-\mathrm{VE}-T>0$; el consumidor pagaría más de $T$ por un impuesto de suma fija equivalente.
\item \textbf{¿Qué aporta la dualidad a la estimación?} Especificar $E$ o $V$ y derivar: demandas coherentes con la maximización sin resolver ningún problema; restricciones contrastables.
\end{enumerate}
