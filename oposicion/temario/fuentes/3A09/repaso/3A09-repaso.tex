\bloque[10]{I. Análisis teórico de la dualidad\label{rep:I}}
\begin{ideaclave}
Maximizar la utilidad con una renta y minimizar el gasto con una utilidad son el mismo problema visto desde 2 lados. $E$ mide la utilidad en dinero, sus derivadas son las hicksianas (\autor{Shephard}) y su concavidad da la ley de la demanda compensada.
\end{ideaclave}

\sub{I.1. Ingredientes comunes\label{rep:I.1}}
\begin{itemize}
\item \concepto{Primal}: recursos dados, máx.\ utilidad; \concepto{dual}: utilidad dada, mín.\ gasto. «Dual» en sentido \emph{estricto}: intercambiar objetivo y restricción.
\item \emph{Ingredientes}: $S\subseteq\mathbb{R}^n_+$; preferencias racionales, continuas, localmente insaciables, estr.\ convexas; $U$ continua, creciente, estr.\ cuasicóncava \vertema{3.A.8}; gasto $\vect{p}\cdot\vect{x}$.
\item \emph{Supuestos}: estático, cierto, información perfecta, precio-aceptante.
\end{itemize}

\toprule

\midrule
Objetivo & $U(\vect{x})$ (máx.) & $\vect{p}\cdot\vect{h}$ (mín.)\\

\sub{I.2. Problemas primal y dual\label{rep:I.2}}
\subsub{a) Problema primal: maximización de la utilidad}
\begin{itemize}
\item \emph{Desarrollo}: existe (\autor{Weierstrass}), global y único; $\lambda$ = \concepto{utilidad marginal de la renta}:
\begin{formula}
\centering $\optim{\max}{\vect{x}}{U(\vect{x})}{\vect{p}\cdot\vect{x}\leq\overline{W}\\ \vect{x}\geq\vect{0}}$\quad$\Rightarrow$\quad $\dfrac{U_i}{p_i}=\lambda$,\ \ $\mathrm{RMS}_{12}=\dfrac{p_1}{p_2}$
\end{formula}
\item \emph{Implicaciones}:
  \begin{itemize}
  \item \concepto{demanda marshalliana}: única; continua; grado 0; ley de \autor{Walras};
  \item \concepto{FIU} $V(\vect{p},\overline{W})$: \emph{cuasiconvexa} (no exige preferencias convexas); identidad de \autor{Roy}.
  \end{itemize}
\item \emph{Valoración}: $V$ en «útiles», solo ordena $\Rightarrow$ ¿y si se da la vuelta?
\end{itemize}
\graficorepaso[0.36\linewidth]{cuasiconvexidad-fiu}{$V$ cuasiconvexa: con el presupuesto promedio no se alcanza más que con el mejor extremo}
\subsub{b) Problema dual: minimización del gasto}
\begin{itemize}
\item \emph{Desarrollo}: $p_i/U_i=\mu$: \emph{misma} tangencia; $\mu^*=\partial E/\partial\overline{U}$, \concepto{coste marginal de la utilidad} $=1/\lambda^*$.
\end{itemize}
\pareja{primal}{Primal: la curva de indiferencia más alta alcanzable con $\overline{W}$}{dual}{Dual: la isogasto más baja que alcanza $\overline{U}$}
\begin{itemize}
\item \concepto{Demanda hicksiana} $\vect{h}(\vect{p},\overline{U})$: grado 0 en $\vect{p}$; \concepto{ley de la demanda compensada} $\partial h_i/\partial p_i\leq0$ $\Rightarrow$ sin \autor{Giffen}. \emph{No observable}.
\item \concepto{Función de gasto}: gasto mínimo para alcanzar $\overline{U}$; utilidad en \emph{euros}:
\funcionclave{E(\vect{p},\overline{U})=\vect{p}\cdot\vect{h}(\vect{p},\overline{U})=\optim{\min}{\vect{h}}{\vect{p}\cdot\vect{h}}{U(\vect{h})\geq\overline{U}\\ \vect{h}\geq\vect{0}}}
\item \emph{Propiedades de $E$}: homog.\ de \emph{grado 1} en $\vect{p}$; creciente en $\overline{U}$; no decreciente en $p_i$; \emph{cóncava} en $\vect{p}$ (el consumidor «pasivo» pagaría $\vect{p}\,'\cdot\vect{h}$; sustituyendo, como mucho eso):
\begin{formula}
\centering $E(\vect{p}\,',\overline{U})\leq\vect{p}\,'\cdot\vect{h}(\vect{p},\overline{U})$;\qquad \autor{Shephard}: $h_i=\dfrac{\partial E(\vect{p},\overline{U})}{\partial p_i}$\\[4pt]
$S_{ij}\equiv\dfrac{\partial^2E}{\partial p_i\,\partial p_j}=\dfrac{\partial h_i}{\partial p_j}$\quad (\concepto{matriz de \autor{Slutsky}})
\end{formula}
  \begin{itemize}
  \item \emph{Lema de \autor{Shephard}} (1953): si $p_i$ sube 1 €, la cesta apenas cambia (envolvente): el gasto sube $h_i$ €.
  \item $S$: \emph{simétrica}; \emph{semidef.\ negativa}; $S\vect{p}=\vect{0}$ (\autor{Euler}).
  \end{itemize}
\end{itemize}
\graficorepaso[0.42\linewidth]{concavidad-fg}{$E$ cóncava: bajo su tangente, de pendiente $h_i$ (\autor{Shephard})}

\toprule
\textit{Primal: $\vect{x}(\vect{p},\overline{W})$, $V(\vect{p},\overline{W})$} & \textit{Dual: $\vect{h}(\vect{p},\overline{U})$, $E(\vect{p},\overline{U})$}\\

\sub{I.3. Relaciones de dualidad\label{rep:I.3}}
\begin{itemize}
\item \concepto{Teorema básico de la dualidad}: con $U$ continua, insaciabilidad local y $\vect{p}\gg\vect{0}$, la cesta del PMU resuelve el PMG con $\overline{U}=U(\vect{x}^*)$, y al revés.
\item \concepto{Coinciden las cestas}, no las funciones:
\begin{formula}
\centering $\vect{x}(\vect{p},\overline{W})\equiv\vect{h}\big(\vect{p},V(\vect{p},\overline{W})\big)$,\qquad $\vect{h}(\vect{p},\overline{U})\equiv\vect{x}\big(\vect{p},E(\vect{p},\overline{U})\big)$\\[3pt]
$E\big(\vect{p},V(\vect{p},\overline{W})\big)\equiv\overline{W}$,\qquad $V\big(\vect{p},E(\vect{p},\overline{U})\big)\equiv\overline{U}$
\end{formula}
  \begin{itemize}
  \item $V$ y $E$, \concepto{inversas} en su 2.º argumento.
  \end{itemize}
\item \emph{Ejemplo} (\autor{Cobb}-\autor{Douglas}): $x_1=\alpha\overline{W}/p_1$ $\to$ $V$ $\to$ $E$ $\to$ $h_1=\alpha E/p_1$.
\end{itemize}

\bloque[5]{II. Sistemas completos de demanda\label{rep:II}}
\begin{ideaclave}
Se elige una $E$ o una $V$ flexible, se derivan demandas estimables y se contrastan las restricciones de la teoría; el AIDS equilibra flexibilidad, coherencia, agregación y sencillez.
\end{ideaclave}

\sub{II.1. Idea\label{rep:II.1}}
\begin{itemize}
\item \concepto{Sistema completo de ecuaciones de demanda}: gasto en \emph{todos} los bienes en función de precios y gasto total, derivado de una optimización.
\item \emph{Fines}:
  \begin{enumerate}
  \item \emph{medir} elasticidades (IVA, índices, pobreza);
  \item \emph{contrastar} la teoría: homogeneidad, simetría, negatividad (\hyperref[rep:III.1]{III.1});
  \item \emph{valorar} el bienestar (VC, VE).
  \end{enumerate}
\item \emph{Dualidad}: se especifica $E$ o $V$ y se deriva (\autor{Shephard}/\autor{Roy}): coherente \emph{por construcción}.
\item \emph{Requisitos} (en tensión): coherencia, flexibilidad, agregación, sencillez.
\end{itemize}

\sub{II.2. Pasos que sigue la literatura\label{rep:II.2}}
\subsub{a) Elección de la forma funcional}
\begin{itemize}
\item \autor{Cobb}-\autor{Douglas}: $s_i=\alpha_i$ constantes; homotética: no recoge \autor{Engel}.
\item \concepto{LES} (\autor{Stone}, 1954; \autor{Stone}-\autor{Geary}): $U=\sum_i\alpha_i\ln(x_i-\gamma_i)$:
\begin{formula}
\centering $p_ix_i=p_i\gamma_i+\alpha_i\big(\overline{W}-\textstyle\sum_jp_j\gamma_j\big)$
\end{formula}
  \begin{itemize}
  \item Subsistencia + renta \emph{supernumeraria}; \autor{Engel} rectas sin pasar por el origen.
  \item \emph{Aditivo}: sin inferiores ni complementarios; \autor{Deaton} (1974) rechaza la aditividad.
  \end{itemize}
\item \concepto{Rotterdam} (\autor{Barten}, \autor{Theil}): \autor{Slutsky} en diferencias; local, sin $E$ explícita.
\item \concepto{Translog} (1975): $\ln V$ flexible de 2.º orden; rechaza la teoría; no lineal, no agrega.
\end{itemize}
\subsubsub{El sistema casi ideal (AIDS; \autor{Deaton}-\autor{Muellbauer}, 1980)}
\begin{itemize}
\item \emph{Supuestos}: $E$ \concepto{PIGLOG}; $a(\vect{p})$ subsistencia, $b(\vect{p})$ saciedad.
\item \emph{Desarrollo}: \autor{Shephard} en logaritmos y $\overline{U}$ despejada:
\begin{formula}
\centering $\ln E(\vect{p},\overline{U})=(1-\overline{U})\ln a(\vect{p})+\overline{U}\ln b(\vect{p})$\\[3pt]
$s_i=\alpha_i+\sum_j\gamma_{ij}\ln p_j+\beta_i\ln\dfrac{\overline{W}}{P}$,\qquad $\varepsilon_{x_i,\overline{W}}=1+\dfrac{\beta_i}{s_i}$
\end{formula}
\item \emph{Implicaciones}: restricciones \emph{lineales}: agregación, homogeneidad $\sum_j\gamma_{ij}=0$, simetría $\gamma_{ij}=\gamma_{ji}$; $\beta_i>0$ lujo; \emph{agregación exacta}.
\item \emph{Extensiones}: \emph{LA-AIDS} (índice de \autor{Stone}: MCO); \concepto{QUAIDS} (1997): $(\ln\overline{W})^2$, rango 3.
\item \emph{Valoración}: «\emph{casi}» ideal: negatividad no garantizada.
\end{itemize}

\subsub{b) Sistema completo, datos, econometría y resultados}
\begin{itemize}
\item \emph{Sistema}: estimación \emph{conjunta}; \concepto{separabilidad débil} y \concepto{presupuestación en 2 etapas}.
\item \emph{Datos}: microdatos de la \concepto{EPF} (INE); valores unitarios (\autor{Deaton}).
\item \emph{Econometría}: se quita una ecuación; gasto endógeno; ceros; demografía.
\item \emph{Resultados}: con series agregadas se rechaza sobre todo la \emph{homogeneidad}; \autor{Deaton}: rigidez, dinámica y, sobre todo, \emph{agregación} $\Rightarrow$ microdatos.
  \begin{itemize}
  \item Usos: IVA, incidencia, índices por hogar, pobreza; España: AIDS con la EPF (\autor{Sanz}-\autor{Romero}, 2012).
  \end{itemize}
\end{itemize}

\sub{II.3. Valoración\label{rep:II.3}}
\begin{itemize}
\item \emph{Aportaciones}: la teoría, instrumento de medida; restricciones, hipótesis contrastables.
\item \emph{Limitaciones}: flexibilidad vs.\ parsimonia; agregación (\autor{Gorman}, SMD); marco estático \vertema{3.A.33}.
\item \emph{Enlace}: ¿procede el sistema de unas preferencias? $\Rightarrow$ integrabilidad.
\end{itemize}

\bloque[12]{III. Aplicaciones de la dualidad\label{rep:III}}
\begin{ideaclave}
$E$ traduce la utilidad a euros: la VC y la VE miden exactamente el cambio de bienestar; el $\Delta$EC las aproxima tanto mejor cuanto menor es el efecto renta (\autor{Willig}), y \autor{Laspeyres} y \autor{Paasche} las acotan con un sesgo de sustitución que sigue en los índices oficiales.
\end{ideaclave}

\sub{III.1. Integrabilidad de las preferencias\label{rep:III.1}}
\begin{itemize}
\item \emph{Pregunta}: dado $\vect{x}(\vect{p},\overline{W})$ observado, ¿existe una $U$ de la que proceda? \autor{Antonelli} (1886); \autor{Hurwicz}-\autor{Uzawa} (1971).
\item \concepto{Teorema de integrabilidad}: sí \emph{si y solo si}: ley de \autor{Walras}; $S$ \emph{simétrica}; $S$ \emph{semidef.\ negativa}:
\begin{formula}
\centering $S_{ij}=\dfrac{\partial x_i}{\partial p_j}+x_j\dfrac{\partial x_i}{\partial\overline{W}}=S_{ji}$\quad (observable)\\[3pt]
$\nabla_{\vect{p}}E(\vect{p},\overline{U})=\vect{x}\big(\vect{p},E(\vect{p},\overline{U})\big)$,\quad $E(\vect{p}^0,\overline{U})=\overline{W}^0$
\end{formula}
\item \emph{Intuición}:
  \begin{itemize}
  \item de la demanda a $E$: solución $\Leftrightarrow$ derivadas cruzadas iguales (\concepto{\autor{Frobenius}}) $\Leftrightarrow$ \emph{simetría};
  \item que sea función de gasto: \emph{concavidad} $\Leftrightarrow$ \emph{negatividad};
  \item de $E$ a $U$: $U(\vect{x}^0)$ = mín.\ de $V$ entre los presupuestos con los que se compra $\vect{x}^0$.
  \end{itemize}
\end{itemize}
\graficorepaso[0.45\linewidth]{recuperar-utilidad}{Recuperar $U$ desde $V$}
\begin{itemize}
\item \emph{Preferencia revelada} \vertema{3.A.8}: axioma débil $\Rightarrow$ $S$ negativa, \emph{no} simétrica; fuerte (\autor{Houthakker}) sí racionaliza.
\item \emph{Relevancia}:
  \begin{itemize}
  \item \emph{teórica}: cierra el círculo; las preferencias no añaden nada a 3 propiedades contrastables;
  \item \emph{empírica}: contraste sin hallar $U$;
  \item \emph{normativa}: $E$ exacta desde la demanda (\autor{Hausman}, 1981).
  \end{itemize}
\end{itemize}

\sub{III.2. Ecuación de \autor{Slutsky} y clasificación\label{rep:III.2}}
\begin{itemize}
\item Derivar $h_i\equiv x_i(\vect{p},E(\vect{p},\overline{U}))$ y usar \autor{Shephard}:
\funcionclave{\underbrace{\frac{\partial x_i(\vect{p},\overline{W})}{\partial p_j}}_{\text{ET}}=\underbrace{\frac{\partial h_i(\vect{p},\overline{U})}{\partial p_j}}_{\text{ES}}\ \underbrace{-\,x_j\,\frac{\partial x_i(\vect{p},\overline{W})}{\partial\overline{W}}}_{\text{ER}}}
  \begin{itemize}
  \item En elasticidades: $\varepsilon_{x_i,p_j}=\varepsilon^{h}_{i,p_j}-s_j\,\varepsilon_{x_i,\overline{W}}$.
  \item ES inobservable; ET y ER observables $\Rightarrow$ $S$ calculable.
  \end{itemize}
\item \emph{Propiedades de $S$}: \concepto{todo bien tiene al menos un sustitutivo neto} (\autor{Hicks}).
\item \emph{Precio propio}: normal $\Rightarrow$ ordinario; inferior: ordinario o \autor{Giffen} ($s_i$ alto). \autor{Veblen}, fuera.
\end{itemize}
{\centering\small\renewcommand{\arraystretch}{1.15}\setlength{\tabcolsep}{3pt}%
\begin{tabular}{@{}ccl@{}}
\toprule
\textit{ES propio} & \textit{ER propio} & \textit{ET propio}\\\midrule
$\leq0$ & $<0$: normal & $\leq0$: ordinario\\
$\leq0$ & $>0$: inferior & ordinario si $|\mathrm{ES}|\geq\mathrm{ER}$\\
 & & $>0$: \autor{Giffen} si $\mathrm{ER}>|\mathrm{ES}|$\\\bottomrule
\end{tabular}\par}
\begin{itemize}
\item \emph{Cruzados}: neta \emph{simétrica} (\autor{Hicks}-\autor{Allen}, 1934); bruta no: los ER cruzados difieren.
  \begin{itemize}
  \item Sustitutivos netos + $i$ inferior $\Rightarrow$ sustitutivos brutos; complementarios netos + $i$ normal $\Rightarrow$ complementarios brutos; el resto, ambiguo.
  \end{itemize}
\item \emph{Pendientes}: normal, marshalliana más plana que la hicksiana $\Rightarrow$ $\mathrm{VC}\leq\Delta\mathrm{EC}\leq\mathrm{VE}$.
\end{itemize}

\sub{III.3. Análisis de bienestar del consumidor individual\label{rep:III.3}}
\subsub{a) Idea}
\begin{itemize}
\item Un consumidor, cambios de \emph{precios} \vertema{3.A.16}; utilidad ordinal: $\overline{U}^1-\overline{U}^0$ no significa nada $\Rightarrow$ euros con $E$.
\end{itemize}
\subsub{b) Medidas: $\Delta$EC, VC y VE}
\begin{itemize}
\item \concepto{Excedente del consumidor} (\autor{Dupuit}, \autor{Marshall}): observable, pero \emph{mezcla} ES y ER.
\item \concepto{Métrica monetaria} (\autor{McKenzie}, \autor{Samuelson}): $E(\bar{\vect{p}},V(\vect{p},\overline{W}))$: FIU en euros.
\item Las 3 medidas:
\begin{formula}
\centering $\Delta\mathrm{EC}=\displaystyle\int_{p_1^1}^{p_1^0}x_1(p_1,\overline{\vect{p}}_{-1},\overline{W})\,dp_1$\\[3pt]
$\mathrm{VC}=\overline{W}-E(\vect{p}^1,\overline{U}^0)=\displaystyle\int_{p_1^1}^{p_1^0}h_1(\cdot,\overline{U}^0)\,dp_1$\\[3pt]
$\mathrm{VE}=E(\vect{p}^0,\overline{U}^1)-\overline{W}=\displaystyle\int_{p_1^1}^{p_1^0}h_1(\cdot,\overline{U}^1)\,dp_1$
\end{formula}
  \begin{itemize}
  \item \concepto{VC}: a precios \emph{finales}, renta que devuelve a la utilidad \emph{inicial}; baja $p$: lo máximo que pagaría.
  \item \concepto{VE}: a precios \emph{iniciales}, renta que lleva a la utilidad \emph{final}; baja $p$: lo mínimo que aceptaría.
  \end{itemize}
\end{itemize}
\pareja{bienestar-vc}{VC: a precios finales, vuelta a $\overline{U}^0$}{bienestar-ve}{VE: a precios iniciales, hasta $\overline{U}^1$}
\graficorepaso[0.52\linewidth]{bienestar-vc-ve}{VC, $\Delta$EC y VE (y $L$, $P$) ante una bajada de $p_1$ de un bien normal}
\begin{itemize}
\item Orden y aproximación:
\begin{formula}
\centering $L\leq\mathrm{VC}\leq\Delta\mathrm{EC}\leq\mathrm{VE}\leq P$\quad (normal, baja $p_1$)\\[3pt]
\autor{Willig} (1976):\ \ $\mathrm{VC}\approx\Delta\mathrm{EC}-\dfrac{\varepsilon_{x_1,\overline{W}}(\Delta\mathrm{EC})^2}{2\overline{W}}$
\end{formula}
  \begin{itemize}
  \item Inferior o subida, se invierte; ER nulo: VC $=\Delta$EC $=$ VE.
  \item Error de \autor{Willig}: 5\,\% de la renta y elasticidad 1: 2,5\,\%; \autor{Hausman} (1981): falla en la pérdida irrecuperable.
  \end{itemize}
\item \emph{¿Cuál usar?} Ordenar políticas: VE; compensar (\autor{Kaldor}-\autor{Hicks} \vertema{3.A.22}): VC. Exceso de gravamen $=-\mathrm{VE}-T$ \vertema{4.B.9}.
\end{itemize}
\subsub{c) Índices de precios}
\begin{itemize}
\item Solo con $\vect{x}^0$ y $\vect{x}^1$: \autor{Laspeyres} $L\leq\mathrm{VC}$, \autor{Paasche} $P\geq\mathrm{VE}$: \concepto{sesgo de sustitución}.
\item \autor{Konüs} (1924), \concepto{índice verdadero de coste de la vida}:
\begin{formula}
\centering $I_K(\overline{U})=\dfrac{E(\vect{p}^1,\overline{U})}{E(\vect{p}^0,\overline{U})}$;\quad $I_L\geq I_K(\overline{U}^0)$,\quad $I_P\leq I_K(\overline{U}^1)$
\end{formula}
  \begin{itemize}
  \item \autor{Laspeyres}: aproximación de 1.\textsuperscript{er} orden, por la concavidad \emph{cota superior}: indexar sobrecompensa.
  \item \autor{Paasche}: cota inferior de $I_K(\overline{U}^1)$.
  \item Acotan índices \emph{distintos}: $I_P\leq I_K\leq I_L$ solo con homotéticas; igualdades solo con \autor{Leontief}.
  \end{itemize}
\end{itemize}
\graficorepaso[0.5\linewidth]{indice-konus-laspeyres}{\autor{Laspeyres} sobrestima: el sesgo de sustitución}
\begin{itemize}
\item \concepto{Superlativos} (\autor{Diewert}, 1976): \autor{Fisher} $\sqrt{I_LI_P}$; \autor{Törnqvist} (translog).
\item \concepto{IPC del INE}: tipo \autor{Laspeyres} encadenado (base 2025) $\Rightarrow$ sobrevalora; \concepto{deflactor del PIB}: \autor{Paasche} encadenado \vertema{3.A.30} \vertema{4.A.1}.
\end{itemize}
