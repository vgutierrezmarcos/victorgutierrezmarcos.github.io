\bloque[2]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: pensiones revalorizadas con la media del IPC (+2,7\,\% en 2026). \emph{¿Basta para quedar igual?} Si los precios suben de forma desigual, se sustituye: importa cuánto cuesta \emph{volver a estar igual de bien}. Utilidad \emph{ordinal} $\Rightarrow$ ¿se puede medir en euros algo que solo sabemos ordenar? $\Rightarrow$ \concepto{función de gasto} $\Rightarrow$ \concepto{dualidad}.
\item \textbf{Relevancia}: \emph{teórica} (\autor{Slutsky} en dos líneas; integrabilidad); \emph{empírica} (elasticidades: IVA, índices, pobreza); \emph{normativa} (VC y VE: coste-beneficio, exceso de gravamen).
\item \textbf{Contextualización}: ligar \emph{preferencias} (no se ven) y \emph{demanda} (se ve). 4 líneas: \emph{dualidad} (\autor{Hotelling} 1932 $\to$ \autor{Shephard} 1953, \autor{McKenzie} 1957); \emph{integrabilidad} (\autor{Antonelli} 1886 $\to$ \autor{Hurwicz}-\autor{Uzawa} 1971); \emph{bienestar} (\autor{Dupuit} 1844 $\to$ \autor{Hicks}, \autor{Willig}); \emph{sistemas} (\autor{Stone} 1954 $\to$ AIDS 1980; Nobel a \autor{Deaton}, 2015).
\item \textbf{«Dual»}: \emph{estricto} = intercambiar objetivo y restricción; \emph{amplio} = papeles invertidos (empresa \vertema{3.A.12}). Aquí, estricto.
\item \textbf{Problemática}: ¿se llega a la misma decisión al «dar la vuelta» al problema? ¿Cómo se estiman los sistemas de demanda? ¿Qué demandas proceden de unas preferencias? ¿Cómo medir en euros el bienestar y qué error cometen el excedente y los índices?
\item \textbf{Estructura}: I. dualidad (herramientas); II. sistemas de demanda (estimación); III. aplicaciones: integrabilidad, \autor{Slutsky}, bienestar. La integrabilidad tras II: responde si lo estimado puede venir de un consumidor racional.
\end{itemize}

\bloque[10]{I. Análisis teórico de la dualidad}
\begin{itemize}
\item \emph{Conclusión anticipada}: máx.\ $U$ con renta dada y mín.\ gasto con utilidad dada $\Rightarrow$ \emph{misma cesta}; de ahí, relaciones para pasar de unas funciones a otras sin resolver más problemas.
\end{itemize}

\sub{I.1. Ingredientes comunes}
\begin{itemize}
\item \emph{Dos enfoques}: \concepto{primal}: recursos dados, máx.\ utilidad; \concepto{dual}: «da la vuelta»: utilidad dada (restricción), mín.\ gasto (objetivo). Mismo problema de fondo.
\item \emph{Conjunto de elección}: $S\subseteq\mathbb{R}^n_+$ no vacío, cerrado y convexo (normalmente $\mathbb{R}^n_+$).
\item \emph{Preferencias} $\succcurlyeq$: completitud, reflexividad, transitividad, continuidad ($\Rightarrow$ $U$ continua, \autor{Debreu}); insaciabilidad local; convexidad estricta; diferenciabilidad.
\item \emph{Utilidad}: $U(\vect{x})$ continua, creciente, estr.\ cuasicóncava \vertema{3.A.8}. \emph{Gasto}: $\vect{p}\cdot\vect{x}$, $\vect{p}\gg\vect{0}$ dados.
\item \emph{Supuestos}: estático, sin incertidumbre, información perfecta, bienes homogéneos, precio-aceptante. Solo cambia qué es objetivo y qué restricción:
\end{itemize}

\sub{I.2. Problemas primal y dual}
\textbf{Problema primal: maximización de la utilidad (PMU)}
\begin{itemize}
\item \emph{a) Idea}: necesidades ilimitadas, recursos escasos; punto de partida de la dualidad \vertema{3.A.8}.
\item \emph{b.i) Supuestos}: los de I.1.
\item \emph{b.ii) Desarrollo}: óptimo que existe (\autor{Weierstrass}), global y único; \autor{Kuhn}-\autor{Tucker} con $\mathcal{L}=U+\lambda(\overline{W}-\vect{p}\cdot\vect{x})$; $\lambda$ = \concepto{utilidad marginal de la renta}:
\end{itemize}
\begin{formula}
\centering $\optim{\max}{\vect{x}}{U(\vect{x})}{\vect{p}\cdot\vect{x}\leq\overline{W}\\ \vect{x}\geq\vect{0}}$\quad$\Rightarrow$\quad $\dfrac{U_i}{p_i}=\lambda$ (\autor{Gossen}),\ \ $\mathrm{RMS}_{12}=\dfrac{p_1}{p_2}$
\end{formula}
\begin{itemize}
\item \emph{b.iii) Implicaciones}:
  \begin{itemize}
  \item \concepto{Demanda marshalliana} $\vect{x}(\vect{p},\overline{W})$: única; continua (\autor{Berge}); sin ilusión monetaria (grado 0); ley de \autor{Walras} (tabla).
  \item \concepto{Función indirecta de utilidad} (FIU) $V(\vect{p},\overline{W})=U(\vect{x}(\vect{p},\overline{W}))$: utilidad máxima; \emph{cuasiconvexa} (con el presupuesto promedio no se alcanza más que con el mejor extremo; no exige preferencias convexas); identidad de \autor{Roy} (1947).
  \end{itemize}
\item \emph{c) Valoración}: $V$ en «útiles», solo ordena $\Rightarrow$ ¿y si se da la vuelta al problema?
\end{itemize}
\textbf{Problema dual: minimización del gasto (PMG)}
\begin{itemize}
\item \emph{a) Idea}: la forma más barata de alcanzar un nivel de utilidad.
\item \emph{b.i) Supuestos}: idénticos a los del primal.
\item \emph{b.ii) Desarrollo}: $\mathcal{L}=\vect{p}\cdot\vect{h}+\mu(\overline{U}-U(\vect{h}))$ $\Rightarrow$ $p_i/U_i=\mu$: \emph{misma} tangencia $\mathrm{RMS}_{12}=p_1/p_2$; $\mu^*=\partial E/\partial\overline{U}$ = \concepto{coste marginal de la utilidad} $=1/\lambda^*$; restricción saturada.
\end{itemize}
\pareja{primal}{Primal: la curva de indiferencia más alta alcanzable con $\overline{W}$}{dual}{Dual: la isogasto más baja que alcanza $\overline{U}$}
\begin{itemize}
\item \emph{b.iii) Implicaciones}:
  \begin{itemize}
  \item \concepto{Demanda hicksiana} (compensada) $\vect{h}(\vect{p},\overline{U})$: única, continua; homog.\ de grado 0 en $\vect{p}$; sin exceso de utilidad; \concepto{ley de la demanda compensada} $\partial h_i/\partial p_i\leq0$ $\Rightarrow$ \autor{Giffen} imposible (no hay efecto renta). \emph{No observable}: depende de $\overline{U}$.
  \item \concepto{Función de gasto} (FG): gasto mínimo para alcanzar $\overline{U}$; mide la utilidad en \emph{euros}:
  \end{itemize}
\end{itemize}
\funcionclave{E(\vect{p},\overline{U})=\vect{p}\cdot\vect{h}(\vect{p},\overline{U})=\optim{\min}{\vect{h}}{\vect{p}\cdot\vect{h}}{U(\vect{h})\geq\overline{U}\\ \vect{h}\geq\vect{0}}}
\begin{itemize}
\item \emph{Propiedades de $E$}: continua; homog.\ de \emph{grado 1} en $\vect{p}$; estr.\ creciente en $\overline{U}$; no decreciente en $p_i$; \emph{cóncava} en $\vect{p}$: el consumidor «pasivo» pagaría $\vect{p}\,'\cdot\vect{h}(\vect{p},\overline{U})$ (lineal); sustituyendo, como mucho eso:
\end{itemize}
\begin{formula}
\centering $E(\vect{p}\,',\overline{U})\leq\vect{p}\,'\cdot\vect{h}(\vect{p},\overline{U})$;\qquad \autor{Shephard}: $h_i(\vect{p},\overline{U})=\dfrac{\partial E(\vect{p},\overline{U})}{\partial p_i}$\\[4pt]
$S_{ij}\equiv\dfrac{\partial^2E}{\partial p_i\,\partial p_j}=\dfrac{\partial h_i(\vect{p},\overline{U})}{\partial p_j}$\quad (\concepto{matriz de \autor{Slutsky}})
\end{formula}
\begin{itemize}
\item \emph{Lema de \autor{Shephard}} (1953; \autor{McKenzie} 1957): si $p_i$ sube 1 €, la cesta apenas cambia en primer orden (envolvente) $\Rightarrow$ el gasto sube $h_i$ €.
\item \emph{Matriz $S$}: \emph{simétrica} (\autor{Schwarz}/\autor{Young}); \emph{semidef.\ negativa} ($E$ cóncava; $S_{ii}\leq0$); $S\vect{p}=\vect{0}$ (\autor{Euler}).
\end{itemize}

\end{tabular}\par}
\begin{itemize}
\item \emph{c) Valoración}: misma tangencia y 2 funciones nuevas: $\vect{h}$ (cumple siempre la ley de la demanda, pero inobservable) y $E$ (utilidad en dinero). Falta ligarlas con las del primal, que sí se observan.
\end{itemize}

\sub{I.3. Relaciones de dualidad}
\begin{itemize}
\item \concepto{Teorema básico de la dualidad}: con $U$ continua, insaciabilidad local y $\vect{p}\gg\vect{0}$: (1) si $\vect{x}^*$ resuelve el PMU con $\overline{W}>0$, resuelve el PMG con $\overline{U}=U(\vect{x}^*)$ y gasto mínimo $\overline{W}$; (2) si $\vect{h}^*$ resuelve el PMG, resuelve el PMU con $\overline{W}=\vect{p}\cdot\vect{h}^*$ y utilidad máx.\ $\overline{U}$.
\item \concepto{Coinciden las cestas}, no las funciones: $\vect{x}$ y $\vect{h}$ dependen de variables distintas ($\overline{W}$, $\overline{U}$).
\end{itemize}
\begin{formula}
\centering $\vect{x}(\vect{p},\overline{W})\equiv\vect{h}\big(\vect{p},V(\vect{p},\overline{W})\big)$,\qquad $\vect{h}(\vect{p},\overline{U})\equiv\vect{x}\big(\vect{p},E(\vect{p},\overline{U})\big)$\\[3pt]
$E\big(\vect{p},V(\vect{p},\overline{W})\big)\equiv\overline{W}$,\qquad $V\big(\vect{p},E(\vect{p},\overline{U})\big)\equiv\overline{U}$
\end{formula}
\begin{itemize}
\item \emph{Lectura}: entre demandas, se sustituye $\overline{U}$ por $V$ o $\overline{W}$ por $E$; $V$ y $E$, \concepto{inversas} en su 2.º argumento: hallada una, la otra se despeja.
\item \emph{Ejemplo} (\autor{Cobb}-\autor{Douglas}): $x_1=\alpha\overline{W}/p_1$ $\to$ $V$ $\to$ invirtiendo, $E$ $\to$ \autor{Shephard}: $h_1=\alpha E/p_1$ (como la marshalliana, con $E$ en lugar de $\overline{W}$).
\end{itemize}
\graficorepaso[0.92\linewidth]{esquema-dualidad}{Esquema de dualidad: problemas, soluciones, funciones valor y relaciones (\autor{Roy}, \autor{Shephard}, inversas, \autor{Slutsky}, integrabilidad)}
\begin{ideaclave}
Maximizar la utilidad con una renta y minimizar el gasto con una utilidad son el mismo problema visto desde dos lados. $E$ mide la utilidad en dinero, sus derivadas son las hicksianas (\autor{Shephard}) y su concavidad da la ley de la demanda compensada. Las flechas del esquema son las herramientas de II y III.
\end{ideaclave}

\bloque[5]{II. Sistemas completos de demanda}
\begin{itemize}
\item \emph{Conclusión anticipada}: se elige una $E$ o una $V$ flexible, se derivan demandas estimables y se contrastan las restricciones de la teoría; el AIDS equilibra flexibilidad, coherencia, agregación y sencillez.
\end{itemize}

\sub{II.1. Idea}
\begin{itemize}
\item \emph{Definición}: \concepto{sistema completo de ecuaciones de demanda}: gasto en \emph{todos} los bienes (o grupos) en función de precios y gasto total, derivado de una optimización.
\item \emph{Fines}: (1) \emph{medir} elasticidades (IVA, índices, pobreza); (2) \emph{contrastar} la teoría (homogeneidad, simetría, negatividad = integrabilidad, III.1); (3) \emph{valorar} el bienestar (VC, VE).
\item \emph{Papel de la dualidad}: se especifica $E$ o $V$ y se deriva (\autor{Shephard}/\autor{Roy}) $\Rightarrow$ coherente con la maximización \emph{por construcción}, sin resolver ningún problema.
\item \emph{Requisitos} (tiran en sentidos opuestos): coherencia con la teoría; flexibilidad (no fijar elasticidades); agregación (la media con la forma individual); sencillez (lineal en los parámetros).
\end{itemize}

\sub{II.2. Pasos que sigue la literatura}
\begin{itemize}
\item \emph{Pasos}: (1) forma funcional; (2) sistema completo; (3) datos; (4) especificación econométrica; (5) estimar y contrastar.
\item \emph{(1) Forma funcional}: compromiso sencillez--flexibilidad; cada sistema supera el límite del anterior. Punto de partida, \concepto{\autor{Cobb}-\autor{Douglas}} $U=\sum_i\alpha_i\ln x_i$: $s_i=\alpha_i$ constantes; $\varepsilon_{\overline{W}}=1$, $\varepsilon_{ii}=-1$, cruzadas 0, $\sigma=1$; homotética $\Rightarrow$ no recoge la ley de \autor{Engel}.
\end{itemize}
\textbf{Sistema lineal de gasto (LES)}
\begin{itemize}
\item \emph{a) Idea}: recoger la ley de \autor{Engel} (necesarios y de lujo) sin perder linealidad.
\item \emph{b) Modelo}: \emph{supuestos}: utilidad de \autor{Stone}-\autor{Geary} $U=\sum_i\alpha_i\ln(x_i-\gamma_i)$, $\alpha_i>0$, $\sum\alpha_i=1$; \emph{desarrollo}: $p_ix_i=p_i\gamma_i+\alpha_i\big(\overline{W}-\sum_jp_j\gamma_j\big)$: subsistencia $\gamma_i$ + renta \emph{supernumeraria} en proporciones fijas; \emph{implicaciones}: curvas de \autor{Engel} rectas que no pasan por el origen.
\item \emph{c) Evidencia}: \autor{Stone} (1954), Reino Unido, primer sistema estimado; \autor{Deaton} (1974) rechaza la aditividad.
\item \emph{d) Valoración}: \emph{aditivo} $\Rightarrow$ sin inferiores ($\alpha_i>0$) ni complementarios netos; elasticidades-precio ligadas a las de renta.
\end{itemize}
\textbf{Modelo de Rotterdam y translog}
\begin{itemize}
\item \concepto{Rotterdam} (\autor{Barten} 1964, \autor{Theil} 1965): \emph{idea}: \autor{Slutsky} en diferencias logarítmicas con coeficientes constantes; \emph{implicaciones}: restricciones lineales; \emph{evidencia}: Países Bajos, rechaza la homogeneidad; \emph{valoración}: local, sin $E$ explícita.
\item \concepto{Translog} (\autor{Christensen}-\autor{Jorgenson}-\autor{Lau}, 1975): \emph{idea}: forma \emph{flexible}, $\ln V$ de 2.º orden en $\ln(p_i/\overline{W})$, participaciones por \autor{Roy}; \emph{evidencia}: EE.\,UU.\ 1929-72, rechaza la teoría y la homoteticidad; \emph{valoración}: no lineal, no agrega.
\end{itemize}
\textbf{Sistema casi ideal de demanda (AIDS)}
\begin{itemize}
\item \emph{a) Idea}: reunir los 4 requisitos (\autor{Deaton}-\autor{Muellbauer}, 1980).
\item \emph{b.i) Supuestos}: $E$ de la clase \concepto{PIGLOG} (\autor{Muellbauer}), compatible con la agregación; $a(\vect{p})$ coste de subsistencia ($\overline{U}=0$), $b(\vect{p})$ de saciedad ($\overline{U}=1$).
\item \emph{b.ii) Desarrollo}: \autor{Shephard} en logaritmos ($s_i=\partial\ln E/\partial\ln p_i$) y $\overline{U}$ despejada de $E=\overline{W}$:
\end{itemize}
\begin{formula}
\centering $\ln E(\vect{p},\overline{U})=(1-\overline{U})\ln a(\vect{p})+\overline{U}\ln b(\vect{p})$\\[3pt]
$s_i=\alpha_i+\sum_j\gamma_{ij}\ln p_j+\beta_i\ln\dfrac{\overline{W}}{P}$,\qquad $\ln P=\ln a(\vect{p})$ (translog)\\[3pt]
$\varepsilon_{x_i,\overline{W}}=1+\dfrac{\beta_i}{s_i}$
\end{formula}
\begin{itemize}
\item \emph{b.iii) Implicaciones}: restricciones \emph{lineales}: agregación $\sum_i\alpha_i=1$, $\sum_i\gamma_{ij}=\sum_i\beta_i=0$; homogeneidad $\sum_j\gamma_{ij}=0$; simetría $\gamma_{ij}=\gamma_{ji}$; negatividad no impuesta. \autor{Engel} de \autor{Working}-\autor{Leser}: $\beta_i>0$ lujo, $\beta_i<0$ necesario. \emph{Agregación exacta} (PIGLOG).
\item \emph{c) Extensiones}: \emph{LA-AIDS}: índice de \autor{Stone} $\ln P^*=\sum_ks_k\ln p_k$ $\Rightarrow$ lineal, MCO ecuación a ecuación. \concepto{QUAIDS} (\autor{Banks}-\autor{Blundell}-\autor{Lewbel}, 1997): término en $(\ln\overline{W})^2$, rango 3 (\autor{Gorman}): un bien, de lujo para renta baja y necesario para alta.
\item \emph{d) Evidencia}: Reino Unido, rechazan la homogeneidad (dinámica).
\item \emph{e) Valoración}: «\emph{casi}» ideal: negatividad no garantizada y \autor{Engel} restrictivas.
\end{itemize}

\begin{itemize}
\item \emph{(2) Sistema completo}: estimación \emph{conjunta}, $\sum_is_i=1$. Agrupar ($\sim n^2$ parámetros): \concepto{separabilidad débil} y \concepto{presupuestación en dos etapas} (\autor{Strotz}, \autor{Gorman}).
\item \emph{(3) Datos}: microdatos de la \concepto{Encuesta de Presupuestos Familiares} (EPF, INE). Precios casi iguales en un corte $\Rightarrow$ años y regiones o valores unitarios (\autor{Deaton}).
\item \emph{(4) Econometría}: covarianzas singulares $\Rightarrow$ se quita una ecuación (MV: \autor{Barten}); gasto endógeno; ceros; demografía.
\item \emph{(5) Resultados}: con series agregadas se rechaza sobre todo la \emph{homogeneidad}; \autor{Deaton}: rigidez, dinámica y, sobre todo, \emph{agregación} $\Rightarrow$ microdatos. Usos: IVA, incidencia, índices por hogar, pobreza; España: AIDS con la EPF (\autor{Sanz}-\autor{Romero}, 2012).
\end{itemize}

\sub{II.3. Valoración}
\begin{itemize}
\item \emph{Aportaciones}: la teoría pasa a ser instrumento de medida; sus restricciones, hipótesis contrastables; AIDS y QUAIDS concilian flexibilidad, agregación y sencillez.
\item \emph{Limitaciones}: (1) flexibilidad vs.\ parsimonia ($n(n+1)/2$ términos cruzados; flexibilidad local); (2) agregación: consumidor representativo solo con forma polar de \autor{Gorman} o PIGLOG; \concepto{teorema de \autor{Sonnenschein}-\autor{Mantel}-\autor{Debreu}}: las restricciones individuales no tienen por qué cumplirse en la media; (3) marco estático: hábitos, duraderos, liquidez \vertema{3.A.33}.
\item \emph{Enlace}: estimar no garantiza que el sistema proceda de unas preferencias $\Rightarrow$ integrabilidad.
\end{itemize}

\bloque[12]{III. Aplicaciones de la dualidad}
\begin{itemize}
\item \emph{Conclusión anticipada}: una aplicación teórica, la \emph{integrabilidad} (el esquema al revés, de la demanda a las preferencias), y dos prácticas: \autor{Slutsky} en dos líneas (separa lo observable de lo que no) y el bienestar en euros con $E$.
\end{itemize}

\sub{III.1. Integrabilidad de las preferencias}
\begin{itemize}
\item \emph{Pregunta}: dado $\vect{x}(\vect{p},\overline{W})$ observado (estimado en II), ¿existe una $U$ de la que proceda? Si sí, se «integran» las demandas. \autor{Antonelli} (1886); \autor{Hurwicz}-\autor{Uzawa} (1971).
\item \emph{Respuesta}: \concepto{teorema de integrabilidad}: $\vect{x}$ continuamente diferenciable es demanda de una $U$ creciente y cuasicóncava \emph{si y solo si}: (1) ley de \autor{Walras}; (2) $S$ \emph{simétrica}; (3) $S$ \emph{semidef.\ negativa} (y homogeneidad). Con 2 bienes, simetría automática. Necesarias (I.2, III.2) y suficientes (teorema).
\end{itemize}
\begin{formula}
\centering $S_{ij}(\vect{p},\overline{W})=\dfrac{\partial x_i}{\partial p_j}+x_j\dfrac{\partial x_i}{\partial\overline{W}}=S_{ji}$\quad (observable)\\[3pt]
$\nabla_{\vect{p}}E(\vect{p},\overline{U})=\vect{x}\big(\vect{p},E(\vect{p},\overline{U})\big)$,\quad $E(\vect{p}^0,\overline{U})=\overline{W}^0$
\end{formula}
\begin{itemize}
\item \emph{Intuición de la demostración} (por dualidad):
  \begin{itemize}
  \item \emph{de la demanda a $E$}: \autor{Shephard} + dualidad $\Rightarrow$ sistema de ecuaciones en derivadas parciales; solución $\Leftrightarrow$ derivadas cruzadas iguales (\concepto{teorema de \autor{Frobenius}}) $\Leftrightarrow$ \emph{simetría} de $S$;
  \item \emph{que sea función de gasto}: homogénea de grado 1 y creciente por construcción; falta la \emph{concavidad}: su hessiana es $S$ $\Rightarrow$ \emph{negatividad};
  \item \emph{de $E$ a $U$}: $E\to V$ invirtiendo; $U(\vect{x}^0)$ = mín.\ de $V$ entre los presupuestos con los que se compra $\vect{x}^0$.
  \end{itemize}
\end{itemize}
\graficorepaso[0.45\linewidth]{recuperar-utilidad}{Recuperar $U$ desde $V$: el mínimo de $V$ entre los presupuestos que permiten comprar $\vect{x}^0$}
\begin{itemize}
\item \emph{Preferencia revelada} \vertema{3.A.8}: axioma débil $\Rightarrow$ $S$ semidef.\ negativa, \emph{no} simétrica; axioma fuerte (\autor{Houthakker}, 1950) sí racionaliza; \autor{Kihlstrom}-\autor{Mas-Colell}-\autor{Sonnenschein} (1976) los unen; \autor{Afriat} (1967), datos finitos.
\item \emph{Relevancia}: \emph{teórica} (cierra el círculo: las preferencias no añaden nada a 3 propiedades contrastables); \emph{empírica} (contraste sin hallar $U$); \emph{normativa} ($E$ exacta desde la demanda: \autor{Hausman}, 1981).
\end{itemize}

\sub{III.2. Ecuación de \autor{Slutsky} y clasificación de los bienes}
\begin{itemize}
\item \emph{Derivación}: derivar $h_i(\vect{p},\overline{U})\equiv x_i\big(\vect{p},E(\vect{p},\overline{U})\big)$ respecto a $p_j$ (regla de la cadena); evaluar en $\overline{U}=V(\vect{p},\overline{W})$: $E=\overline{W}$ y, por \autor{Shephard}, $\partial E/\partial p_j=h_j=x_j$:
\end{itemize}
\funcionclave{\underbrace{\frac{\partial x_i(\vect{p},\overline{W})}{\partial p_j}}_{\text{ET}}=\underbrace{\frac{\partial h_i(\vect{p},\overline{U})}{\partial p_j}}_{\text{ES}}\ \underbrace{-\,x_j\,\frac{\partial x_i(\vect{p},\overline{W})}{\partial\overline{W}}}_{\text{ER}}}
\begin{formula}
\centering En elasticidades:\quad $\varepsilon_{x_i,p_j}=\varepsilon^{h}_{i,p_j}-s_j\,\varepsilon_{x_i,\overline{W}}$
\end{formula}
\begin{itemize}
\item \emph{Lectura}: efecto sustitución (ES, utilidad constante, \autor{Hicks}) inobservable; ET y efecto renta (ER) observables $\Rightarrow$ $S$ calculable con datos. ER pesa más cuanto mayor es $s_j$.
\item \emph{Propiedades de $S$}: simétrica; semidef.\ negativa ($S_{ii}\leq0$); $S\vect{p}=\vect{0}$ $\Rightarrow$ $\sum_{j\neq i}p_jS_{ij}=-p_iS_{ii}\geq0$: \concepto{todo bien tiene al menos un sustitutivo neto} (\autor{Hicks}); con 2 bienes, sustitutivos netos.
\item \emph{Precio propio}: \textcolor{tceeverde}{normal} $\Rightarrow$ \textcolor{tceeazul}{ordinario} (ley de la demanda); \textcolor{tceedorado}{inferior}: ordinario si domina el ES, \textcolor{tceerojo}{\autor{Giffen}} si domina el ER ($s_i$ alto). \autor{Giffen} $\Rightarrow$ inferior, no al revés. \autor{Veblen}, fuera (precio en las preferencias).
\end{itemize}
{\centering\small\renewcommand{\arraystretch}{1.15}\setlength{\tabcolsep}{3pt}%
\begin{tabular}{@{}ccl@{}}
\toprule
\textit{ES propio} & \textit{ER propio} & \textit{ET propio}\\\midrule
$\leq0$ & \textcolor{tceeverde}{$<0$: normal} & \textcolor{tceeazul}{$\leq0$: ordinario}\\
$\leq0$ & $=0$: frontera & \textcolor{tceeazul}{$\leq0$: ordinario}\\
$\leq0$ & \textcolor{tceedorado}{$>0$: inferior} & \textcolor{tceeazul}{ordinario} si $|\mathrm{ES}|\geq\mathrm{ER}$\\
 & & \textcolor{tceerojo}{$>0$: \autor{Giffen}} si $\mathrm{ER}>|\mathrm{ES}|$\\\bottomrule
\end{tabular}\par}
\begin{itemize}
\item \emph{Precios cruzados}: relación \emph{neta} (signo de $S_{ij}$) \emph{simétrica} (\autor{Hicks}-\autor{Allen}, 1934); \emph{bruta} (signo de $\partial x_i/\partial p_j$) no: los ER cruzados difieren. Sustitutivos netos + $i$ \textcolor{tceedorado}{inferior} $\Rightarrow$ sustitutivos brutos; complementarios netos + $i$ \textcolor{tceeverde}{normal} $\Rightarrow$ complementarios brutos; los otros casos, ambiguos (sustitutivos netos y complementarios brutos, posible).
\end{itemize}
\begin{itemize}
\item \emph{Pendientes} en $(x_i,p_i)$: bien normal, marshalliana más plana (más elástica) que la hicksiana; inferior, más inclinada; sin ER, coinciden $\Rightarrow$ puente con el bienestar: normal y baja el precio, $\mathrm{VC}\leq\Delta\mathrm{EC}\leq\mathrm{VE}$.
\end{itemize}

\sub{III.3. Análisis de bienestar del consumidor individual}
\begin{itemize}
\item \emph{Idea}: \concepto{economía del bienestar}: deseabilidad de las asignaciones. Aquí, \emph{un} consumidor y cambios de \emph{precios} (equilibrio parcial \vertema{3.A.16}): evita comparar personas. Usuario: el decisor público.
\item \emph{Problema}: baja $p_1$ $\Rightarrow$ de $\overline{U}^0$ a $\overline{U}^1$. ¿Cuánto mejora? Utilidad ordinal: $\overline{U}^1-\overline{U}^0$ no significa nada $\Rightarrow$ medida en euros con $E$. Notación: $\vect{p}^0\to\vect{p}^1$, $\overline{W}$ fija, $\overline{U}^k=V(\vect{p}^k,\overline{W})$.
\end{itemize}
\begin{itemize}
\item \concepto{Variación del excedente del consumidor} (\autor{Dupuit} 1844, \autor{Marshall} 1890): área a la izquierda de la marshalliana entre los dos precios. Observable, pero \emph{mezcla} ES y ER. Exacta si el ER es nulo (cuasilineal); con varios precios, depende del camino.
\item \concepto{Métrica monetaria} (\autor{McKenzie}, \autor{Samuelson}): $m(\bar{\vect{p}};\vect{p},\overline{W})\equiv E\big(\bar{\vect{p}},V(\vect{p},\overline{W})\big)$: una FIU en euros $\Rightarrow$ sus diferencias sí tienen sentido. VC: referencia $\vect{p}^1$; VE: $\vect{p}^0$ (\autor{Hicks}).
\end{itemize}
\begin{formula}
\centering $\Delta\mathrm{EC}=\displaystyle\int_{p_1^1}^{p_1^0}x_1(p_1,\overline{\vect{p}}_{-1},\overline{W})\,dp_1$\\[3pt]
$\mathrm{VC}=\overline{W}-E(\vect{p}^1,\overline{U}^0)=\displaystyle\int_{p_1^1}^{p_1^0}h_1(p_1,\overline{\vect{p}}_{-1},\overline{U}^0)\,dp_1$\\[3pt]
$\mathrm{VE}=E(\vect{p}^0,\overline{U}^1)-\overline{W}=\displaystyle\int_{p_1^1}^{p_1^0}h_1(p_1,\overline{\vect{p}}_{-1},\overline{U}^1)\,dp_1$
\end{formula}
\begin{itemize}
\item \concepto{Variación compensatoria} (VC): a precios \emph{finales}, renta que devuelve a la utilidad \emph{inicial}. Baja $p$: lo \emph{máximo que pagaría}; sube: $-$VC, compensación mínima. Área bajo la hicksiana \emph{inicial}.
\item \concepto{Variación equivalente} (VE): a precios \emph{iniciales}, renta que lleva a la utilidad \emph{final}. Baja $p$: lo \emph{mínimo que aceptaría} por renunciar; sube: $-$VE, lo máximo que pagaría por evitarla. Área bajo la hicksiana \emph{final}.
\end{itemize}
\graficorepaso[0.6\linewidth]{bienestar-vc-ve}{VC, $\Delta$EC y VE (y $L$, $P$) ante una bajada de $p_1$ de un bien normal}
\begin{formula}
\centering $L\leq\mathrm{VC}\leq\Delta\mathrm{EC}\leq\mathrm{VE}\leq P$\quad (bien normal, baja $p_1$)\\[3pt]
\autor{Willig} (1976):\ \ $\mathrm{VC}\approx\Delta\mathrm{EC}-\dfrac{\varepsilon_{x_1,\overline{W}}(\Delta\mathrm{EC})^2}{2\overline{W}}$,\ \ $\mathrm{VE}\approx\Delta\mathrm{EC}+\dfrac{\varepsilon_{x_1,\overline{W}}(\Delta\mathrm{EC})^2}{2\overline{W}}$
\end{formula}
\begin{itemize}
\item \emph{Ordenación}: inferior, se invierte; si sube el precio, se invierte en valor absoluto; ER nulo: VC $=\Delta$EC $=$ VE.
\item \emph{\autor{Willig}}: error relativo $\approx|\varepsilon_{x_1,\overline{W}}|\,|\Delta\mathrm{EC}|/2\overline{W}$ (5\,\% de la renta y elasticidad 1: 2,5\,\%) $\Rightarrow$ justifica el excedente en equilibrio parcial y coste-beneficio. \autor{Hausman} (1981): falla en la pérdida irrecuperable (2.º orden) $\Rightarrow$ $E$ exacta resolviendo la ecuación de la integrabilidad.
\item \emph{¿Cuál usar?} \emph{Ordenar políticas}: VE (base fija $\vect{p}^0$; la VC puede ordenar mal). \emph{Compensar} (criterio de \autor{Kaldor}-\autor{Hicks} \vertema{3.A.22}): VC (precios que regirán). Exceso de gravamen: $\mathrm{EG}=-\mathrm{VE}-T>0$ \vertema{4.B.9}.
\item \emph{Información incompleta} (solo $\vect{x}^0$, $\vect{x}^1$): variaciones de \autor{Laspeyres} $L=(\vect{p}^0-\vect{p}^1)\cdot\vect{x}^0\leq\mathrm{VC}$ y de \autor{Paasche} $P=(\vect{p}^0-\vect{p}^1)\cdot\vect{x}^1\geq\mathrm{VE}$, para cualquier bien: \concepto{sesgo de sustitución} (las cestas fijas ignoran que se reoptimiza).
\end{itemize}
\begin{formula}
\centering \autor{Konüs} (1924):\ \ $I_K(\overline{U})=\dfrac{E(\vect{p}^1,\overline{U})}{E(\vect{p}^0,\overline{U})}$\\[3pt]
$I_L=\dfrac{\vect{p}^1\cdot\vect{x}^0}{\vect{p}^0\cdot\vect{x}^0}\geq I_K(\overline{U}^0)$,\qquad $I_P=\dfrac{\vect{p}^1\cdot\vect{x}^1}{\vect{p}^0\cdot\vect{x}^1}\leq I_K(\overline{U}^1)$
\end{formula}
\begin{itemize}
\item \concepto{Índice verdadero de coste de la vida} (\autor{Konüs}): $I_K(\overline{U}^0)$, versión de la VC; $I_K(\overline{U}^1)$, de la VE. Exige $E$.
\item \concepto{\autor{Laspeyres}}: aproximación de 1.\textsuperscript{er} orden de $E(\vect{p}^1,\overline{U}^0)$ $\Rightarrow$ por la concavidad, \emph{cota superior}: sobrestima (indexar con $I_L$ sobrecompensa). \concepto{\autor{Paasche}}: cota inferior de $I_K(\overline{U}^1)$. Acotan índices \emph{distintos}: $I_P\leq I_K\leq I_L$ solo con homotéticas; igualdades solo con \autor{Leontief} ($S=0$).
\end{itemize}
\graficorepaso[0.5\linewidth]{indice-konus-laspeyres}{\autor{Laspeyres} sobrestima: el sesgo de sustitución}
\begin{itemize}
\item \concepto{Índices superlativos} (\autor{Diewert}, 1976): exactos para formas flexibles: \autor{Fisher} $I_F=\sqrt{I_LI_P}$ ($E$ cuadrática homogénea); \autor{Törnqvist} $\ln I_T=\sum_i\tfrac12(s_i^0+s_i^1)\ln(p_i^1/p_i^0)$ (translog): puente con II.
\item \concepto{IPC del INE}: índice \emph{de tipo \autor{Laspeyres}} encadenado (base 2025; ponderaciones anuales: reducen el sesgo, no lo eliminan) $\Rightarrow$ sobrevalora. \concepto{Deflactor del PIB}: \autor{Paasche} encadenado $\Rightarrow$ infravalora \vertema{3.A.30}\vertema{4.A.1}.
\end{itemize}
\begin{ideaclave}
$E$ traduce la utilidad a euros: la VC y la VE miden exactamente el cambio de bienestar; el $\Delta$EC las aproxima tanto mejor cuanto menor es el efecto renta (\autor{Willig}), y \autor{Laspeyres} y \autor{Paasche} las acotan con un sesgo de sustitución que sigue en los índices oficiales.
\end{ideaclave}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}: $\vect{h}$, $E$ y dualidad: un problema da las 4 funciones (I); AIDS y microdatos (II); integrabilidad, \autor{Slutsky} y bienestar: VC, VE; $\Delta$EC aproxima, \autor{Laspeyres}-\autor{Paasche} acotan (III).
\item \textbf{Relevancia}: de lo inobservable (utilidad) a restricciones sobre lo observable y bienestar en euros.
\item \textbf{Extensiones}: empresa \vertema{3.A.12}; parcial \vertema{3.A.16}; bienestar \vertema{3.A.22}; coste-beneficio \vertema{4.B.6}; gravamen \vertema{4.B.9}; IPC \vertema{4.A.1}; riesgo \vertema{3.A.10}; intertemporal \vertema{3.A.29}.
\item \textbf{Opinión}: la teoría más abstracta, muy útil en la práctica; límites: consumidor estático y agregación (sumar exige juicios de valor); herencia de \autor{Deaton}: microdatos y heterogeneidad.
\item \textbf{Idea final}: ¿euros para algo que solo se ordena? Sí: lo que cuesta alcanzar la utilidad. Pensionista: si su cesta es como la del IPC, la subida sobrecompensa algo (\autor{Laspeyres}); si es distinta, puede quedarse corta; la exacta, la VC. Sumar personas exige juicios de valor.
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Por qué $\mu^*=1/\lambda^*$?} Derivando $E(\vect{p},V(\vect{p},\overline{W}))=\overline{W}$ respecto a $\overline{W}$: $\frac{\partial E}{\partial\overline{U}}\frac{\partial V}{\partial\overline{W}}=1$.
\item \textbf{¿Por qué la hicksiana no admite bienes \autor{Giffen}?} No tiene efecto renta y $E$ es cóncava: $\partial h_i/\partial p_i\leq0$.
\item \textbf{¿Qué condiciones da la integrabilidad?} Ley de \autor{Walras}, simetría y negatividad de $S$; con 2 bienes la simetría es automática.
\item \textbf{¿Por qué es «casi ideal» el AIDS?} Flexible, coherente, agrega exactamente y lineal (LA-AIDS); pero negatividad no garantizada y \autor{Engel} restrictivas.
\item \textbf{¿VC o VE?} VE para ordenar políticas (base fija); VC para compensar a los perjudicados.
\item \textbf{¿Por qué el IPC sobrestima el coste de la vida?} \autor{Laspeyres} fija la cesta e ignora la sustitución; la concavidad de $E$ lo hace cota superior.
\end{enumerate}

\begin{trampa}
\begin{itemize}
\item $E$: homog.\ de grado \emph{1} en $\vect{p}$; $\vect{h}$: grado \emph{0} en $\vect{p}$; $V$ y $\vect{x}$: grado 0 en $(\vect{p},\overline{W})$.
\item \autor{Shephard} sin signo menos; \autor{Roy}, con signo menos.
\item $S$: \emph{primeras} derivadas de $\vect{h}$, \emph{segundas} de $E$.
\item $V$ \emph{cuasiconvexa} (no cuasicóncava), sin depender de la convexidad de las preferencias; $E$ \emph{cóncava} en $\vect{p}$.
\item VC: utilidad \emph{inicial}, precios \emph{finales}; VE: utilidad \emph{final}, precios \emph{iniciales}. ER nulo: VC $=\Delta$EC $=$ VE, pero $L\neq P$ si hay sustitución.
\item \autor{Laspeyres} y \autor{Paasche} acotan índices verdaderos \emph{distintos}; $I_P\leq I_K\leq I_L$ solo con homotéticas.
\item Axioma débil $\Rightarrow$ $S$ semidef.\ negativa, \emph{no} simétrica.
\item \autor{Giffen} $\Rightarrow$ inferior; inferior $\not\Rightarrow$ \autor{Giffen}.
\end{itemize}
\end{trampa}
