% Esquema de dualidad: problemas primal y dual, soluciones, funciones valor y
% relaciones entre ellas (rehecho a partir del cuadro resumen del original, con
% el lema de Shephard sin signo menos y las identidades inversas bien escritas).
% Notación del canon: vectores con flecha; primal (marshalliana) en morado y
% dual (hicksiana, variable de elección h) en verde.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usetikzlibrary{fit}
\begin{document}
\begin{tikzpicture}[tcee,
    cajap/.style={draw=tceemorado,very thick,fill=tceemorado!8,rounded corners=3pt,align=center,
                  minimum width=4.2cm,minimum height=1.2cm,inner sep=5pt},
    cajad/.style={cajap,draw=tceeverde,fill=tceeverde!8},
    une/.style={->,thick},
    rel/.style={->,thick,tceegris},
    nota/.style={font=\footnotesize,inner sep=2pt},
    notag/.style={nota,text=tceegris}]
  % Capas (vídeo): primal y sus funciones (1-3), dual y las suyas (4-6), relaciones (7-13).
  % columnas: primal en x = 0, dual en x = 13; filas en y = 0, -3,6, -7,2
  % 1: problema primal
  \capa{1}{
    \node[text=tceemorado,align=center] at (0,1.65) {\textbf{Problema primal}\\ \footnotesize maximización de la utilidad};
    \node[inner sep=2pt,anchor=base west] (PU) at (0.05,0.3) {$U(\vect{x})$};
    \node[inner sep=2pt,anchor=base east] (Pmax) at (0,0.3) {$\max\limits_{\vect{x}}$};
    \node[inner sep=2pt,anchor=base] (Psa) at (0,-0.45) {s.a.\ $\vect{p}\cdot\vect{x}\leq\overline{W},\ \vect{x}\geq\vect{0}$};
    \begin{scope}[on background layer]
      \node[cajap,fit=(Pmax)(PU)(Psa)] (P) {};
    \end{scope}
  }
  % 4: problema dual (la variable de elección es h)
  \capa{4}{
    \node[text=tceeverde,align=center] at (13,1.65) {\textbf{Problema dual}\\ \footnotesize minimización del gasto};
    \node[inner sep=2pt,anchor=base west] (DU) at (13.05,0.3) {$\vect{p}\cdot\vect{h}$};
    \node[inner sep=2pt,anchor=base east] (Dmin) at (13,0.3) {$\min\limits_{\vect{h}}$};
    \node[inner sep=2pt,anchor=base] (Dsa) at (13,-0.45) {s.a.\ $U(\vect{h})\geq\overline{U},\ \vect{h}\geq\vect{0}$};
    \begin{scope}[on background layer]
      \node[cajad,fit=(Dmin)(DU)(Dsa)] (D) {};
    \end{scope}
  }
  % 7: teorema básico de la dualidad
  \capa{7}{
    \draw[tceegris,thick,dashed,rounded corners=4pt] (2.45,1.25) rectangle (10.55,-8.75);
    \node[notag] at (6.5,-8.5) {\itshape Relaciones de dualidad};
    \draw[<->,very thick] (P.east) -- node[above,align=center] {$\overline{W}=E^*,\ \overline{U}=U^*$} node[below,nota] {teorema básico de la dualidad} (P.east -| D.west);
  }
  % 2: demanda marshalliana
  \capa{2}{
    \node[cajap] (X) at (0,-3.6) {\textbf{SCEDM}\\ $\vect{x}^*=\vect{x}(\vect{p},\overline{W})$};
    \draw[une] (P.south) -- node[right,nota] {resolviendo} (X.north);
  }
  % 3: función indirecta de utilidad
  \capa{3}{
    \node[cajap] (V) at (0,-7.2) {\textbf{FIU}\\ $U^*=V(\vect{p},\overline{W})=U(\vect{x}^*)$};
    \draw[une] (X.south) -- node[right,nota,align=left] {función\\ valor} (V.north);
  }
  % 8: identidad de Roy
  \capa{8}{
    \draw[une,tceemorado] ([xshift=-1.1cm]V.north) -- node[left,nota,align=right,text=tceemorado] {Identidad de \textsc{Roy}\\ $x_i(\vect{p},\overline{W})=-\dfrac{\partial V(\vect{p},\overline{W})/\partial p_i}{\partial V(\vect{p},\overline{W})/\partial\overline{W}}$} ([xshift=-1.1cm]X.south);
  }
  % 5: demanda hicksiana
  \capa{5}{
    \node[cajad] (H) at (13,-3.6) {\textbf{SCEDC}\\ $\vect{h}^*=\vect{h}(\vect{p},\overline{U})$};
    \draw[une] (D.south) -- node[left,nota] {resolviendo} (H.north);
  }
  % 6: función de gasto
  \capa{6}{
    \node[cajad] (E) at (13,-7.2) {\textbf{FG}\\ $E^*=E(\vect{p},\overline{U})=\vect{p}\cdot\vect{h}^*$};
    \draw[une] (H.south) -- node[left,nota,align=right] {función\\ valor} (E.north);
  }
  % 9: lema de Shephard (sin signo menos)
  \capa{9}{
    \draw[une,tceeverde] ([xshift=1.1cm]E.north) -- node[right,nota,align=left,text=tceeverde] {Lema de \textsc{Shephard}\\ $h_i(\vect{p},\overline{U})=\dfrac{\partial E(\vect{p},\overline{U})}{\partial p_i}$} ([xshift=1.1cm]H.south);
  }
  % 10: las funciones valor son inversas
  \capa{10}{
    \draw[rel,black] ([yshift=3mm]E.west) -- node[above,nota] {$V\big(\vect{p},E(\vect{p},\overline{U})\big)\equiv\overline{U}$} ([yshift=3mm]V.east);
    \draw[rel,black] ([yshift=-3mm]V.east) -- node[below,nota] {$E\big(\vect{p},V(\vect{p},\overline{W})\big)\equiv\overline{W}$} ([yshift=-3mm]E.west);
    \node[notag] at (6.5,-7.2) {inversas (dualidad entre funciones valor)};
  }
  % 11: relaciones entre demandas (diagonales)
  \capa{11}{
    \draw[rel,dashed] (V.north east) -- node[pos=0.78,sloped,notag,anchor=south,inner sep=2pt,yshift=6pt] {$\vect{x}(\vect{p},\overline{W})\equiv\vect{h}\big(\vect{p},V(\vect{p},\overline{W})\big)$} (H.south west);
    \draw[rel,dashed,<-] (X.south east) -- node[pos=0.22,sloped,notag,anchor=south,inner sep=2pt,yshift=6pt] {$\vect{h}(\vect{p},\overline{U})\equiv\vect{x}\big(\vect{p},E(\vect{p},\overline{U})\big)$} (E.north west);
  }
  % 12: ecuación de Slutsky
  \capa{12}{
    \draw[<->,thick] (X.east) -- (H.west);
    \node[nota,above] at (6.5,-3.55) {$\underbrace{\dfrac{\partial x_i(\vect{p},\overline{W})}{\partial p_j}}_{\text{efecto total}}=\underbrace{\dfrac{\partial h_i(\vect{p},\overline{U})}{\partial p_j}}_{\text{efecto sustitución}}\underbrace{-\,x_j(\vect{p},\overline{W})\,\dfrac{\partial x_i(\vect{p},\overline{W})}{\partial\overline{W}}}_{\text{efecto renta}}$};
    \node[notag,below] at (6.5,-3.65) {ecuación de \textsc{Slutsky}};
  }
  % 13: integrabilidad: de la demanda marshalliana a la utilidad
  \capa{13}{
    \draw[rel,dashed] ([xshift=0.4cm]X.north west) to[out=150,in=180,looseness=1.2] node[pos=0.5,left,notag,align=right] {integrabilidad\\ (III.1)} (P.west |- PU);
  }
\end{tikzpicture}
\end{document}
