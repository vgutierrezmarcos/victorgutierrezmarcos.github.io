% U* = Ū = 3 y x* = h* = (2; 1,5). Isogastos no tangentes e/p2 = 4 y 2.

\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,
    resto/.style={thick,tceegris!60}]
  % 1: ejes
  \capa{1}{\ejes{$x_1$}{$x_2$}{6.2}{4.2}}
  % 2: primal: recta de balance y conjunto presupuestario
  \capa{2}{\fill[area] (0,0) -- (0,3) -- (4,0) -- cycle;}
  \capa{2}{\draw[curva1] (0,3) -- (4,0);}
  \capa{2}{\node[etiqueta,below,text=tceemorado] (Wp) at (4,0) {$\overline{W}/p_1$};}
  \capa{2}{\node[etiqueta,text=tceemorado] at (0.95,0.4) {$\vect{p}\cdot\vect{x}\leq\overline{W}$};}
  % 3: primal: mapa de indiferencia (las no tangentes, en gris)
  \capa{3}{\draw[resto] plot[domain=0.45:5.6,samples=80,smooth] (\x,{1.6/\x}) node[etiqueta,right] {$U^0$};}
  \capa{3}{\draw[resto] plot[domain=1.35:5.6,samples=80,smooth] (\x,{5/\x}) node[etiqueta,right] {$U^2$};}
  \capa{3}{\draw[curva2] plot[domain=0.8:5.6,samples=80,smooth] (\x,{3/\x}) node[etiqueta,right] (Ul) {$U^*$};}
  % 4: primal: tangencia x*
  \capa{4}{\node[punto,fill=tceemorado] (X) at (2,1.5) {};}
  \capa{8}{\node[etiqueta,anchor=south west,text=tceemorado] (Xl) at (2.6,2.6) {$\vect{x}^*=\vect{x}(\vect{p},\overline{W})$};}
  \capa{8}{\draw[guia,solid] (Xl.south west) -- (X);}
  % 5: dual: la misma curva como restricción Ū e isogastos (las no tangentes, en gris)
  \capa{5}{\node[etiqueta,anchor=west,text=tceeazul,inner sep=0pt] at (Ul.east) {$=\overline{U}$};}
  \capa{5}{\draw[resto,dashed] (0,4) node[etiqueta,left,font=\footnotesize] {$e_1/p_2$} -- (5.33,0);}
  \capa{5}{\draw[resto,dashed] (0,2) node[etiqueta,left,font=\footnotesize] {$e_2/p_2$} -- (2.67,0);}
  \capa{5}{\draw[curva3,dash pattern=on 5pt off 4pt] (0,3) -- (4,0);}
  % 6: dual: tangencia h*
  \capa{6}{\node[punto,fill=tceeverde,inner sep=2.2pt] at (2,1.5) {};}
  \capa{6}{\node[punto,fill=tceemorado,inner sep=1.1pt] at (2,1.5) {};}
  \capa{8}{\node[etiqueta,anchor=south west,text=tceeverde] at (Xl.north west) {$\vect{h}^*=\vect{h}(\vect{p},\overline{U})$};}
  \capa{6}{\proyeccion{(X)}{$x_1^*=h_1^*$}{$x_2^*=h_2^*$}}
  % 7: W = E*; 8: etiquetas x* = x(p,W), h* = h(p,Ū) y recuadro del teorema básico
  % (el vídeo dibuja hasta la 7: las funciones y el teorema están en el esquema de dualidad)
  \capa{7}{\node[etiqueta,below,text=tceeverde,inner sep=1pt] at (Wp.south) {$=E^*/p_1$};}
  \capa{8}{\node[draw=tceerojo,thick,rounded corners=2pt,fill=white,inner sep=3pt,anchor=north] at (3.1,-1.15) {\textcolor{tceerojo}{misma cesta:} $\vect{x}^*=\vect{h}^*$ \quad con \quad $\overline{W}=E^*$,\ \ $\overline{U}=U^*$};}
\end{tikzpicture}
