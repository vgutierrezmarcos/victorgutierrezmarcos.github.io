% Escala: demanda p = a - bQ, coste marginal c igual; cantidades en (a-c)/b.
%   R_B: q_B = 1/2 - q_A/2, de (0; 1/2) a (1; 0)
%   R_A: q_A = 1/2 - q_B/2 -> q_B = 1 - 2 q_A, de (0; 1) a (1/2; 0)
%   C = (1/3; 1/3); isobeneficio de A: q_B = 1 - q_A - B/q_A, máximo sobre R_A
%   por C: B = 1/9; tangente a R_B en S = (1/2; 1/4): B = 1/8 (pendiente -1/2)
%   s_A* = (a-c)/4: R_A': q_A = 5/8 - q_B/2 -> q_B = 5/4 - 2 q_A, corta en S

\begin{tikzpicture}[tcee,x=8cm,y=8cm,
    declare function={iso(\k,\q)=1-\q-\k/\q;}]
  \useasboundingbox (-0.2,-0.2) rectangle (1.4,1.19);
  % 1: ejes
  \capa{1}{\ejes{$q_A$}{$q_B$}{1.18}{1.15}}
  % 2: función de reacción de la empresa extranjera, R_B(q_A)
  \capa{2}{\draw[curva2] (0,0.5) -- (1,0);}
  \capa{2}{\node[etiqueta,text=tceeazul,rotate={atan(-0.5)},anchor=south] at (0.84,0.08) {$R_B(q_A)$};}
  \capa{2}{\node[etiqueta,left] at (0,0.5) {$\frac{a-c}{2b}$};}
  \capa{2}{\node[etiqueta,below] at (1,0) {$\frac{a-c}{b}$};}
  % 3: función de reacción de la empresa nacional, R_A(q_B)
  \capa{3}{\draw[curva1] (0,1) -- (0.5,0);}
  \capa{3}{\node[etiqueta,text=tceemorado,rotate={atan(-2)},anchor=south,yshift=2pt] at (0.06,0.88) {$R_A(q_B)$};}
  \capa{3}{\node[etiqueta,left] at (0,1) {$\frac{a-c}{b}$};}
  \capa{3}{\node[etiqueta,below] at (0.5,0) {$\frac{a-c}{2b}$};}
  % 4: equilibrio de Cournot sin política
  \capa{4}{\node[punto,label={[etiqueta]above right:$C$}] (C) at ({1/3},{1/3}) {};}
  \capa{4}{\proyeccion{(C)}{$q^C$}{$q^C$}}
  % 5: isobeneficio de A por C: máximo sobre R_A
  \capa{5}{\draw[curva1,thin] plot[domain=0.75:0.15,samples=80] (\x,{iso(1/9,\x)}) node[etiqueta,left,text=tceemorado] {$B_A^C$};}
  % 6: isobeneficio de A tangente a R_B en S (bienestar nacional)
  \capa{6}{\draw[curva1,thin] plot[domain=0.81:0.19,samples=80] (\x,{iso(1/8,\x)}) node[etiqueta,below,text=tceemorado] {$B_A^S$};}
  \capa{6}{\draw[->,thin,tceemorado] (0.62,0.31) -- (0.62,0.21) node[etiqueta,right,midway,text=tceemorado] {más $B_A$};}
  % 7: la subvención desplaza R_A hacia fuera
  \capa{7}{\draw[curvanueva] (0.075,1.1) -- (0.625,0);}
  \capa{7}{\node[etiqueta,right,text=tceerojo] at (0.09,1.11) {$R_A(q_B;s_A)$};}
  \capa{7}{\draw[flecha] (0.2,0.6) -- (0.315,0.6) node[etiqueta,above,midway,text=tceerojo] {$s_A$};}
  % 8: nuevo equilibrio S, el de Stackelberg
  \capa{8}{\node[punto,label={[etiqueta]above right:$S$}] (S) at (0.5,0.25) {};}
  \capa{8}{\draw[guia] (0.5,0.25) -- (0.5,0) node[below=15pt] {$q_A^S$};}
  \capa{8}{\draw[guia] (0.5,0.25) -- (0,0.25) node[left] {$q_B^S$};}
  \capa{8}{\draw[flecha,shorten <=3pt,shorten >=3pt] (C) -- (S);}
\end{tikzpicture}
