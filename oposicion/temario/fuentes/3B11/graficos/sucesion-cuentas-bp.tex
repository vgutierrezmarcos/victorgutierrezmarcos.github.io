\hyphenpenalty=10000 \exhyphenpenalty=10000
% La balanza de pagos como sucesión de cuentas (MBP6, cap. 2; MBP7, cap. 2):
% cada flecha lleva el saldo que pasa a la cuenta siguiente (morado); entre
% corchetes, el nombre del MBP7. Colores como el panorama: lo no financiero en
% verde, lo financiero y la posición en azul, las otras variaciones en dorado.
% Capas: PII inicial -> bienes y servicios -> renta primaria -> renta
% secundaria (saldo CC) -> capital (capacidad o necesidad) -> financiera y EO
% -> otras variaciones -> PII final -> arrastre de la posición inicial.
\begin{tikzpicture}[tcee, font=\footnotesize,
  cuenta/.style={draw=#1, fill=#1!12, line width=0.8pt, rounded corners=2pt,
    align=center, text width=5.2cm, minimum height=0.85cm, inner sep=3pt},
  nofin/.style={cuenta=tceeverde},
  fin/.style={cuenta=tceeazul},
  pii/.style={cuenta=tceeazul, fill=tceeazul!22},
  ov/.style={cuenta=tceedorado},
  paso/.style={->, thick, tceegris!80!black},
  saldo/.style={font=\footnotesize\bfseries, text=tceemorado, anchor=west, align=left, inner sep=2pt},
  mbp/.style={font=\scriptsize\itshape, text=tceegris!80!black}]
  \foreach \n/\y in {p0/0, c1/-1.7, c2/-3.4, c3/-5.1, c4/-6.8, c5/-8.5, c6/-10.2, p1/-11.9}
    \coordinate (\n) at (0,\y);
  % Marco fijo
  \path (-4.3,-12.7) rectangle (8.9,0.75);
  % 1: posición inicial
  \capa{1}{\node[pii] at (p0) {\textbf{Posición de inversión internacional inicial}\\ $\mathrm{PIIN}_{t-1}$};}
  % 2: bienes y servicios
  \capa{2}{\node[nofin] at (c1) {\textbf{Cuenta de bienes y servicios}\\ {\scriptsize\itshape\color{tceegris!80!black}[MBP7: cuentas de bienes y de servicios]}};
    \draw[paso] ($(c1)+(0,-0.45)$) -- node[saldo, right] {saldo de bienes y servicios: $X-M$} ($(c2)+(0,0.45)$);}
  % 3: renta primaria
  \capa{3}{\node[nofin] at (c2) {\textbf{Cuenta de renta primaria}\\ {\scriptsize\itshape\color{tceegris!80!black}[MBP7: renta obtenida]}};
    \draw[paso] ($(c2)+(0,-0.45)$) -- node[saldo, right] {$+\,\mathrm{RP}$} ($(c3)+(0,0.45)$);}
  % 4: renta secundaria y saldo corriente
  \capa{4}{\node[nofin] at (c3) {\textbf{Cuenta de renta secundaria}\\ {\scriptsize\itshape\color{tceegris!80!black}[MBP7: renta de transferencias]}};
    \draw[paso] ($(c3)+(0,-0.45)$) -- node[saldo, right] {$+\,\mathrm{RS}$: saldo de la cuenta corriente,\\ $\mathrm{CC}=(X-M)+\mathrm{RP}+\mathrm{RS}$} ($(c4)+(0,0.45)$);}
  % 5: cuenta de capital
  \capa{5}{\node[nofin] at (c4) {\textbf{Cuenta de capital}};
    \draw[paso] ($(c4)+(0,-0.45)$) -- node[saldo, right] {$+\,\mathrm{CK}$: capacidad ($+$) o necesidad ($-$)\\ de financiación, $\mathrm{CC}+\mathrm{CK}$} ($(c5)+(0,0.45)$);}
  % 6: cuenta financiera y errores y omisiones
  \capa{6}{\node[fin] at (c5) {\textbf{Cuenta financiera}\\ ${\mathrm{CF}=\mathrm{VNA}-\mathrm{VNP}}$};
    \node[draw=tceegris, dashed, rounded corners=2pt, fill=tceegris!10, align=center, text width=2.6cm, font=\scriptsize, inner sep=2pt]
      (eo) at (5.6,-8.5) {errores y omisiones\\ $\mathrm{EO}$ {\scriptsize\itshape\color{tceegris!80!black}[MBP7: discrepancia estadística]}};
    \draw[paso, dashed] (eo.west) -- ($(c5)+(2.75,0)$);
    \draw[paso] ($(c5)+(0,-0.45)$) -- node[saldo, right] {$\mathrm{CF}=\mathrm{CC}+\mathrm{CK}+\mathrm{EO}$} ($(c6)+(0,0.45)$);}
  % 7: otras variaciones
  \capa{7}{\node[ov] at (c6) {\textbf{Cuenta de otras variaciones}\\ revalorizaciones y otros cambios de volumen};
    \draw[paso] ($(c6)+(0,-0.45)$) -- node[saldo, right] {$+\,\mathrm{OV}$} ($(p1)+(0,0.45)$);}
  % 8: posición final
  \capa{8}{\node[pii] at (p1) {\textbf{Posición de inversión internacional final}\\ ${\mathrm{PIIN}_t=\mathrm{PIIN}_{t-1}+\mathrm{CF}_t+\mathrm{OV}_t}$};}
  % 9: la posición inicial se arrastra hasta la final
  \capa{9}{\draw[paso, tceeazul, dashed] ($(p0)+(-2.75,0)$) -- ++(-0.9,0) |- ($(p1)+(-2.75,0)$);
    \node[rotate=90, font=\scriptsize\itshape, text=tceeazul] at (-3.95,-3.4) {la posición inicial pasa a la final};}
\end{tikzpicture}
\end{document}
