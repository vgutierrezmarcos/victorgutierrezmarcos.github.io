% Tipos de interés oficiales del BCE en escalón (facilidad de depósito, OPF y
% facilidad marginal de crédito) y tipo a un día del mercado (eonia hasta
% 2021, €STR desde 2022, media mensual), 1/1/1999-14/11/2026.
% Datos: ../datos/tipos-bce.csv (BCE, FM: DFR, MRR_FR, MRR_MBR —tipo mínimo de
% puja del 28/6/2000 al 14/10/2008—, MLFR) y ../datos/tipo-un-dia.csv (BCE, FM
% EONIA y EST €STR); consulta 14/11/2026. Eje x en años decimales.
%   facilidad de depósito: 4,00 del 20/9/2023 al 11/6/2024 (máximo);
%   −0,50 del 18/9/2019 al 26/7/2022 (mínimo); 2,50 desde el 16/9/2026
%   primera subida de 2011 (Trichet): 13/4/2011, OPF de 1,00 a 1,25
%   tipos negativos de la facilidad de depósito: 11/6/2014-26/7/2022
% Paleta «dic»: depósito dicdeposito (gruesa), OPF dicopf, facilidad marginal
% dicfmc, tipo a un día gris fino.
% Capas (CRITERIOS §7): sombreado de tipos negativos → cero → facilidad
% marginal y OPF → facilidad de depósito → tipo a un día → puntos → rótulos.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\newcommand{\tipos}{../datos/tipos-bce.csv}
\newcommand{\undia}{../datos/tipo-un-dia.csv}
\begin{document}
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=11.5cm, height=6.2cm, scale only axis,
      xmin=1998.8, xmax=2027.2, xtick={1999,2003,...,2027},
      ymin=-1, ymax=6.5, ytick={-1,0,...,6},
      axis lines=left, tick align=outside, clip=false,
      ymajorgrids, grid style={tceegris!20},
      ticklabel style={font=\footnotesize},
      x tick label style={/pgf/number format/1000 sep={}},
      /pgf/number format/use comma,
      ylabel={\%}, label style={font=\footnotesize},
      legend style={at={(0.5,-0.12)}, anchor=north, draw=none, font=\footnotesize,
        legend columns=2, /tikz/every even column/.append style={column sep=12pt}},
      legend cell align=left,
      dicapas,
    ]
    % 1: periodo de tipos negativos de la facilidad de depósito (11/6/2014-26/7/2022)
    \capa{1}{\dicsombra[fill=dicreferencia, fill opacity=0.10]{2014.4411}{-1}{2022.5671}{6.5}}
    \capa{1}{\dicrotulo[anchor=north, font=\scriptsize, text=dicreferencia]{(2018.5,6.4)}{tipos negativos}}
    % 2: la línea del cero
    \capa{2}{\dicraya{(1998.8,0) (2027.2,0)}}
    % 3: el corredor: facilidad marginal de crédito (techo) y OPF
    \capa{3}{\addplot[const plot, diclinea=dicfmc, thick] table[col sep=comma, comment chars={\#}, x=x, y=facilidad_marginal] {\tipos};
      \addlegendentry{Facilidad marginal de crédito}}
    \capa{4}{\addplot[const plot, diclinea=dicopf, thick] table[col sep=comma, comment chars={\#}, x=x, y=opf] {\tipos};
      \addlegendentry{OPF (fijo o mínimo de puja)}}
    % 5: facilidad de depósito (suelo del corredor; el tipo que guía la política desde 2015)
    \capa{5}{\addplot[const plot, diclinea=dicdeposito, line width=1.6pt] table[col sep=comma, comment chars={\#}, x=x, y=facilidad_deposito] {\tipos};
      \addlegendentry{Facilidad de depósito}}
    % 6: tipo a un día del mercado (eonia hasta 2021; €STR desde 2022)
    \capa{6}{\addplot[thin, dicotro] table[col sep=comma, comment chars={\#}, x=x, y=tipo] {\undia};
      \addlegendentry{Eonia (hasta 2021) y €STR (desde 2022), media mensual}}
    % 7: hitos de la facilidad de depósito y de las OPF
    \capa{7}{\addplot[dicpuntos=dicdeposito, forget plot] coordinates {(2023.72,4) (2020.5,-0.5) (2026.71,2.5)};}
    \capa{7}{\addplot[dicpuntos=dicopf, forget plot] coordinates {(2011.28,1.25)};}
    % 8: rótulos
    \capa{8}{\dicrotulo[anchor=east, font=\scriptsize, text=dicdeposito, align=right]{(2022.7,4.3)}{4,00~\%\\(9/2023-6/2024)}}
    \capa{8}{\dicrotulo[anchor=north, font=\scriptsize, text=dicdeposito, align=center]{(2020.5,-0.6)}{$-0{,}50$~\% (2019-2022)}}
    \capa{8}{\dicrotulo[anchor=south, font=\scriptsize, text=dicopf, align=center]{(2011.28,2.35)}{subidas\\de 2011}}
    \capa{8}{\dicrotulo[anchor=west, font=\scriptsize, text=dicdeposito]{(2026.9,2.5)}{2,50}}
  \end{axis}
\end{tikzpicture}
\end{document}
