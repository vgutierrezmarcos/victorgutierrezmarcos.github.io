% El problema n-1 y la asimetría del SME: 2 mercados de dinero. A la izquierda,
% el país ancla A (Alemania) fija su oferta monetaria y con ella i_A; a la
% derecha, el país B mantiene la paridad: su tipo es i_A + delta^e y su oferta
% monetaria se ajusta (intervención, reservas) hasta donde lo pide su demanda.
% Perturbación: A contrae su oferta (reunificación, 1990-1992): sube i_A, sube
% la recta de B y B tiene que contraer su oferta aunque esté en recesión.
% Capas: ejes A -> demanda A -> oferta A e i_A -> ejes B -> demanda B ->
% recta de paridad y oferta B -> contracción en A -> nuevo i_A -> recta nueva y
% contracción forzosa en B.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=1cm,y=1cm]
  \path (-1.2,-0.9) rectangle (14.6,5.6);
  % ---- País A (ancla) ----
  % 1: ejes
  \capa{1}{\ejes{$M_A/P_A$}{$i_A$}{5.6}{5}}
  \capa{1}{\node[font=\small\bfseries,text=tceemorado] at (2.8,5.3) {A: el ancla (Alemania)};}
  % 2: demanda de dinero de A
  \capa{2}{\draw[curva2,domain=0.9:5.2,samples=60] plot (\x,{4/\x}) node[etiqueta,right,text=tceeazul] {$L_A$};}
  % 3: oferta de A y su tipo
  \capa{3}{\draw[curva1] (3.2,0) -- (3.2,4.3) node[etiqueta,above,text=tceemorado] {$M^S_A$};}
  \capa{3}{\node[punto] at (3.2,1.25) {};}
  \capa{3}{\draw[guia] (3.2,1.25) -- (0,1.25) node[left] {$i_A$};}
  % ---- País B ----
  % 4: ejes
  \begin{scope}[xshift=8cm]
  \capa{4}{\ejes{$M_B/P_B$}{$i_B$}{5.6}{5}}
  \capa{4}{\node[font=\small\bfseries,text=tceemorado] at (2.8,5.3) {B: mantiene la paridad};}
  % 5: demanda de dinero de B
  \capa{5}{\draw[curva2,domain=1.15:5.2,samples=60] plot (\x,{5.2/\x}) node[etiqueta,right,text=tceeazul] {$L_B$};}
  % 6: la paridad fija el tipo de B; la oferta de B se ajusta
  \capa{6}{\draw[curva4] (0,1.65) -- (5.3,1.65) node[etiqueta,right,text=tceedorado] {$i_A+\delta^e$};}
  \capa{6}{\draw[curva1] (3.15,0) -- (3.15,4.3) node[etiqueta,above,text=tceemorado] {$M^S_B$};}
  \capa{6}{\node[punto] at (3.15,1.65) {};}
  \end{scope}
  % ---- Perturbación en A ----
  % 7: el ancla contrae su oferta
  \capa{7}{\draw[curvanueva] (2.0,0) -- (2.0,4.3) node[etiqueta,above,text=tceerojo] {$M^{S\prime}_A$};}
  \capa{7}{\draw[flecha] (3.1,3.5) -- (2.1,3.5);}
  % 8: sube su tipo
  \capa{8}{\node[punto] at (2.0,2.0) {};}
  \capa{8}{\draw[guia] (2.0,2.0) -- (0,2.0) node[left] {$i_A'$};}
  % 9: B tiene que seguirle
  \begin{scope}[xshift=8cm]
  \capa{9}{\draw[curvanueva] (0,2.4) -- (5.3,2.4) node[etiqueta,right,text=tceerojo] {$i_A'+\delta^e$};}
  \capa{9}{\draw[curvanueva] (2.17,0) -- (2.17,4.3) node[etiqueta,above,text=tceerojo] {$M^{S\prime}_B$};}
  \capa{9}{\draw[flecha] (3.05,3.5) -- (2.27,3.5);}
  \capa{9}{\node[punto] at (2.17,2.4) {};}
  \capa{9}{\node[etiqueta,text=tceerojo,anchor=west,align=left,font=\footnotesize] at (3.25,3.5) {pierde reservas:\\contracción forzosa};}
  \end{scope}
\end{tikzpicture}
\end{document}
