% Oferta de trabajo de hoy en el modelo de Lucas y Rapping (1969).
% Elaboración propia. Solución cerrada con u = ln C + b ln(1-L):
%   L_1 = 1 - k [1 + w_2/((1+r) w_1)],  k = b/((1+b)(1+beta))
% Valores: b = 1, beta = 0,96, r = 0,04 -> k = 1/3,92 = 0,2551; asíntota 1 - k = 0,7449.
% L_1 > 0 si w_1 > w_2/((1+r)(1/k - 1)) = w_2/3,0368:
%   w_2 = 1 (bajo): desde w_1 = 0,3293;  w_2 = 2 (alto): desde w_1 = 0,6586.
% Se dibuja w_1 entre ese umbral y 4. Escala: 1 hora (L_1 = 1) = 6 cm; w_1 = 1 -> 1,1 cm.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=6cm,y=1.1cm,
    declare function={lu(\w,\v)=1-(1/3.92)*(1+\v/(1.04*\w));}]
  % 1: ejes
  \capa{1}{\ejes{$L_1$}{$w_1$}{1.18}{4.6}}
  \capa{1}{\draw (1,0.06) -- (1,-0.06) node[below] {$1$};}
  \capa{1}{\draw[guia] (0.7449,0) node[below,text=black] {$1-\frac{b}{(1+b)(1+\beta)}$} -- (0.7449,4.4);}
  % 2: oferta de hoy con un salario de mañana bajo
  \capa{2}{\draw[curva1] plot[domain=0.3293:4,samples=120,smooth] ({lu(\x,1)},\x)
    ;}
  \capa{2}{\node[etiqueta,anchor=west,text=tceemorado] at (0.57,0.95) {$w_2$ bajo};}
  % 3: con un salario de mañana alto, menos horas hoy a cada w_1
  \capa{3}{\draw[curvanueva] plot[domain=0.6586:4,samples=120,smooth] ({lu(\x,2)},\x)
    ;}
  \capa{3}{\node[etiqueta,anchor=east,text=tceerojo] at (0.5,2.6) {$w_2$ alto};}
  \capa{3}{\draw[flecha] ({lu(1.6,1)-0.015},1.6) -- ({lu(1.6,2)+0.015},1.6);}
\end{tikzpicture}
\end{document}
