\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: «\emph{el tiempo es oro}»: 24 horas al día.
  \begin{itemize}
  \item ¿Cuántas se venden, a qué edad, tras cuántos años de estudio?
  \item ¿Cuánto vale una hora? En el margen, el salario; pero cambia con la persona, el momento y lo invertido en sí mismo.
  \item Nobel 2023, \autor{Goldin}: 2 siglos de mujeres en el mercado laboral.
  \end{itemize}
\item \textbf{Relevancia}:
  \begin{itemize}
  \item \emph{micro}: el salario, principal renta de los hogares;
  \item \emph{macro}: principal factor; producción, distribución y paro (\dato{10,18\,\%}{INE, EPA 3T 2026});
  \item \emph{política}: impuestos sobre el trabajo \vertema{4.B.10}, salario mínimo \vertema{4.A.16}, educación \vertema{4.A.2}.
  \end{itemize}
\item \textbf{Contextualización}:
  \begin{itemize}
  \item \concepto{Economía laboral}: servicios de trabajo por salarios. Enfoque micro y \emph{neoclásico} (competencia, salarios flexibles) $\Rightarrow$ $w=\mathrm{PMg}_L$, sin paro involuntario.
  \item \autor{Marshall} (1890), \autor{Hicks} (1932); eclipsado por \autor{Keynes} (1936) \vertema{3.A.4}.
  \item Años 60, vuelta a la oferta: capital humano (\autor{Mincer}, 1958; \autor{Schultz}, 1961; \autor{Becker}, 1962-1964) y oferta intertemporal (\autor{Lucas}-\autor{Rapping}, 1969 $\to$ \autor{Kydland}-\autor{Prescott}, 1982).
  \item Revolución de la credibilidad: Nobel 2021 y 2023.
  \end{itemize}
\item \textbf{Problemática}:
  \begin{itemize}
  \item ¿Cuánto se trabaja, a cuántos se contrata, qué salario?
  \item ¿Más horas si sube $w$? ¿Transitorio o permanente?
  \item ¿Por qué no todos cobran igual? ¿Rinde estudiar y cómo se mide?
  \end{itemize}
\item \textbf{Estructura}: de lo sencillo a lo rico; evidencia al final de cada modelo.
  \begin{itemize}
  \item \hyperref[rep:I]{I}: modelo estático (oferta, demanda, equilibrio);
  \item \hyperref[rep:II]{II}: la oferta en el tiempo;
  \item \hyperref[rep:III]{III}: se levanta la homogeneidad del trabajo (\autor{Becker}, \autor{Mincer}).
  \end{itemize}
\end{itemize}

\bloque[10]{I. Oferta y demanda de trabajo neoclásicas\label{rep:I}}
\begin{ideaclave}
Con competencia y salarios flexibles, el mercado se vacía: el salario real iguala la valoración marginal del ocio y la productividad marginal del trabajo; no hay paro involuntario salvo con salario mínimo.
\end{ideaclave}
\begin{itemize}
\item \textbf{8 supuestos} (y dónde se levantan): 1 racionalidad; 2 información perfecta (\autor{Spence}; \vertema{3.A.26} \vertema{3.A.27}); 3 trabajo homogéneo (\hyperref[rep:III]{III}); 4 precio-aceptación (monopsonio, sindicatos); 5 libre participación; 6 salarios flexibles (salario mínimo; \vertema{3.A.28}); 7 movilidad sin costes (formación específica, \emph{insiders}); 8 sin externalidades \vertema{3.A.44}.
\end{itemize}

\sub{I.1. Oferta de trabajo\label{rep:I.1}}
\subsub{a) Idea}
\begin{itemize}
\item Oferta = \emph{demanda de ocio} del revés \vertema{3.A.8}: quién participa ($w^R$), cuántas horas (tangencia), por qué no siempre crece (efectos renta).
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item $U(C,O)$ de buen comportamiento; \concepto{$\mathrm{RMS}_{OC}$} $=U_O/U_C$, decreciente.
\item Dotación $O_0$, $L=O_0-O$; \concepto{renta no laboral} $\overline{W}\geq0$; $p=1$.
\item \concepto{Renta completa} $wO_0+\overline{W}$: $w$ a los 2 lados (precio del ocio y valor de la dotación).
\item \emph{Problema} (\autor{Kuhn}-\autor{Tucker}; $\lambda$ = utilidad marginal de la renta):
\begin{formula}
\centering $\optim{\max}{C,O}{U(C,O)}{C+w\,O\leq w\,O_0+\overline{W},\ 0\leq O\leq O_0}$
\end{formula}
\item \emph{Solución interior}: valoración del ocio = su coste de oportunidad:
\funcionclave{\mathrm{RMS}_{OC}=\frac{U_O}{U_C}=w,\qquad C^*+w\,O^*=w\,O_0+\overline{W}}
\item \concepto{Salario de reserva}: pendiente de la curva de indiferencia en la dotación $A=(O_0,\overline{W})$:
\begin{formula}
\centering $w^R=\dfrac{U_O(\overline{W},O_0)}{U_C(\overline{W},O_0)}$;\ ocio normal $\Rightarrow$ $\partial w^R/\partial\overline{W}>0$
\end{formula}
  \begin{itemize}
  \item $w\leq w^R$ $\Rightarrow$ $L=0$ (esquina). ↑$\overline{W}$: solo efecto renta, $\partial L/\partial\overline{W}<0$: desincentiva participar.
  \item Más aún una \emph{prestación por desempleo}: ↑$w^R$, nadie elige pocas horas, \emph{trampa del desempleo} \vertema{4.B.10}.
  \end{itemize}
\item ↑$w$, \emph{3 efectos} (la recta gira sobre $A$):
  \begin{itemize}
  \item \concepto{sustitución} $E\to E'$: ↑$L$ siempre (oferta hicksiana);
  \item \concepto{renta ordinario} $E'\to E''$: menos poder adquisitivo: ↑$L$ si ocio normal;
  \item \concepto{renta dotación} $E''\to E_1$: revaloriza $O_0$: ↓$L$. Total \emph{ambiguo}.
  \end{itemize}
\end{itemize}
\graficorepaso[0.55\linewidth]{efectos-salario}{↑$w$: sustitución, renta ordinario y renta dotación}
\begin{itemize}
\item \autor{Slutsky} con renta-dotación: renta global $\propto L$ (lo que se \emph{vende}):
\begin{formula}
\centering $\dfrac{\partial L}{\partial w}=\underbrace{-\dfrac{\partial h_O}{\partial w}}_{(+)}+\underbrace{L\,\dfrac{\partial L}{\partial\overline{W}}}_{(-)\text{ ocio normal}}$
\end{formula}
\item \emph{Prima por horas extra} (solo sobre $L>L_n$): casi solo sustitución $\Rightarrow$ ↑$L$ sin ambigüedad.
\end{itemize}
\begin{itemize}
\item \emph{Oferta individual}: $L=0$ bajo $w^R$; creciente con $w$ moderado; \emph{hacia atrás} con $w$ alto (\autor{Robbins}, 1930).
\item \emph{Agregada}: sin agente representativo (\autor{Gorman}, 1953); \concepto{margen intensivo} (horas) + \concepto{extensivo} (participación).
\end{itemize}
\subsub{c) Extensiones}
\begin{itemize}
\item \emph{Producción doméstica} (\autor{Becker}, 1965; \autor{Gronau}, 1977): $f'(L_{\mathrm{dom}})=w$ y $U_O/U_C=w$.
\item \emph{Familia}: \concepto{trabajador añadido}; modelo \emph{unitario} vs \emph{colectivo} (\autor{Chiappori}, 1992: importa quién cobra).
\end{itemize}
\subsub{d) Evidencia empírica}
\begin{itemize}
\item 3 elasticidades; ocio normal $\Rightarrow$ $\varepsilon^F\geq\varepsilon^h\geq\varepsilon$ (\autor{Keane}, 2011).
  \begin{itemize}
  \item Varones, pequeñas (hicksiana $\approx0{,}3$); mujeres casadas, mayores. \autor{Chetty} (2011): 0,30 intensiva, 0,26 extensiva.
  \end{itemize}
\item \concepto{U de \autor{Goldin}} (1995), participación de las casadas:
  \begin{itemize}
  \item cae con la industrialización (efecto renta del marido, estigma);
  \item sube con educación y servicios (sustitución);
  \item se acelera desde los 70 con la píldora (\autor{Goldin}-\autor{Katz}, 2002).
  \end{itemize}
\end{itemize}
\subsub{e) Valoración}
\begin{itemize}
\item + participación, horas poco elásticas, fiscalidad \vertema{4.B.10}; $-$ estático, horas libres.
\end{itemize}

\sub{I.2. Demanda de trabajo\label{rep:I.2}}
\subsub{a) Idea}
\begin{itemize}
\item \concepto{Demanda derivada} (\autor{Hicks}, 1932): $z_L(p,\vect{w})$ de la teoría de la empresa \vertema{3.A.12}.
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item \emph{Supuestos}: max $B$; precio-aceptante en $p$, $w_L$, $w_K$; $F_L,F_K>0$, $F_{LL},F_{KK}<0$, $F_{LK}>0$.
\item \emph{Corto plazo}: $\max_L\ B=p\,F(L,\overline{K})-w_LL-w_K\overline{K}$:
\funcionclave{\mathrm{VPMg}_L\equiv p\,F_L(L,\overline{K})=w_L\ \Longleftrightarrow\ F_L=\frac{w_L}{p}\equiv w}
  \begin{itemize}
  \item CSO $pF_{LL}<0$; $L^D(\overset{-}{w},\overset{+}{F_L})$: tramo decreciente del $\mathrm{VPMg}_L$.
  \end{itemize}
\item \emph{Largo plazo}: $\max_{L,K}\ pF(L,K)-w_LL-w_KK$:
  \begin{itemize}
  \item $pF_L=w_L$, $pF_K=w_K$ $\Rightarrow$ $\mathrm{RMST}_{LK}=w_L/w_K$;
  \item \concepto{demanda no condicionada} $z_L(\overset{+}{p},\overset{-}{w_L},\overset{?}{w_K})$.
  \end{itemize}
\item \emph{Corto vs largo} (↓$w_L$): $A\to B$; más $K$ desplaza el $\mathrm{VPMg}_L$: $B\to C$ $\Rightarrow$ L/p \emph{más elástica} (\autor{Le Chatelier}).
\item Sustitución + escala, los 2 $\leq0$ $\Rightarrow$ \emph{no hay factores \autor{Giffen}}.
\item \emph{Mercado}: todas a la vez $\Rightarrow$ ↓$p$, ↓$\mathrm{VPMg}_L$: de $A$ a $C$, no a $B$; suma de \concepto{cuasidemandas}: \emph{menos elástica}.
\end{itemize}
\pareja{demanda-lp-cp}{Corto y largo plazo: la de largo, más elástica}{demanda-agregada}{Mercado: cuasidemandas, de $A$ a $C$}
\begin{itemize}
\item \emph{Implicaciones}: elasticidad, la cifra clave; \autor{Marshall}-\autor{Hicks}: $\varepsilon_{L^D,w_L}=-(1-s_L)\sigma-s_L|\varepsilon_{q,p}|$.
\end{itemize}
\subsub{c) Extensiones: poder de mercado}
\begin{itemize}
\item $\mathrm{IPMg}_L=\mathrm{GMg}_L$; 4 casos \vertema{3.A.17}: menos $L$ y $w$, \emph{sin paro involuntario} (\emph{explotación}, \autor{Robinson}, 1933).
\end{itemize}
\subsub{d) Evidencia y e) valoración}
\begin{itemize}
\item 0,15-0,75, central 0,3 (\autor{Hamermesh}, 1993). Mariel (\autor{Card}, 1990): +7\,\% activos, sin efectos; \autor{Borjas} (2017): caídas sin secundaria.
\item Estática: sin costes de ajuste \vertema{3.A.26}.
\end{itemize}

\sub{I.3. Equilibrio\label{rep:I.3}}
\begin{itemize}
\item $L^S(w^c)=L^D(w^c)$ $\Rightarrow$ $U_O/U_C=w^c=F_L(L^c,\overline{K})$.
  \begin{itemize}
  \item Existencia; unicidad y estabilidad no garantizadas (oferta hacia atrás) \vertema{3.A.21}.
  \item Sin \concepto{paro involuntario} (sí \emph{voluntario}, $w^R>w^c$); eficiente (1TFEB) \vertema{3.A.22}.
  \end{itemize}
\item \autor{Marshall}: equilibrio \emph{temporal} (oferta fija, sube solo $w$) y \emph{normal} (entra oferta: ↑$L$, $w$ baja algo).
\item \concepto{Salario mínimo} $\underline{w}>w^c$: manda el lado corto; paro involuntario $L^S(\underline{w})-L^D(\underline{w})$, el único del modelo.
\end{itemize}
\pareja{oferta-individual}{Oferta individual: $w^R$ y tramo hacia atrás}{equilibrio-trabajo}{Equilibrio y salario mínimo: paro involuntario}

\sub{I.4. Valoración: monopsonio y paro\label{rep:I.4}}
\begin{itemize}
\item Supuesto más frágil, el 4: \concepto{monopsonio} (único empleador o fricciones); contrata hasta $\mathrm{VPMg}_L=$ \concepto{gasto marginal}:
  \begin{itemize}
  \item[]
\begin{formula}
\centering $\mathrm{VPMg}_L=w_L(L)\big(1+\tfrac{1}{\varepsilon_{L,w_L}}\big)\Leftrightarrow\tfrac{\mathrm{VPMg}_L-w_L}{w_L}=\tfrac{1}{\varepsilon_{L,w_L}}$
\end{formula}
  \item $L^M<L^c$, $w^M<w^c$, \emph{sin paro involuntario}; salario mínimo en $(w^M,w^c]$: ↑$w$ y ↑$L$.
  \end{itemize}
\item \autor{Card}-\autor{Krueger} (1994): Nueva Jersey (4,25 $\to$ 5,05 \$/h) vs Pensilvania: sin caída del empleo. \autor{Manning} (2003).
\end{itemize}
\pareja{monopsonio}{Monopsonio: menos $L$ y menos $w$}{salario-minimo}{Salario mínimo con monopsonio: ↑$L$ hasta $w^c$}
\begin{itemize}
\item \concepto{Sindicato monopolista} (\autor{Dunlop}, 1944): $w^S>w^c$: \emph{sí} paro involuntario.
\item \emph{El paro existe} (España $>$10\,\%): levantar 2 y 7 \vertema{3.A.26}; 6 y 4 \vertema{3.A.27} \vertema{3.A.28}.
\end{itemize}

\bloque[6,5]{II. Oferta intertemporal: \autor{Lucas} y \autor{Rapping} (1969)\label{rep:II}}
\begin{ideaclave}
Se trabaja más cuando trabajar compensa más: los cambios transitorios del salario mueven las horas mucho más que los permanentes; el mercado de trabajo, pieza central del ciclo.
\end{ideaclave}

\sub{II.1. Idea\label{rep:II.1}}
\begin{itemize}
\item Horas muy volátiles con $w$ estable: se \emph{elige cuándo trabajar} (ocio de hoy y de mañana, sustitutivos). «\emph{Hacer el agosto}».
\item Microfundamenta la oferta (como el consumo \vertema{3.A.33}); mercado siempre vaciado $\Rightarrow$ nueva macro clásica \vertema{3.A.6}, ciclos reales \vertema{3.A.42}.
\end{itemize}

\sub{II.2. Modelo\label{rep:II.2}}
\subsub{a) Supuestos}
\begin{itemize}
\item Infinitos periodos, sin incertidumbre; $U$ separable, $\beta=1/(1+\rho)$; $O_t+L_t=1$; $w_t$, $\overline{W}_t$ conocidos; activos $b_t$ a $r$.
\end{itemize}
\subsub{b) Desarrollo}
\begin{formula}
\centering $\optim{\max}{C_t,O_t,b_{t+1}}{\textstyle\sum_{t=0}^{\infty}\beta^tU(C_t,O_t)}{C_t+w_tO_t+b_{t+1}\leq w_t+(1+r)b_t+\overline{W}_t\\ \lim_{T\to\infty}b_T/(1+r)^T\geq0}$
\end{formula}
\begin{itemize}
\item 3 CPO:
  \begin{enumerate}
  \item intratemporal: $U_O(t)/U_C(t)=w_t$ ($\mathrm{RMS}_{OC}=w_t$);
  \item \autor{Euler}: $U_C(t)/U_C(t+1)=\beta(1+r)$ \vertema{3.A.29};
  \item intertemporal del ocio:
\funcionclave{\frac{U_O(t)}{U_O(t+1)}=\frac{1+r}{1+\rho}\cdot\frac{w_t}{w_{t+1}}}
  \end{enumerate}
\item \concepto{Salario relativo intertemporal}: $L_t/L_{t+1}=f\big(\overset{+}{w_t/w_{t+1}},\overset{+}{r},\overset{-}{\rho}\big)$.
\end{itemize}
\subsub{c) Implicaciones}
{\centering\small\renewcommand{\arraystretch}{1.05}\setlength{\tabcolsep}{3pt}%
\begin{tabular}{@{}lcl@{}}
\toprule
\rowcolor{tceeazulpalido}\textbf{Cambio} & $L_t/L_{t+1}$ & \textbf{Por qué}\\
\midrule
$w_t,w_{t+1}$↑ permanente & $=$ & sin sustitución; nivel: efecto riqueza\\
$w_t$↑ transitorio & \textcolor{tceeverde}{↑} & \concepto{sustitución intertemporal}\\
$r$↑ & \textcolor{tceeverde}{↑} & trabajar y ahorrar hoy rinde más\\
$\rho$↑ (impaciencia) & \textcolor{tceerojo}{↓} & el ocio de hoy vale más\\
\bottomrule
\end{tabular}\par}
\begin{itemize}
\item \emph{Política}: rebajas \emph{temporales} de cotizaciones o impuestos mueven más las horas que las permanentes.
\end{itemize}

\sub{II.3. Extensiones\label{rep:II.3}}
\begin{itemize}
\item \concepto{Elasticidad de \autor{Frisch}}: con $\lambda$ (utilidad marginal de la riqueza) constante; cambios previstos o transitorios. Con $u=\ln C_t-\chi L_t^{1+\varphi}/(1+\varphi)$: $\varepsilon^F=1/\varphi$.
\item \emph{Ciclo vital}: $w$ en U invertida $\Rightarrow$ horas con la \autor{Frisch}; jubilación (\autor{Stock}-\autor{Wise}, 1990; \autor{Gruber}-\autor{Wise}, 1999).
\item \emph{Ciclos reales} (\autor{Kydland}-\autor{Prescott}, 1982): ↑productividad $\Rightarrow$ riqueza (↓$L$) vs sustitución intertemporal (↑$L$).
  \begin{itemize}
  \item Choque \emph{transitorio}: domina sustitución $\Rightarrow$ ciclo; estabilizar no mejora el bienestar.
  \end{itemize}
\item \emph{Trabajo indivisible} (\autor{Hansen}, 1985; \autor{Rogerson}, 1988): ajuste extensivo $\Rightarrow$ elasticidad agregada alta con individual baja.
\end{itemize}

\sub{II.4. Evidencia\label{rep:II.4}}
\begin{itemize}
\item \autor{Frisch} micro pequeña: \autor{MaCurdy} (1981) 0,15; \autor{Chetty} (2011) 0,54 horas, 0,28 participación; mayor en mujeres y con la edad.
\item Mensajeros de Zúrich (\autor{Fehr}-\autor{Goette}, 2007), taxistas (\autor{Farber}, 2015): la sustitución existe.
\item \emph{Problema de escala}: ciclos reales necesitan $\approx2{,}8$ agregada vs 0,8 micro; participación 0,28 vs 2,3.
\end{itemize}

\sub{II.5. Valoración\label{rep:II.5}}
\begin{itemize}
\item + propagación del ciclo, impuestos temporales; $-$ elasticidades altas, paro «elegido» \vertema{3.A.28}; trabajadores iguales $\Rightarrow$ \hyperref[rep:III]{III}.
\end{itemize}

\bloque[10]{III. Capital humano y función de ingresos\label{rep:III}}
\begin{ideaclave}
La educación es una inversión que se paga con salarios más altos toda la vida laboral; \autor{Mincer} mide su rendimiento ($\approx9$\,\% por año) y los experimentos naturales muestran que la mayor parte la causa lo aprendido, no solo quién estudia.
\end{ideaclave}

\sub{III.1. Usos del capital humano\label{rep:III.1}}
\begin{itemize}
\item \concepto{Capital humano}: conocimientos, habilidades y rasgos que hacen más productiva a una persona; \emph{inseparable} de ella (no sirve de garantía). Supuesto 3.
\item 5 enfoques (\autor{Acemoglu}-\autor{Autor}, 2011): \autor{Becker}; \autor{Gardner}; \autor{Nelson}-\autor{Phelps} \vertema{3.A.44}; \autor{Bowles}-\autor{Gintis}; \autor{Spence} (1973, \emph{señal}) \vertema{3.A.13}.
\item \autor{Becker}: educación $\to$ productividad $\to$ salario; \autor{Spence}: productividad $\to$ educación $\to$ salario. Las distingue la evidencia.
\end{itemize}

\sub{III.2. Teoría del capital humano de \autor{Becker} (1964)\label{rep:III.2}}
\subsub{a) Idea}
\begin{itemize}
\item Inversión: costes directos + \emph{coste de oportunidad} + esfuerzo; rinde salario más alto. «\emph{El saber no ocupa lugar}»… pero sí tiempo.
\end{itemize}
\subsub{b) Modelo: cuánto estudiar}
\begin{itemize}
\item $S$ años sin cobrar, coste $c$; $w'>0$, $w''<0$; mercado de capitales perfecto (\concepto{separación de \autor{Fisher}}).
  \begin{itemize}
  \item $\max V(S)=\int_S^Tw(S)e^{-rt}dt-\int_0^Sce^{-rt}dt$; con $T\to\infty$, \concepto{tasa marginal de rendimiento} $=r$:
\begin{formula}
\centering $\rho(S^*)\equiv\dfrac{w'(S^*)}{w(S^*)+c}=r$
\end{formula}
  \end{itemize}
\item \autor{Becker} (1967): demanda $\rho(S)$ decreciente; oferta de fondos creciente y escalonada.
\end{itemize}
\pareja{becker-inversion}{Demanda y oferta de fondos para estudiar}{fisher-capital-humano}{\autor{Fisher}: la formación no depende de la paciencia}
\begin{itemize}
\item \emph{Quién}: los más hábiles y los de más recursos; liquidez $\Rightarrow$ se rompe la separación. \emph{Cuándo}: joven. Con $c=0$: $w'/w=r$ (\autor{Mincer}).
\end{itemize}
\graficorepaso[0.62\linewidth]{perfiles-edad}{Perfiles edad-ingresos (\autor{McConnell}-\autor{Brue}): estudiar si VAN $>0$}
\subsub{c) Formación general y específica}
\begin{itemize}
\item \concepto{General} (vale en toda empresa) y \concepto{específica} (solo en la que la da); coste $c$, $\mathrm{VPMg}^0<\mathrm{VPMg}^1$.
\item \emph{General}: \emph{la paga el trabajador}, $w_1=\mathrm{VPMg}^0-c$ (aprendices, MIR).
\item \emph{Específica}: ninguno la paga solo (el otro se quedaría la inversión) $\Rightarrow$ se reparte, $\gamma\in(0,1)$ \vertema{3.A.27}:
\begin{formula}
\centering $w_2=\mathrm{VPMg}^0+\gamma\big(\mathrm{VPMg}^1-\mathrm{VPMg}^0\big)$
\end{formula}
  \begin{itemize}
  \item Menos rotación; salario con antigüedad; «último en entrar, primero en salir»; \emph{insiders}.
  \end{itemize}
\item \autor{Acemoglu}-\autor{Pischke} (1998): mercados imperfectos $\Rightarrow$ la empresa paga también la general (Alemania).
\end{itemize}
\subsub{d) Valoración}
\begin{itemize}
\item + edad, antigüedad, quién paga; $-$ otros determinantes (discriminación), información perfecta, liquidez.
\end{itemize}

\sub{III.3. Función de ingresos: \autor{Mincer}\label{rep:III.3}}
\subsub{a) Idea y b) modelo}
\begin{itemize}
\item \autor{Mincer} (1974): la ecuación más estimada de la economía laboral.
\item \emph{Escolarización} (1958): solo coste de oportunidad; mismo VP para todo $S$ $\Rightarrow$ $\ln w(S)=\ln w(0)+rS$.
\item \emph{Experiencia}: \autor{Ben-Porath} (1967), $\dot h=a(1-u)h$; la fracción para formarse cae linealmente $\Rightarrow$ $\ln w$ cuadrático y cóncavo en $X$:
\funcionclave{\ln w_i=\beta_0+\beta_1S_i+\beta_2X_i+\beta_3X_i^2+\varepsilon_i}
  \begin{itemize}
  \item $\beta_1\approx r$: \emph{rendimiento privado} (semielasticidad); $\beta_2>0$, $\beta_3<0$; máximo en $X=-\beta_2/(2\beta_3)$.
  \end{itemize}
\end{itemize}
\subsub{c) Extensiones: brechas salariales}
\begin{itemize}
\item \concepto{\autor{Oaxaca}-\autor{Blinder}} (1973): brecha entre grupos A y B:
\begin{formula}
\centering $\overline{\ln w}_{\mathrm A}-\overline{\ln w}_{\mathrm B}=\underbrace{(\bar{\vect{x}}_{\mathrm A}-\bar{\vect{x}}_{\mathrm B})\cdot\hat{\vect{\beta}}_{\mathrm A}}_{\text{explicada}}+\underbrace{\bar{\vect{x}}_{\mathrm B}\cdot(\hat{\vect{\beta}}_{\mathrm A}-\hat{\vect{\beta}}_{\mathrm B})}_{\text{no explicada}}$
\end{formula}
  \begin{itemize}
  \item No explicada: discriminación y lo no observado. España, EES 2010: 20 puntos, 12 explicados (\autor{Murillo} et al., 2017).
  \end{itemize}
\item \autor{Goldin} (2014): brecha dentro de cada ocupación (jornadas largas y rígidas). \concepto{Penalización por hijos} (\autor{Kleven}): $\approx-20$\,\%.
\end{itemize}
\subsub{d) Evidencia empírica}
\begin{itemize}
\item \emph{MCO}: $\approx9$\,\% por año (\autor{Psacharopoulos}-\autor{Patrinos}, 2018); superiores $\approx15$\,\%.
\item \emph{¿Causal?} \concepto{Sesgo de habilidad} \emph{sobreestima}; error de medida \emph{subestima} (\autor{Griliches}, 1977).
\item \emph{Experimentos naturales} (Nobel 2021: \autor{Card}; \autor{Angrist}-\autor{Imbens}), variable instrumental:
  \begin{itemize}
  \item trimestre de nacimiento, 7-10\,\% (\autor{Angrist}-\autor{Krueger}, 1991);
  \item distancia a la universidad, 25-60\,\% más que MCO (\autor{Card}, 1995);
  \item gemelos, 12-16\,\% (\autor{Ashenfelter}-\autor{Krueger}, 1994);
  \item \concepto{LATE} (\autor{Imbens}-\autor{Angrist}, 1994): efecto en quienes cambian por el instrumento.
  \end{itemize}
\item VI $\geq$ MCO (\autor{Card}, 1999): mueven a quienes tenían más difícil estudiar $\Rightarrow$ la educación causa buena parte de la prima: \autor{Spence} no lo explica todo.
\item \emph{España}: perfiles cóncavos, más pendientes cuanto más estudio (EES 2022).
\end{itemize}
\subsub{e) Valoración}
\begin{itemize}
\item + describe bien los salarios; $-$ tasa de rendimiento solo con supuestos fuertes; pesan los títulos (\autor{Heckman} et al., 2006); el rendimiento \emph{social} puede ser mayor \vertema{3.A.43}.
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Hay paro involuntario?} Competencia y monopsonio: no; salario mínimo o sindicato: sí.
\item \textbf{¿Por qué la oferta se vuelve hacia atrás?} Efecto renta global $\propto L$: con $w$ alto domina.
\item \textbf{¿\autor{Frisch} frente a hicksiana?} $\lambda$ constante (transitorios) vs $U$ constante.
\item \textbf{¿Quién paga la formación?} General: el trabajador; específica: se reparte.
\item \textbf{¿Por qué VI $\geq$ MCO?} LATE de quienes tenían más difícil estudiar, con más rendimiento.
\item \textbf{¿Por qué lo transitorio mueve más?} Apenas cambia la riqueza de toda la vida: domina la sustitución intertemporal.
\item \textbf{¿Es $\beta_1$ una tasa de rendimiento?} Solo sin costes directos y con igual vida laboral.
\end{enumerate}
\begin{trampa}
\begin{itemize}
\item Renta dotación ↓$L$; prima por horas extra: ↑$L$ sin ambigüedad.

\item Monopsonio: salario mínimo en $(w^M,w^c]$ \emph{sube} el empleo.
\item Sesgo de habilidad \emph{sobre}estima $\beta_1$; error de medida \emph{sub}estima.
\item $\varepsilon^F\geq\varepsilon^h\geq\varepsilon$ (ocio normal); \autor{Frisch} $=1/\varphi$.
\item \autor{Lucas}-\autor{Rapping}: $r$↑ $\Rightarrow$ ↑$L_t/L_{t+1}$; $\rho$↑ $\Rightarrow$ ↓.
\end{itemize}
\end{trampa}
