% Variante para la ficha de repaso (letra mayor, se reduce a ancho de columna) de superficie-utilidad.tex.
% Autor: Víctor Gutiérrez Marcos
% Superficie de utilidad U(x1,x2) = (x1·x2)^(1/2) cortada a los niveles U = 1, 2 y 3,
% y proyección de los cortes sobre el plano (x1,x2): el mapa de curvas de indiferencia.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\pgfplotsset{
  superficie/.style={surf,shader=faceted,fill opacity=0.55,draw=tceemorado!40,line width=0.15pt,
    colormap={tcee}{rgb255=(226,239,217) rgb255=(167,137,189)}},
  corte/.style={curvanueva,line width=2.4pt},
  proyeccion/.style={curva1,line width=2.4pt},
}
\begin{document}
\begin{tikzpicture}[tcee,font=\LARGE,etiqueta/.style={font=\LARGE,inner sep=2pt},eje/.append style={line width=1.4pt}]
  % ---------- panel izquierdo: la superficie en tres dimensiones ----------
  % perspectiva del gráfico de cuasiconcavidad (vista por defecto de pgfplots, acimut 30°,
  % elevación 28°): origen a la izquierda, x1 a la derecha algo hacia abajo,
  % x2 en profundidad hacia arriba a la derecha, U vertical
  \begin{axis}[name=tresd,x={(1.02cm,-0.2805cm)},y={(0.5865cm,0.476cm)},z={(0cm,1.037cm)},axis lines=none,
      xmin=0,xmax=4.4,ymin=0,ymax=4.4,zmin=0,zmax=4.4,clip=false]
    % 1: ejes
    \capa{1}{\draw[eje] (axis cs:0,0,0) -- (axis cs:4.8,0,0) node[below left=-1pt] {$x_1$};}
    \capa{1}{\draw[eje] (axis cs:0,0,0) -- (axis cs:0,6.2,0) coordinate (finx2);}
    \capa{1}{\draw[eje] (axis cs:0,0,0) -- (axis cs:0,0,4.6) node[above] {$U$};}
    \capa{1}{\draw[guia] (axis cs:4,0,0) -- (axis cs:4,4,0) -- (axis cs:0,4,0);} % borde del plano de las cestas
    % 2: la superficie de utilidad
    \capa{2}{\addplot3[superficie,z buffer=sort,domain=0:4,y domain=0:4,samples=9,samples y=9] {sqrt(x*y)};}
    \capa{2}{\node[etiqueta,anchor=west] at (axis cs:4.05,4,4) {$U(x_1,x_2)$};}
    % el rótulo de x2 (paso 1) va después de la superficie para que no quede tapado
    \capa{1}{\node[right] at (finx2) {$x_2$};}
    % 3-5: cortes a los niveles 1, 2 y 3 y su proyección en el plano
    \capa{3}{\addplot3[corte,domain=0.25:4,samples=40,samples y=0] (x,{1/x},1);}
    \capa{3}{\addplot3[proyeccion,domain=0.25:4,samples=40,samples y=0] (x,{1/x},0);}
    \capa{3}{\draw[guia] (axis cs:1,1,1) -- (axis cs:1,1,0);}
    \capa{3}{\node[etiqueta,text=tceerojo,anchor=south east] at (axis cs:0.25,4,1) {$U=1$};}
    \capa{4}{\addplot3[corte,domain=1:4,samples=40,samples y=0] (x,{4/x},2);}
    \capa{4}{\addplot3[proyeccion,domain=1:4,samples=40,samples y=0] (x,{4/x},0);}
    \capa{4}{\draw[guia] (axis cs:2,2,2) -- (axis cs:2,2,0);}
    \capa{4}{\node[etiqueta,text=tceerojo,anchor=south west] at (axis cs:4.05,1,2) {$U=2$};}
    \capa{5}{\addplot3[corte,domain=2.25:4,samples=40,samples y=0] (x,{9/x},3);}
    \capa{5}{\addplot3[proyeccion,domain=2.25:4,samples=40,samples y=0] (x,{9/x},0);}
    \capa{5}{\draw[guia] (axis cs:3,3,3) -- (axis cs:3,3,0);}
    \capa{5}{\node[etiqueta,text=tceerojo,anchor=west] at (axis cs:4.05,2.25,3) {$U=3$};}
  \end{axis}
\end{tikzpicture}
\end{document}
