\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: \emph{¿puede una teoría explicar lo que elegimos sin ver lo que queremos?}
  \begin{itemize}
  \item \autor{Marshall} (1890): «el estudio de la humanidad en los asuntos ordinarios de la vida»; pocas decisiones más ordinarias que qué comprar.
  \end{itemize}
\item \textbf{Relevancia}:
  \begin{itemize}
  \item \emph{Micro}: paso previo al equilibrio de mercado \vertema{3.A.16}.
  \item \emph{Macro}: consumo de los hogares, \dato{54,1\,\% del PIB}{España, 2025}.
  \end{itemize}
\item \textbf{Contextualización}:
  \begin{itemize}
  \item Utilidad (\autor{Bernoulli} 1738, \autor{Gossen} 1854) $\to$ marginalistas (1871-74) $\to$ \autor{Marshall}, cardinal.
  \item \autor{Edgeworth} (1881), curvas de indiferencia; \autor{Pareto} (1906), ordinal $\to$ \autor{Slutsky} (1915), \autor{Hicks}-\autor{Allen} (1934), \autor{Debreu} (1959).
  \item Hacia lo observable: \autor{Samuelson}, \autor{Becker}, \autor{Rosen}.
  \item \concepto{Individualismo metodológico}; \concepto{consumidor}: demanda productos, ofrece factores, optimiza con un presupuesto.
  \end{itemize}
\item \textbf{Problemática}: ¿cómo se obtiene la demanda y cuándo hay demanda de mercado? ¿Cómo varía con renta y precios? ¿Puede construirse desde lo observable?
\item \textbf{Estructura}: \hyperref[rep:I]{I}, fundamentos; \hyperref[rep:II]{II}, estática comparativa (\autor{Slutsky}); \hyperref[rep:III]{III}, 3~desarrollos hacia lo observable.
\end{itemize}

\bloque[11]{I. Fundamentos de la elección racional\label{rep:I}}
\begin{ideaclave}
Con pocos axiomas y un presupuesto lineal sale una demanda con propiedades contrastables; agregarla en una de mercado exige condiciones muy restrictivas.
\end{ideaclave}
\graficorepaso[0.5\linewidth]{esquema-ingredientes-repaso}{De las preferencias a la demanda de mercado}
\begin{itemize}
\item \emph{Supuestos}: estático y cierto; información perfecta; precio-aceptante; bienes divisibles y homogéneos; sin oferta de trabajo \vertema{3.A.25} ni decisiones intertemporales \vertema{3.A.29}.
\end{itemize}

\sub{I.1. Ingredientes\label{rep:I.1}}
\subsub{a) Conjunto de elección y preferencias}
\begin{itemize}
\item \concepto{Conjunto de elección} $S\subseteq\mathbb{R}^n_+$: no vacío, cerrado, acotado inferiormente (no superiormente: la escasez la pone el presupuesto), convexo.
\item \concepto{Preferencia débil} $\succcurlyeq$ (primitiva) $\to$ \concepto{estricta} $\succ$ e \concepto{indiferencia} $\sim$.
\item \concepto{Conjuntos de contorno}: «al menos tan bueno como» $B(\vect{x}^0)$; «no mejor que» $H(\vect{x}^0)$; de indiferencia $I=B\cap H$.
\item \concepto{Axiomas} (\autor{Debreu}, 1959): 1-3 = \concepto{preorden completo} (\emph{racionalidad}): $\sim$ particiona $S$ en conjuntos que no se cortan.
\end{itemize}
\pareja{esquema-axiomas-repaso}{7 axiomas: qué garantiza cada uno}{conjuntos-contorno}{$B(\vect{x}^0)$, $H(\vect{x}^0)$ y su frontera}
\begin{itemize}
\item \concepto{Teorema de \autor{Debreu}}: 1-4 $\Rightarrow$ existe $U$ continua; no hacen falta monotonía ni convexidad.
  \begin{itemize}
  \item \concepto{Preferencias lexicográficas}: racionales, no continuas $\Rightarrow$ sin $U$.
  \end{itemize}
\item \concepto{Grados de deseabilidad}: monotonía estricta $\Rightarrow$ monotonía $\Rightarrow$ insaciabilidad local (se agota la renta) $\Rightarrow$ no saturación global.
\end{itemize}
\subsub{b) Función de utilidad}
\begin{itemize}
\item \concepto{Función de utilidad}: «numeración» de los conjuntos de indiferencia:
\begin{formula}
\centering $\vect{x}^1\succcurlyeq\vect{x}^2\ \Leftrightarrow\ U(\vect{x}^1)\geq U(\vect{x}^2)$
\end{formula}
  \begin{itemize}
  \item \concepto{Ordinal}: cualquier transformación creciente; \concepto{cardinal}: solo afines positivas \vertema{3.A.10}.
  \item Propiedades: continua (4); creciente (5); cuasicóncava (6; estrictamente, 6$'$); no única.
  \end{itemize}
\item \concepto{Curva de indiferencia}: conjunto de nivel de $U$.
  \begin{itemize}
  \item No se cortan (3); continuas (4); más lejos, mejor, y pendiente negativa (5); convexas (6); sin vértices (7).
  \end{itemize}
\end{itemize}
\graficorepaso[0.55\linewidth]{superficie-utilidad-repaso}{Superficie $\sqrt{x_1x_2}$ cortada a $U=1,2,3$: su proyección, el mapa de indiferencia}
\begin{itemize}
\item \concepto{Relación marginal de sustitución} y \concepto{elasticidad de sustitución}:
\begin{formula}
\centering $\mathrm{RMS}_{12}\equiv-\left.\dfrac{dx_2}{dx_1}\right|_{\overline{U}}=\dfrac{U_1}{U_2}$,\qquad $\sigma\equiv\dfrac{d\ln(x_2/x_1)}{d\ln\mathrm{RMS}_{12}}$
\end{formula}
  \begin{itemize}
  \item RMS: bien 2~que se cede por una unidad más del 1; decreciente. $\sigma$: número puro, curvatura.
  \end{itemize}
\item \concepto{CES}: $\sigma\to0$ \autor{Leontief}; $\sigma\to1$ \autor{Cobb}-\autor{Douglas} (gasto fijo, $\varepsilon_{x,p}=-1$); $\sigma\to\infty$ lineal.
\end{itemize}
\graficorepaso[0.45\linewidth]{ces-curvas-indiferencia}{Curvas CES por $(1,1)$ según $\sigma$}
\subsub{c) Restricción presupuestaria}
\begin{itemize}
\item \concepto{Conjunto presupuestario walrasiano} $B(\vect{p},\overline{W})=\{\vect{x}\in S:\vect{p}\cdot\vect{x}\leq\overline{W}\}$: no vacío, compacto si $\vect{p}\gg\vect{0}$, convexo.
\item \concepto{Recta de balance}: pendiente $-p_1/p_2$.
\end{itemize}

\sub{I.2. Maximización de la utilidad: el equilibrio\label{rep:I.2}}
\subsub{a) Planteamiento y condiciones de \autor{Kuhn}-\autor{Tucker}}
\begin{itemize}
\item \concepto{Problema de maximización de la utilidad} (PMU, primal):
\funcionclave{\optim{\max}{x_1,\dots,x_n}{U(x_1,\dots,x_n)}{\textstyle\sum_{i}p_ix_i\leq\overline{W}\\ x_i\geq0,\quad i=1,\dots,n}}
\item \emph{Teoremas}: existencia (\autor{Weierstrass}); unicidad (estr.\ cuasiconcavidad); CPO suficientes (\autor{Arrow}-\autor{Enthoven}, 1961).
\item Lagrangiano y condiciones:
\begin{formula}
\centering $\mathcal{L}=U(\vect{x})+\lambda\big(\overline{W}-\vect{p}\cdot\vect{x}\big)$\\[2pt]
$U_i-\lambda p_i\leq0,\ \ x_i\geq0,\ \ x_i(U_i-\lambda p_i)=0$\\
$\overline{W}-\vect{p}\cdot\vect{x}\geq0,\ \ \lambda\geq0,\ \ \lambda(\overline{W}-\vect{p}\cdot\vect{x})=0$
\end{formula}
  \begin{itemize}
  \item $\lambda$: \concepto{utilidad marginal de la renta}; insaciabilidad local $\Rightarrow$ $\lambda>0$; saturación interior $\Rightarrow$ $\lambda=0$.
  \end{itemize}
\end{itemize}
\subsub{b) Solución interior y de esquina}
\begin{itemize}
\item Interior y esquina:
\begin{formula}
\centering \concepto{Interior} (2.ª ley de \autor{Gossen}): $\dfrac{U_1}{p_1}=\dots=\dfrac{U_n}{p_n}=\lambda\ \Rightarrow\ \mathrm{RMS}_{12}=\dfrac{p_1}{p_2}$\\[2pt]
\concepto{Esquina} ($x_2^*=0$): $\dfrac{U_1}{p_1}=\lambda\geq\dfrac{U_2}{p_2}\ \Rightarrow\ \mathrm{RMS}_{12}\geq\dfrac{p_1}{p_2}$
\end{formula}
\end{itemize}
\graficorepaso[0.6\linewidth]{solucion-interior-esquina-repaso}{Solución interior (tangencia) y de esquina ($x_2^*=0$)}
\begin{itemize}
\item \emph{Sin convexidad}: la tangencia no basta; esquinas, soluciones múltiples, demanda que salta. Cóncavas $\Rightarrow$ siempre esquina.
\end{itemize}

\sub{I.3. Solución: demanda, FIU y mercado\label{rep:I.3}}
\subsub{a) Demanda marshalliana}
\begin{itemize}
\item \concepto{Sistema completo de ecuaciones de demanda marshalliana}:
\funcionclave{\vect{x}^*=\vect{x}(\vect{p},\overline{W})=\big(x_1(\vect{p},\overline{W}),\dots,x_n(\vect{p},\overline{W})\big)}
\item \emph{Propiedades}: única; continua (\autor{Berge}); \concepto{homogénea de grado 0} (sin \emph{ilusión monetaria}); \concepto{ley de \autor{Walras}} $\vect{p}\cdot\vect{x}=\overline{W}$; diferenciable.
\end{itemize}
\subsub{b) Función indirecta de utilidad}
\begin{itemize}
\item $V(\vect{p},\overline{W})=U(\vect{x}(\vect{p},\overline{W}))$: homogénea de grado 0; creciente en $\overline{W}$, no creciente en $\vect{p}$; \emph{cuasiconvexa}.
\item \concepto{Identidad de \autor{Roy}} (1947), por la envolvente:
\begin{formula}
\centering $x_i(\vect{p},\overline{W})=-\dfrac{\partial V/\partial p_i}{\partial V/\partial\overline{W}}$
\end{formula}
\end{itemize}
\subsub{c) Demanda de mercado: agregación}
\begin{itemize}
\item $X_i=\sum_hx_{hi}(\vect{p},\overline{W}_h)$, \concepto{suma horizontal}.
\item \emph{Primer problema}: depende de \emph{toda} la distribución de la renta.
  \begin{itemize}
  \item Solo de $\sum_h\overline{W}_h$ $\Leftrightarrow$ \concepto{forma polar de \autor{Gorman}}, $V_h=a_h(\vect{p})+b(\vect{p})\overline{W}_h$ $\Leftrightarrow$ curvas de \autor{Engel} \emph{rectas y paralelas} $\Rightarrow$ \concepto{consumidor representativo}.
  \item Casos: \emph{homotéticas idénticas}; \emph{cuasilineales}.
  \end{itemize}
\end{itemize}
\graficorepaso[0.42\linewidth]{engel-gorman}{\autor{Gorman}: misma pendiente $\Rightarrow$ redistribuir no cambia la demanda agregada}
\begin{itemize}
\item \emph{Efecto renta nulo} (cuasilineales): marshalliana = hicksiana, excedente exacto \vertema{3.A.16}; pero sin inferiores ni \autor{Giffen}. \autor{Marshall}: UMg del dinero constante.
\item \emph{Segundo problema}: no se heredan el axioma débil ni la simetría de \autor{Slutsky}: \concepto{teorema SMD} (1972-74): ni unicidad ni estabilidad \vertema{3.A.21}.
\end{itemize}

\bloque[8]{II. Estática comparativa\label{rep:II}}
\begin{ideaclave}
Un cambio de precio tiene un ES, siempre no positivo, y un ER, cuyo signo depende de que el bien sea normal o inferior: la ley de la demanda vale siempre para la compensada y para la marshalliana solo si el ER no domina (\autor{Giffen}). Netas, simétricas; brutas, no.
\end{ideaclave}
\begin{itemize}
\item \concepto{Estática comparativa}: comparar óptimos; \concepto{elasticidades}: números puros y medida \emph{local}.
\end{itemize}

\sub{II.1. Variaciones en la renta\label{rep:II.1}}
\subsub{a) Curva de renta-consumo y de \autor{Engel}}
\begin{itemize}
\item \concepto{Curva de renta-consumo} (homotéticas, rayo) $\to$ \concepto{curva de \autor{Engel}} $x_i(\bar{\vect{p}},\overline{W})$; pendiente = \concepto{efecto renta}.
\end{itemize}
\graficorepaso[0.7\linewidth]{renta-consumo-engel}{CRC y curva de \autor{Engel}: normal hasta cierta renta, inferior después}
\subsub{b) Elasticidad-renta y ley de \autor{Engel}}
\begin{itemize}
\item $\varepsilon_{x_i,\overline{W}}$: $<0$ \emph{inferior}; $=0$ independiente; $(0,1)$ \emph{normal necesario}; $>1$ \emph{normal de lujo}.
\item \concepto{Ley de \autor{Engel}} (1857): el peso de los alimentos cae con la renta. España: \dato{19,6\,\% (Q1) frente a 12,3\,\% (Q5)}{INE, EPF 2025}.
\end{itemize}
\subsub{c) Agregación de \autor{Engel}}
\begin{itemize}
\item De la ley de \autor{Walras}:
\begin{formula}
\centering $\displaystyle\sum_i s_i\,\varepsilon_{x_i,\overline{W}}=1$,\quad $s_i=\dfrac{p_ix_i}{\overline{W}}$
\end{formula}
  \begin{itemize}
  \item No todos inferiores; alguno con $\varepsilon\geq1$.
  \end{itemize}
\end{itemize}

\sub{II.2. Variaciones en los precios\label{rep:II.2}}
\subsub{a) Curva de precio-consumo y demanda}
\begin{itemize}
\item La recta gira $\to$ \concepto{curva de precio-consumo} $\to$ \concepto{curva de demanda}.
\item \emph{Elasticidad-precio propio}: $<0$ \emph{ordinario} ($<-1$ elástica; $(-1,0)$ inelástica); $=0$ rígida; $>0$ \emph{\autor{Giffen}}.
\end{itemize}
\subsub{b) Ecuación de \autor{Slutsky}}
\begin{itemize}
\item \concepto{Demanda compensada} (hicksiana) $h_i(\vect{p},\overline{U})$: minimización del gasto, dual del PMU \vertema{3.A.9}.
\item \concepto{Ecuación de \autor{Slutsky}} (1915), con $\overline{U}=V(\vect{p},\overline{W})$:
\begin{formula}
\centering $\underbrace{\dfrac{\partial x_i(\vect{p},\overline{W})}{\partial p_j}}_{\mathrm{ET}}=\underbrace{\dfrac{\partial h_i(\vect{p},\overline{U})}{\partial p_j}}_{\mathrm{ES}}\underbrace{-\,x_j(\vect{p},\overline{W})\dfrac{\partial x_i(\vect{p},\overline{W})}{\partial\overline{W}}}_{\mathrm{ER}}$\\[1pt]
Elasticidades: $\varepsilon_{x_i,p_j}=\varepsilon^h_{i,p_j}-s_j\,\varepsilon_{x_i,\overline{W}}$
\end{formula}
\item \concepto{Efecto sustitución}: solo el precio relativo; propio \emph{siempre} $\leq0$ (concavidad del gasto).
\item \concepto{Efecto renta}: propio $<0$ si normal, $>0$ si inferior; crece con $s_i$.
\item \concepto{Clasificación}:
  \begin{itemize}
  \item normal $\Rightarrow$ ordinario (\emph{ley de la demanda});
  \item inferior: ordinario si domina el ES, \autor{Giffen} si domina el ER (y pesa mucho): arroz en Hunan (\autor{Jensen}-\autor{Miller}, 2008).
  \end{itemize}
\end{itemize}
\pareja{slutsky-propio}{Signos de \autor{Slutsky} con $j=i$}{elasticidades-renta-precio-repaso}{Todo normal es ordinario; todo \autor{Giffen} es inferior}
\begin{itemize}
\item \emph{Gráficamente} ($p_1\downarrow$): ES $A\to B$; ER $B\to C$. Marshalliana por $A$ y $C$; hicksiana por $A$ y $B$, más inclinada si es normal.
\end{itemize}
\graficorepaso[0.4\linewidth]{efecto-sustitucion-renta}{ES ($A\to B$), ER ($B\to C$); marshalliana y hicksiana}
\begin{itemize}
\item \emph{Fuera del modelo}: \concepto{efecto \autor{Veblen}} (1899): el precio entra en las preferencias.
\end{itemize}
\subsub{c) Relaciones cruzadas}
\begin{itemize}
\item \emph{Brutas}: $\varepsilon_{x_i,p_j}>0$ \concepto{sustitutivo bruto}; $<0$ \concepto{complementario bruto}: \emph{no simétricas}.
\item \emph{Netas}, por el ES cruzado: \concepto{sustitutivos} o \concepto{complementarios netos}: \emph{simétricas} ($\partial h_i/\partial p_j=\partial h_j/\partial p_i$).
  \begin{itemize}
  \item Homogeneidad de $h$ $\Rightarrow$ todo bien tiene un sustitutivo neto; con 2~bienes, ambos lo son.
  \end{itemize}
\end{itemize}
\subsub{d) Agregación de \autor{Cournot} y restricciones}
\begin{itemize}
\item Restricciones sobre las elasticidades:
\begin{formula}
\centering \concepto{\autor{Cournot}}: $\displaystyle\sum_is_i\,\varepsilon_{x_i,p_j}=-s_j$;\quad \autor{Euler}: $\displaystyle\sum_j\varepsilon_{x_i,p_j}+\varepsilon_{x_i,\overline{W}}=0$
\end{formula}
\item Agregaciones, de la ley de \autor{Walras}; homogeneidad, del presupuesto; simetría y negatividad, de \emph{maximizar} \vertema{3.A.9}.
\end{itemize}

\sub{II.3. Valoración\label{rep:II.3}}
\begin{itemize}
\item \emph{Logros}: \concepto{restricciones contrastables}: homogeneidad, \autor{Walras}, agregaciones, signo del ES.
\item \emph{Limitaciones} $\to$ \hyperref[rep:III]{III}:
  \begin{enumerate}
  \item preferencias inobservables $\to$ preferencia revelada;
  \item sin tiempo $\to$ producción doméstica;
  \item bienes homogéneos $\to$ precios hedónicos;
  \item estático y cierto \vertema{3.A.10} \vertema{3.A.29}.
  \end{enumerate}
\end{itemize}

\bloque[7,5]{III. Otros desarrollos de la demanda\label{rep:III}}
\begin{ideaclave}
Los 3~desarrollos acercan la teoría a lo observable: la preferencia revelada reconstruye las preferencias desde las elecciones y las contrasta (\autor{Afriat}); la producción doméstica añade el coste del tiempo; los precios hedónicos miden cuánto se paga por cada característica.
\end{ideaclave}

\sub{III.1. La teoría de la preferencia revelada\label{rep:III.1}}
\begin{itemize}
\item \emph{Idea}: \autor{Samuelson} (1938, 1948; Nobel 1970): partir de las \emph{elecciones}. \emph{Inductivo}.
\item \emph{Supuestos}: consumidor \emph{individual}; una cesta por cada $(\vect{p},\overline{W})$; ley de \autor{Walras}.
\item \emph{Desarrollo}: $\vect{x}^0$ \concepto{directamente preferida} a $\vect{x}^1$ si se eligió pudiendo comprar $\vect{x}^1$.
  \begin{itemize}
  \item \concepto{Axioma débil} (ADPR):
  \end{itemize}
\begin{formula}
\centering $\vect{p}^0\cdot\vect{x}^1\leq\vect{p}^0\cdot\vect{x}^0,\ \vect{x}^1\neq\vect{x}^0\ \Rightarrow\ \vect{p}^1\cdot\vect{x}^0>\vect{p}^1\cdot\vect{x}^1$
\end{formula}
  \begin{itemize}
  \item \concepto{Axioma fuerte} (AFPR, \autor{Houthakker}, 1950): transitividad de la revelada; AFPR $\Rightarrow$ ADPR.
  \end{itemize}
\end{itemize}
\graficorepaso[0.7\linewidth]{adpr-efecto-sustitucion}{ADPR (izq.) y signo del efecto sustitución: si baja $p_1$ y se compensa la renta, $x_1$ no baja}
\begin{itemize}
\item \emph{Implicaciones} del ADPR, sin utilidad:
  \begin{enumerate}
  \item homogeneidad de grado 0;
  \item \concepto{ley de la demanda compensada}: $(\vect{p}^1-\vect{p}^0)\cdot(\vect{x}^1-\vect{x}^0)\leq0$;
  \item matriz de \autor{Slutsky} semidefinida negativa, \emph{no} simétrica (con 2~bienes, basta: \autor{Rose}, 1958).
  \end{enumerate}
\item \concepto{Teorema de \autor{Houthakker}}: AFPR $\Leftrightarrow$ preferencias racionales: revelada y ordinal, equivalentes.
\item \emph{Extensiones}: \concepto{teorema de \autor{Afriat}} (1967): datos \emph{finitos} compatibles con $U$ $\Leftrightarrow$ GARP; índices de \autor{Laspeyres} y \autor{Paasche} \vertema{3.A.9}.
\item \emph{Evidencia}: \autor{Varian} (1982): contraste \emph{no paramétrico} del GARP.
\item \emph{Valoración}: demanda sin utilidad y contrastable; identifica elección y bienestar (\autor{Sen}, 1973).
\end{itemize}

\sub{III.2. La producción doméstica\label{rep:III.2}}
\begin{itemize}
\item \emph{Idea}: \autor{Becker} (1965; Nobel 1992): el hogar, una fábrica de \emph{commodities} con bienes y \emph{tiempo}. \autor{Lancaster} (1966): características $\vect{z}=B\vect{x}$ \vertema{3.A.18}.
\item \emph{Supuestos}: $U(Z_1,\dots,Z_m)$, $Z_i=f_i(x_i,t_i)$; tiempo $T$ entre trabajo y consumo:
\begin{formula}
\centering $\optim{\max}{\{x_i,t_i\},\,t_w}{U(Z_1,\dots,Z_m)}{Z_i=f_i(x_i,t_i)\\ \textstyle\sum_ip_ix_i=w\,t_w+\overline{W}\\ \textstyle\sum_it_i+t_w=T}$
\end{formula}
\item \emph{Desarrollo}: \concepto{renta completa} $\sum_i(p_ix_i+w\,t_i)=wT+\overline{W}$; \concepto{precio sombra} $\pi_i=p_ib_i+w\,\ell_i$.
\item \emph{Implicaciones}: $w\uparrow$ $\Rightarrow$ se sustituye tiempo por bienes (comida preparada, guarderías): mujer en el mercado de trabajo \vertema{3.A.25}.
\item \emph{Extensiones y valoración}: \autor{Gronau} (1977): trabajo doméstico $\neq$ ocio; educación, matrimonio, fecundidad; el PIB omite la producción doméstica.
\end{itemize}

\sub{III.3. Precios hedónicos\label{rep:III.3}}
\begin{itemize}
\item \emph{Idea}: bienes \emph{diferenciados} = «paquetes» de características; el \concepto{método de los precios hedónicos} descompone el precio. \autor{Court} (1939), \autor{Griliches} (1961), \autor{Rosen} (1974).
\item \emph{Supuestos}: mercado competitivo; informados; $U(\vect{z},y)$; una unidad.
\item \emph{Desarrollo}: \concepto{función hedónica} $p(\vect{z})$; \concepto{precio implícito} $\partial p/\partial z_j$:
\begin{formula}
\centering $\optim{\max}{\vect{z},\,y}{U(\vect{z},y)}{p(\vect{z})+y=\overline{W}}\ \Rightarrow\ \dfrac{\partial U/\partial z_j}{\partial U/\partial y}=\dfrac{\partial p(\vect{z})}{\partial z_j}$
\end{formula}
  \begin{itemize}
  \item \concepto{Función de puja} $\Theta$ y \concepto{de oferta} $\phi$; \concepto{equilibrio}: $p(\vect{z})$, \emph{envolvente} de ambas.
  \end{itemize}
\end{itemize}
\graficorepaso[0.5\linewidth]{funcion-hedonica-rosen}{$p(\vect{z})$, envolvente de pujas y ofertas (\autor{Rosen}, 1974)}
\begin{itemize}
\item \emph{Implicaciones}: precio implícito = valoración marginal = coste marginal; $p(\vect{z})$, lugar de equilibrios.
\item \emph{Extensiones}: bienes sin mercado (ruido, vivir frente al Retiro) \vertema{4.B.6}; salarios hedónicos.
\item \emph{Evidencia}: 2~etapas, $\ln p_v=\beta_0+\sum_j\beta_jz_{vj}+u_v$; IPC: \autor{Boskin} (1996), sesgo \dato{1,1 p./año}{EE.\,UU.}; IPV del INE.
\item \emph{Valoración}: forma funcional; 2.ª etapa mal identificada; equilibrio dudoso en vivienda.
\end{itemize}
