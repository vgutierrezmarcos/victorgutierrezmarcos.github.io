\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: \emph{¿Puede una teoría explicar lo que elegimos sin ver lo que queremos?} \autor{Marshall} (\emph{Principios}, 1890): la economía, «el estudio de la humanidad en los asuntos ordinarios de la vida»; pocas decisiones más ordinarias que qué comprar.
\item \textbf{Relevancia}: \emph{micro}, paso previo al equilibrio de mercado \vertema{3.A.16}; \emph{macro}, base del consumo agregado: consumo de los hogares, \dato{54,1\,\% del PIB}{España, 2025}.
\item \textbf{Contextualización}: utilidad (\autor{Bernoulli} 1738, \autor{Dupuit} 1844, \autor{Gossen} 1854) $\to$ revolución marginalista (\autor{Jevons}, \autor{Menger} 1871; \autor{Walras} 1874) $\to$ \autor{Marshall}, cardinal $\to$ \autor{Edgeworth} (1881), curvas de indiferencia; \autor{Pareto} (1906), ordinal $\to$ \autor{Slutsky} (1915), \autor{Hicks}-\autor{Allen} (1934), \autor{Debreu} (1959) $\to$ hacia lo observable: \autor{Samuelson}, \autor{Becker}, \autor{Rosen}.
\item \textbf{Problemática}: ¿cómo se obtiene la demanda de preferencias y presupuesto, y cuándo hay demanda de mercado? ¿Cómo varía con renta y precios? ¿Puede construirse desde lo observable?
\item \textbf{Estructura}: I. fundamentos (hasta la demanda de mercado); II. estática comparativa, con \autor{Slutsky} en el centro; III. 3 desarrollos hacia lo observable.
\end{itemize}
\begin{itemize}
\item \concepto{Individualismo metodológico}: lo social, explicado desde metas, creencias y acciones de los individuos.
\item \concepto{Consumidor}: demanda productos, ofrece factores; optimiza sujeto a un presupuesto.
\item \concepto{Leyes de \autor{Gossen}}: (1) saturación; (2) igualdad de las UMg ponderadas (I.2).
\end{itemize}

\bloque[11]{I. Fundamentos de la elección racional}
\begin{itemize}
\item \emph{Conclusión anticipada}: con pocos axiomas y un presupuesto lineal sale una demanda con propiedades contrastables; agregarla en una de mercado exige condiciones muy restrictivas.
\end{itemize}
\graficorepaso[0.52\linewidth]{esquema-ingredientes-repaso}{De las preferencias a la demanda de mercado}
\begin{itemize}
\item \emph{Supuestos}: estático y cierto; información perfecta; precio-aceptante; bienes divisibles y homogéneos; sin oferta de trabajo \vertema{3.A.25} ni decisiones intertemporales \vertema{3.A.29}.
\end{itemize}

\sub{I.1. Ingredientes}
\begin{itemize}
\item \concepto{Conjunto de elección} $S\subseteq\mathbb{R}^n_+$ (restricciones físicas): no vacío, cerrado, acotado inferiormente (no superiormente: la escasez la pone el presupuesto), convexo ($\Rightarrow$ divisibilidad).
\item \concepto{Preferencia débil} $\succcurlyeq\ \subseteq S\times S$ (primitiva) $\to$ \concepto{estricta} $\succ$ (sí y no al revés) e \concepto{indiferencia} $\sim$ (en los dos sentidos).
\item \concepto{Conjuntos de contorno}: «al menos tan bueno como» $B(\vect{x}^0)=\{\vect{x}\succcurlyeq\vect{x}^0\}$; «no mejor que» $H(\vect{x}^0)$; de indiferencia $I(\vect{x}^0)=B\cap H$, su frontera común.
\end{itemize}
\begin{itemize}
\item \concepto{Axiomas} (\autor{Debreu}, 1959): 1-3 = \concepto{preorden completo} (\emph{racionalidad}): $\sim$, relación de equivalencia $\Rightarrow$ particiona $S$ en conjuntos de indiferencia que no se cortan.
\end{itemize}
\pareja{esquema-axiomas-repaso}{7 axiomas: qué garantiza cada uno}{conjuntos-contorno}{$B(\vect{x}^0)$, $H(\vect{x}^0)$ y su frontera}
\begin{itemize}
\item \concepto{Teorema de \autor{Debreu}}: 1-4 $\Rightarrow$ existe $U$ continua; no hacen falta monotonía, convexidad ni diferenciabilidad. \concepto{Preferencias lexicográficas}: racionales, no continuas $\Rightarrow$ sin $U$.
\item \concepto{Grados de deseabilidad}: monotonía estricta ($\vect{x}\geq\vect{y}$, $\vect{x}\neq\vect{y}$) $\Rightarrow$ monotonía ($\vect{x}\gg\vect{y}$) $\Rightarrow$ insaciabilidad local (se agota la renta) $\Rightarrow$ no saturación global.
\end{itemize}
\begin{formula}
\centering Representación: $\vect{x}^1\succcurlyeq\vect{x}^2\ \Leftrightarrow\ U(\vect{x}^1)\geq U(\vect{x}^2)$
\end{formula}
\begin{itemize}
\item \concepto{Función de utilidad}: «numeración» de los conjuntos de indiferencia. \concepto{Ordinal}: vale cualquier transformación creciente; \concepto{cardinal}: solo afines positivas (utilidad esperada \vertema{3.A.10}).
\item \concepto{Propiedades de $U$} (preferencias regulares): continua (4); creciente (5); cuasicóncava, contornos superiores convexos (6; estrictamente, 6$'$); no única.
\item \concepto{Curva de indiferencia}: conjunto de nivel de $U$; mapa = cortes de la «colina» proyectados.
\end{itemize}
\pareja{superficie-utilidad-repaso}{Superficie $\sqrt{x_1x_2}$ cortada a $U=1,2,3$}{propiedades-curvas-indiferencia}{Propiedades y axioma que da cada una}
\begin{formula}
\centering $\mathrm{RMS}_{12}\equiv-\left.\dfrac{dx_2}{dx_1}\right|_{\overline{U}}=\dfrac{U_1}{U_2}$,\qquad $\sigma\equiv\dfrac{d\ln(x_2/x_1)}{d\ln\mathrm{RMS}_{12}}$
\end{formula}
\begin{itemize}
\item \concepto{Relación marginal de sustitución} (RMS): bien 2 que se cede por una unidad más del 1 sin cambiar $U$; decreciente. \concepto{Elasticidad de sustitución} $\sigma$: número puro, curvatura.
\item \concepto{CES} (elasticidad de sustitución constante): $\sigma\to0$ \autor{Leontief} $\min\{x_1/a;x_2/b\}$; $\sigma\to1$ \autor{Cobb}-\autor{Douglas}; $\sigma\to\infty$ lineal $ax_1+bx_2$.
\item \concepto{Conjunto presupuestario walrasiano} $B(\vect{p},\overline{W})=\{\vect{x}\in S:\vect{p}\cdot\vect{x}\leq\overline{W}\}$: no vacío, compacto si $\vect{p}\gg\vect{0}$, convexo. \concepto{Recta de balance}: pendiente $-p_1/p_2$.
\end{itemize}
\pareja{ces-curvas-indiferencia}{Curvas CES según $\sigma$}{conjunto-presupuestario}{$B(\vect{p},\overline{W})$; si $p_1\uparrow$, gira hacia dentro}

\sub{I.2. Maximización de la utilidad: el equilibrio}
\begin{itemize}
\item \concepto{Problema de maximización de la utilidad} (PMU, primal):
\end{itemize}
\funcionclave{\optim{\max}{x_1,\dots,x_n}{U(x_1,\dots,x_n)}{\textstyle\sum_{i}p_ix_i\leq\overline{W}\\ x_i\geq0,\quad i=1,\dots,n}}
\begin{itemize}
\item \emph{Teoremas}: existencia (\autor{Weierstrass}); unicidad (estr.\ cuasiconcavidad); CPO suficientes (\autor{Arrow}-\autor{Enthoven}, 1961).
\end{itemize}
\begin{formula}
\centering $\mathcal{L}=U(\vect{x})+\lambda\big(\overline{W}-\vect{p}\cdot\vect{x}\big)$\\[2pt]
\autor{Kuhn}-\autor{Tucker}: $U_i-\lambda p_i\leq0,\ \ x_i\geq0,\ \ x_i(U_i-\lambda p_i)=0$\\
$\overline{W}-\vect{p}\cdot\vect{x}\geq0,\ \ \lambda\geq0,\ \ \lambda(\overline{W}-\vect{p}\cdot\vect{x})=0$
\end{formula}
\begin{itemize}
\item \emph{Multiplicador}: $\lambda$, \concepto{utilidad marginal de la renta} (precio sombra); insaciabilidad local $\Rightarrow$ $\lambda>0$ y se agota la renta; saturación interior $\Rightarrow$ $\lambda=0$.
\end{itemize}
\begin{formula}
\centering \concepto{Interior} (2.ª ley de \autor{Gossen}): $\dfrac{U_1}{p_1}=\dots=\dfrac{U_n}{p_n}=\lambda\ \Rightarrow\ \mathrm{RMS}_{12}=\dfrac{p_1}{p_2}$\\[2pt]
\concepto{Esquina} ($x_2^*=0$): $\dfrac{U_1}{p_1}=\lambda\geq\dfrac{U_2}{p_2}\ \Rightarrow\ \mathrm{RMS}_{12}\geq\dfrac{p_1}{p_2}$
\end{formula}
\graficorepaso[0.62\linewidth]{solucion-interior-esquina-repaso}{Solución interior (tangencia) y de esquina ($x_2^*=0$)}
\begin{itemize}
\item \emph{Sin convexidad}: la tangencia no basta (puede ser un mínimo); esquinas, soluciones múltiples, demanda que salta. Cóncavas $\Rightarrow$ siempre esquina.
\end{itemize}

\sub{I.3. Solución: demanda, FIU y mercado}
\begin{itemize}
\item \concepto{Sistema completo de ecuaciones de demanda marshalliana} (SCEDM; ordinaria o walrasiana):
\end{itemize}
\funcionclave{\vect{x}^*=\vect{x}(\vect{p},\overline{W})=\big(x_1(\vect{p},\overline{W}),\dots,x_n(\vect{p},\overline{W})\big)}
\begin{itemize}
\item \emph{Propiedades}: única; continua (\autor{Berge}); \concepto{homogénea de grado 0}, $\vect{x}(\kappa\vect{p},\kappa\overline{W})=\vect{x}(\vect{p},\overline{W})$ (sin \emph{ilusión monetaria}); \concepto{ley de \autor{Walras}} $\vect{p}\cdot\vect{x}(\vect{p},\overline{W})=\overline{W}$; diferenciable.
\item \emph{Función valor}: \concepto{función indirecta de utilidad} (FIU) $V(\vect{p},\overline{W})=U(\vect{x}(\vect{p},\overline{W}))$: homogénea de grado 0; creciente en $\overline{W}$, no creciente en $\vect{p}$; \emph{cuasiconvexa}.
\end{itemize}
\begin{formula}
\centering \concepto{Identidad de \autor{Roy}} (1947):\ \ $x_i(\vect{p},\overline{W})=-\dfrac{\partial V/\partial p_i}{\partial V/\partial\overline{W}}$\ \ (envolvente)
\end{formula}
\begin{itemize}
\item \emph{Demanda de mercado}: $X_i=\sum_hx_{hi}(\vect{p},\overline{W}_h)$, \concepto{suma horizontal}.
\item \emph{Primer problema}: depende de \emph{toda} la distribución de la renta. Solo de $\sum_h\overline{W}_h$ $\Leftrightarrow$ \concepto{forma polar de \autor{Gorman}} (1953, 1961), $V_h=a_h(\vect{p})+b(\vect{p})\overline{W}_h$ con $b$ común $\Leftrightarrow$ curvas de \autor{Engel} \emph{rectas y paralelas} (mismo efecto renta) $\Rightarrow$ \concepto{consumidor representativo}. Casos: \emph{homotéticas idénticas} ($a_h=0$); \emph{cuasilineales} $U_h=x_0+\phi_h(\cdot)$.
\end{itemize}
\graficorepaso[0.42\linewidth]{engel-gorman}{\autor{Gorman}: misma pendiente $\Rightarrow$ redistribuir no cambia la demanda agregada}
\begin{itemize}
\item \emph{Efecto renta nulo} (cuasilineales): \emph{ventajas}, marshalliana = hicksiana, excedente exacto \vertema{3.A.16}; \emph{problemas}, sin inferiores ni \autor{Giffen}, contra la ley de \autor{Engel}.
\item \emph{Segundo problema}: se heredan continuidad, homogeneidad y ley de \autor{Walras}; no el axioma débil ni la simetría y negatividad de \autor{Slutsky}. \concepto{Teorema de \autor{Sonnenschein}-\autor{Mantel}-\autor{Debreu}} (SMD, 1972-74): ni unicidad ni estabilidad del equilibrio general \vertema{3.A.21}.
\end{itemize}
\begin{ideaclave}
La demanda de mercado depende solo de precios y renta total, como la de un consumidor representativo, si las curvas de \autor{Engel} son rectas y paralelas (\autor{Gorman}). El efecto renta nulo es el caso más cómodo y el más restrictivo.
\end{ideaclave}

\bloque[8]{II. Estática comparativa}
\begin{itemize}
\item \emph{Pregunta}: cómo cambia la elección al cambiar renta y precios.
\item \emph{Método}: \concepto{estática comparativa}, comparar óptimos sin el ajuste, con \concepto{elasticidades}: números puros y medida \emph{local} (un bien, normal a una renta e inferior a otra).
\end{itemize}

\sub{II.1. Variaciones en la renta}
\begin{itemize}
\item \emph{Gráfico}: \concepto{curva de renta-consumo} (CRC); homotéticas, rayo. Al plano $(\overline{W},x_i)$: \concepto{curva de \autor{Engel}} $x_i(\bar{\vect{p}},\overline{W})$; pendiente = \concepto{efecto renta} $\partial x_i/\partial\overline{W}$.
\item \emph{Elasticidad-renta} $\varepsilon_{x_i,\overline{W}}=\frac{\partial x_i}{\partial\overline{W}}\frac{\overline{W}}{x_i}$: $<0$ \binferior{inferior}; $=0$ independiente; $(0,1)$ \bnormal{normal necesario} (pierde peso); $>1$ \bnormal{normal de lujo}.
\item \emph{Evidencia}: \concepto{ley de \autor{Engel}} (1857): el peso de los alimentos cae con la renta (necesarios). España: \dato{19,6\,\% (Q1) frente a 12,3\,\% (Q5)}{INE, EPF 2025}.
\end{itemize}
\pareja{renta-consumo-engel}{CRC y curva de \autor{Engel}: \bnormal{normal}, luego \binferior{inferior}}{epf-alimentos-quintil}{Alimentos por quintil de gasto (INE, EPF 2025)}
\begin{formula}
\centering \concepto{Agregación de \autor{Engel}}: $\displaystyle\sum_i s_i\,\varepsilon_{x_i,\overline{W}}=1$,\quad $s_i=\dfrac{p_ix_i}{\overline{W}}$
\end{formula}
\begin{itemize}
\item \emph{Implicación}: no todos inferiores ni de elasticidad nula; alguno con $\varepsilon\geq1$.
\end{itemize}

\sub{II.2. Variaciones en los precios}
\begin{itemize}
\item \emph{Gráfico}: la recta gira $\to$ \concepto{curva de precio-consumo} (CPC) $\to$ \concepto{curva de demanda} $x_1(p_1;p_2,\overline{W})$; otros parámetros la desplazan.
\item \emph{Elasticidad-precio propio}: $<0$ \bordinario{ordinario} ($<-1$ elástica: $p\uparrow$ $\Rightarrow$ gasto $\downarrow$; $(-1,0)$ inelástica); $=0$ rígida; $>0$ \bgiffen{\autor{Giffen}}.
\item \emph{Compensar la renta}: \concepto{demanda compensada} (hicksiana) $h_i(\vect{p},\overline{U})$, que mantiene $\overline{U}$; resuelve la minimización del gasto, dual del PMU \vertema{3.A.9}.
\end{itemize}
\begin{formula}
\centering \concepto{Ecuación de \autor{Slutsky}} (1915), con $\overline{U}=V(\vect{p},\overline{W})$:\\[1pt]
$\underbrace{\dfrac{\partial x_i(\vect{p},\overline{W})}{\partial p_j}}_{\mathrm{ET}}=\underbrace{\dfrac{\partial h_i(\vect{p},\overline{U})}{\partial p_j}}_{\mathrm{ES}}\underbrace{-\,x_j(\vect{p},\overline{W})\dfrac{\partial x_i(\vect{p},\overline{W})}{\partial\overline{W}}}_{\mathrm{ER}}$\\[1pt]
Elasticidades: $\varepsilon_{x_i,p_j}=\varepsilon^h_{i,p_j}-s_j\,\varepsilon_{x_i,\overline{W}}$
\end{formula}
\begin{itemize}
\item \concepto{Efecto sustitución} (ES): solo el precio relativo, $U$ constante; propio \emph{siempre} $\leq0$ (concavidad del gasto).
\item \concepto{Efecto renta} (ER): poder adquisitivo; propio $<0$ si normal, $>0$ si inferior; crece con el peso $s_i$.
\item \concepto{Clasificación}: \bnormal{normal} $\Rightarrow$ \bordinario{ordinario} (\emph{ley de la demanda}); \binferior{inferior}: \bordinario{ordinario} si domina el ES, \bgiffen{\autor{Giffen}} si domina el ER (y pesa mucho). Arroz en Hunan (\autor{Jensen}-\autor{Miller}, 2008).
\end{itemize}
\pareja{slutsky-propio}{Signos de \autor{Slutsky} con $j=i$}{elasticidades-renta-precio-repaso}{Todo \bnormal{normal} es \bordinario{ordinario}; todo \bgiffen{\autor{Giffen}}, \binferior{inferior}}
\begin{itemize}
\item \emph{Gráficamente} ($p_1\downarrow$): ES $A\to B$ (misma curva, nueva pendiente); ER $B\to C$ (desplazamiento paralelo). Marshalliana por $A$ y $C$; hicksiana por $A$ y $B$, más inclinada si es normal.
\end{itemize}
\graficorepaso[0.36\linewidth]{efecto-sustitucion-renta}{ES ($A\to B$), ER ($B\to C$); marshalliana y hicksiana}
\begin{itemize}
\item \emph{Fuera del modelo}: \concepto{efecto \autor{Veblen}} (1899), el precio entra en las preferencias $\Rightarrow$ no se aplica \autor{Slutsky}.
\item \emph{Elasticidad cruzada}: $\varepsilon_{x_i,p_j}>0$ \bnormal{\concepto{sustitutivo bruto}}; $=0$ independiente; $<0$ \binferior{\concepto{complementario bruto}}: \emph{no simétricas} (incluyen el ER).
\item \emph{Relaciones netas}, por el ES cruzado: \bnormal{\concepto{sustitutivos netos}} $\partial h_i/\partial p_j>0$, \binferior{\concepto{complementarios netos}} $<0$: \emph{simétricas} ($\partial h_i/\partial p_j=\partial h_j/\partial p_i$). Homogeneidad de $h$: $\sum_jp_j\,\partial h_i/\partial p_j=0$ $\Rightarrow$ todo bien tiene un sustitutivo neto; con 2 bienes, ambos lo son.
\end{itemize}
\begin{formula}
\centering \concepto{Agregación de \autor{Cournot}}: $\displaystyle\sum_is_i\,\varepsilon_{x_i,p_j}=-s_j$\\[2pt]
Homogeneidad (\autor{Euler}): $\displaystyle\sum_j\varepsilon_{x_i,p_j}+\varepsilon_{x_i,\overline{W}}=0$
\end{formula}
\begin{itemize}
\item \emph{Origen}: agregaciones, de la ley de \autor{Walras}; homogeneidad, del presupuesto; simetría y negatividad, de \emph{maximizar} \vertema{3.A.9}.
\end{itemize}
\begin{ideaclave}
Un cambio de precio tiene un ES, siempre no positivo, y un ER, cuyo signo depende de que el bien sea normal o inferior: la ley de la demanda vale siempre para la compensada y para la marshalliana solo si el ER no domina (\autor{Giffen}). Netas, simétricas; brutas, no.
\end{ideaclave}

\sub{II.3. Valoración}
\begin{itemize}
\item \emph{Logros}: \concepto{restricciones contrastables}: homogeneidad, ley de \autor{Walras}, agregaciones de \autor{Engel} y \autor{Cournot}, signo del ES.
\item \emph{Limitaciones} $\to$ III: (1) preferencias inobservables $\to$ preferencia revelada; (2) sin tiempo $\to$ producción doméstica; (3) bienes homogéneos, sin calidad $\to$ precios hedónicos; (4) estático y cierto \vertema{3.A.10}\vertema{3.A.29}.
\end{itemize}

\bloque[7,5]{III. Otros desarrollos de la demanda}
\begin{itemize}
\item \emph{Conclusión anticipada}: los 3 desarrollos responden a esas limitaciones y acercan la teoría a lo observable.

\sub{III.1. La teoría de la preferencia revelada}
\begin{itemize}
\item \emph{Idea}: \autor{Samuelson} (1938, 1948; Nobel 1970): partir de las \emph{elecciones}, no de preferencias inobservables. \emph{Inductivo} (la ordinal, deductiva).
\item \emph{Supuestos}: consumidor \emph{individual} (no el representativo); una cesta por cada $(\vect{p}\gg\vect{0},\overline{W})$ (\emph{función} de demanda); ley de \autor{Walras}.
\item \emph{Desarrollo}: $\vect{x}^0$ \concepto{directamente preferida} a $\vect{x}^1$ si se eligió pudiendo comprar $\vect{x}^1$.
\end{itemize}
\begin{formula}
\centering \concepto{Axioma débil} (ADPR): $\vect{p}^0\cdot\vect{x}^1\leq\vect{p}^0\cdot\vect{x}^0,\ \vect{x}^1\neq\vect{x}^0\ \Rightarrow\ \vect{p}^1\cdot\vect{x}^0>\vect{p}^1\cdot\vect{x}^1$
\end{formula}
\begin{itemize}
\item[] Coherencia. \concepto{Indirectamente preferida}: por una cadena. \concepto{Axioma fuerte} (AFPR, \autor{Houthakker}, 1950): transitividad de la revelada; AFPR $\Rightarrow$ ADPR.
\end{itemize}
\pareja{adpr-efecto-sustitucion}{ADPR y signo del efecto sustitución}{afpr}{AFPR: $\vect{x}^0$ indirectamente preferida a $\vect{x}^2$}
\begin{itemize}
\item \emph{Implicaciones} del ADPR, sin utilidad: (1) homogeneidad de grado 0; (2) \concepto{ley de la demanda compensada}, con $\overline{W}^1=\vect{p}^1\cdot\vect{x}^0$:
\end{itemize}
\begin{formula}
\centering $(\vect{p}^1-\vect{p}^0)\cdot(\vect{x}^1-\vect{x}^0)\leq0$;\quad solo $p_1$: $\Delta p_1\,\Delta x_1\leq0$
\end{formula}
\begin{itemize}
\item[] (3) matriz de \autor{Slutsky} semidefinida negativa, \emph{no} simétrica $\Rightarrow$ con más de 2 bienes no basta para una utilidad (con 2, sí: \autor{Rose}, 1958). \concepto{Teorema de \autor{Houthakker}}: AFPR $\Leftrightarrow$ demanda generada por preferencias racionales $\Rightarrow$ revelada y ordinal, equivalentes. \concepto{Curvas de indiferencia} (\autor{Samuelson}, 1948): entre las cestas reveladas peores y las mejores; más datos, más acotada.
\item \emph{Extensiones}: \concepto{teorema de \autor{Afriat}} (1967): datos \emph{finitos} compatibles con $U$ continua, creciente y cóncava $\Leftrightarrow$ axioma generalizado (GARP); índices de \autor{Laspeyres} y \autor{Paasche} \vertema{3.A.9}.
\item \emph{Evidencia}: \autor{Varian} (1982): contraste \emph{no paramétrico} del GARP con hogares y experimentos.
\item \emph{Valoración}: demanda sin utilidad y contrastable; pero identifica elección y bienestar (\autor{Sen}, 1973); sin tiempo $\to$ III.2.
\end{itemize}

\sub{III.2. La producción doméstica}
\begin{itemize}
\item \emph{Idea}: \concepto{tecnologías del consumo}: los bienes son \emph{insumos}. \autor{Becker} (1965; Nobel 1992): el hogar, una fábrica de \emph{commodities} con bienes y \emph{tiempo}. \autor{Lancaster} (1966): características $\vect{z}=B\vect{x}$ (anexo; diferenciación de productos \vertema{3.A.18}).
\item \emph{Supuestos}: $U(Z_1,\dots,Z_m)$, $Z_i=f_i(x_i,t_i)$; tiempo $T$ entre trabajo $t_w$ (salario $w$) y cada $t_i$; renta no laboral $\overline{W}$.
\item \emph{Desarrollo}:
\end{itemize}
\begin{formula}
\centering $\optim{\max}{\{x_i,t_i\},\,t_w}{U(Z_1,\dots,Z_m)}{Z_i=f_i(x_i,t_i)\\ \textstyle\sum_ip_ix_i=w\,t_w+\overline{W}\\ \textstyle\sum_it_i+t_w=T}$\\[3pt]
\concepto{Renta completa}: $\sum_i(p_ix_i+w\,t_i)=wT+\overline{W}$\\
$x_i=b_iZ_i$, $t_i=\ell_iZ_i$ $\Rightarrow$ \concepto{precio sombra} $\pi_i=p_ib_i+w\,\ell_i$
\end{formula}
\begin{itemize}
\item \emph{Implicaciones}: el salario es un precio del consumo; $w\uparrow$ $\Rightarrow$ se sustituye tiempo por bienes (comida preparada, guarderías): mujer en el mercado de trabajo, externalización \vertema{3.A.25}; nuevos efectos sustitución.
\item \emph{Extensiones}: \autor{Gronau} (1977): trabajo doméstico $\neq$ ocio.
\item \emph{Valoración}: educación, matrimonio, fecundidad; el PIB omite la producción doméstica; sin calidad $\to$ III.3.
\end{itemize}

\sub{III.3. Precios hedónicos}
\begin{itemize}
\item \emph{Idea}: bienes \emph{diferenciados} = «paquetes» de características (vivienda, coche); el \concepto{método de los precios hedónicos} descompone el precio en el valor implícito de cada una. Base: \autor{Lancaster}. \autor{Court} (1939), \autor{Griliches} (1961, índices), \autor{Rosen} (1974, teoría).
\item \emph{Supuestos}: mercado competitivo de un producto diferenciado; precio-aceptantes informados; $U(\vect{z},y)$, $y$ numerario; una sola unidad.
\item \emph{Desarrollo}: \concepto{función hedónica} $p(\vect{z})$; \concepto{precio implícito} $\partial p/\partial z_j$.
\end{itemize}
\begin{formula}
\centering $\optim{\max}{\vect{z},\,y}{U(\vect{z},y)}{p(\vect{z})+y=\overline{W}}\ \Rightarrow\ \dfrac{\partial U/\partial z_j}{\partial U/\partial y}=\dfrac{\partial p(\vect{z})}{\partial z_j}$
\end{formula}
\begin{itemize}
\item[] La tangencia de I.2 con presupuesto no lineal. \concepto{Función de puja} $\Theta(\vect{z};\overline{U},\overline{W})$ (lo máximo que pagaría) y \concepto{de oferta} $\phi(\vect{z};B)$ (lo mínimo que aceptaría). \concepto{Equilibrio}: $p(\vect{z})$, \emph{envolvente} de ambas, tangentes en cada variedad vendida.
\end{itemize}
\graficorepaso[0.48\linewidth]{funcion-hedonica-rosen}{$p(\vect{z})$, envolvente de pujas y ofertas (\autor{Rosen}, 1974)}
\begin{itemize}
\item \emph{Implicaciones}: precio implícito = valoración marginal del comprador = coste marginal del vendedor; $p(\vect{z})$ no es demanda ni oferta: lugar de equilibrios.
\item \emph{Extensiones}: bienes sin mercado (aire limpio, ruido, vivir frente al Retiro; análisis coste-beneficio \vertema{4.B.6}); salarios hedónicos (valor estadístico de la vida).
\item \emph{Evidencia}: 2 etapas: (1) $\ln p_v=\beta_0+\sum_j\beta_jz_{vj}+u_v$ $\to$ precios implícitos; (2) demandas de características, mal identificada. IPC: \autor{Boskin} (1996), sesgo \dato{1,1 p./año}{EE.\,UU.}; \emph{Índice de Precios de Vivienda} del INE (desde 2008), con regresión hedónica.
\item \emph{Valoración}: forma funcional y variables omitidas; 2.ª etapa mal identificada; equilibrio competitivo e informado (dudoso en vivienda); precios marginales, no para cambios grandes.
\end{itemize}
\begin{ideaclave}
Los 3 desarrollos acercan la teoría a lo observable: la preferencia revelada reconstruye las preferencias desde las elecciones y las contrasta (\autor{Afriat}); la producción doméstica añade el coste del tiempo; los precios hedónicos miden cuánto se paga por cada característica.
\end{ideaclave}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}: de pocos axiomas y un presupuesto, una demanda homogénea de grado 0 que agota la renta; mercado solo con \autor{Gorman} (I); \autor{Slutsky} explica normales, inferiores y \autor{Giffen}, brutas y netas (II); 3 desarrollos hacia los datos (III).
\item \textbf{Relevancia}: de las construcciones más acabadas (ida y vuelta: \autor{Houthakker}, integrabilidad); se usa a diario, de las elasticidades a los índices de precios.
\item \textbf{Extensiones}: dualidad, bienestar y sistemas de demanda \vertema{3.A.9}; incertidumbre \vertema{3.A.10}; intertemporal \vertema{3.A.29}; equilibrio parcial y general \vertema{3.A.16}\vertema{3.A.21}; oferta de trabajo \vertema{3.A.25}; producción, misma estructura \vertema{3.A.11}.
\item \textbf{Opinión}: lógica más sólida que evidencia: homogeneidad y simetría rechazadas con datos agregados (\autor{Deaton}-\autor{Muellbauer}, 1980); \emph{como si} (\autor{Friedman}, 1953); conducta: \autor{Simon} (1955), \autor{Kahneman}-\autor{Tversky} (1979), \autor{Thaler}; «tonto racional» (\autor{Sen}, 1977). La amplían más que sustituirla.
\item \textbf{Idea final}: sí, con matices: las elecciones delatan las preferencias y basta pedirles coherencia para una demanda refutable. Primer ladrillo: precios, inflación con calidad cambiante, bienestar; el riesgo, el tiempo y los sesgos, para el resto de la teoría.
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Qué axiomas hacen falta para que exista $U$?} Los 1-4 (\autor{Debreu}). Monotonía y convexidad dan sus propiedades; 6$'$ y 7 simplifican.
\item \textbf{¿Puede un bien \bnormal{normal} ser \bgiffen{\autor{Giffen}}?} No: normal $\Rightarrow$ ordinario. \autor{Giffen} exige ser \binferior{inferior} y pesar mucho en el gasto.
\item \textbf{¿Por qué netas simétricas y brutas no?} ES simétrico (hessiana del gasto); el ER cruzado $-x_j\,\partial x_i/\partial\overline{W}$, no.
\item \textbf{¿Cuándo hay consumidor representativo?} Forma polar de \autor{Gorman}: curvas de \autor{Engel} rectas y paralelas.
\item \textbf{¿Equivalen preferencia revelada y teoría ordinal?} Sí con el AFPR (\autor{Houthakker}); el ADPR basta solo con 2 bienes.
\item \textbf{¿Por qué $p(\vect{z})$ no es una curva de demanda?} Es la envolvente de pujas y ofertas: un lugar de equilibrios.
\item \textbf{¿Por qué es homogénea de grado 0?} Con $\kappa\vect{p},\kappa\overline{W}$ el presupuesto no cambia: sin ilusión monetaria.
\item \textbf{¿Cuándo es $\lambda=0$?} Con un punto de saturación dentro del presupuesto: la restricción no es operativa.
\end{enumerate}

\begin{trampa}
\begin{itemize}
\item La \emph{convexidad} no exige UMg decrecientes ni estas la garantizan (depende de $U_{12}$); la UMg decreciente no es ordinal ($x_1x_2$ y $(x_1x_2)^2$): importa la RMS decreciente.
\item ER $=-x_i\,\partial x_i/\partial\overline{W}$, de signo opuesto a $\partial x_i/\partial\overline{W}$. Todo \bgiffen{\autor{Giffen}} es \binferior{inferior}, no al revés. \autor{Veblen} $\neq$ \autor{Giffen}.
\item ADPR $\Rightarrow$ \autor{Slutsky} semidefinida negativa, \emph{no} simétrica. AFPR $\Rightarrow$ ADPR.
\item Esquina: $\mathrm{RMS}_{12}\geq p_1/p_2$, no igualdad.
\item \autor{Gorman}: rectas y \emph{paralelas}; por el origen, solo homotéticas.
\item Bienes, males y neutrales: signo de la UMg, que no dice nada de la curvatura. 2 males: curvas decrecientes, mejores hacia el origen; neutral: rectas verticales u horizontales.
\item $U=xy$: RMS $y/x$ (convexa); $ax+by$: lineal; $x^2+y^2$: cóncava $\Rightarrow$ esquina.
\item FIU \emph{cuasiconvexa} en $(\vect{p},\overline{W})$, aunque las preferencias no sean convexas.
\end{itemize}
\end{trampa}
