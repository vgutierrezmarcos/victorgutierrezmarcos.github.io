\bloque[3]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: \autor{Marshall} (1890), «el estudio de la humanidad en los asuntos ordinarios de la vida» $\Rightarrow$ \concepto{individualismo metodológico}.
\item \textbf{Relevancia}: micro, paso previo al equilibrio de mercado; macro, consumo de los hogares \dato{54,1\,\% del PIB}{España, 2025}.
\item \textbf{Contexto}: \autor{Bernoulli}, \autor{Gossen} $\to$ marginalistas $\to$ \autor{Marshall} cardinal $\to$ \autor{Pareto} ordinal $\to$ \autor{Slutsky}, \autor{Hicks}-\autor{Allen}, \autor{Debreu} $\to$ \autor{Samuelson}, \autor{Becker}, \autor{Rosen}.
\item \textbf{Problemática}: ¿cómo se obtiene la demanda? ¿Cómo varía con renta y precios? ¿Puede construirse desde lo observable?
\item \textbf{Estructura}: I. fundamentos; II. estática comparativa; III. otros desarrollos.
\end{itemize}

\bloque[10]{I. Elección racional}
\graficorepaso{esquema-ingredientes-repaso}{Ingredientes: de las preferencias a la demanda de mercado}
\begin{itemize}
\item \textbf{Supuestos}: estático, sin incertidumbre; información perfecta; precio-aceptante, bienes divisibles; consumo separado de trabajo e intertemporal.
\end{itemize}

\sub{I.1. Ingredientes}
\begin{itemize}
\item \concepto{Conjunto de elección} $S\subseteq\mathbb{R}^n_+$: no vacío, cerrado, acotado inferiormente, convexo.
\item \concepto{Preferencia débil} $\succcurlyeq$ (primitiva) $\to$ estricta $\succ$ e indiferencia $\sim$. Conjuntos «al menos tan bueno como» $B(x^0)$, «no mejor que» $H(x^0)$; $I(x^0)=B\cap H$.
\end{itemize}
\graficorepaso{esquema-axiomas-repaso}{Axiomas: qué garantiza cada uno y su papel}
\begin{itemize}
\item \concepto{Teorema de Debreu}: 1-4 $\Rightarrow$ existe $U$ continua. \concepto{Lexicográficas}: racionales, no continuas, sin utilidad. Deseabilidad: monot.\ estricta $\Rightarrow$ monot.\ $\Rightarrow$ insaciab.\ local $\Rightarrow$ no saturación.
\item \concepto{Función de utilidad}: $x^1\succcurlyeq x^2\Leftrightarrow U(x^1)\geq U(x^2)$; «numera» los conjuntos de indiferencia; \concepto{ordinal} (transf.\ monótonas). \concepto{Curvas de indiferencia} = conjuntos de nivel de $U$: no se cortan (3), pendiente negativa (5), convexas (6).
\end{itemize}
\graficorepaso[0.72\linewidth]{superficie-utilidad-repaso}{Superficie de utilidad cortada a $U=1,2,3$: su proyección es el mapa de curvas de indiferencia}
\begin{formula}
\centering $RMS_{12}\equiv-\left.\dfrac{dx_2}{dx_1}\right|_{\overline{U}}=\dfrac{U_1}{U_2}$\\[1pt] $\sigma\equiv\dfrac{d\ln(x_2/x_1)}{d\ln RMS_{12}}$
\end{formula}
\begin{itemize}
\item \concepto{CES}: $\sigma\to0$ \autor{Leontief}, $\sigma\to1$ Cobb-Douglas, $\sigma\to\infty$ lineal.
\item \concepto{Conjunto presupuestario} $B(p,\overline{W})=\{x\in S: p\cdot x\leq\overline{W}\}$: compacto si $p\gg0$, convexo. Recta de balance, pendiente $-p_1/p_2$.
\end{itemize}

\sub{I.2. PMU: el equilibrio}
\begin{itemize}
\item $\max U(x)$ s.a.\ $p\cdot x\leq\overline{W}$, $x\geq0$: existe (\autor{Weierstrass}), única (estr.\ cuasiconcavidad), CPO suficientes (\autor{Arrow}-\autor{Enthoven}).
\item \concepto{Kuhn-Tucker}: $U_i-\lambda p_i\leq0$, $x_i\geq0$, holgura complementaria. $\lambda$ = UMg de la renta; $\lambda>0$ por insaciabilidad local (con saturación interior, $\lambda=0$).
\end{itemize}
\begin{formula}
Interior (\autor{Gossen}): $\dfrac{U_1}{p_1}=\dots=\dfrac{U_n}{p_n}=\lambda$\\ $\Rightarrow RMS_{12}=p_1/p_2$.\ \ Esquina ($x_2^*=0$): $RMS_{12}\geq p_1/p_2$
\end{formula}
\graficorepaso{solucion-interior-esquina-repaso}{Solución interior (tangencia) y de esquina}

\sub{I.3. Solución: demanda y mercado}
\begin{itemize}
\item \concepto{Demanda marshalliana} $x(p,\overline{W})$: única, continua (\autor{Berge}), \textbf{homogénea de grado 0} (sin ilusión monetaria), \textbf{ley de Walras} $p\cdot x=\overline{W}$, diferenciable.
\item \concepto{FIU} $V(p,\overline{W})$: homogénea de grado 0, creciente en $\overline{W}$, no creciente en $p$, cuasiconvexa. \autor{Roy}: $x_i=-\dfrac{\partial V/\partial p_i}{\partial V/\partial\overline{W}}$.
\item \concepto{Demanda de mercado}: suma horizontal; depende de la distribución de la renta. Solo de $\sum_h\overline{W}_h$ $\Leftrightarrow$ \concepto{forma polar de Gorman} $V_h=a_h(p)+b(p)\overline{W}_h$ $\Leftrightarrow$ curvas de Engel \textbf{rectas y paralelas} $\Rightarrow$ consumidor representativo. Casos: homotéticas idénticas; cuasilineales (\textbf{efecto renta nulo}: excedente exacto, pero sin inferiores ni Giffen).
\item Se heredan homogeneidad y ley de Walras, no el ADPR ni Slutsky: \concepto{SMD} $\Rightarrow$ ni unicidad ni estabilidad \vertema{3.A.21}.
\end{itemize}

\bloque[7]{II. Estática comparativa}
Comparar óptimos sin el ajuste; \concepto{elasticidades}: números puros y medida \emph{local}.

\sub{II.1. Variaciones en la renta}
\begin{itemize}
\item \concepto{Curva de renta-consumo} $\to$ \concepto{curva de Engel} $x_i(\bar p,\overline{W})$; pendiente = efecto renta.
\item $\varepsilon_{x_i,\overline{W}}$: $<0$ inferior; $=0$ independiente; $(0,1)$ necesario; $>1$ lujo.
\item \concepto{Ley de Engel} (1857): alimentos \dato{19,6\,\% (Q1) frente a 12,3\,\% (Q5)}{INE, EPF 2025}.
\end{itemize}
\begin{formula}
Agregación de \autor{Engel}: $\sum_i s_i\,\varepsilon_{x_i,\overline{W}}=1$,\ \ $s_i=p_ix_i/\overline{W}$
\end{formula}

\sub{II.2. Variaciones en los precios}
\begin{itemize}
\item \concepto{Curva de precio-consumo} $\to$ \concepto{curva de demanda}. $\varepsilon_{x_i,p_i}$: $<0$ ordinario ($<-1$ elástica); $=0$ rígida; $>0$ \autor{Giffen}.
\item \concepto{Hicksiana} $h_i(p,\overline{U})$: mantiene la utilidad \vertema{3.A.9}.
\end{itemize}
\begin{formula}
\autor{Slutsky}: $\underbrace{\dfrac{\partial x_i}{\partial p_j}}_{ET}=\underbrace{\dfrac{\partial h_i}{\partial p_j}}_{ES}\ \underbrace{-\,x_j\dfrac{\partial x_i}{\partial\overline{W}}}_{ER}$\\ Con $j=i$, $ES\leq0$.
\end{formula}
\begin{itemize}
\item Normal $\Rightarrow$ ordinario (\emph{ley de la demanda}). Inferior: ordinario si domina ES; \concepto{Giffen} si domina ER (arroz en Hunan, \autor{Jensen}-\autor{Miller}, 2008). \concepto{Veblen}: precio en las preferencias, fuera del modelo.
\end{itemize}
\graficorepaso{elasticidades-renta-precio-repaso}{Todo normal es ordinario; todo Giffen es inferior}
\graficorepaso[0.62\linewidth]{efecto-sustitucion-renta}{ES ($A\to B$) y ER ($B\to C$); hicksiana por $A$ y $B$}
\begin{itemize}
\item $\varepsilon_{x_i,p_j}>0$ \concepto{sustitutivo bruto}, $<0$ \concepto{complementario bruto}: \textbf{no simétricas}. Netos según $\partial h_i/\partial p_j$: \textbf{simétricos}; todo bien tiene un sustitutivo neto.
\end{itemize}
\begin{formula}
Homog.: $\sum_j\varepsilon_{x_i,p_j}+\varepsilon_{x_i,\overline{W}}=0$\qquad\autor{Cournot}: $\sum_i s_i\,\varepsilon_{x_i,p_j}=-s_j$
\end{formula}

\sub{II.3. Valoración}
\begin{itemize}
\item \concepto{Restricciones contrastables}: homogeneidad y agregaciones (presupuesto); simetría y negatividad (maximización). Límites $\to$ III: preferencias inobservables; sin tiempo; sin características; estático.
\end{itemize}

\bloque[7]{III. Otros desarrollos}
\sub{III.1. Preferencia revelada}
\begin{itemize}
\item \autor{Samuelson} (1938, 1948): de las \emph{elecciones} (inductivo), no de preferencias inobservables.
\end{itemize}
\begin{formula}
\textbf{ADPR}: $p^0\cdot x^1\leq p^0\cdot x^0,\ x^1\neq x^0\Rightarrow p^1\cdot x^0>p^1\cdot x^1$
\end{formula}
\begin{itemize}
\item ADPR $\Rightarrow$ homogeneidad; \concepto{ley de la demanda compensada} (ES $\leq0$ sin utilidad); Slutsky semidefinida negativa, \emph{no} simétrica (con 2 bienes basta, \autor{Rose}).
\item \concepto{AFPR} (\autor{Houthakker}, 1950): transitividad; AFPR $\Leftrightarrow$ preferencias racionales. \concepto{Afriat} (1967): datos finitos, GARP $\Rightarrow$ contraste no paramétrico. Crítica: elección $\neq$ bienestar (\autor{Sen}).
\end{itemize}
\graficorepaso[0.6\linewidth]{preferencia-revelada-indiferencia}{Curva de indiferencia acotada por la preferencia revelada}

\sub{III.2. Producción doméstica}
\begin{itemize}
\item \autor{Becker} (1965): el hogar produce \emph{commodities} $Z_i=f_i(x_i,t_i)$ con bienes y \emph{tiempo}. \autor{Lancaster} (1966): características $z=Bx$ (anexo; \vertema{3.A.18}).
\end{itemize}
\begin{formula}
Renta completa: $\sum_i(p_ix_i+w\,t_i)=wT+V$;\ \ $\pi_i=p_ib_i+w\,\tau_i$
\end{formula}
\begin{itemize}
\item $w\uparrow\Rightarrow$ se sustituye tiempo por bienes. \autor{Gronau} (1977): trabajo doméstico $\neq$ ocio.
\end{itemize}

\sub{III.3. Precios hedónicos}
\begin{itemize}
\item \concepto{Función hedónica} $p(z)$; \concepto{precio implícito} $\partial p/\partial z_j$. \autor{Court} (1939), \autor{Griliches} (1961), \autor{Rosen} (1974).
\item \concepto{Rosen}: $\dfrac{U_{z_j}}{U_y}=\dfrac{\partial p}{\partial z_j}$; $p(z)$, \textbf{envolvente} de curvas de puja y de oferta (no es demanda ni oferta).
\item Estimación en dos etapas; la 2.ª, mal identificada. Aplicaciones: IPC con ajuste de calidad (\autor{Boskin}: sesgo \dato{1,1 p.p./año}{EE.\,UU., 1996}), \textbf{IPV} del INE, bienes sin mercado \vertema{4.B.6}, salarios hedónicos.
\end{itemize}
\begin{ideaclave}
Los tres desarrollos acercan la teoría a lo observable: elecciones, coste del tiempo y valor de cada característica.
\end{ideaclave}

\bloque[3]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}: de unos pocos axiomas a la demanda marshalliana y, con Gorman, a la de mercado (I); Slutsky explica la clasificación de los bienes (II); tres desarrollos hacia lo observable (III).
\item \textbf{Valoración}: construcción lógica muy acabada (ida y vuelta: Houthakker, integrabilidad). Empíricamente discutible: homogeneidad y simetría rechazadas con datos agregados (\autor{Deaton}-\autor{Muellbauer}, 1980); \emph{como si} (\autor{Friedman}, 1953); economía conductual (\autor{Simon}, \autor{Kahneman}-\autor{Tversky}, \autor{Thaler}); «tonto racional» (\autor{Sen}, 1977).
\item \textbf{Extensiones}: dualidad, bienestar y sistemas de demanda \vertema{3.A.9}; incertidumbre \vertema{3.A.10}; intertemporal \vertema{3.A.29}; equilibrio \vertema{3.A.16}\vertema{3.A.21}; trabajo \vertema{3.A.25}; producción \vertema{3.A.11}.
\item \textbf{Idea final}: primer ladrillo del análisis de mercados: precios, inflación con calidad cambiante y bienestar.
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Qué axiomas hacen falta para que exista $U$?} Los 1-4 (Debreu). Monotonía y convexidad dan sus propiedades; 6$'$ y 7 simplifican.
\item \textbf{¿Puede un bien normal ser Giffen?} No: normal $\Rightarrow$ ordinario. Giffen exige ser inferior y pesar mucho en el gasto.
\item \textbf{¿Por qué las relaciones netas son simétricas y las brutas no?} La matriz de Slutsky es simétrica; el ER cruzado ($-x_j\,\partial x_i/\partial\overline{W}$) no tiene por qué coincidir con el de $j$ ante $p_i$.
\item \textbf{¿Cuándo hay consumidor representativo?} Forma polar de Gorman: curvas de Engel rectas y paralelas.
\item \textbf{¿Equivalen preferencia revelada y teoría ordinal?} Sí con el AFPR (Houthakker); el ADPR solo basta con dos bienes.
\item \textbf{¿Por qué $p(z)$ no es una curva de demanda?} Es la envolvente de pujas y ofertas: un lugar de equilibrios.
\item \textbf{¿Qué restricciones son contrastables?} Homogeneidad, agregaciones de Engel y Cournot (presupuesto), simetría y negatividad de Slutsky (maximización).
\item \textbf{¿Por qué la demanda es homogénea de grado 0?} Si se multiplican $p$ y $\overline{W}$ por $k>0$ el conjunto presupuestario no cambia: sin ilusión monetaria; por \autor{Euler}, $\sum_j\varepsilon_{x_i,p_j}+\varepsilon_{x_i,\overline{W}}=0$.
\item \textbf{¿Qué aporta Becker?} El coste de oportunidad del tiempo: el salario es un precio del consumo y, si sube, se sustituye tiempo por bienes.
\end{enumerate}

\begin{trampa}
\begin{itemize}
\item La \textbf{convexidad} no exige UMg decrecientes ni estas la garantizan; la UMg decreciente no es ordinal ($xy$ y $(xy)^2$): importa la RMS decreciente.
\item Cuasiconcavidad y RMS son ordinales; concavidad y signo de $U_{ii}$, cardinales.
\item Para que exista $U$ \emph{no} hacen falta monotonía, convexidad ni diferenciabilidad.
\item Lexicográficas: preorden completo y monotonía estricta, generan demanda, pero \emph{sin} función de utilidad.
\item Insaciabilidad local basta para la ley de Walras ($\lambda>0$).
\item Signo de Slutsky: ER $=-x_i\,\partial x_i/\partial\overline{W}$, opuesto al de $\partial x_i/\partial\overline{W}$. Todo Giffen es inferior, no al revés. Veblen $\neq$ Giffen.
\item ADPR $\Rightarrow$ Slutsky semidefinida negativa, \emph{no} simétrica.
\item La agregación conserva homogeneidad y ley de Walras, no el ADPR.
\item La «ley de la escasez» no la formula \autor{Gossen}.
\item Esquina: $RMS_{12}\geq p_1/p_2$, no igualdad.
\item La clasificación de un bien es \emph{local}: normal a una renta, inferior a otra.
\item Gorman: curvas de Engel rectas y \emph{paralelas}; que pasen por el origen solo en las homotéticas.
\end{itemize}
\end{trampa}
