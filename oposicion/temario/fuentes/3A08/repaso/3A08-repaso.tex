\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: \autor{Marshall} (1890), «el estudio de la humanidad en los asuntos ordinarios de la vida» $\Rightarrow$ \concepto{individualismo metodológico}.
\item \textbf{Relevancia}: micro, paso previo al equilibrio de mercado; macro, consumo de los hogares \dato{54,1\,\% del PIB}{España, 2025}.
\item \textbf{Contexto}: \autor{Bernoulli}, \autor{Gossen} $\to$ marginalistas (años 70 del s.~XIX) $\to$ \autor{Marshall} cardinal $\to$ \autor{Pareto} ordinal $\to$ \autor{Slutsky}, \autor{Hicks}-\autor{Allen}, \autor{Debreu} $\to$ \autor{Samuelson}, \autor{Becker}, \autor{Rosen}.
\item \textbf{Problemática}: ¿cómo se obtiene la demanda? ¿Cómo varía con renta y precios? ¿Puede construirse desde lo observable?
\item \textbf{Estructura}: 3 partes: I. fundamentos; II. estática comparativa; III. otros desarrollos.
\end{itemize}

\bloque[11]{I. Elección racional}
\graficorepaso{esquema-ingredientes-repaso}{Ingredientes: de las preferencias a la demanda de mercado}
\begin{itemize}
\item \textbf{Supuestos}: estático, sin incertidumbre; información perfecta; precio-aceptante, bienes divisibles; consumo separado de trabajo e intertemporal; bienes homogéneos.
\end{itemize}

\sub{I.1. Ingredientes}
\begin{itemize}
\item \concepto{Conjunto de elección} $S\subseteq\mathbb{R}^n_+$: no vacío, cerrado, acotado inferiormente, convexo ($\tau\vect{x}'+(1-\tau)\vect{x}''\in S$, $\tau\in[0,1]$).
\item \concepto{Preferencia débil} $\succcurlyeq$ (primitiva) $\to$ estricta $\succ$ e indiferencia $\sim$. Conjuntos «al menos tan bueno como» $B(\vect{x}^0)$, «no mejor que» $H(\vect{x}^0)$; $I(\vect{x}^0)=B\cap H$.
\end{itemize}
\graficorepaso{esquema-axiomas-repaso}{7 axiomas: qué garantiza cada uno y su papel}
\begin{itemize}
\item \concepto{Teorema de \autor{Debreu}}: 1-4 $\Rightarrow$ existe $U$ continua. \concepto{Lexicográficas}: racionales, no continuas, sin utilidad. Deseabilidad: monot.\ estricta $\Rightarrow$ monot.\ $\Rightarrow$ insaciab.\ local $\Rightarrow$ no saturación.
\item \concepto{Función de utilidad}: $\vect{x}^1\succcurlyeq\vect{x}^2\Leftrightarrow U(\vect{x}^1)\geq U(\vect{x}^2)$; «numera» los conjuntos de indiferencia; \concepto{ordinal} (transf.\ monótonas). \concepto{Curvas de indiferencia} = conjuntos de nivel de $U(\vect{x})$: no se cortan (3), pendiente negativa (5), convexas (6).
\end{itemize}
\graficorepaso[0.85\linewidth]{propiedades-curvas-indiferencia}{Propiedades de las curvas de indiferencia y el axioma del que sale cada una}
\graficorepaso[0.65\linewidth]{superficie-utilidad-repaso}{Superficie de utilidad cortada a $U=1,2,3$: su proyección es el mapa de curvas de indiferencia}
\begin{formula}
\centering $\mathrm{RMS}_{12}\equiv-\left.\dfrac{dx_2}{dx_1}\right|_{\overline{U}}=\dfrac{U_1}{U_2}$\\[1pt] $\sigma\equiv\dfrac{d\ln(x_2/x_1)}{d\ln\mathrm{RMS}_{12}}$
\end{formula}
\begin{itemize}
\item \concepto{CES}: $\sigma\to0$ \autor{Leontief}, $\sigma\to1$ \autor{Cobb}-\autor{Douglas}, $\sigma\to\infty$ lineal.
\end{itemize}
\graficorepaso[0.5\linewidth]{ces-curvas-indiferencia}{Curvas de indiferencia CES por $(1,1)$ según $\sigma$}
\begin{itemize}
\item \concepto{Conjunto presupuestario} $B(\vect{p},\overline{W})=\{\vect{x}\in S: \vect{p}\cdot\vect{x}\leq\overline{W}\}$: compacto si $\vect{p}\gg\vect{0}$, convexo. Recta de balance, pendiente $-p_1/p_2$.
\end{itemize}
\graficorepaso{conjunto-presupuestario}{Conjunto presupuestario; $\vect{p}$, perpendicular a la recta; si $p_1\uparrow$, gira hacia dentro}

\sub{I.2. PMU: el equilibrio}
\begin{formula}
\centering $\begin{array}{@{}r@{\quad}l@{}}\displaystyle\max_{\vect{x}} & U(\vect{x})\\[1pt] \text{s.a.} & \vect{p}\cdot\vect{x}\leq\overline{W}\\[1pt] & \vect{x}\geq\vect{0}\end{array}$
\end{formula}
\begin{itemize}
\item Existe (\autor{Weierstrass}), única (estr.\ cuasiconcavidad), CPO suficientes (\autor{Arrow}-\autor{Enthoven}).
\item \concepto{\autor{Kuhn}-\autor{Tucker}}: $U_i-\lambda p_i\leq0$, $x_i\geq0$, holgura complementaria. $\lambda$ = UMg de la renta; $\lambda>0$ por insaciabilidad local (con saturación interior, $\lambda=0$).
\end{itemize}
\begin{formula}
Interior (\autor{Gossen}): $\dfrac{U_1}{p_1}=\dots=\dfrac{U_n}{p_n}=\lambda$\\ $\Rightarrow \mathrm{RMS}_{12}=p_1/p_2$.\ \ Esquina ($x_2^*=0$): $\mathrm{RMS}_{12}\geq p_1/p_2$
\end{formula}
\graficorepaso{solucion-interior-esquina}{Solución interior (tangencia) y de esquina ($x_2^*=0$)}

\sub{I.3. Solución: demanda y mercado}
\begin{itemize}
\item \concepto{Demanda marshalliana} $\vect{x}(\vect{p},\overline{W})$: única, continua (\autor{Berge}), \textbf{homogénea de grado 0} (sin ilusión monetaria), \textbf{ley de \autor{Walras}} $\vect{p}\cdot\vect{x}=\overline{W}$, diferenciable.
\item \concepto{FIU} $V(\vect{p},\overline{W})$: homogénea de grado 0, creciente en $\overline{W}$, no creciente en $\vect{p}$, cuasiconvexa. \autor{Roy}: \mbox{$x_i=-\dfrac{\partial V/\partial p_i}{\partial V/\partial\overline{W}}$}.
\item \concepto{Demanda de mercado}: suma horizontal; depende de la distribución de la renta. Solo de $\sum_h\overline{W}_h$ $\Leftrightarrow$ \concepto{forma polar de \autor{Gorman}} $V_h=a_h(\vect{p})+b(\vect{p})\overline{W}_h$ $\Leftrightarrow$ curvas de \autor{Engel} \textbf{rectas y paralelas} $\Rightarrow$ consumidor representativo. Casos: homotéticas idénticas; cuasilineales (\textbf{efecto renta nulo}: excedente exacto, pero sin inferiores ni \autor{Giffen}).
\end{itemize}
\graficorepaso[0.6\linewidth]{engel-gorman}{\autor{Gorman}: misma pendiente $\Rightarrow$ redistribuir la renta no cambia la demanda agregada}
\begin{itemize}
\item Se heredan homogeneidad y ley de \autor{Walras}, no el ADPR ni \autor{Slutsky}: \concepto{SMD} $\Rightarrow$ ni unicidad ni estabilidad \vertema{3.A.21}.
\end{itemize}

\bloque[8]{II. Estática comparativa}
Comparar óptimos sin el ajuste; \concepto{elasticidades}: números puros y medida \emph{local}.

\sub{II.1. Variaciones en la renta}
\begin{itemize}
\item \concepto{Curva de renta-consumo} $\to$ \concepto{curva de \autor{Engel}} $x_i(\bar p,\overline{W})$; pendiente = efecto renta.
\item $\varepsilon_{x_i,\overline{W}}$: $<0$ \binferior{inferior}; $=0$ independiente; $(0,1)$ \bnormal{normal necesario}; $>1$ \bnormal{normal de lujo}.
\item \concepto{Ley de \autor{Engel}} (1857): alimentos \dato{19,6\,\% (Q1) frente a 12,3\,\% (Q5)}{INE, EPF 2025}.
\end{itemize}
\graficorepaso{renta-consumo-engel}{CRC y curva de \autor{Engel}: \bnormal{normal} hasta cierta renta, \binferior{inferior} después}
\begin{formula}
Agregación de \autor{Engel}: $\sum_i s_i\,\varepsilon_{x_i,\overline{W}}=1$,\ \ $s_i=p_ix_i/\overline{W}$
\end{formula}

\sub{II.2. Variaciones en los precios}
\begin{itemize}
\item \concepto{Curva de precio-consumo} $\to$ \concepto{curva de demanda}. $\varepsilon_{x_i,p_i}$: $<0$ \bordinario{ordinario} ($<-1$ elástica); $=0$ rígida; $>0$ \bgiffen{\autor{Giffen}}.
\item \concepto{Hicksiana} $h_i(\vect{p},\overline{U})$: mantiene la utilidad \vertema{3.A.9}.
\end{itemize}
\begin{formula}
\centering \autor{Slutsky}, con $\overline{U}=V(\vect{p},\overline{W})$:\\[1pt]
$\underbrace{\dfrac{\partial x_i(\vect{p},\overline{W})}{\partial p_j}}_{\mathrm{ET}}=\underbrace{\dfrac{\partial h_i(\vect{p},\overline{U})}{\partial p_j}}_{\mathrm{ES}\,=\,S_{ij}}\underbrace{-\,x_j(\vect{p},\overline{W})\dfrac{\partial x_i(\vect{p},\overline{W})}{\partial\overline{W}}}_{\mathrm{ER}}$\\[1pt]
Con $j=i$: $S_{ii}\leq0$.
\end{formula}
\graficorepaso[0.8\linewidth]{slutsky-propio}{Signos de \autor{Slutsky} con $j=i$}
\begin{itemize}
\item \bnormal{Normal} $\Rightarrow$ \bordinario{ordinario} (\emph{ley de la demanda}). \binferior{Inferior}: \bordinario{ordinario} si domina ES; \bgiffen{\concepto{\autor{Giffen}}} si domina ER (arroz en Hunan, \autor{Jensen}-\autor{Miller}, 2008). \concepto{\autor{Veblen}}: precio en las preferencias, fuera del modelo.
\end{itemize}
\graficorepaso{elasticidades-renta-precio-repaso}{Todo \bnormal{normal} es \bordinario{ordinario}; todo \bgiffen{\autor{Giffen}} es \binferior{inferior}}
\graficorepaso[0.55\linewidth]{efecto-sustitucion-renta}{ES ($A\to B$) y ER ($B\to C$); hicksiana por $A$ y $B$}
\begin{itemize}
\item $\varepsilon_{x_i,p_j}>0$ \bnormal{\concepto{sustitutivo bruto}}, $<0$ \binferior{\concepto{complementario bruto}}: \textbf{no simétricas}. Netos según $S_{ij}=\partial h_i/\partial p_j$: \textbf{simétricos} ($S_{ij}=S_{ji}$); todo bien tiene un \bnormal{sustitutivo neto} (con 2 bienes, ambos lo son).
\end{itemize}
\begin{formula}
\centering Homogeneidad: $\sum_j\varepsilon_{x_i,p_j}+\varepsilon_{x_i,\overline{W}}=0$\\ Agregación de \autor{Cournot}: $\sum_i s_i\,\varepsilon_{x_i,p_j}=-s_j$
\end{formula}

\sub{II.3. Valoración}
\begin{itemize}
\item \concepto{Restricciones contrastables}: homogeneidad y agregaciones (presupuesto); simetría y negatividad (maximización). Límites $\to$ III: preferencias inobservables; sin tiempo; sin características; estático.
\end{itemize}

\bloque[7,5]{III. Otros desarrollos}
\sub{III.1. Preferencia revelada}
\begin{itemize}
\item \autor{Samuelson} (1938, 1948): de las \emph{elecciones} (inductivo), no de preferencias inobservables. Supuestos: consumidor individual, una cesta por $(\vect{p},\overline{W})$, ley de \autor{Walras}.
\end{itemize}
\begin{formula}
\centering \textbf{ADPR}: $\vect{p}^0\cdot\vect{x}^1\leq\vect{p}^0\cdot\vect{x}^0,\ \vect{x}^1\neq\vect{x}^0$\\ $\Rightarrow\ \vect{p}^1\cdot\vect{x}^0>\vect{p}^1\cdot\vect{x}^1$\\[1pt] Ley de la demanda compensada:\\ $(\vect{p}^1-\vect{p}^0)\cdot(\vect{x}^1-\vect{x}^0)\leq0$
\end{formula}
\begin{itemize}
\item ADPR $\Rightarrow$ homogeneidad; \concepto{ley de la demanda compensada} (ES $\leq0$ sin utilidad); matriz de \autor{Slutsky} $[S_{ij}]$ semidefinida negativa, \emph{no} simétrica (con 2 bienes basta, \autor{Rose}).
\end{itemize}
\graficorepaso{adpr-efecto-sustitucion}{ADPR y ley de la demanda compensada: si baja $p_1$ y se compensa la renta, $x_1$ no baja}
\begin{itemize}
\item \concepto{AFPR} (\autor{Houthakker}, 1950): transitividad; AFPR $\Leftrightarrow$ preferencias racionales. \concepto{\autor{Afriat}} (1967): datos finitos, GARP. \autor{Varian} (1982): contraste no paramétrico práctico (hogares, experimentos). Crítica: elección $\neq$ bienestar (\autor{Sen}).
\end{itemize}
\graficorepaso[0.75\linewidth]{afpr}{AFPR: la preferencia revelada indirecta (transitividad)}
\graficorepaso[0.6\linewidth]{preferencia-revelada-indiferencia}{Curva de indiferencia acotada por la preferencia revelada}

\sub{III.2. Producción doméstica}
\begin{itemize}
\item \autor{Becker} (1965): el hogar produce \emph{commodities} $Z_i=f_i(x_i,t_i)$ con bienes y \emph{tiempo}. \autor{Lancaster} (1966): características $\vect{z}=B\vect{x}$ (anexo; \vertema{3.A.18}).
\end{itemize}
\begin{formula}
Renta completa: $\sum_i(p_ix_i+w\,t_i)=wT+\overline{W}$\\ $x_i=b_iZ_i$, $t_i=\ell_iZ_i$ $\Rightarrow$ $\pi_i=p_ib_i+w\,\ell_i$
\end{formula}
\begin{itemize}
\item $w\uparrow\Rightarrow$ se sustituye tiempo por bienes (mujer en el mercado de trabajo, externalización). \autor{Gronau} (1977): trabajo doméstico $\neq$ ocio.
\end{itemize}

\sub{III.3. Precios hedónicos}
\begin{itemize}
\item \concepto{Función hedónica} $p(\vect{z})$; \concepto{precio implícito} $\partial p/\partial z_j$. \autor{Court} (1939), \autor{Griliches} (1961), \autor{Rosen} (1974).
\item \textbf{Supuestos}: mercado competitivo de un producto diferenciado; precio-aceptantes informados; $U(\vect{z},y)$ y una sola unidad.
\item \concepto{\autor{Rosen}}: $\dfrac{U_{z_j}}{U_y}=\dfrac{\partial p(\vect{z})}{\partial z_j}$; $p(\vect{z})$, \textbf{envolvente} de curvas de puja $\Theta$ y de oferta $\phi$ (no es demanda ni oferta).
\item Estimación en 2 etapas; la 2.ª, mal identificada. Aplicaciones: IPC con ajuste de calidad (\autor{Boskin}: sesgo \dato{1,1 p.p./año}{EE.\,UU., 1996}), \textbf{IPV} del INE (desde 2008), bienes sin mercado \vertema{4.B.6}, salarios hedónicos.
\end{itemize}
\graficorepaso[0.65\linewidth]{funcion-hedonica-rosen}{$p(\vect{z})$, envolvente de pujas y ofertas}
\begin{ideaclave}
Los 3 desarrollos acercan la teoría a lo observable: elecciones, coste del tiempo y valor de cada característica.
\end{ideaclave}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}: de unos pocos axiomas a la demanda marshalliana y, con \autor{Gorman}, a la de mercado (I); \autor{Slutsky} explica la clasificación de los bienes (II); 3 desarrollos hacia lo observable (III).

\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Qué axiomas hacen falta para que exista $U$?} Los 1-4 (\autor{Debreu}). Monotonía y convexidad dan sus propiedades; 6$'$ y 7 simplifican.
\item \textbf{¿Puede un bien \bnormal{normal} ser \bgiffen{\autor{Giffen}}?} No: normal $\Rightarrow$ ordinario. \autor{Giffen} exige ser \binferior{inferior} y pesar mucho en el gasto.
\item \textbf{¿Por qué las relaciones netas son simétricas y las brutas no?} $S_{ij}=S_{ji}$ (hessiana de $E$, \autor{Young}); el ER cruzado ($-x_j\,\partial x_i/\partial\overline{W}$) no tiene por qué coincidir con el de $j$ ante $p_i$.
\item \textbf{¿Cuándo hay consumidor representativo?} Forma polar de \autor{Gorman}: curvas de \autor{Engel} rectas y paralelas.
\item \textbf{¿Equivalen preferencia revelada y teoría ordinal?} Sí con el AFPR (\autor{Houthakker}); el ADPR solo basta con 2 bienes.
\item \textbf{¿Por qué $p(\vect{z})$ no es una curva de demanda?} Es la envolvente de pujas y ofertas: un lugar de equilibrios.
\item \textbf{¿Qué restricciones son contrastables?} Homogeneidad, agregaciones de \autor{Engel} y \autor{Cournot} (presupuesto), simetría y negatividad de \autor{Slutsky} (maximización).
\item \textbf{¿Por qué la demanda es homogénea de grado 0?} Si se multiplican $\vect{p}$ y $\overline{W}$ por $\kappa>0$ el conjunto presupuestario no cambia: sin ilusión monetaria; por \autor{Euler}, $\sum_j\varepsilon_{x_i,p_j}+\varepsilon_{x_i,\overline{W}}=0$.
\item \textbf{¿Qué aporta \autor{Becker}?} El coste de oportunidad del tiempo: el salario es un precio del consumo y, si sube, se sustituye tiempo por bienes.
\item \textbf{¿Qué es el efecto dotación?} Se pide por vender más de lo que se pagaría por comprar (tazas: más del doble; \autor{Kahneman}, \autor{Knetsch} y \autor{Thaler}, 1990): las curvas de indiferencia dependerían de la dotación.
\end{enumerate}

\begin{trampa}
\begin{itemize}
\item La \textbf{convexidad} no exige UMg decrecientes ni estas la garantizan; la UMg decreciente no es ordinal ($x_1x_2$ y $(x_1x_2)^2$): importa la RMS decreciente.
\item Cuasiconcavidad y RMS son ordinales; concavidad y signo de $U_{ii}$, cardinales.
\item Para que exista $U$ \emph{no} hacen falta monotonía, convexidad ni diferenciabilidad.
\item Insaciabilidad local basta para la ley de \autor{Walras} ($\lambda>0$).
\item Signo de \autor{Slutsky}: ER $=-x_i\,\partial x_i/\partial\overline{W}$, opuesto al de $\partial x_i/\partial\overline{W}$. Todo \bgiffen{\autor{Giffen}} es \binferior{inferior}, no al revés. \autor{Veblen} $\neq$ \autor{Giffen}.
\item ADPR $\Rightarrow$ \autor{Slutsky} semidefinida negativa, \emph{no} simétrica.
\item La agregación conserva homogeneidad y ley de \autor{Walras}, no el ADPR.
\item Esquina: $\mathrm{RMS}_{12}\geq p_1/p_2$, no igualdad.
\item La «ley de la escasez» no la formula \autor{Gossen}.
\item La clasificación de un bien es \emph{local}: normal a una renta, inferior a otra.
\item \autor{Gorman}: curvas de \autor{Engel} rectas y \emph{paralelas}; que pasen por el origen solo en las homotéticas.
