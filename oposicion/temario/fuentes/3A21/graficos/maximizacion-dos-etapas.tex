% Maximización del beneficio en 2 etapas (3.A.21, anexo A).
% Coste (para unos precios de los factores w dados): C(q) = 4 + 2,5q - 0,8q^2 + 0,1q^3.
% CMg = 2,5 - 1,6q + 0,3q^2 (mínimo en q = 2,67); CMe = 4/q + 2,5 - 0,8q + 0,1q^2
%   (mínimo en q = 4,85, CMe = 1,797, donde lo corta el CMg).
% Precio p = 4: p = CMg en el tramo creciente en q* = 6,147; CMe(q*) = 2,012;
%   beneficio B* = (4 - 2,012)·6,147 = 12,22.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.8cm,y=0.8cm,
    declare function={cme(\q)=4/\q+2.5-0.8*\q+0.1*\q^2; cmg(\q)=2.5-1.6*\q+0.3*\q^2;}]
  % 1: ejes
  \capa{1}{\ejes{$q$}{$p,\ \mathrm{CMg},\ \mathrm{CMe}$}{8.6}{6.4}}
  % 2: coste medio en U
  \capa{2}{\draw[curva1] plot[domain=1:7.9,samples=100] (\x,{cme(\x)})
    node[etiqueta,right] {$\mathrm{CMe}(\vect{w},q)$};}
  % 3: coste marginal, que corta al CMe en su mínimo
  \capa{3}{\draw[curva2] plot[domain=0.5:6.8,samples=100] (\x,{cmg(\x)})
    node[etiqueta,above] {$\mathrm{CMg}(\vect{w},q)$};}
  % 4: precio y punto E (p = CMg en el tramo creciente)
  \capa{4}{\draw[curva4] (0,4) node[etiqueta,left,text=black] {$p$} -- (8.3,4);}
  \capa{4}{\node[punto,label={[etiqueta]above left:$E$}] (E) at (6.147,4) {};}
  % 5: proyecciones: q* = q(p, w) y CMe(w, q*)
  \capa{5}{\draw[guia] (E) -- (6.147,0) node[below,text=black] {$q^*=q(p,\vect{w})$};}
  \capa{5}{\node[punto] (C) at (6.147,2.012) {};}
  \capa{5}{\draw[guia] (C) -- (0,2.012) node[left,text=black] {$\mathrm{CMe}(\vect{w},q^*)$};}
  % 6: beneficio B* (rectángulo entre p y CMe(q*) hasta q*)
  \capa{6}{\begin{scope}[on background layer]
      \fill[pattern=north east lines,pattern color=tceemorado!55] (0,2.012) rectangle (6.147,4);
    \end{scope}}
  \capa{6}{\draw[->,thin] (3.4,5.3) node[etiqueta,above] {$B^*=B(p,\vect{w})$} -- (3.6,3.3);}
\end{tikzpicture}
\end{document}
