% Estabilidad en un mercado: ajuste walrasiano (precios) y marshalliano (cantidades)
% con oferta y demanda decrecientes (3.A.21, anexo D).
% Equilibrio en (q*, p*) = (3, 3) en los 2 paneles. Rectas: inclinada p = 3 - 1,5(q - 3),
%   plana p = 3 - 0,3(q - 3).
% Izquierda: D inclinada, S plana. En p = 3,6: x = 2,6 > q = 1 (z > 0, p sube:
%   se aleja); en p = 2,55: x = 3,3 < q = 4,5 (z < 0, p baja: se aleja). Walras inestable.
%   En q = 0,9: p^D = 6,15 > p^S = 3,63 (q sube); en q = 4,9: p^D = 0,15 < p^S = 2,43
%   (q baja). Marshall estable.
% Derecha: S inclinada, D plana: al revés (Walras estable, Marshall inestable).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.75cm,y=0.75cm,
    declare function={inc(\q)=3-1.5*(\q-3); pla(\q)=3-0.3*(\q-3);},
    estable/.style={->,thick,tceeverde},
    inestable/.style={->,thick,tceerojo}]
  % ---------------- Panel izquierdo: oferta más plana que la demanda ----------------
  % 1: ejes
  \capa{1}{\ejes{$q$}{$p$}{6.6}{6.6}}
  \capa{1}{\node[font=\small\bfseries,align=center] at (3.3,7.4) {Oferta más plana que la demanda};}
  % 2: demanda (inclinada) y oferta (plana), ambas decrecientes, y el equilibrio
  \capa{2}{\draw[curva1] (0.9,6.15) node[etiqueta,above] {$D$} -- (5,0);}
  \capa{2}{\draw[curva3] (0,3.9) -- (6,2.1) node[etiqueta,right] {$S$};}
  \capa{2}{\node[punto] (E1) at (3,3) {};}
  \capa{2}{\proyeccion{(E1)}{$q^*$}{$p^*$}}
  % 3: ajuste walrasiano (por precios): el exceso de demanda aleja el precio
  \capa{3}{\draw[guia,solid,semithick] (1,3.6) -- (2.6,3.6) node[midway,above,etiqueta,text=black] {$z>0$};}
  \capa{3}{\draw[guia,solid,semithick] (3.3,2.55) -- (4.5,2.55) node[pos=0.8,below,etiqueta,text=black] {$z<0$};}
  \capa{3}{\draw[inestable] (0.2,3.1) -- (0.2,3.7);}
  \capa{3}{\draw[inestable] (0.2,2.9) -- (0.2,2.2);}
  \capa{3}{\node[etiqueta,text=tceerojo,anchor=west] at (3.2,5.7) {\textsc{Walras}: inestable};}
  % 4: ajuste marshalliano (por cantidades): la diferencia p^D - p^S acerca la cantidad
  \capa{4}{\draw[guia,solid,semithick] (0.9,3.63) -- (0.9,6.15);}
  \capa{4}{\draw[guia,solid,semithick] (4.9,0.15) -- (4.9,2.43);}
  \capa{4}{\draw[estable] (2.2,0.25) -- (2.85,0.25);}
  \capa{4}{\draw[estable] (3.8,0.25) -- (3.15,0.25);}
  \capa{4}{\node[etiqueta,text=tceeverde,anchor=west] at (3.2,5.1) {\textsc{Marshall}: estable};}
  % ---------------- Panel derecho: oferta más inclinada que la demanda ----------------
  \begin{scope}[xshift=7.4cm]
    % 5: ejes
    \capa{5}{\ejes{$q$}{$p$}{6.6}{6.6}}
    \capa{5}{\node[font=\small\bfseries,align=center] at (3.3,7.4) {Oferta más inclinada que la demanda};}
    % 6: oferta (inclinada) y demanda (plana) y el equilibrio
    \capa{6}{\draw[curva1] (0,3.9) -- (6,2.1) node[etiqueta,right] {$D$};}
    \capa{6}{\draw[curva3] (0.9,6.15) node[etiqueta,above] {$S$} -- (5,0);}
    \capa{6}{\node[punto] (E2) at (3,3) {};}
    \capa{6}{\proyeccion{(E2)}{$q^*$}{$p^*$}}
    % 7: ajuste walrasiano: el exceso de oferta (demanda) baja (sube) el precio hacia p*
    \capa{7}{\draw[guia,solid,semithick] (1,3.6) -- (2.6,3.6) node[midway,above,etiqueta,text=black] {$z<0$};}
    \capa{7}{\draw[guia,solid,semithick] (3.3,2.55) -- (4.5,2.55) node[pos=0.8,below,etiqueta,text=black] {$z>0$};}
    \capa{7}{\draw[estable] (0.2,3.7) -- (0.2,3.1);}
    \capa{7}{\draw[estable] (0.2,2.2) -- (0.2,2.9);}
    \capa{7}{\node[etiqueta,text=tceeverde,anchor=west] at (3.2,5.7) {\textsc{Walras}: estable};}
    % 8: ajuste marshalliano: la cantidad se aleja de q*
    \capa{8}{\draw[guia,solid,semithick] (0.9,3.63) -- (0.9,6.15);}
    \capa{8}{\draw[guia,solid,semithick] (4.9,0.15) -- (4.9,2.43);}
    \capa{8}{\draw[inestable] (2.85,0.25) -- (2.2,0.25);}
    \capa{8}{\draw[inestable] (3.15,0.25) -- (3.8,0.25);}
    \capa{8}{\node[etiqueta,text=tceerojo,anchor=west] at (3.2,5.1) {\textsc{Marshall}: inestable};}
  \end{scope}
\end{tikzpicture}
