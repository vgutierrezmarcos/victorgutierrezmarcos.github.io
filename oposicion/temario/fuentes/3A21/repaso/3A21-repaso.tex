\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: «cada uno arrima el ascua a su sardina».
  \begin{itemize}
  \item \autor{Bastiat} (\emph{Sofismas}, 1845): París, 1~millón de habitantes, se abastece cada día sin que nadie lo organice.
  \item ¿Planes compatibles solo con interés propio y precios? ¿Es bueno el resultado? Mano invisible (\autor{Smith}, 1776).
  \item \concepto{Equilibrio parcial} \vertema{3.A.16} frente a \concepto{general}: todos los mercados, interdependientes, a la vez.
  \end{itemize}
\item \textbf{Relevancia}: \emph{teórica} (bienestar, H-O, incidencia); \emph{práctica} (MEGC); \emph{de política}: cuándo coordinan los precios y cuándo no \vertema{3.A.23}.
\item \textbf{Contextualización}:
  \begin{itemize}
  \item \autor{Quesnay} (1758), \autor{Smith} $\to$ \autor{L. Walras} (1874): \emph{rareté}, tanteo, numerario; cuenta ecuaciones;
  \item \autor{Edgeworth} (1881), \autor{Pareto} (1906) $\to$ \autor{Cassel} (1918), \autor{Wald} (1935-36), von~\autor{Neumann} (1937);
  \item \autor{Arrow}-\autor{Debreu} y \autor{McKenzie} (1954), \autor{Debreu} (1959) $\to$ límites: estabilidad, núcleo, SMD, \autor{Radner}.
  \end{itemize}
\item \textbf{Problemática}: ¿qué es un EGC? ¿Existe, es único, es estable? ¿Es eficiente, justo? ¿Y al relajar supuestos?
\item \textbf{Estructura}: coordinación \emph{posible}, \emph{determinada}, \emph{alcanzable}, \emph{deseable}, \emph{robusta}:
  \begin{itemize}
  \item \hyperref[rep:I]{I}, modelo; \hyperref[rep:II]{II}, propiedades; \hyperref[rep:III]{III}, cada extensión relaja un supuesto de \hyperref[rep:I]{I} que \hyperref[rep:II]{II} necesitaba.
  \end{itemize}
\end{itemize}

\bloque[11]{I. Construcción del modelo\label{rep:I}}
\begin{ideaclave}
Si cada agente optimiza a precios dados, todo el sistema se resume en $\vect{z}(\vect{p})$; el equilibrio vacía todos los mercados a la vez y la ley de \autor{Walras} deja solo precios relativos.
\end{ideaclave}

\sub{I.1. Supuestos\label{rep:I.1}}
\begin{itemize}
\item \concepto{Equilibrio general à la \autor{Walras}}: tantos mercados como mercancías; en cada uno, demanda, oferta y condición de vaciado.
\item \concepto{Competencia perfecta} (racionales, información perfecta, homogéneos, precio-aceptantes): $\vect{p}=(p_X,p_Y,w_L,w_K)$; \concepto{mercados completos}.
\item \concepto{2×2×2×2~estático} (\autor{Bator}, 1957):
  \begin{itemize}
  \item 2~bienes $X,Y$; 2~factores $L,K$, con $\overline{L},\overline{K}$ de los consumidores;
  \item 2~consumidores: preferencias completas, transitivas, continuas, estr.\ monótonas y estr.\ convexas;
  \item 2~empresas, $F_j$ neoclásica, conjunto \emph{convexo}: rendim.\ constantes o decrecientes; crecientes, fuera (\hyperref[rep:III.2]{III.2});
  \item renta de cada consumidor:
\begin{formula}
\centering $\overline{W}_i=w_L\overline{L}_i+w_K\overline{K}_i+\sum_j\theta_{ij}B_j$,\quad $\sum_i\theta_{ij}=1$
\end{formula}
  \end{itemize}
\item \concepto{Variables}: exógenas, preferencias, tecnologías y dotaciones; endógenas, 4~precios (3~relativos) y cantidades.
\item \concepto{Tanteo} (\emph{tâtonnement}): precio $\uparrow$ con exceso de demanda, $\downarrow$ con exceso de oferta; ajustan los precios.
\end{itemize}

\sub{I.2. Desarrollo\label{rep:I.2}}
\subsub{a) Productores y oferta}
\begin{itemize}
\item \emph{Problema}: cada empresa maximiza el beneficio a precios dados:
\begin{formula}
\centering $\optim{\max}{L_j,K_j}{p_jF_j(L_j,K_j)-w_LL_j-w_KK_j}{L_j\geq0,\ K_j\geq0}$
\end{formula}
\item \emph{CPO}: $p_j\mathrm{PMg}^j_L=w_L$, $p_j\mathrm{PMg}^j_K=w_K$ $\Rightarrow$ tasa subjetiva = objetiva:
\begin{formula}
\centering $\mathrm{RMST}^X_{LK}=\dfrac{w_L}{w_K}=\mathrm{RMST}^Y_{LK}$;\qquad $\mathrm{RMT}_{XY}=\dfrac{\mathrm{CMg}_X}{\mathrm{CMg}_Y}=\dfrac{p_X}{p_Y}$
\end{formula}
\item \emph{Resultado}: $L_j,K_j,q_j(p_j,\vect{w})$; con RCE, la escala la fija el pleno empleo.
\item \emph{Interpretación gráfica}:
  \begin{itemize}
  \item isocoste común tangente a las 2~isocuantas $\Rightarrow$ \concepto{curva de contrato} (eficiencia productiva) $\to$ \concepto{FPP};
  \item $p_X/p_Y$ elige el punto de la FPP.
  \end{itemize}
\end{itemize}
\graficorepaso[0.8\linewidth]{caja-produccion-precios}{Caja de la producción: isocoste $w_L/w_K$ en la curva de contrato; $p_X/p_Y$ tangente a la FPP}
\subsub{b) Consumidores y demanda}
\begin{itemize}
\item \emph{Problema}: $\max U_i$ s.a.\ $p_XX_i+p_YY_i=\overline{W}_i$ $\Rightarrow$ $\mathrm{RMS}^A_{XY}=p_X/p_Y=\mathrm{RMS}^B_{XY}$.
\item \emph{Resultado}: demandas de $\vect{p}$ únicas, continuas, homogéneas de grado 0 (sin ilusión monetaria) y que agotan la renta (\autor{Walras} individual); ofrecen toda la dotación.
\end{itemize}
\subsubsub{La demanda de mercado}
\begin{itemize}
\item \emph{Suma horizontal} de las individuales, con la renta de cada uno a esos precios.
\item \emph{Hereda}: continuidad, homogeneidad 0, ley de \autor{Walras}, cota inferior $\Rightarrow$ bastan para la existencia.
\item \emph{No hereda}: independencia del reparto de la renta (salvo \autor{Gorman}), ley de la demanda, \autor{Slutsky}, axioma débil.
\item \concepto{Efecto renta-dotación}: si $p_k\uparrow$, el vendedor neto ($x_{ik}<\omega_{ik}$) se enriquece $\Rightarrow$ su demanda puede $\uparrow$ sin ser \autor{Giffen}.
\end{itemize}
\subsub{c) Función de exceso de demanda}
\begin{itemize}
\item \concepto{Exceso de demanda}: demanda $-$ oferta ($z_k>0$ falta, $<0$ sobra):
\funcionclave{\vect{z}(\vect{p})=\sum_{i}\vect{x}_i\big(\vect{p},\overline{W}_i(\vect{p})\big)-\sum_{i}\vect{\omega}_i-\sum_{j}\vect{y}_j(\vect{p})}
\item \emph{Ventaja}: todo el modelo en función de los precios.
\item \emph{Propiedades}: (1) continua; (2) homogénea de grado 0; (3) ley de \autor{Walras}; (4) acotada inferiormente y \emph{condición de frontera} ($z_k\to\infty$ si $p_k\to0$).
\item \emph{Tipos de bienes} (\autor{Schlesinger}, \autor{Wald}, 1935):
  \begin{itemize}
  \item \concepto{económicos}: $p^*_k>0$, $z_k(\vect{p}^*)=0$ ($z_k$ corta el eje);
  \item \concepto{libres}: $p^*_k=0$, $z_k(\vect{p}^*)\leq0$ (el aire; no corta).
  \end{itemize}
\end{itemize}
\graficorepaso[0.6\linewidth]{bienes-libres}{Bien económico y bien libre: mercado y exceso de demanda}
\subsub{d) Ley de \autor{Walras}}
\begin{itemize}
\item \emph{Enunciado}: a \emph{cualquier} precio, el valor de los excesos de demanda suma 0:
\funcionclave{\vect{p}\cdot\vect{z}(\vect{p})\equiv\sum_{k=1}^{n}p_k\,z_k(\vect{p})=0}
\item \emph{Demostración}: sumar las restricciones presupuestarias saturadas (no saciedad local) con $\sum_i\theta_{ij}=1$.\findemo
\item \emph{Implicaciones}:
  \begin{itemize}
  \item mercados interdependientes; si $n-1$ se vacían y $p_n>0$, el $n$-ésimo también;
  \item con homogeneidad, solo $n-1$ precios relativos: \concepto{numerario} ($p_1=1$) o símplice; sin dinero, precios absolutos indeterminados.
  \end{itemize}
\item \emph{Naturaleza}: \emph{identidad}, no condición de equilibrio; $\neq$ ley de \autor{Say} (\autor{Lange}, 1942: bienes sin dinero $\equiv0$).
\end{itemize}

\sub{I.3. Equilibrio general competitivo\label{rep:I.3}}
\begin{itemize}
\item \concepto{EGC} (walrasiano): $\vect{p}^*\geq\vect{0}$ y $(\vect{x}^*_i,\vect{y}^*_j)$; consumidores y empresas optimizan y todos los mercados se vacían:
\funcionclave{z_k(\vect{p}^*)\leq0\quad\text{y}\quad p^*_k\,z_k(\vect{p}^*)=0,\quad k=1,\dots,n}
\item \emph{En el 2×2×2×2}: $\mathrm{RMS}^A=\mathrm{RMS}^B=p_X/p_Y=\mathrm{RMT}$; $\mathrm{RMST}^X=\mathrm{RMST}^Y=w_L/w_K$.
\item \emph{Formación de precios}: \concepto{subastador walrasiano}:
  \begin{itemize}
  \item anuncia precios, recoge planes, sube donde falta y baja donde sobra; \emph{sin transacciones fuera del equilibrio};
  \item \autor{Walras} no habla de subastador («\emph{on}»): lo personifica la literatura.
  \end{itemize}
\end{itemize}
\subsub{Interpretación gráfica: la caja de \autor{Edgeworth} del intercambio}
\begin{itemize}
\item Lados $\overline{X},\overline{Y}$; recta de precios por $\vect{\omega}$; al girarla, las tangencias dan las \concepto{curvas de oferta-demanda} $\mathrm{OD}_A$, $\mathrm{OD}_B$.
\item Fuera del equilibrio: exceso de demanda de $X$ = exceso de oferta de $Y$ de igual valor $\Rightarrow$ $p_X/p_Y\uparrow$.
\item \emph{Equilibrio} $E$: $\mathrm{OD}_A\cap\mathrm{OD}_B$; $U^*_A$ y $U^*_B$ tangentes, dentro de la \concepto{lente}.
\end{itemize}
\graficorepaso[0.7\linewidth]{caja-intercambio-equilibrio}{Curvas de oferta-demanda y equilibrio $E$ en la caja de intercambio}
\begin{itemize}
\item \emph{Valoración}: el subastador es la pieza más irreal $\Rightarrow$ III busca otros fundamentos.
\end{itemize}

\bloque[9]{II. Análisis positivo y normativo\label{rep:II}}
\begin{ideaclave}
Con supuestos razonables el EGC \emph{existe}; solo con supuestos fuertes es \emph{único} y \emph{estable} (SMD); en cambio, es \emph{eficiente} con muy pocos.
\end{ideaclave}

\begin{itemize}
\item 3~preguntas, cada una presupone la anterior: \concepto{existencia} (¿algún $\vect{p}$ vacía todo?), \concepto{unicidad} (¿solo uno? si no, no predice) y \concepto{estabilidad} (¿lo alcanza el mercado?).
\end{itemize}
\sub{II.1. Análisis positivo\label{rep:II.1}}
\subsub{a) Existencia\label{rep:II.1.1}}
\begin{itemize}
\item \emph{Idea}: \autor{Walras} cuenta ecuaciones e incógnitas: ni necesario ni suficiente. \autor{Wald} (1935-36), primeras pruebas; \autor{Arrow}-\autor{Debreu} y \autor{McKenzie} (1954), punto fijo.
\item \emph{Supuestos}: $\vect{z}$ continua, homogénea de grado 0~y ley de \autor{Walras} (detrás: convexidad y no saciedad local).
\item \concepto{Teorema de \autor{Brouwer}} (1911): $\Delta$ no vacío, compacto y convexo, $\vect{f}:\Delta\to\Delta$ continua $\Rightarrow$ $\exists\,\vect{p}^*=\vect{f}(\vect{p}^*)$. Correspondencias: \autor{Kakutani} (1941).
\end{itemize}
\graficorepaso[0.42\linewidth]{brouwer}{\autor{Brouwer} en 1~dimensión: la gráfica corta la diagonal}
\begin{itemize}
\item \emph{Desarrollo (intuición en 3~pasos)}:
  \begin{enumerate}
  \item \emph{conjunto}: por homogeneidad, \concepto{símplice} $\Delta=\{\vect{p}\geq\vect{0}:\sum_kp_k=1\}$;
  \item \emph{función}: el subastador sube los precios con exceso de demanda y baja los demás sin salir de $\Delta$:
\begin{formula}
\centering $f_k(\vect{p})=\dfrac{\max\{0,\,p_k+\kappa_kz_k(\vect{p})\}}{\sum_l\max\{0,\,p_l+\kappa_lz_l(\vect{p})\}}$
\end{formula}
  \item \emph{punto fijo = equilibrio}: ningún precio se mueve; precio nulo $\Rightarrow$ $z_k\leq0$ (libre); por \autor{Walras}, precio positivo $\Rightarrow$ $z_k=0$.
  \end{enumerate}
\item \emph{Implicaciones}: coordinación por precios \emph{lógicamente posible}; prueba \emph{no constructiva}.
\item \emph{Valoración}: cruciales la convexidad y la ausencia de poder de mercado (\hyperref[rep:III.2]{III.2}); nada dice del número ni de cómo se alcanza.
\end{itemize}
\subsub{b) Unicidad\label{rep:II.1.2}}
\begin{itemize}
\item \emph{Idea}: existencia $\not\Rightarrow$ unicidad: con $p_Y=1$, $z_X$ corta un nº impar de veces (1, 3…).
  \begin{itemize}
  \item \concepto{Economías regulares} (\autor{Debreu}, 1970): salvo medida nula, nº finito; impar (\concepto{teorema del índice}, \autor{Dierker}, 1972).
  \end{itemize}
\end{itemize}
\graficorepaso[0.5\linewidth]{equilibrios-multiples}{$z_X$ con 3~equilibrios; tanteo: extremos estables, intermedio inestable}
\begin{itemize}
\item \emph{Supuestos}: los de la existencia + frontera + forma de $\vect{z}$ (2~condiciones suficientes):
  \begin{itemize}
  \item \concepto{sustitutivos brutos}: $\partial z_k/\partial p_l>0$ $\forall l\neq k$; «brutos»: con efectos renta y renta-dotación;
  \item \concepto{axioma débil agregado}: la demanda de mercado como la de un consumidor coherente.
  \end{itemize}
\item \emph{Intuición (sustitutivos brutos)}:
  \begin{itemize}
  \item con 2~equilibrios, el bien que \emph{menos} se encarece cuesta lo mismo y los demás más $\Rightarrow$ los agentes se pasan a él, $z_k>0$: no es equilibrio;
  \item con 2~bienes, $z_X(p_X,1)$ siempre decreciente.
  \end{itemize}
\item \emph{Intuición (axioma débil)}: el «consumidor agregado» elegiría $A$ pudiendo pagar $B$ y viceversa.
\item \emph{Condición común} (MWG §17.F), base también de la estabilidad:
\begin{formula}
\centering $\vect{p}^*\cdot\vect{z}(\vect{p})>0\quad\forall\,\vect{p}\not\propto\vect{p}^*$
\end{formula}
\item \emph{Valoración}: condiciones muy restrictivas (excluyen complementarios; que no se redistribuya renta).
\item \concepto{Teorema SMD} (\autor{Sonnenschein} 1972-73, \autor{Mantel} 1974, \autor{Debreu} 1974):
  \begin{itemize}
  \item toda función continua, homogénea de grado 0~y con ley de \autor{Walras} es el exceso de demanda de alguna economía (precios $>0$);
  \item la racionalidad individual no impone nada más: cualquier nº impar de equilibrios y cualquier dinámica.
  \end{itemize}
\end{itemize}
\subsub{c) Estabilidad\label{rep:II.1.3}}
\begin{itemize}
\item \emph{Idea}: fuerzas que llevan al equilibrio: \concepto{local} o \concepto{global}; argumento para la mano invisible (si no, estabilización, \autor{Musgrave}); ajuste \concepto{walrasiano} (precios).
\item \emph{Supuestos}: tanteo continuo, sin intercambios fuera del equilibrio:
\begin{formula}
\centering $\dot p_k=\kappa_k\,z_k(\vect{p}),\quad\kappa_k>0$\\[3pt]
$\dfrac{d}{dt}\Big[\tfrac12\sum_k\dfrac{(p_k-p^*_k)^2}{\kappa_k}\Big]=\underbrace{\vect{p}\cdot\vect{z}(\vect{p})}_{=0}-\vect{p}^*\cdot\vect{z}(\vect{p})$
\end{formula}
\item \emph{Intuición}: \concepto{función de \autor{Liapunov}}:
  \begin{itemize}
  \item si la distancia baja siempre fuera del equilibrio, se llega;
  \item baja si $\vect{p}^*\cdot\vect{z}(\vect{p})>0$: la condición de la unicidad.
  \end{itemize}
\item \emph{Implicaciones}: \concepto{\autor{Arrow}-\autor{Block}-\autor{Hurwicz}} (1959): sustitutivos brutos $\Rightarrow$ \emph{globalmente estable}.
\item \emph{Valoración}: \concepto{contraejemplo de \autor{Scarf}} (1960): 3~bienes complementarios, equilibrio único que el tanteo no alcanza; tanteo ficticio; desequilibrio (\autor{Clower}, \autor{Leijonhufvud}) \vertema{3.A.4}.
\end{itemize}

\sub{II.2. Análisis normativo\label{rep:II.2}}
\subsub{a) Optimalidad de \autor{Pareto}}
\begin{itemize}
\item \concepto{Óptimo de \autor{Pareto}} (1906): asignación factible que no admite mejorar a uno sin empeorar a otro.
  \begin{itemize}
  \item Ordinal, pocos juicios de valor; \emph{incompleto}: infinitos óptimos. En la caja, \concepto{curva de contrato} $O_AO_B$.
  \item \autor{Lerner} (1934), \autor{Lange} (1942); \autor{Arrow} y \autor{Debreu} (1951).
  \end{itemize}
\end{itemize}
\graficorepaso[0.45\linewidth]{caja-intercambio-teoremas}{1TFEB: $E$ en la curva de contrato; 2TFEB: de $\vect{\omega}$ a $\vect{\omega}'$ para alcanzar $E'$}
\subsub{b) Teoremas fundamentales de la economía del bienestar}
\begin{itemize}
\item \concepto{1TFEB}: con no saciedad local, \emph{todo EGC es óptimo de \autor{Pareto}}; sin convexidad; todos igualan sus relaciones marginales a los mismos precios.
  \begin{itemize}
  \item \emph{Supuestos}: mercados completos, precio-aceptantes, precios comunes (sin impuestos), información perfecta y simétrica; si fallan, fallos de mercado \vertema{3.A.23} \vertema{3.A.13}.
  \item Mano invisible formalizada; no dice que sea \emph{justo}.
  \end{itemize}
\item \concepto{2TFEB}: con convexidad, \emph{todo óptimo es alcanzable como EGC} tras redistribución de suma fija (de $\vect{\omega}$ a $\vect{\omega}'$).
  \begin{itemize}
  \item Separa eficiencia y equidad; pero la suma fija no es factible (información privada) $\Rightarrow$ vuelve el dilema \vertema{3.A.22}.
  \end{itemize}
\end{itemize}

\bloque[6,5]{III. Otros desarrollos y aplicaciones\label{rep:III}}
\begin{ideaclave}
El equilibrio walrasiano resiste quitar el subastador con muchos agentes (núcleo), pero no el poder de mercado, las no convexidades, el dinero sin valor intrínseco ni los mercados incompletos.
\end{ideaclave}
\begin{itemize}
\item Cada extensión relaja un supuesto de \hyperref[rep:I]{I} que \hyperref[rep:II]{II} necesitaba.
\end{itemize}

\sub{III.1. Intercambio de \autor{Edgeworth} (1881)\label{rep:III.1}}
\begin{itemize}
\item \emph{Idea}: sin precios dados; negociación libre con recontratación (\emph{recontracting}).
\item \emph{Desarrollo}: \emph{racionalidad individual} (dentro de la lente) + \emph{eficiencia} (curva de contrato) $\Rightarrow$ tramo $CC'$.
\item \emph{Implicaciones}: con 2~agentes, \emph{indeterminado} (poder de negociación); qué punto: \concepto{teoría de la negociación} (\autor{Nash}, \autor{Rubinstein}) \vertema{3.A.14}.
\item \emph{Valoración}: conjetura de \autor{Edgeworth}: con más agentes, el tramo se estrecha hasta $E$.
\end{itemize}

\sub{III.2. Núcleo, estrategia y no convexidades\label{rep:III.2}}
\subsub{a) El núcleo (\autor{Debreu}-\autor{Scarf}, 1963)}
\begin{itemize}
\item \emph{Idea}: fundamentar el equilibrio sin subastador con juegos cooperativos (\autor{Gillies}, 1959; \autor{Shubik}, 1959); utilidad no transferible.
\item \emph{Supuestos}: $I$ consumidores; cualquier \concepto{coalición} reparte sus dotaciones; acuerdos vinculantes y sin coste.
\item \emph{Desarrollo}: una coalición \concepto{bloquea} si con sus recursos mejora a alguno sin empeorar a ninguno; \concepto{núcleo}: asignaciones no bloqueadas ($CC'$).
\end{itemize}
\graficorepaso[0.58\linewidth]{caja-intercambio-nucleo}{Núcleo: tramo $CC'$ que se encoge hacia $E$ al replicar la economía}
\begin{itemize}
\item \emph{Implicaciones}:
  \begin{enumerate}
  \item el equilibrio walrasiano está en el núcleo (no saciedad local);
  \item \concepto{teorema del límite}: replicando $r$ veces (preferencias estr.\ convexas), el núcleo se encoge hasta los equilibrios walrasianos: conjetura de \autor{Edgeworth}; \autor{Aumann} (1964), continuo de agentes: núcleo = equilibrios;
  \item precio-aceptación como \emph{resultado}.
  \end{enumerate}
\item \emph{Extensiones}: \concepto{valor de \autor{Shapley}}, mismas asignaciones (\autor{Aumann}, 1975).
\item \emph{Valoración}: exige muchos agentes, convexidad, coaliciones gratis; no es un proceso.
\end{itemize}
\subsub{b) Comportamiento estratégico}
\begin{itemize}
\item Pocos agentes $\Rightarrow$ influyen en los precios: monopolio en la caja ($B$ elige $M$ sobre $\mathrm{OD}_A$), ineficiente.
\item \autor{Cournot}-\autor{Walras} (\autor{Gabszewicz}-\autor{Vial}, 1972): depende del numerario; el poder se diluye al crecer la economía (\autor{Roberts}-\autor{Postlewaite}, 1976).
\end{itemize}
\subsub{c) No convexidades y equilibrios múltiples}
\begin{itemize}
\item Sin convexidad, saltos:
  \begin{itemize}
  \item preferencias no convexas: con muchos consumidores, equilibrios aproximados (\autor{Starr}, 1969);
  \item \emph{rendim.\ crecientes}: $\mathrm{CMg}<\mathrm{CMe}$ $\Rightarrow$ pérdidas con $p=\mathrm{CMg}$ $\to$ monopolio natural \vertema{3.A.17}; no se diluyen.
  \end{itemize}
\item \emph{Equilibrios múltiples}: \concepto{fallos de coordinación} (nadie invierte porque nadie compra; \autor{Cooper}-\autor{John}, 1988) \vertema{3.A.4}.
\end{itemize}

\sub{III.3. Dinero (\autor{Patinkin}, 1956)\label{rep:III.3}}
\begin{itemize}
\item \emph{Idea}: en \autor{Arrow}-\autor{Debreu}, solo numerario: no da utilidad ni hace falta para cambiar (subastador) ni para guardar valor (todo se acuerda hoy).
\item \emph{Conceptos}: \concepto{dicotomía clásica} (real $\to$ precios relativos; monetario $\to$ nivel de precios; dinero, velo); \concepto{neutralidad}: $\Delta M$ sin efectos reales.
\item \emph{Desarrollo}:
  \begin{enumerate}
  \item \autor{Walras} con dinero:
\begin{formula}
\centering $\sum_kp_kz_k(\vect{p})+\big(M^D(\vect{p})-\overline{M}\big)\equiv0$
\end{formula}
  \item dicotomía: postulado de homogeneidad + regla de Cambridge $M^D_i=\varphi\,\vect{p}\cdot\vect{\omega}_i$;
  \item inválida (\autor{Lange}, \autor{Patinkin}): duplicar $\vect{p}$ vacía los bienes pero no el dinero;
  \item \concepto{efecto saldos reales}: $\overline{M}_i/P$ en la demanda de bienes.
  \end{enumerate}
\item \emph{Implicaciones}: $P$ determinado; homogeneidad 0~en $(\vect{p},\overline{M})$ $\Rightarrow$ \emph{neutralidad sin dicotomía}.
  \begin{itemize}
  \item \autor{Archibald}-\autor{Lipsey} (1958): dicotomía válida en el estacionario; efecto \autor{Pigou}-\autor{Patinkin} \vertema{3.A.4}.
  \end{itemize}
\item \emph{Valoración}: \concepto{problema de \autor{Hahn}} (1965): siempre hay un equilibrio con dinero sin valor.
  \begin{itemize}
  \item 4~vías: generaciones solapadas, \emph{cash in advance}, búsqueda, dinero en la utilidad \vertema{3.A.35}.
  \end{itemize}
\end{itemize}

\sub{III.4. Incertidumbre (\autor{Arrow} 1953, \autor{Debreu} 1959)\label{rep:III.4}}
\begin{itemize}
\item \emph{Idea}: más bienes, no otra teoría: por fecha y \emph{estado de la naturaleza}. Paraguas si llueve $\neq$ paraguas si no.
\item \emph{Supuestos}: \concepto{bien contingente} $x_{ks}$, se paga hoy (póliza); $n\cdot S$ mercados hoy (\concepto{completos}).
\item \emph{Desarrollo}: el modelo de \hyperref[rep:I]{I} con $n\cdot S$ bienes (información simétrica, preferencias convexas): existe, con \concepto{precios de \autor{Arrow}-\autor{Debreu}}, y valen los 2~TFEB:
\begin{formula}
\centering $\dfrac{\pi_s\,u_i'(x_{is})}{\pi_{s'}\,u_i'(x_{is'})}=\dfrac{p_s}{p_{s'}}\quad\forall\,i$
\end{formula}
  \begin{itemize}
  \item sin riesgo agregado, $p_s/p_{s'}=\pi_s/\pi_{s'}$ y seguro completo.
  \end{itemize}
\item \emph{Implicaciones}:
  \begin{itemize}
  \item riesgo repartido de forma eficiente;
  \item \concepto{títulos de \autor{Arrow}} (1953): 1 € si ocurre $s$; $n+S$ mercados en vez de $n\cdot S$ (1.000~bienes y 10~estados: 1.010~frente a 10.000);
  \item \autor{Radner} (1972): equilibrio de planes, precios y expectativas.
  \end{itemize}
\item \emph{Extensiones}: \concepto{mercados incompletos}:
  \begin{itemize}
  \item no eficiente ni restringidamente (\autor{Hart}, 1975);
  \item puede no existir (activos reales); indeterminado (activos nominales);
  \item generaciones solapadas: ineficiencia dinámica \vertema{3.A.29}.
  \end{itemize}
\item \emph{Evidencia y valoración}: faltan mercados (renta del trabajo, PIB; \autor{Shiller}, 1993) por costes de transacción e información asimétrica \vertema{3.A.13}.
\end{itemize}

\sub{III.5. Modelos computables (\autor{Scarf})\label{rep:III.5}}
\begin{itemize}
\item \emph{Idea}: \concepto{MEGC} (CGE): equilibrio numérico con muchos sectores, hogares y países; algoritmo de \autor{Scarf} (1967; con \autor{Hansen}, 1973) + ordenadores.
\item \emph{Metodología} (\autor{Shoven}-\autor{Whalley}, 1992): formas funcionales, \emph{calibración} a una matriz de contabilidad social, simulación = estática comparativa.
\item \emph{Usos}: comercio (TLCAN infravaloró el comercio; UE-Mercosur con GTAP); clima (DICE, \autor{Nordhaus}); macro (REMS, EREMS) \vertema{3.A.6}.
\item \emph{Valoración}: dependen de elasticidades y cierres; unicidad no garantizada.
  \begin{itemize}
  \item Recogen los efectos indirectos que el análisis mercado a mercado no ve.
  \end{itemize}
\end{itemize}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}:
  \begin{itemize}
  \item existe; no tiene por qué ser único ni estable (SMD), pero es eficiente;
  \item resiste el núcleo, no el poder de mercado, las no convexidades, los mercados incompletos ni el dinero;
  \item \autor{Walras} formaliza la mano invisible de \autor{Smith}.
  \end{itemize}
\item \textbf{Relevancia}: lenguaje común: macro DSGE \vertema{3.A.6}, comercio \vertema{3.B.5}, clima \vertema{3.B.31}, MEGC.
\item \textbf{Extensiones}: \vertema{3.A.16} \vertema{3.A.23} \vertema{3.A.13} \vertema{3.A.35}; críticas: austriacos, desequilibrio, agregación (\autor{Kirman}), conductual.
\item \textbf{Opinión}: no describe ninguna economía real ni lo pretende: es una referencia exacta; «no es de la benevolencia del carnicero…» (\autor{Smith}, 1776); cada supuesto incumplido señala dónde puede fallar el mercado.
\item \textbf{Idea final}: volviendo a \autor{Bastiat}, aunque cada uno arrime el ascua a su sardina, los precios consiguen que se asen todas: el modelo enseña por qué el mercado funciona cuando lo hace.
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Por qué no basta contar ecuaciones e incógnitas?} No linealidad (0~o varias soluciones) y precios negativos.
\item \textbf{¿Papel de la convexidad?} Sin ella $\vect{z}$ salta y no vale \autor{Brouwer}; el 1TFEB no la necesita; el 2TFEB, sí.
\item \textbf{¿SMD contradice la existencia?} No: sobreviven continuidad, homogeneidad 0~y \autor{Walras}, justo lo que basta.
\item \textbf{¿Ley de \autor{Walras} o de \autor{Say}?} \autor{Walras}: identidad con todos los mercados; \autor{Say}: bienes sin dinero $\equiv0$.
\item \textbf{¿Por qué la sustitución bruta da unicidad y estabilidad?} Implica $\vect{p}^*\cdot\vect{z}(\vect{p})>0$ fuera de $\vect{p}^*$ (ABH, 1959).
\item \textbf{¿Por qué el equilibrio walrasiano está en el núcleo?} Si una coalición mejorara con sus dotaciones, alguno habría comprado algo mejor a $\vect{p}^*$.
\item \textbf{¿Por qué cuesta meter el dinero?} No da utilidad ni hace falta; siempre hay equilibrio sin valor (\autor{Hahn}).
\end{enumerate}

\begin{trampa}
\begin{itemize}
\item Ley de \autor{Walras}: identidad a \emph{cualquier} precio.
\item 1TFEB sin convexidad; 2TFEB con convexidad y suma fija.
\item \autor{Scarf} (1960): 3~bienes \emph{complementarios}, equilibrio \emph{único} pero inestable.
\item Títulos de \autor{Arrow}: $n+S$ mercados, no $n\cdot S$.
\end{itemize}
\end{trampa}
