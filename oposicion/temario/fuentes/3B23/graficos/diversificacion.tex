% Riesgo de la cartera frente al número de títulos (modelo diagonal de Sharpe,
% cartera equiponderada): sigma_p^2(n) = s + A/n, con s = beta media^2 sigma_M^2.
% Escala: n -> 0,2 cm por título; s = 1,4; A = 3,6 (en n = 1, 5 cm).
\begin{tikzpicture}[tcee,scale=0.9]
  % 1: ejes y rótulos
  \capa{1}{
    \draw[eje] (0,0) -- (9,0) node[right] {$n$};
    \node[etiqueta,font=\footnotesize] at (4.5,-0.85) {número de títulos};
    \draw[eje] (0,0) -- (0,5.6) node[left] {$\sigma_p^2$};
    \node[below left] at (0,0) {$0$};
    \foreach \n in {1,10,20,30,40}
      \draw (0.2*\n,0.06) -- (0.2*\n,-0.06) node[below,font=\footnotesize] {$\n$};
  }
  % 2: asíntota del riesgo sistemático
  \capa{2}{
    \draw[guia,thick,tceeazul] (0,1.4) -- (8.6,1.4);
    \node[etiqueta,left,text=tceeazul] at (0,1.4) {$\bar\beta^{\,2}\sigma_M^2$};
  }
  % 3: curva del riesgo total
  \capa{3}{
    \draw[curva1] plot[domain=1:43,samples=120] ({0.2*\x},{1.4+3.6/\x});
    \node[etiqueta,right,text=tceemorado] at ({0.2*1.6},{1.4+3.6/1.6}) {riesgo total};
  }
  % 4: zonas sombreadas (debajo de las líneas)
  \capa{4}{
    \begin{pgfonlayer}{background}
      \fill[area2] plot[domain=1:43,samples=120] ({0.2*\x},{1.4+3.6/\x}) -- (8.6,1.4) -- (0.2,1.4) -- cycle;
      \fill[area,fill=tceeazul] (0,0) rectangle (8.6,1.4);
    \end{pgfonlayer}
  }
  % 5: rótulos de las zonas
  \capa{5}{
    \node[etiqueta,align=left,anchor=south west,text=tceedorado!70!black] (esp) at (3.4,2.35) {riesgo específico\\(diversificable)};
    \draw[->,thin,tceedorado!70!black] (esp.south west) ++(0.1,0.05) to[out=-150,in=60] (1.75,1.65);
    \node[etiqueta,align=center,text=tceeazul] at (4.4,0.7) {riesgo sistemático (no diversificable)};
  }
\end{tikzpicture}
