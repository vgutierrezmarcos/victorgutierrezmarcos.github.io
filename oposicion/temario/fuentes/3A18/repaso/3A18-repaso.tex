\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: «para gustos, los colores». Coca-Cola: más de 200 marcas de bebidas.
  \begin{itemize}
  \item ¿Por qué no venden todas lo mismo al mismo precio? ¿Cuánto pagamos por la variedad? ¿Demasiada o poca?
  \end{itemize}
\item \textbf{Relevancia}:
  \begin{itemize}
  \item empírica: variedades importadas por EE.\,UU.\ = 2,6\,\% del PIB;
  \item teórica: \autor{Dixit}-\autor{Stiglitz} en macro \vertema{3.A.7}, comercio \vertema{3.B.6}, crecimiento \vertema{3.A.44};
  \item normativa: falla el 1TFEB \vertema{3.A.22}; competencia \vertema{3.B.39}.
  \end{itemize}
\item \textbf{Contextualización}: \autor{Hotelling} (1929) $\to$ \autor{Chamberlin}, \autor{Robinson} (1933) $\to$ \autor{Lancaster}, \autor{Dixit}-\autor{Stiglitz} + teoría de juegos.
  \begin{itemize}
  \item \concepto{Diferenciación}: sustitutivos cercanos no perfectos; \concepto{horizontal} (gustos) o \concepto{vertical} (calidad).
  \item \concepto{No direcciones} (preferencia por la variedad) frente a \concepto{direcciones} (características, variedad ideal).
  \item «Mapa a escala 1:1» (\autor{J.\ Robinson}): supuestos fuertes.
  \end{itemize}
\item \textbf{Problemática}: ¿cuánto poder da la diferenciación? ¿Resuelve la paradoja de \autor{Bertrand}? ¿Cuántas variedades hay y cuántas debería haber?
\item \textbf{Estructura}: \hyperref[rep:I]{I}, muchas empresas y libre entrada; \hyperref[rep:II]{II}, pocas empresas, un modelo por decisión.
\end{itemize}

\bloque[10,5]{I. Competencia monopolística: no direcciones\label{rep:I}}
\begin{ideaclave}
Cada empresa es monopolista de su variedad y su margen depende de lo sustituibles que sean las variedades; la libre entrada se lleva los beneficios, no la ineficiencia: el exceso de capacidad es el precio de la variedad.
\end{ideaclave}

\sub{I.1. Caracterización\label{rep:I.1}}
\begin{itemize}
\item Se levanta el \textbf{3}: \textbf{producto diferenciado}, elasticidad cruzada alta pero finita.
  \begin{itemize}
  \item el 4, en parte: muchos, sin interacción estratégica, pero cada uno fija el precio de su variedad;
  \item se mantienen racionalidad, información y libre entrada.
  \end{itemize}
\item \emph{Consecuencia}: \concepto{poder de mercado}, $\mathcal{L}=1/|\varepsilon|$ \vertema{3.A.17} con la elasticidad \emph{de su variedad}.
\item \emph{Tecnología común}: $C(q)=\mathrm{CF}+c\,q$, CMe decreciente: el CF explica por qué no existe toda variedad.
\end{itemize}

\sub{I.2. Modelo de \autor{Chamberlin} (1933)\label{rep:I.2}}
\subsub{a) Idea}
\begin{itemize}
\item Cada una, monopolista de su variedad; el conjunto se parece a la competencia.
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item \emph{Supuestos}: \emph{grupo} de sustitutivos; muchas empresas \emph{simétricas} (\concepto{supuesto heroico}); \emph{miopía}.
\item \emph{2 demandas}:
  \begin{itemize}
  \item \concepto{percibida} $d$: si las demás no mueven su precio; muy elástica;
  \item \concepto{efectiva} $D$ (cuota $1/n$): cuando todas lo mueven; menos elástica.
  \end{itemize}
\item \emph{CP}: $\mathrm{IMg}_d=\mathrm{CMg}$; $d$ se desliza por $D$; beneficios o pérdidas.
\item \emph{LP}: entrada $\Rightarrow$ $D$ (y $d$) a la izquierda; por la miopía, guerra de precios, pérdidas y salidas, hasta la \emph{tangencia} de $d$ con el CMe:
\begin{formula}
\centering $p^*=\mathrm{CMe}(q^*)>\mathrm{CMg}(q^*)=\mathrm{IMg}_d(q^*)$,\quad $B^*=0$
\end{formula}
  \begin{itemize}
  \item 3 condiciones: $d$ tangente al CMe ($B=0$); $\mathrm{IMg}_d=\mathrm{CMg}$; $d$ corta a $D$ (lo que cree vender es lo que vende).
  \end{itemize}
\end{itemize}
\graficorepaso[0.98\linewidth]{chamberlin}{a) $d$ y $D$; b) CP con beneficios; c) tangencia a LP}
\begin{itemize}
\item \concepto{Teorema del exceso de capacidad}: $q^*<\bar{q}$ (escala mínima eficiente).
  \begin{itemize}
  \item «demasiadas empresas, cada una demasiado pequeña» (\autor{Kaldor}, 1935);
  \item ineficiencia \emph{asignativa} ($p>\mathrm{CMg}$) y \emph{productiva} ($\mathrm{CMe}>\min\mathrm{CMe}$);
  \item pero es \emph{el precio de la variedad}, que no puede medir.
  \end{itemize}
\end{itemize}
\subsub{c) Valoración}
\begin{itemize}
\item Reconcilia libre entrada y escala no agotada (\autor{Sraffa}, 1926).
\item Primero en tratar la diferenciación y los costes de venta como decisiones.
\item Críticas: grupo mal definido (\autor{Stigler}, 1949); simetría incoherente (\autor{Kaldor}); miopía; costes hundidos; no modeliza la demanda.
\end{itemize}

\sub{I.3. Modelo de \autor{Dixit} y \autor{Stiglitz} (1977)\label{rep:I.3}}
\subsub{a) Idea}
\begin{itemize}
\item Formaliza a \autor{Chamberlin} con un \concepto{consumidor representativo} que prefiere la variedad $\Rightarrow$ se compara con el óptimo.
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item \emph{Supuestos}: una variedad por empresa; numerario $x_0$; renta $\overline{W}$; \concepto{agregador} CES \vertema{3.A.11}:
\begin{formula}
\centering $U=x_0^{\,1-\alpha}\,y^{\alpha}$,\quad $y\equiv\Big[\sum_{i=1}^{n}x_i^{\rho}\Big]^{1/\rho}$,\quad $\rho=\dfrac{\sigma-1}{\sigma}$,\ $\sigma>1$
\end{formula}
  \begin{itemize}
  \item $\sigma$: \concepto{elasticidad de sustitución} entre variedades; $\alpha$: gasto en el grupo;
  \item \concepto{preferencia por la variedad}: $x_i=X/n$ $\Rightarrow$ $y=n^{1/(\sigma-1)}X$; más intensa cuanto \emph{menor} $\sigma$;
  \item $\sigma>1$: demanda de cada variedad elástica (si no, no hay precio óptimo).
  \end{itemize}
\item \emph{Consumidor}, en 2 etapas: \concepto{demanda de cada variedad} con el \concepto{índice de precios} $P$:
\funcionclave{x_i=\frac{\alpha\overline{W}}{P}\Big(\frac{p_i}{P}\Big)^{-\sigma},\quad P\equiv\Big[\sum_{j}p_j^{1-\sigma}\Big]^{\frac{1}{1-\sigma}}}
  \begin{itemize}
  \item $\uparrow$ con el mercado ($\alpha\overline{W}$) y con $P$; $\downarrow$ con $p_i$;
  \item muchas variedades: $p_i$ no mueve $P$ $\Rightarrow$ elasticidad $\sigma$ (exacta: $-\sigma+(\sigma-1)s_i$): la $d$ de \autor{Chamberlin}; la $D$, unitaria;
  \item precios iguales: $P=p\,n^{1/(1-\sigma)}$: la entrada baja $P$ y quita demanda a las demás.
  \end{itemize}
\item \emph{Empresa}: máx.\ $(p_i-c)\,x_i-\mathrm{CF}$ con $P$ dado $\Rightarrow$ \concepto{regla del margen}:
\funcionclave{\mathcal{L}=\frac{p^*-c}{p^*}=\frac{1}{\sigma}\ \Longleftrightarrow\ p^*=\frac{\sigma}{\sigma-1}\,c}
  \begin{itemize}
  \item margen constante, mayor con más preferencia por la variedad ($\downarrow\sigma$).
  \end{itemize}
\item \emph{Libre entrada}: $(p^*-c)\,q^*=\mathrm{CF}$ y $n\,p^*q^*=\alpha\overline{W}$:
\begin{formula}
\centering $q^*=\dfrac{(\sigma-1)\,\mathrm{CF}}{c}$,\qquad $n^*=\dfrac{\alpha\overline{W}}{\sigma\,\mathrm{CF}}$
\end{formula}
  \begin{itemize}
  \item $q^*$ \emph{no depende del tamaño del mercado}; $n^*$ $\uparrow$ con él, $\downarrow$ con CF y $\sigma$.
  \end{itemize}
\end{itemize}
\graficorepaso[0.98\linewidth]{dixit-stiglitz}{a) beneficios; b) la entrada desplaza la demanda con $p=\mu c$; c) tangencia}
\begin{itemize}
\item \emph{Implicaciones positivas}: $p>\mathrm{CMg}$ con $B=0$: el poder está en el \emph{margen}, que paga el CF.
\item \emph{Normativas}: exceso de capacidad, pero ya cuenta la variedad.
  \begin{itemize}
  \item CES: equilibrio = \concepto{óptimo restringido} (sin subvenciones, cada una cubre costes);
  \item \concepto{primer óptimo} ($p=\mathrm{CMg}$, CF con suma fija): \emph{misma} $q$, \emph{más} variedades, $n^{\mathrm{o}}/n^*=1/[\rho(1-\alpha)+\alpha]>1$ \vertema{3.A.22};
  \item en general (\autor{Spence} 1976, \autor{Mankiw}-\autor{Whinston} 1986), 2 fuerzas:
    \begin{itemize}
    \item \concepto{excedente no apropiable}: quien lanza una variedad no cobra todo lo que vale $\Rightarrow$ pocas;
    \item \concepto{robo de clientes}: parte de lo que gana se lo quita a otras $\Rightarrow$ demasiadas.
    \end{itemize}
  \end{itemize}
\end{itemize}
\subsub{c) Extensiones}
\begin{itemize}
\item Comercio intraindustrial (\autor{Krugman}, 1979; \autor{Melitz}, 2003) \vertema{3.B.6}; nueva economía keynesiana (\autor{Blanchard}-\autor{Kiyotaki}, 1987) \vertema{3.A.7}; crecimiento (\autor{Romer}, 1990) \vertema{3.A.44}.
\end{itemize}
\subsub{d) Evidencia empírica}
\begin{itemize}
\item \autor{Broda}-\autor{Weinstein} (2006): variedades importadas por EE.\,UU.\ $\times3$ (1972-2001).
  \begin{itemize}
  \item el IPC sobrestimó la inflación importada 1,2 p.p./año; la variedad vale 2,6\,\% del PIB.
  \end{itemize}
\end{itemize}
\subsub{e) Valoración}
\begin{itemize}
\item Soluciones cerradas; juzga el número de variedades.
\item Críticas: no dice en qué características se compite; elasticidades cruzadas iguales; margen constante.
\item Competencia \emph{global}; si es \emph{local} y entre pocas $\Rightarrow$ \hyperref[rep:II]{II}.
\end{itemize}

\bloque[16]{II. El oligopolio con productos diferenciados\label{rep:II}}
\begin{ideaclave}
La diferenciación es la salida a la paradoja de \autor{Bertrand}: las empresas se diferencian a propósito para suavizar la competencia en precios, y el mercado no garantiza ni la diferenciación ni el número de empresas óptimos.
\end{ideaclave}

\sub{II.1. Caracterización\label{rep:II.1}}
\begin{itemize}
\item Se levantan el \textbf{3} (diferenciado), el \textbf{4} (pocos, \emph{interdependientes}) y el \textbf{5} (barreras), salvo en \autor{Salop}.
\item \concepto{Interdependencia estratégica} $\Rightarrow$ juegos \vertema{3.A.14}:
  \begin{itemize}
  \item estáticos: \concepto{equilibrio de \autor{Nash}};
  \item en 2 etapas: \concepto{perfecto en subjuegos}, por inducción hacia atrás (primero los precios).
  \end{itemize}
\end{itemize}

\sub{II.2. Las decisiones: un modelo para cada una\label{rep:II.2}}
\begin{itemize}
\item \emph{Mapa}: precio $\to$ localización $\to$ entrada (horizontal) $\to$ calidad (vertical) $\to$ publicidad y marcas.
  \begin{itemize}
  \item cada modelo fija las demás decisiones y responde a la pregunta que deja el anterior;
  \item el precio, siempre la última etapa (lo más fácil de cambiar) y la primera que se resuelve (\autor{Tirole}, 1988).
  \end{itemize}
\item \emph{Resultados}:
  \begin{itemize}
  \item precio (ciudad lineal): $p^*=c+t$, salida a la paradoja;
  \item localización: diferenciación máxima;
  \item entrada (\autor{Salop}): el doble de las óptimas;
  \item calidad: diferenciación máxima y finitud;
  \item publicidad: $A/(pq)=\mathcal{L}\,\varepsilon_{q,A}$; marcas, barrera.
  \end{itemize}
\end{itemize}
\subsub{a) Justificación: \autor{Lancaster} (1966)}
\begin{itemize}
\item No se demandan bienes, sino sus \concepto{características}: el bien $i$ da $a_{ki}$ de la $k$ («tecnología del consumo»); $z_k=\sum_ia_{ki}x_i$, $U(\vect{z})$ \vertema{3.A.8}.
\item Explica fidelidad, nichos y publicidad $\Rightarrow$ \concepto{enfoque de direcciones} y \concepto{competencia local}.
\item Inmanejable con muchas características $\Rightarrow$ una sola: la «localización».
\end{itemize}

\subsub{b) Precio: ciudad lineal de \autor{Hotelling} (1929)}
\begin{itemize}
\item \emph{Idea}: 2 tiendas en una calle; primer modelo de \concepto{diferenciación espacial} (la calle: sabor, horario, ideología).
\item \emph{Supuestos}:
  \begin{itemize}
  \item masa 1 uniforme en $[0,1]$ ($x$: variedad ideal); empresas en 0 y 1, CMg $c$;
  \item demanda unitaria; utilidad $v-p_i-t\,d^2$ ($t$: \concepto{coste de transporte}, pérdida por no consumir la ideal); \autor{Bertrand}, estático.
  \end{itemize}
\item \emph{Desarrollo}: \concepto{consumidor indiferente} $\tilde{x}$:
\begin{formula}
\centering $x_1=\tilde{x}=\dfrac12+\dfrac{p_2-p_1}{2t}$\ $\Rightarrow$\ $R_i(p_j)=\dfrac{p_j+c+t}{2}$\\[3pt]
$p_1^*=p_2^*=c+t$,\quad $B_i^*=t/2$
\end{formula}
\end{itemize}
\pareja{hotelling}{El consumidor indiferente reparte el mercado}{hotelling-reaccion}{Reacción en precios: equilibrio en $c+t$}
\begin{itemize}
\item \emph{Implicaciones}:
  \begin{itemize}
  \item \emph{salida a la paradoja de \autor{Bertrand}}: margen $t$; si $t\to0$, $p=c$;
  \item precios = \concepto{complementos estratégicos} (reacción creciente);
  \item valoración: localizaciones exógenas.
  \end{itemize}
\item \emph{Sin direcciones} (\autor{Singh}-\autor{Vives}, 1984): $p_i=\alpha-\beta q_i-\gamma q_j$, $\gamma<\beta$.
  \begin{itemize}
  \item \autor{Bertrand}: $p^B=c+\frac{(\alpha-c)(\beta-\gamma)}{2\beta-\gamma}>c$, margen $\uparrow$ al $\downarrow\gamma$;
  \item \autor{Cournot} (sustitutivos estratégicos): $p^C>p^B$; prefieren comprometerse a cantidades;
  \item $\gamma$ exógeno: la ciudad lineal le da origen ($t$, heterogeneidad de gustos).
  \end{itemize}
\end{itemize}

\subsub{c) Localización: diferenciación máxima}
\begin{itemize}
\item \autor{Hotelling}: \concepto{diferenciación mínima} (las 2 en el centro).
\item \autor{d'Aspremont}-\autor{Gabszewicz}-\autor{Thisse} (1979): falla si eligen precios.
  \begin{itemize}
  \item costes lineales: demanda discontinua, sin equilibrio si están cerca;
  \item cuadráticos: a los extremos.
  \end{itemize}
\item \emph{Modelo}: costes cuadráticos; 1) localizaciones $0\le a\le b\le1$; 2) precios:
\begin{formula}
\centering $p_1^*=c+\dfrac{t(b-a)(2+a+b)}{3}$,\quad $p_2^*=c+\dfrac{t(b-a)(4-a-b)}{3}$
\end{formula}
  \begin{itemize}
  \item en la 1.ª etapa (envolvente: el propio precio no cuenta):
  \end{itemize}
\begin{formula}
\centering $\dfrac{dB_1}{da}=(p_1^*-c)\Big[\underbrace{\tfrac{\partial x_1}{\partial a}}_{\text{demanda }(+)}+\underbrace{\tfrac{\partial x_1}{\partial p_2}\tfrac{dp_2^*}{da}}_{\text{estratégico }(-)}\Big]<0$
\end{formula}
  \begin{itemize}
  \item \concepto{efecto demanda}: acercarse al centro da clientes;
  \item \concepto{efecto estratégico}: acercarse a la rival baja su precio; domina.
  \end{itemize}
\end{itemize}
\graficorepaso[0.6\linewidth]{hotelling-localizacion}{Equilibrio en 0 y 1; óptimo en $\tfrac14$ y $\tfrac34$}
\begin{itemize}
\item \emph{Implicaciones}: \concepto{diferenciación máxima}, $a=0$, $b=1$: alejarse para \emph{suavizar la competencia en precios}.
  \begin{itemize}
  \item con precios fijos, solo el efecto demanda $\Rightarrow$ centro;
  \item bienestar: el precio es transferencia; óptimo en $1/4$ y $3/4$, transporte $t/48$ frente a $t/12$: el mercado \emph{diferencia demasiado}.
  \end{itemize}
\item \emph{Extensión}: votante mediano (\autor{Downs}, 1957) \vertema{3.A.24}.
\item \emph{Valoración}: sensible a los supuestos; agrupamiento si la demanda se concentra en el centro, por aglomeración o sin competencia en precios.
\end{itemize}

\subsub{d) Entrada: ciudad circular de \autor{Salop} (1979)}
\begin{itemize}
\item Competencia monopolística con competencia \emph{local}: ¿demasiadas o pocas?
\item \emph{Supuestos}: círculo de perímetro 1; costes lineales $t\,d$; CF de entrada; 1) entran $n$ equidistantes; 2) precios.
\item \emph{Desarrollo}: $x_i=1/n+(p-p_i)/t$:
\begin{formula}
\centering $p^*=c+\dfrac tn$,\quad $B^*=\dfrac{t}{n^2}-\mathrm{CF}$\ $\overset{B=0}{\Longrightarrow}$\ $n^*=\sqrt{\dfrac{t}{\mathrm{CF}}}$
\end{formula}
  \begin{itemize}
  \item planificador: $\min_n\ n\,\mathrm{CF}+t/(4n)$ $\Rightarrow$ $n^{\mathrm{o}}=\tfrac12\sqrt{t/\mathrm{CF}}=n^*/2$.
  \end{itemize}
\end{itemize}
\graficorepaso[0.4\linewidth]{salop}{$n$ empresas equidistantes}
\begin{itemize}
\item \emph{Implicaciones}: \concepto{entrada excesiva} (el doble); domina el \emph{robo de clientes}.
  \begin{itemize}
  \item $\uparrow$CF $\Rightarrow$ $\downarrow n$, $\uparrow$margen; $\uparrow t$ $\Rightarrow$ $\uparrow$ los 2; $\mathrm{CF}\to0$ $\Rightarrow$ $p\to c$;
  \item en \autor{Dixit}-\autor{Stiglitz} los 2 efectos se compensan (óptimo restringido).
  \end{itemize}
\item \emph{Valoración}: canónico de la entrada; localización simétrica supuesta; demanda unitaria.
\end{itemize}

\subsub{e) Calidad: \autor{Shaked}-\autor{Sutton} (1982-83)}
\begin{itemize}
\item \emph{Supuestos}: $U=\theta s-p$; $\theta$ uniforme en $[\underline{\theta},\overline{\theta}]$; calidad en CF (I+D), no en CMg; 1) calidades; 2) precios.
\item \emph{Desarrollo}: con $\Delta s=s_A-s_B$:
\begin{formula}
\centering $p_{\mathrm{A}}^*=c+\dfrac{2\overline{\theta}-\underline{\theta}}{3}\Delta s$,\quad $p_{\mathrm{B}}^*=c+\dfrac{\overline{\theta}-2\underline{\theta}}{3}\Delta s$
\end{formula}
  \begin{itemize}
  \item si $\overline{\theta}>2\underline{\theta}$ (bastante heterogeneidad); si no, la baja no vende;
  \item 1.ª etapa: márgenes $\propto\Delta s$ $\Rightarrow$ alta elige $\overline{s}$, baja $\underline{s}$; igual calidad $\Rightarrow$ \autor{Bertrand}.
  \end{itemize}
\end{itemize}
\graficorepaso[0.6\linewidth]{shaked-sutton}{El indiferente reparte las valoraciones}
\begin{itemize}
\item \emph{Implicaciones}: \concepto{diferenciación máxima de calidades} (como \autor{Hotelling}); pugna por el liderazgo: la alta gana más.
\item \concepto{Propiedad de finitud}: $n$ acotado \emph{aunque el mercado crezca}: \concepto{oligopolio natural}.
  \begin{itemize}
  \item contraste con \autor{Salop}: $n$ según la heterogeneidad, no el tamaño.
  \end{itemize}
\item \emph{Evidencia}: \autor{Sutton} (1991): con costes hundidos endógenos (publicidad, I+D), la concentración no baja al crecer el mercado; sin ellos, sí.
\item \emph{Valoración}: con la calidad en costes variables o con \autor{Cournot}, puede no haber finitud.
\end{itemize}

\subsub{f) Publicidad y proliferación de marcas}
\begin{itemize}
\item \emph{Idea}: crear y proteger la diferenciación sin cambiar el producto. España, 2025: 12.745 M€ en publicidad.
\item \autor{Dorfman}-\autor{Steiner} (1954): elige $p$ y $A$ con $q(p,A)$; CPO: \autor{Lerner} y $(p-\mathrm{CMg})\,\partial q/\partial A=1$:
\begin{formula}
\centering $\dfrac{A}{p\,q}=\dfrac{\varepsilon_{q,A}}{|\varepsilon_{q,p}|}=\mathcal{L}\,\varepsilon_{q,A}$
\end{formula}
  \begin{itemize}
  \item más publicidad cuanto más eficaz y más poder; sin poder, ninguna.
  \end{itemize}
\item \emph{3 visiones} (\autor{Bagwell}, 2007):
  \begin{itemize}
  \item informativa (\autor{Nelson}, 1974): más competencia;
  \item persuasiva: fidelidad, barrera;
  \item complementaria (\autor{Becker}-\autor{Murphy}, 1993).
  \end{itemize}
\item Señal de calidad (\autor{Milgrom}-\autor{Roberts}, 1986) \vertema{3.A.13}.
\item \concepto{Proliferación de marcas}: ocupar los nichos del círculo: \emph{barrera estratégica} \vertema{3.A.15}.
  \begin{itemize}
  \item creíble solo si retirar marcas cuesta (\autor{Judd}, 1985);
  \item cereales en EE.\,UU.\ (\autor{Schmalensee}, 1978): 6 mayores, más de 80 marcas sin entrada con éxito; FTC 1972, desestimado 1981 \vertema{3.B.39}.
  \end{itemize}
\item \emph{Valoración}: regla sencilla y contrastable; el bienestar depende de la visión correcta, y la evidencia no elige una.
\end{itemize}
\begin{ideaclave}
Márgenes: $1/\sigma$; $t$ en la ciudad lineal; $t/n$ en la circular; $\propto\Delta s$ en vertical; $A/(pq)=\mathcal{L}\,\varepsilon_{q,A}$.
\end{ideaclave}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}:
  \begin{itemize}
  \item margen según lo distintas que sean las variedades;
  \item libre entrada: $B=0$, pero exceso de capacidad o de entrada;
  \item pocas empresas: diferenciación máxima; publicidad y marcas la protegen;
  \item bienestar ambiguo: margen y escala frente a variedad.
  \end{itemize}
\item \textbf{Relevancia}: \vertema{3.B.6} \vertema{3.A.7} \vertema{3.A.44} \vertema{3.A.24} \vertema{3.B.39}.
\item \textbf{Extensiones}: colusión \vertema{3.A.19}, barreras \vertema{3.A.15}, mercado relevante \vertema{3.A.20}.
\item \textbf{Opinión}: importa menos cómo se modelice la demanda que lo \emph{local} de la competencia.
  \begin{itemize}
  \item global (\autor{Dixit}-\autor{Stiglitz}): la variedad compensa buena parte de la ineficiencia;
  \item local: cada una protege su nicho, más lejos del óptimo;
  \item no castigar la diferenciación, sino las estrategias que la usan para cerrar el paso.
  \end{itemize}
\item \textbf{Idea final}: para gustos, los colores, y de esa diversidad nace el poder de mercado.
  \begin{itemize}
  \item pagamos la variedad con un margen; el mercado unas veces se queda corto y otras se pasa;
  \item hay colores para todos los gustos, pero no necesariamente los que la sociedad elegiría.
  \end{itemize}
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Beneficio nulo y poder de mercado?} Sí: el poder es el margen ($1/\sigma$), que paga el CF.
\item \textbf{¿Demasiadas o pocas variedades?} Excedente no apropiable (pocas) frente a robo de clientes (demasiadas); \autor{Salop}: el doble.
\item \textbf{¿Por qué costes cuadráticos?} Con lineales la demanda es discontinua y puede no haber equilibrio.
\item \textbf{¿Por qué con calidades $n$ no crece?} La calidad cuesta en CF: finitud.
\item \textbf{¿\autor{Lerner} y publicidad?} $A/(pq)=\mathcal{L}\,\varepsilon_{q,A}$.
\end{enumerate}
