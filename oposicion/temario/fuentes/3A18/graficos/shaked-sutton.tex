% Diferenciación vertical (3.A.18, III.1, Shaked y Sutton).
% U = theta s - p, theta uniforme en [theta_min, theta_max] = [0,4; 1,4] (longitud 1;
% 2 theta_min < theta_max < 4 theta_min: duopolio natural).
% Calidades s_B = 1 < s_A = 2 (Delta s = 1), c = 0: precios de equilibrio (eq. 10)
%   p_A = (2 theta_max - theta_min)/3 = 0,8 y p_B = (theta_max - 2 theta_min)/3 = 0,2.
%   Consumidor indiferente: theta~ = (p_A - p_B)/(s_A - s_B) = 0,6, con U = 0,4.
%   x_B = theta~ - theta_min = 0,2; x_A = theta_max - theta~ = 0,8 (mercado
%   cubierto: U_A(0,4) = 0 y U_B(0,4) = 0,2).
% Capa final: las calidades en los extremos de [s_min, s_max] (diferenciación máxima).
% Elaboración propia a partir de Shaked y Sutton (1982) y Tirole (1988, cap. 7).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=6cm,y=1.8cm,
    declare function={ub(\t)=\t-0.2; ua(\t)=2*\t-0.8;}]
  \def\tl{0.4}\def\th{1.4}\def\tt{0.6}\def\ut{0.4}
  \useasboundingbox (-0.1,-2.05) rectangle (2.0,2.45);
  % 1: ejes y segmento de valoraciones de la calidad
  \capa{1}{\draw[eje] (0,0) -- (1.7,0) node[below left=2pt and 0pt] {$\theta$};}
  \capa{1}{\draw[eje] (0,0) -- (0,2.3) node[left] {$U$};}
  \capa{1}{\node[below left] at (0,0) {$0$};}
  \capa{1}{\draw[line width=2.2pt,black] (\tl,0) -- (\th,0);}
  \capa{1}{\node[punto] at (\tl,0) {};}
  \capa{1}{\node[punto] at (\th,0) {};}
  \capa{1}{\node[below] at (\tl,0) {$\underline{\theta}$};}
  \capa{1}{\node[below] at (\th,0) {$\overline{\theta}$};}
  \capa{1}{\node[etiqueta,font=\footnotesize,align=left,anchor=north west] at (0.06,2.25) {valoraciones\\ uniformes en $[\underline{\theta},\overline{\theta}]$,\\ $\overline{\theta}-\underline{\theta}=1$};}
  % 2: utilidad neta de la calidad baja (menos pendiente)
  \capa{2}{\draw[curva2] plot[domain=\tl:\th,samples=2] (\x,{ub(\x)});}
  \capa{2}{\node[etiqueta,text=tceeazul,anchor=west] at (\th,{ub(\th)}) {$\theta\,s_{\mathrm{B}}-p_{\mathrm{B}}$};}
  % 3: utilidad neta de la calidad alta (más pendiente)
  \capa{3}{\draw[curva1] plot[domain=\tl:\th,samples=2] (\x,{ua(\x)});}
  \capa{3}{\node[etiqueta,text=tceemorado,anchor=west] at (\th,{ua(\th)}) {$\theta\,s_{\mathrm{A}}-p_{\mathrm{A}}$};}
  % 4: consumidor indiferente
  \capa{4}{\node[punto] (I) at (\tt,\ut) {};}
  \capa{4}{\draw[guia] (I) -- (\tt,0);}
  \capa{4}{\node[below] at (\tt,0) {$\tilde{\theta}$};}
  \capa{4}{\node[etiqueta,font=\footnotesize,align=center,anchor=south east] at (0.58,0.47) {consumidor\\ indiferente};}
  % 5: quién compra cada calidad
  \capa{5}{\draw[decorate,decoration={brace,mirror,amplitude=5pt},thick,tceeazul] (\tl,-0.3) -- (\tt-0.01,-0.3);}
  \capa{5}{\node[etiqueta,font=\footnotesize,align=center,text=tceeazul,below] at ({(\tl+\tt)/2},-0.42) {compran la\\ calidad baja:\\ $x_{\mathrm{B}}=\tilde{\theta}-\underline{\theta}$};}
  \capa{5}{\draw[decorate,decoration={brace,mirror,amplitude=5pt},thick,tceemorado] (\tt+0.01,-0.3) -- (\th,-0.3);}
  \capa{5}{\node[etiqueta,font=\footnotesize,align=center,text=tceemorado,below] at ({(\tt+\th)/2},-0.42) {compran la\\ calidad alta:\\ $x_{\mathrm{A}}=\overline{\theta}-\tilde{\theta}$};}
  % 6: segmento de calidades: diferenciación máxima
  \capa{6}{\draw[thick,black] (\tl,-1.6) -- (\th,-1.6);}
  \capa{6}{\node[etiqueta,anchor=east] at (\tl-0.06,-1.6) {calidad $s$};}
  \capa{6}{\node[punto,fill=tceeazul] at (\tl,-1.6) {};}
  \capa{6}{\node[punto,fill=tceemorado] at (\th,-1.6) {};}
  \capa{6}{\node[above,text=tceeazul] at (\tl,-1.6) {$s_{\mathrm{B}}$};}
  \capa{6}{\node[above,text=tceemorado] at (\th,-1.6) {$s_{\mathrm{A}}$};}
  \capa{6}{\node[below] at (\tl,-1.6) {$\underline{s}$};}
  \capa{6}{\node[below] at (\th,-1.6) {$\overline{s}$};}
  \capa{6}{\draw[<->,thick,tceerojo] (\tl+0.04,-1.72) -- (\th-0.04,-1.72);}
  \capa{6}{\node[etiqueta,font=\footnotesize,text=tceerojo,below] at ({(\tl+\th)/2},-1.74) {diferenciación máxima};}
\end{tikzpicture}
