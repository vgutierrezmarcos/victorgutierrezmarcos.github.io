% Demanda de características (Lancaster, 1966): espacio de dos características,
% un rayo por bien (proporciones fijas a_{ki}), sobre cada rayo el punto que
% permite la renta (x_i = W/p_i unidades del bien i), frontera de eficiencia que
% los une y tangencia con la curva de indiferencia más alta.
% Parte de fuentes/3A08/graficos/lancaster-caracteristicas.tex (mismas
% coordenadas), sin la subida del precio de un bien. Rehace la Imagen 8 del tema
% original.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,scale=1.05]
  \useasboundingbox (-0.5,-0.5) rectangle (7.0,5.9);
  % 1: ejes (las características)
  \capa{1}{\ejes{$z_1$}{$z_2$}{6.4}{5.4}}
  % 2: un rayo por bien, con la proporción de características que contiene
  \capa{2}{\draw[guia] (0,0) -- (1.44,5.4) node[etiqueta,above,text=tceegris] {bien 1};}
  \capa{2}{\draw[guia] (0,0) -- (4.52,4.8) node[etiqueta,above right,text=tceegris] {bien 2};}
  \capa{2}{\draw[guia] (0,0) -- (6.2,1.49) node[etiqueta,right,text=tceegris] {bien 3};}
  % 3: lo que permite la renta sobre cada rayo, x_i = W/p_i
  \capa{3}{\node[punto,label={[etiqueta]left:$\overline{W}/p_1$}] (a) at (1.2,4.5) {};}
  \capa{3}{\node[punto,label={[etiqueta,yshift=-5pt,xshift=-4pt]below:$\overline{W}/p_2$}] (b) at (3.2,3.4) {};}
  \capa{3}{\node[punto,label={[etiqueta]below:$\overline{W}/p_3$}] (c) at (5,1.2) {};}
  % 4: frontera de eficiencia y conjunto alcanzable
  \capa{4}{\fill[area] (0,0) -- (1.2,4.5) -- (3.2,3.4) -- (5,1.2) -- cycle;}
  \capa{4}{\draw[curva1] (a) -- (b) -- (c);}
  % 5: curva de indiferencia tangente a la frontera en E (pendiente -0,55 en x = 2,2)
  \capa{5}{\draw[curva3] plot[domain=0.8:6.2,samples=60] (\x,{3.95*(2.2/\x)^0.306}) node[etiqueta,right] {$U^*$};}
  \capa{5}{\node[punto,label={[etiqueta]above right:$E$}] at (2.2,3.95) {};}
\end{tikzpicture}
