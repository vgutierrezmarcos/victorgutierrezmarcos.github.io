% Funciones de reacción con producto diferenciado (Singh y Vives, 1984, con la
% demanda lineal de Bowley). Rehace la Imagen 10 del tema original (a partir de
% Pindyck y Rubinfeld) y añade el panel de cantidades.
% Parámetros: alpha = 10, c = 2, beta = 1, gamma = 0,6.
%   Cournot:  R_i(q_j) = (alpha - c - gamma q_j)/(2 beta) = 4 - 0,3 q_j
%             q^C = (alpha - c)/(2 beta + gamma) = 8/2,6 = 40/13
%   Bertrand: R_i(p_j) = (alpha(beta-gamma) + beta c + gamma p_j)/(2 beta) = 3 + 0,3 p_j
%             p^B = c + (alpha-c)(beta-gamma)/(2 beta - gamma) = 2 + 3,2/1,4 = 30/7 > c = 2
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.62cm,y=0.62cm,
    declare function={rq(\q)=4-0.3*\q; rp(\p)=3+0.3*\p;}]
  \useasboundingbox (-1.3,-1.3) rectangle (20.6,9.6);
  % ---------------- Izquierda: cantidades (Cournot) ----------------
  % 1: ejes
  \capa{1}{\ejes{$q_1$}{$q_2$}{7.8}{7.8}}
  % 2: reacción de la empresa 1, q_1 = R_1(q_2): decreciente (más inclinada en el plano)
  \capa{2}{\draw[curva1] plot[domain=0:7.2] ({rq(\x)},\x) node[etiqueta,above] {$R_1(q_2)$};}
  % 3: reacción de la empresa 2, q_2 = R_2(q_1)
  \capa{3}{\draw[curva2] plot[domain=0:7.2] (\x,{rq(\x)}) node[etiqueta,right] {$R_2(q_1)$};}
  \capa{3}{\node[etiqueta,align=center] at (3.9,8.9) {cantidades:\\ \emph{sustitutivos estratégicos}};}
  % 4: equilibrio de Cournot
  \capa{4}{\node[punto,label={[etiqueta]above right:$C$}] (C) at ({40/13},{40/13}) {};}
  \capa{4}{\proyeccion{(C)}{$q^C$}{$q^C$}}
  % ---------------- Derecha: precios (Bertrand) ----------------
  \begin{scope}[xshift=6.6cm]
  % 5: ejes, nivel del coste marginal c y diagonal p_1 = p_2
  \capa{5}{\ejes{$p_1$}{$p_2$}{7.8}{7.8}}
  \capa{5}{\draw[guia] (0,0) -- (7.2,7.2) node[etiqueta,above,text=tceegris] {$p_1=p_2$};}
  \capa{5}{\node[punto,fill=tceegris,inner sep=1pt] (c) at (2,2) {};}
  \capa{5}{\proyeccion{(c)}{$c$}{$c$}}
  % 6: reacción de la empresa 1, p_1 = R_1(p_2): creciente
  \capa{6}{\draw[curva1] plot[domain=0:7.2] ({rp(\x)},\x) node[etiqueta,above] {$R_1(p_2)$};}
  % 7: reacción de la empresa 2, p_2 = R_2(p_1)
  \capa{7}{\draw[curva2] plot[domain=0:7.2] (\x,{rp(\x)}) node[etiqueta,right] {$R_2(p_1)$};}
  \capa{7}{\node[etiqueta,align=center] at (3.9,8.9) {precios:\\ \emph{complementos estratégicos}};}
  % 8: equilibrio de Bertrand, sobre la diagonal y por encima de c
  \capa{8}{\node[punto,label={[etiqueta]below right:$B$}] (B) at ({30/7},{30/7}) {};}
  \capa{8}{\proyeccion{(B)}{$p^B$}{$p^B$}}
  \end{scope}
\end{tikzpicture}
\end{document}
