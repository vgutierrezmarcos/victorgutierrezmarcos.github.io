% Demanda de la empresa 1 con costes de transporte lineales (3.A.18, nota del
% anexo de Hotelling; d'Aspremont, Gabszewicz y Thisse, 1979). Empresas en
% a = 0,25 y b = 0,75, t = 2 y p_2 = 2, así que t (b - a) = 1:
%   p_1 > p_2 + t(b-a) = 3      -> x_1 = 0 (la 2 se queda también con el patio
%                                   trasero de la 1: salto de a a 0);
%   |p_1 - p_2| < t(b-a)         -> x_1 = (a+b)/2 + (p_2 - p_1)/(2t), de b a a;
%   p_1 < p_2 - t(b-a) = 1       -> x_1 = 1 (salto de b a 1).
% Elaboración propia a partir de d'Aspremont, Gabszewicz y Thisse (1979).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=1.3cm,y=3cm,
    abierto/.style={circle,draw=tceemorado,fill=white,thick,inner sep=1.5pt},
    cerrado/.style={circle,fill=tceemorado,inner sep=1.8pt}]
  \def\pb{1}\def\pa{3}
  \useasboundingbox (-0.75,-0.24) rectangle (4.75,1.3);
  % Ejes
  \capa{1}{\ejes{$p_1$}{$x_1$}{4.6}{1.25}}
  % Precio muy alto: la empresa 1 no vende
  \capa{2}{\draw[curva1] (\pa,0) -- (4.3,0);}
  \capa{2}{\draw[guia] (\pa,0.25) -- (\pa,0) node[below,text=black] {$p_2+t\,(b-a)$};}
  \capa{2}{\node[abierto] at (\pa,0) {};}
  % Tramo continuo: consumidor indiferente entre a y b
  \capa{3}{\draw[curva1] (\pb,0.75) -- (\pa,0.25);}
  \capa{3}{\draw[guia] (\pa,0.25) -- (0,0.25) node[left,text=black] {$a$};}
  \capa{3}{\draw[guia] (\pb,0.75) -- (0,0.75) node[left,text=black] {$b$};}
  \capa{3}{\node[cerrado] at (\pa,0.25) {};}
  \capa{3}{\node[cerrado] at (\pb,0.75) {};}
  % Precio bajo: salto a x_1 = 1 (se queda con el patio trasero de la rival)
  \capa{4}{\draw[curva1] (0,1) -- (\pb,1);}
  \capa{4}{\node[left] at (0,1) {$1$};}
  \capa{4}{\draw[guia] (\pb,1) -- (\pb,0) node[below,text=black] {$p_2-t\,(b-a)$};}
  \capa{4}{\node[abierto] at (\pb,1) {};}
  \capa{4}{\node[cerrado] at (\pb,0.75) {};}
\end{tikzpicture}
\end{document}
