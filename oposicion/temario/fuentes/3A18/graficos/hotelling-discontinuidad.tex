% Ciudad lineal con costes de transporte lineales (3.A.18, anexo:hotelling):
% empresa 1 en a y empresa 2 en b, precio total p_i + t |x - loc_i|, con t = 2.
% a) a = 0,25, b = 0,8, p_1 = 0,5, p_2 = 0,8: las V se cortan en
%    x~ = [(p_2 - p_1)/t + a + b]/2 = 0,6, con precio total 1,2.
% b) a = 0,35, b = 0,75, p_1 = 0,3, p_2 = 1,3: p_1 + t (b - a) = 1,1 < p_2, la V
%    de la 1 queda por debajo de la de la 2 en todo [0,1] (también en su
%    «patio trasero»): no hay consumidor indiferente y la 1 se queda todo.
% Elaboración propia a partir de d'Aspremont, Gabszewicz y Thisse (1979).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=6.5cm,y=1.5cm,
    llave/.style={decorate,decoration={brace,amplitude=3pt},thin,tceegris},
    llaveabajo/.style={decorate,decoration={brace,mirror,amplitude=5pt}},
    nota/.style={etiqueta,font=\footnotesize,align=center},
    titulo/.style={font=\small,anchor=south},
    rotulo/.style={below,text height=1.5ex,text depth=0.25ex},
    eti/.style={font=\footnotesize,sloped,above}]
  \useasboundingbox (-0.12,-1.3) rectangle (2.38,2.45);
  % ======================= a) Hay consumidor indiferente =======================
  % 1: ciudad lineal
  \capa{1}{\node[titulo] at (0.5,2.05) {a) Hay consumidor indiferente};}
  \capa{1}{\draw[line width=1.6pt] (0,0) -- (1,0);}
  \capa{1}{\node[rotulo] at (0,0) {$0$};}
  \capa{1}{\node[rotulo] at (1,0) {$1$};}
  % 2: empresa 1 en a y su precio
  \capa{2}{\node[rotulo] at (0.25,0) {$a$};}
  \capa{2}{\draw[curva1] (0.25,0) -- (0.25,0.5);}
  \capa{2}{\draw[llave] (0.26,0.5) -- (0.26,0) node[midway,right=3pt,text=black] {$p_1$};}
  % 3: precio total de la empresa 1
  \capa{3}{\draw[curva1] (0,1.0) -- (0.25,0.5) -- (0.7,1.4) node[eti,pos=0.35,text=tceemorado] {$p_1+t\,|x-a|$};}
  % 4: empresa 2 en b y su precio
  \capa{4}{\node[rotulo] at (0.8,0) {$b$};}
  \capa{4}{\draw[curva2] (0.8,0) -- (0.8,0.8);}
  \capa{4}{\draw[llave] (0.81,0.8) -- (0.81,0) node[midway,right=3pt,text=black] {$p_2$};}
  % 5: precio total de la empresa 2
  \capa{5}{\draw[curva2] (0.5,1.4) -- (0.8,0.8) -- (1,1.2) node[eti,pos=0.4,text=tceeazul] {$p_2+t\,|x-b|$};}
  % 6: consumidor indiferente
  \capa{6}{\draw[guia] (0.6,1.2) -- (0.6,0) node[rotulo,text=black] {$\tilde{x}$};}
  \capa{6}{\node[punto] at (0.6,1.2) {};}
  \capa{6}{\node[nota] at (0.6,-0.62) {consumidor\\ indiferente};}
  % 7: mercados cautivos
  \capa{7}{\draw[llaveabajo] (0.005,-0.42) -- (0.245,-0.42) node[nota,midway,below=6pt] {mercado\\ cautivo de 1};}
  \capa{7}{\draw[llaveabajo] (0.805,-0.42) -- (0.995,-0.42) node[nota,midway,below=6pt] {mercado\\ cautivo de 2};}
  % ================= b) La empresa 1 se queda todo el mercado =================
  \begin{scope}[xshift=8.3cm]
    % 8: ciudad lineal
    \capa{8}{\node[titulo] at (0.5,2.05) {b) La empresa 1 se queda todo el mercado};}
    \capa{8}{\draw[line width=1.6pt] (0,0) -- (1,0);}
    \capa{8}{\node[rotulo] at (0,0) {$0$};}
    \capa{8}{\node[rotulo] at (1,0) {$1$};}
    % 9: empresa 1 en a con un precio muy bajo
    \capa{9}{\node[rotulo] at (0.35,0) {$a$};}
    \capa{9}{\draw[curva1] (0.35,0) -- (0.35,0.3);}
    \capa{9}{\draw[llave] (0.36,0.3) -- (0.36,0) node[midway,right=3pt,text=black] {$p_1$};}
    % 10: precio total de la empresa 1
    \capa{10}{\draw[curva1] (0.05,0.9) -- (0.35,0.3) -- (1,1.6) node[eti,pos=0.3,text=tceemorado] {$p_1+t\,|x-a|$};}
    % 11: empresa 2 en b y su precio
    \capa{11}{\node[rotulo] at (0.75,0) {$b$};}
    \capa{11}{\draw[curva2] (0.75,0) -- (0.75,1.3);}
    \capa{11}{\draw[llave] (0.76,1.3) -- (0.76,0) node[midway,right=3pt,text=black] {$p_2$};}
    % 12: precio total de la empresa 2, siempre por encima
    \capa{12}{\draw[curva2] (0.4,2.0) -- (0.75,1.3) -- (1,1.8) node[eti,pos=0.4,text=tceeazul] {$p_2+t\,|x-b|$};}
    % 13: también en b compensa comprar a 1: todos le compran
    \capa{13}{\node[punto] at (0.75,1.1) {};}
    \capa{13}{\draw[llaveabajo] (0.005,-0.42) -- (0.995,-0.42) node[nota,midway,below=6pt] {todos compran a 1: $p_1+t\,(b-a)<p_2$};}
  \end{scope}
\end{tikzpicture}
\end{document}
