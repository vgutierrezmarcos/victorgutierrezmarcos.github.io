% El juego del ultimátum (Güth, Schmittberger y Schwarze, 1982) con 10 euros.
% El jugador 1 ofrece x en [0, 10] al 2 (el abanico recoge todas las ofertas);
% el 2 acepta, pagos (10 - x, x), o rechaza, (0, 0). Pagos (1, 2).
% Inducción hacia atrás con jugadores que solo miran su dinero: el 2 acepta
% cualquier x > 0 (x > 0) y el 1 ofrece lo mínimo. ENPS: ofrecer lo mínimo; aceptar.
% En el laboratorio, muchas ofertas se acercan a la mitad (Güth y otros, 1982).
% Mismo estilo que juego-entrada.tex. Elaboración propia.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,
    nodo/.style={circle,draw=tceemorado,thick,fill=tceesalvia,inner sep=2.2pt},
    terminal/.style={circle,fill=tceemorado,inner sep=1.2pt},
    rama/.style={thick,tceemorado},
    jugador/.style={etiqueta,font=\small\itshape},
    accion/.style={etiqueta,font=\footnotesize},
    pago/.style={etiqueta,below=2pt,font=\small},
    elegida/.style={line width=1.8pt,tceerojo}]
  \useasboundingbox (-4.3,-4.35) rectangle (5.2,0.55);
  % 1: nodo del jugador 1, que propone el reparto
  \capa{1}{\node[nodo,label={[jugador]right:{\upshape\textbf{1}} (propone)}] (P) at (0,0) {};}
  % 2: el abanico de ofertas, de 0 a 10 euros para el 2
  \capa{2}{\fill[tceesalvia] (P.center) -- (-2.8,-1.9) -- (2.8,-1.9) -- cycle;}
  \capa{2}{\draw[rama] (-2.8,-1.9) -- (P.center) -- (2.8,-1.9);}
  \capa{2}{\draw[rama] (-2.8,-1.9) -- (2.8,-1.9);}
  \capa{2}{\node[accion,left] at (-2.8,-1.9) {$x=0$};}
  \capa{2}{\node[accion,right] at (2.8,-1.9) {$x=10$};}
  % 3: una oferta cualquiera, x euros para el 2, y su nodo
  \capa{3}{\draw[rama] (P) -- node[accion,right,pos=0.6] {$x$} (1.3,-1.9);}
  \capa{3}{\node[nodo,label={[jugador]above right=1pt:{\upshape\textbf{2}}}] (R) at (1.3,-1.9) {};}
  % 4: el 2 acepta o rechaza
  \capa{4}{\node[terminal] (A) at (-0.2,-3.6) {};}
  \capa{4}{\draw[rama] (R) -- node[accion,left,pos=0.45] {Aceptar} (A);}
  \capa{4}{\node[pago] at (A) {$(10-x,\,x)$};}
  \capa{4}{\node[terminal] (N) at (2.8,-3.6) {};}
  \capa{4}{\draw[rama] (R) -- node[accion,right,pos=0.45] {Rechazar} (N);}
  \capa{4}{\node[pago] at (N) {$(0,\,0)$};}
  % 5: si solo mira su dinero, el 2 acepta cualquier x > 0
  \capa{5}{\draw[elegida] (R) -- (A);}
  % 6: anticipándolo, el 1 ofrece lo mínimo
  \capa{6}{\node[punto,fill=tceerojo] at (-2.55,-1.9) {};}
  \capa{6}{\node[etiqueta,text=tceerojo,anchor=north west,align=left] at (-4.2,-2.35) {ENPS: el 1 ofrece\\lo mínimo; el 2 acepta};}
  % 7: en el laboratorio, las ofertas se acercan a la mitad
  \capa{7}{\draw[line width=1.6pt,tceedorado] (-0.6,-1.9) -- (0,-1.9);}
  \capa{7}{\node[etiqueta,text=tceedorado,anchor=south,font=\footnotesize,align=center] at (-1.05,-1.75) {laboratorio:\\cerca de la mitad};}
\end{tikzpicture}
\end{document}
