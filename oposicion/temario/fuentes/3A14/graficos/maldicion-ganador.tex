% La maldición del ganador con valor común. Cada pujador estima el valor V con un
% error uniforme en [-a, a]: s_i ~ U[V - a, V + a], con densidad 1/(2a).
% Con n pujadores, la estimación más alta (la del que gana si puja su estimación)
% tiene densidad n (s - V + a)^(n-1) / (2a)^n y media V + a (n - 1)/(n + 1).
% Aquí V = 3, a = 2, n = 5: densidad de cada estimación 0,25; media de la más alta
% 3 + 2(4/6) = 4,33. Eje vertical multiplicado por 2 para que se vea.
% Elaboración propia a partir de Wilson (1977) y Krishna (2010, cap. 6).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=1.45cm,y=1.25cm]
  \useasboundingbox (-0.5,-0.65) rectangle (7.1,3.6);
  % 1: ejes: estimación del valor y densidad
  \capa{1}{\draw[eje] (0,0) -- (6.6,0) node[right] {$s_i$};}
  \capa{1}{\draw[eje] (0,0) -- (0,3.1) node[above] {densidad};}
  \capa{1}{\node[below left] at (0,0) {$0$};}
  % 2: el valor real, igual para todos y desconocido
  \capa{2}{\draw[guia] (3,0) node[below,text=black] {$V$} -- (3,2.95);}
  \capa{2}{\node[etiqueta,above,font=\footnotesize] at (3,2.95) {valor real};}
  % 3: la estimación de cada pujador: acierta en promedio
  \capa{3}{\draw[curva1] (1,0) -- (1,0.5) -- (5,0.5) -- (5,0);}
  \capa{3}{\node[below] at (1,0) {$V-a$};}
  \capa{3}{\node[below] at (5,0) {$V+a$};}
  \capa{3}{\node[etiqueta,text=tceemorado,anchor=south west,font=\footnotesize] at (1.05,0.5) {cada estimación};}
  % 4: la estimación más alta de 5 pujadores, la del que gana
  \capa{4}{\draw[curva2,domain=1:5,samples=60] plot (\x,{2.5*((\x-1)/4)^4});}
  \capa{4}{\node[etiqueta,text=tceeazul,anchor=west,align=left,font=\footnotesize] at (5.1,2.5) {la más alta\\de $n=5$};}
  % 5: en promedio, la estimación del ganador supera el valor
  \capa{5}{\draw[guia] (4.333,0) -- (4.333,2.35);}
  \capa{5}{\node[etiqueta,font=\footnotesize,anchor=east] at (4.3,2.2) {$V+a\,\frac{n-1}{n+1}$};}
  % 6: la maldición del ganador: lo que paga de más quien puja su estimación
  \capa{6}{\draw[flecha,<->] (3,1.45) -- (4.333,1.45);}
  \capa{6}{\node[etiqueta,text=tceerojo,above,font=\footnotesize,align=center] at (3.67,1.45) {maldición\\del ganador};}
\end{tikzpicture}
\end{document}
