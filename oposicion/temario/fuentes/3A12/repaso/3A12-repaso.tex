\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: «\emph{La práctica hace al maestro}», y pocas industrias lo demuestran mejor que la fotovoltaica.
  \begin{itemize}
  \item Coste nivelado $0{,}41\to0{,}044$ \$/kWh (2010-2025, $-89$\,\%); módulos $-27$\,\% por cada duplicación de la producción acumulada.
  \item España: primera por potencia (más de 41.500~MW), 18,4\,\% de la electricidad en 2025 \vertema{4.A.5}.
  \item \emph{¿Cuánto se debe a la escala, cuánto al aprendizaje y cuánto a los precios de los factores?}
  \end{itemize}
\item \textbf{Relevancia}:
  \begin{itemize}
  \item \emph{Teórica}: puente tecnología--mercado: oferta, cuántas empresas caben.
  \item \emph{Empírica}: tecnología desde costes y precios.
  \item \emph{Política}: regulación de redes (comparar eficiencias), predación de \autor{Areeda}-\autor{Turner}, comercio intraindustrial \vertema{3.B.6}.
  \end{itemize}
\item \textbf{Contextualización}:
  \begin{itemize}
  \item 3~restricciones: \emph{técnicas} \vertema{3.A.11}, \emph{económicas} (costes), \emph{de mercado} \vertema{3.A.16}.
  \item \concepto{Coste económico} = coste de oportunidad $\neq$ contable; \concepto{función de costes} $C(\vect{w},q)$: coste mínimo de producir $q$ a precios $\vect{w}$.
  \item \autor{Marshall} (1890), \autor{Wicksell} (CMe en U), \autor{Viner} (1931) $\to$ dualidad (\autor{Hotelling} 1932, \autor{Shephard} 1953, \autor{McFadden}) $\to$ estimación (\autor{Nerlove}, translog) y eficiencia (\autor{Farrell}, 1957); críticas: \autor{Stigler} (1939), \autor{Leibenstein} (1966).
  \item «Dual»: estricto = intercambiar objetivo y restricción; MB--MC no lo es (lleva ingresos): entre comillas.
  \end{itemize}
\item \textbf{Problemática}: ¿qué contiene $C$? ¿Cómo responde a $q$, $\vect{w}$ y plazo? ¿Cómo se estima? ¿CMe en U? ¿Minimizan costes las empresas?
\item \textbf{Estructura}: \hyperref[rep:I]{I}, «dualidad»; \hyperref[rep:II]{II}, estática comparativa, plazos y empírica; \hyperref[rep:III]{III}, extensiones y críticas.
\end{itemize}

\bloque[10,5]{I. «Dualidad» MB--MC. Función de costes\label{rep:I}}
\begin{ideaclave}
Minimizar el coste de cada $q$ y elegir $p=\mathrm{CMg}$ = maximizar el beneficio. $C$ es cóncava y homogénea de grado 1~en $\vect{w}$, sus derivadas son las demandas condicionadas (\autor{Shephard}) y contiene la misma información que la tecnología: basta conocer los costes para conocer la empresa.
\end{ideaclave}

\sub{I.1. Ingredientes comunes\label{rep:I.1}}
\begin{itemize}
\item \concepto{Eficiencia técnica} (no hay otro proceso con menos factores para igual $q$) no basta: deja toda la isocuanta $\Rightarrow$ hacen falta precios $\Rightarrow$ \concepto{eficiencia económica}: cada $q$ de la forma más barata.
\item \emph{Ingredientes}: $\mathcal{Y}$, $q=F(\vect{z})$, $V(q)=\{\vect{z}\geq\vect{0}:F(\vect{z})\geq q\}$; $F$ \emph{estr.\ cuasicóncava} (MB exige además \emph{estr.\ cóncava} $\Rightarrow$ $C$ existe con cualquier rendimiento, $B$ no); $\vect{w}\gg\vect{0}$ y, en MB, $p>0$. \emph{Supuestos}: estático, información perfecta, un producto y una planta, \emph{largo plazo}.
\end{itemize}

\toprule

\midrule

\sub{I.2. Problemas primal y dual\label{rep:I.2}}
\subsub{a) Maximización del beneficio en una etapa}
\begin{itemize}
\item \emph{MB en una etapa}: contiene toda la información, pero solo vale para la empresa precio-aceptante en todo.
\begin{formula}
\centering $\optim{\max}{\vect{z}}{p\,F(\vect{z})-\vect{w}\cdot\vect{z}}{\vect{z}\geq\vect{0}}$\qquad $\Rightarrow$\quad $p\,\mathrm{PMg}_i=w_i$,\ \ $\mathrm{RMST}_{LK}=\dfrac{w_L}{w_K}$
\end{formula}
\item Valor del PMg = precio del factor. Con RCE, $B^*=0$ y escala indeterminada; con crecientes, no acotado. \concepto{Demandas no condicionadas} $\vect{z}(p,\vect{w})$ y \concepto{oferta} $q(p,\vect{w})$: únicas, continuas (\autor{Berge}), homog.\ de grado 0~en $(p,\vect{w})$.
\funcionclave{B(p,\vect{w})=\optim{\max}{\vect{z}}{p\,F(\vect{z})-\vect{w}\cdot\vect{z}}{\vect{z}\geq\vect{0}}}
\item \concepto{Propiedades de $B$}: continua; homog.\ de grado 1~en $(p,\vect{w})$; no decreciente en $p$ y no creciente en $w_i$; \emph{convexa} en $(p,\vect{w})$ (reoptimizar gana al menos lo lineal); lema de \autor{Hotelling}.
\begin{formula}
\centering \autor{Hotelling} (1932):\quad $\dfrac{\partial B}{\partial p}=q(p,\vect{w})$,\qquad $\dfrac{\partial B}{\partial w_i}=-z_i(p,\vect{w})$\quad (análogo de \autor{Roy})
\end{formula}
\item $D^2B$ \emph{simétrica} y \emph{semidef.\ positiva}:
  \begin{itemize}
  \item $\partial q/\partial p\geq0$: \emph{ley de la oferta};
  \item $\partial z_i(p,\vect{w})/\partial w_i\leq0$: sin restricción presupuestaria no hay efecto renta $\Rightarrow$ \emph{sin factores \autor{Giffen}}; base de la demanda derivada de trabajo \vertema{3.A.25}.
  \end{itemize}
\item \emph{Valoración}: todo en un paso, pero solo precio-aceptante y rendimientos decrecientes $\Rightarrow$ aislar lo que no depende de los ingresos: MC.
\end{itemize}
\subsub{b) Problema dual: minimización del coste}
\begin{itemize}
\item \emph{MC}: fijado $q$, ¿combinación más barata? Vale para competencia, monopolio u oligopolio (si precio-aceptantes en factores; \autor{Shephard}, 1953).
\begin{formula}
\centering $\optim{\min}{\vect{z}}{\vect{w}\cdot\vect{z}}{F(\vect{z})\geq q\\ \vect{z}\geq\vect{0}}$\qquad $\mathcal{L}=\vect{w}\cdot\vect{z}+\mu\big(q-F(\vect{z})\big)$\\[4pt]
$\dfrac{w_L}{\mathrm{PMg}_L}=\dfrac{w_K}{\mathrm{PMg}_K}=\mu$,\qquad $\mathrm{RMST}_{LK}=\dfrac{w_L}{w_K}$
\end{formula}
\item \emph{Misma} tangencia que en MB. $\mu^*=\partial C/\partial q=$ \concepto{coste marginal} (envolvente). CSO: $|\overline{H}|>0$.
\item \concepto{Demandas condicionadas} $\vect{z}(\vect{w},q)$, análogas a las \emph{hicksianas}: únicas; continuas; homog.\ de grado 0~en $\vect{w}$ (no en $q$); $F(\vect{z}(\vect{w},q))=q$; $(\vect{w}'-\vect{w})\cdot(\vect{z}'-\vect{z})\leq0$.
\funcionclave{C(\vect{w},q)=\vect{w}\cdot\vect{z}(\vect{w},q)=\optim{\min}{\vect{z}}{\vect{w}\cdot\vect{z}}{F(\vect{z})\geq q\\ \vect{z}\geq\vect{0}}}
\item \concepto{Propiedades de $C$}: continua, $C(\vect{w},0)=0$; homog.\ de grado 1~en $\vect{w}$; estr.\ creciente en $q$; no decreciente en $w_i$; \emph{cóncava en $\vect{w}$} (la empresa «pasiva» pagaría un coste lineal; sustituyendo, paga como mucho eso); \autor{Shephard}; matriz de sustitución.
\funcionclave{\text{\autor{Shephard} (1953):}\quad z_i(\vect{w},q)=\frac{\partial C(\vect{w},q)}{\partial w_i}}
\item $S(\vect{w},q)=D^2_{\vect{w}}C$, $S_{ij}=\partial z_i(\vect{w},q)/\partial w_j$: \emph{simétrica} (\autor{Schwarz}/\autor{Young}); \emph{semidef.\ negativa} ($S_{ii}\leq0$, ley de la demanda condicionada); $S\vect{w}=\vect{0}$ (\autor{Euler}) $\Rightarrow$ con 2~factores, \emph{sustitutivos netos}.
\end{itemize}
\graficorepaso[0.5\linewidth]{mc}{MC: la isocoste más baja que alcanza la isocuanta $q$}
\begin{itemize}
\item \emph{Ejemplos} \vertema{3.A.11}: \autor{Leontief}, coste lineal en $q$; Cobb-Douglas, participaciones constantes; homotética, $C=\phi(q)\,c(\vect{w})$.
\end{itemize}

\sub{I.3. Relaciones de «dualidad»\label{rep:I.3}}
\subsub{a) Dos etapas: de $C$ a $B$}
\begin{itemize}
\item \concepto{MB en dos etapas}: 1.º mín.\ coste de cada $q$ $\to C$; 2.º elegir $q$:
\begin{formula}
\centering $B(p,\vect{w})=\optim{\max}{q}{p\,q-C(\vect{w},q)}{q\geq0}$\quad$\Rightarrow$\quad $p=\mathrm{CMg}$,\ \ $\dfrac{\partial^2C}{\partial q^2}\geq0$\\[4pt]
$\vect{z}(p,\vect{w})\equiv\vect{z}\big(\vect{w},q(p,\vect{w})\big)$;\quad $B(p,\vect{w})\equiv p\,q(p,\vect{w})-C\big(\vect{w},q(p,\vect{w})\big)$
\end{formula}
\item Oferta = tramo creciente del CMg sobre el CMe evitable \vertema{3.A.16}. $B$ desde $C$ sin volver a minimizar; lemas coherentes: $\partial B/\partial w_i=-\partial C/\partial w_i$.
\item \emph{Asimetría}: $C$ con cualquier rendimiento, $B$ no $\Rightarrow$ $C$, más general.
\end{itemize}
\graficorepaso[0.9\linewidth]{esquema-dualidad-empresa}{Esquema de dualidad de la empresa: MB (izq.), MC (dcha.), dos etapas (arriba) y tecnología (abajo)}
\subsub{b) $C\leftrightarrow F$ y $B\leftrightarrow\mathcal{Y}$}
\begin{itemize}
\item \concepto{$C\leftrightarrow F$} (\autor{Shephard} 1953, \autor{McFadden} 1978): $V^*(q)=\{\vect{z}\geq\vect{0}:\vect{w}\cdot\vect{z}\geq C(\vect{w},q)\ \forall\vect{w}\gg\vect{0}\}$ = envolvente de las isocostes mínimas. $V$ convexo + eliminación gratuita $\Rightarrow$ $V^*=V$; si no, su envoltura convexa (la que importa para la conducta).
\item Toda $C$ continua, no decreciente, homog.\ de grado 1~y cóncava en $\vect{w}$ y creciente en $q$ es función de costes de alguna tecnología (\autor{Shephard}, \autor{Uzawa}): la «integrabilidad» $\Rightarrow$ se estima $C$ sin escribir $F$.
\item \concepto{$B\leftrightarrow\mathcal{Y}$}: $B$ = \concepto{función soporte} de $\mathcal{Y}$; si $\mathcal{Y}$ es convexo, cerrado y con eliminación gratuita, es la intersección de los semiespacios de las isobeneficio máximas.
\item Tecnología, costes y beneficio: 3~descripciones equivalentes; la de costes, más general, es la que se estima.
\end{itemize}

\bloque[12]{II. Estática comparativa, plazos y aplicaciones empíricas\label{rep:II}}
\begin{ideaclave}
La dualidad convierte la teoría de los costes en un instrumento empírico: con costes, precios y producción se estiman escala y sustitución sin conocer la tecnología, y con fronteras, la distancia al coste mínimo. Los datos matizan: costes en L e ineficiencias persistentes.
\end{ideaclave}

\sub{II.1. Estática comparativa\label{rep:II.1}}
\subsub{a) Variaciones de la producción (escala)}
\begin{itemize}
\item \emph{$\Delta q$, demandas}: \concepto{senda de expansión}: combinaciones de mínimo coste al $\uparrow q$ con $\vect{w}$ fijo; RMST constante $\Rightarrow$ isoclina (análogo de la curva renta-consumo); homotética $\Rightarrow$ rayo.
\item Factor \concepto{normal} si $\partial z_i(\vect{w},q)/\partial q>0$; \concepto{inferior} si $<0$ (senda hacia atrás). No todos inferiores: al menos uno, normal.
\end{itemize}
\pareja{senda-expansion}{Senda de expansión, con trabajo inferior entre $B$ y $C$}{costes-largo}{CMe: pendiente del rayo; CMg: de la tangente}
\begin{itemize}
\item \emph{$\Delta q$, costes}: $\mathrm{CMe}=C/q$ (rayo), $\mathrm{CMg}=\partial C/\partial q$ (tangente); CMg corta al CMe en su mínimo.
\begin{formula}
\centering $\dfrac{d\,\mathrm{CMe}}{dq}=\dfrac{\mathrm{CMg}-\mathrm{CMe}}{q}$;\qquad $\varepsilon_{C,q}\equiv\dfrac{\partial C}{\partial q}\dfrac{q}{C}=\dfrac{\mathrm{CMg}}{\mathrm{CMe}}$\\[5pt]
\autor{Hanoch} (1975):\ \ $\boxed{\varepsilon_{q,\lambda}\big(\vect{z}(\vect{w},q)\big)=\dfrac{\mathrm{CMe}}{\mathrm{CMg}}=\dfrac{1}{\varepsilon_{C,q}}}$
\end{formula}
\item $\varepsilon_{C,q}<1$: \concepto{economías de escala} (CMe $\downarrow$); $=1$: CMe mínimo (\emph{escala mínima eficiente}); $>1$: \concepto{deseconomías}. Grado: $1-\varepsilon_{C,q}$.
\item \emph{Rendimientos} (tecnología, rayo, $\varepsilon_{q,\lambda}=\sum_i\mathrm{PMg}_i/\mathrm{PMe}_i$) $\neq$ \emph{economías} (costes, senda):
  \begin{itemize}
  \item inversas en cada punto de la senda (\autor{Hanoch}), \emph{sin} homoteticidad, con rendimientos \emph{locales};
  \item \emph{globales}: solo $\Rightarrow$ (crecientes en todo punto dan economías en todo $q$, no al revés).
  \end{itemize}
\item CMe LP en U $\Leftarrow$ rendim.\ \emph{variables} («en S», \autor{Wicksell}), no proporciones variables (CP). Homogéneas: CMe monótono; RCE: tamaño indeterminado.
\end{itemize}
\subsub{b) Variaciones de los precios de los factores (sustitución)}
\begin{itemize}
\item \emph{$\Delta\vect{w}$, demandas}: $w_L\uparrow$ $\Rightarrow$ menos $z_L$, más $z_K$ en la misma isocuanta: \concepto{demanda condicionada} decreciente.
\item ¿Factores \autor{Giffen}? No: concavidad de $C$ y convexidad de $B$. Sí inferiores.
\item \autor{Slutsky} de factores, derivando $z_i(p,\vect{w})\equiv z_i(\vect{w},q(p,\vect{w}))$:
\begin{formula}
\centering $\underbrace{\dfrac{\partial z_i(p,\vect{w})}{\partial w_j}}_{\text{total}}=\underbrace{\dfrac{\partial z_i(\vect{w},q)}{\partial w_j}}_{\text{sustitución}}+\underbrace{\dfrac{\partial z_i(\vect{w},q)}{\partial q}\dfrac{\partial q(p,\vect{w})}{\partial w_j}}_{\text{escala}}$
\end{formula}
\item \concepto{Efecto escala} (o producción) sustituye al efecto renta; con CMg creciente, el propio es $\leq0$ y \emph{refuerza} al de sustitución $\Rightarrow$ no condicionada \emph{más elástica} (\autor{Le Chatelier}-\autor{Samuelson}).
\item \emph{$\Delta\vect{w}$, costes}: todos los $w_i$ en igual proporción $\Rightarrow$ $C$ igual (grado~1); uno solo: $\varepsilon_{C,w_i}=s_i$ (\autor{Shephard}). $\partial\mathrm{CMg}/\partial w_i=\partial z_i/\partial q$: normal $\Rightarrow$ CMg $\uparrow$; inferior $\Rightarrow$ CMg $\downarrow$ y la producción sube.
\end{itemize}

\sub{II.2. Análisis de plazos\label{rep:II.2}}
\begin{itemize}
\item \autor{Marshall}: \concepto{corto plazo} (algún factor fijo, la planta) y \concepto{largo plazo} (todos variables; costes \emph{planeados}, \emph{ex ante}).
\begin{formula}
\centering $\optim{\min}{z_L}{w_Lz_L+w_K\overline{z}_K}{F(z_L,\overline{z}_K)\geq q\\ z_L\geq0}$\\[4pt]
$\boxed{C(\vect{w},q;\overline{z}_K)=\underbrace{w_Lz_L(\vect{w},q;\overline{z}_K)}_{\mathrm{CV}(q)}+\underbrace{w_K\overline{z}_K}_{\mathrm{CF}}}\ \geq C(\vect{w},q)$\\[4pt]
$\mathrm{CMg}=\dfrac{w_L}{\mathrm{PMg}_L}$,\quad $\mathrm{CVMe}=\dfrac{w_L}{\mathrm{PMe}_L}$,\quad $\varepsilon_{\mathrm{CV},q}=\dfrac{\mathrm{CMg}}{\mathrm{CVMe}}=\dfrac{1}{\varepsilon_{q,z_L}}$
\end{formula}
\item $q^1$: lo ideal es $A$ (senda), con $\overline{z}_K$ hay que ir a $B$; en $q^2$, $D$: coinciden. Económicamente eficiente solo si $\overline{z}_K=z_K(\vect{w},q)$.
\item CMg = imagen invertida del PMg $\Rightarrow$ U por la \concepto{ley de proporciones variables}; CMe baja por el reparto del CF: \emph{no} son economías de escala.
\item \emph{Mínimo de explotación} = mín.\ CVMe (cierre a CP); \emph{óptimo de explotación} = mín.\ CMe (beneficio).
\end{itemize}
{\centering\small\renewcommand{\arraystretch}{1.15}\setlength{\tabcolsep}{3pt}%
\begin{tabular}{@{}llcc@{}}
\toprule
\textit{Tramo} & \textit{Costes} & $\varepsilon_{q,z_L}$ & $\varepsilon_{\mathrm{CV},q}$\\
\midrule
\textcolor{tceedorado}{Etapa I} & CMg $<$ CVMe $\downarrow$ & $>1$ & $(0,1)$\\
Margen extensivo & mín.\ CVMe (mín.\ explot.) & $=1$ & $=1$\\
\textcolor{tceeverde}{Etapa II} & CMg $>$ CVMe $\uparrow$ (óptimo expl.) & $(0,1)$ & $>1$\\

\bottomrule
\end{tabular}\par}
\begin{itemize}
\item \emph{LP}: CT y CMe = \concepto{envolvente} de las de CP. Izq.\ del mín.: exceso de capacidad; dcha.: sobrecarga. Solo en el mín.\ coinciden los cuatro (error de \autor{Viner}, 1931). CMg$_{\mathrm{CP}}$ corta al de LP desde abajo.
\end{itemize}
\graficorepaso[0.45\linewidth]{envolvente}{Envolvente: CT (arriba) y CMe, CMg (abajo); tangencias en $q_1,q_2,q_3$}
\begin{formula}
\centering \autor{Le Chatelier}-\autor{Samuelson} (1947):\\[2pt] $\dfrac{\partial z_i^{\mathrm{LP}}}{\partial w_i}\leq\dfrac{\partial z_i^{\mathrm{CP}}}{\partial w_i}\leq0$,\qquad $\dfrac{\partial q^{\mathrm{LP}}}{\partial p}\geq\dfrac{\partial q^{\mathrm{CP}}}{\partial p}\geq0$
\end{formula}
\begin{itemize}
\item Más restricciones $\Rightarrow$ menos respuesta (químico \autor{Le Chatelier}, 1884).
\end{itemize}
\begin{ideaclave}
LP: los costes reflejan los rendimientos a escala ($\varepsilon_{q,\lambda}=1/\varepsilon_{C,q}$ en el óptimo); CP: los del factor variable. La curva de LP envuelve a las de CP, y la empresa responde más cuantos menos factores fijos tenga.
\end{ideaclave}

\sub{II.3. Estimación por dualidad\label{rep:II.3}}
\begin{itemize}
\item \emph{Idea}: estimar $F$ exige cantidades elegidas por la empresa; si $\vect{w}$ y $q$ son exógenos (eléctrica regulada) $\Rightarrow$ estimar $C$ (\autor{Nerlove}, 1963) y, por \autor{Shephard}, las demandas.
\item CD (\autor{Nerlove}): impone $\sigma=1$ y escala igual a todo tamaño.
\begin{formula}
\raggedright \concepto{Translog de costes} (\autor{Christensen}-\autor{Jorgenson}-\autor{Lau} 1973):\\[2pt]
$\ln C=\alpha_0+\alpha_q\ln q+\tfrac12\beta_{qq}(\ln q)^2+\sum_i\alpha_i\ln w_i$\\ \hspace*{2.5em}$+\tfrac12\sum_i\sum_j\beta_{ij}\ln w_i\ln w_j+\sum_i\beta_{iq}\ln w_i\ln q$
\end{formula}
\funcionclave{s_i=\frac{\partial\ln C}{\partial\ln w_i}=\alpha_i+\sum_j\beta_{ij}\ln w_j+\beta_{iq}\ln q,\qquad\sum_is_i=1}
\item \concepto{Forma flexible}: 2.º orden, no impone $\sigma$ ni rendimientos; simetría y homogeneidad, lineales. Se estima el sistema coste + participaciones (sin una). Da $\varepsilon_{C,q}=\partial\ln C/\partial\ln q$, que varía con $q$ (U o L), y la sustitución ($\beta_{ij}$).
\item Restricciones \emph{contrastables} (homogeneidad, simetría, concavidad: la integrabilidad). \autor{Berndt}-\autor{Wood} (1975): energía-capital complementarios, energía-trabajo sustitutivos. \emph{Valoración}: locales; si no minimiza, media y no frontera.
\end{itemize}

\sub{II.4. Economías de escala en los datos\label{rep:II.4}}
\begin{itemize}
\item \emph{Métodos} (\autor{Stigler}, 1958): estadístico, ingeniería, \emph{superviviente} (tamaños que ganan cuota).
\item \emph{Electricidad}: \autor{Christensen}-\autor{Greene} (1976), translog: en 1970, el grueso en el tramo \emph{plano} $\Rightarrow$ competencia en generación sin perder escala \vertema{3.A.17}.
\item \emph{Transporte aéreo}: las aerolíneas pequeñas, más caras por menor \emph{densidad} de tráfico, no por escala: economías de densidad $\neq$ de escala.
\end{itemize}
\graficorepaso[0.5\linewidth]{christensen-greene-1970}{CMe y producción de las eléctricas de EE.\,UU.\ en 1970 (123~empresas)}
\begin{itemize}
\item \emph{Forma}: CMe decreciente hasta la \concepto{escala mínima eficiente} y casi constante después: en \textbf{L} más que en U (\autor{Stigler}); economías que se agotan pronto.
\end{itemize}

\sub{II.5. Medición de la eficiencia\label{rep:II.5}}
\begin{itemize}
\item \emph{Idea}: $C$ es una \emph{frontera}; \autor{Farrell} (1957): distancia a las mejores prácticas. Isocuanta unitaria $SS'$, empresa en $P$, $Q$ en la isocuanta y $R$ en la isocoste, sobre el rayo $OP$:
\funcionclave{\mathrm{EfT}=\frac{OQ}{OP},\quad \mathrm{EfA}=\frac{OR}{OQ},\quad \mathrm{EfE}=\frac{OR}{OP}=\mathrm{EfT}\cdot\mathrm{EfA}}
\item \concepto{Técnica}: reducir todos los factores (ineficiencia X); \concepto{asignativa}: $\mathrm{RMST}=w_L/w_K$; \concepto{económica}: $C/C^{\mathrm{obs}}=1/(1+\chi)$. Las tres, de 0~a 1.
\end{itemize}
\graficorepaso[0.42\linewidth]{farrell}{Eficiencia técnica, asignativa y económica de \autor{Farrell}}
\begin{itemize}
\item \concepto{Frontera estocástica} (SFA, 1977): $\ln C_j^{\mathrm{obs}}=\ln C(\vect{w}_j,q_j)+v_j+u_j$, $v_j$ ruido, $u_j\geq0$ ineficiencia; $\mathrm{EfE}_j=e^{-u_j}$; paramétrica, separa ruido; exige forma y distribución.
\item \concepto{DEA} (\autor{Charnes}-\autor{Cooper}-\autor{Rhodes}, 1978): programación lineal, envolvente por tramos; sin forma ni ruido (todo es ineficiencia); multiproducto inmediato; sensible a atípicos.
\item \emph{Evidencia}: banco mediano, 84\,\% de sus costes (BCE); regulación por comparación (\autor{Shleifer}, 1985). Eficiencia \emph{relativa}; no dice \emph{por qué} \vertema{3.A.15}.
\end{itemize}

\bloque[4]{III. Extensiones y críticas\label{rep:III}}
\sub{III.1. Estructura de mercado\label{rep:III.1}}
\begin{itemize}
\item \emph{Competencia perfecta}: $p=\mathrm{CMg}$ creciente si cubre CVMe (CP) o CMe (LP); muchas empresas si las deseconomías llegan pronto \vertema{3.A.16}.
\item \concepto{Monopolio natural} $\Leftrightarrow$ \concepto{subaditividad}: $C\big(\sum_jq_j\big)<\sum_jC(q_j)$ (\autor{Baumol}, 1977) \vertema{3.A.17}:
  \begin{itemize}
  \item uniproducto: economías de escala \emph{suficientes}, no necesarias (subaditiva con CMe creciente entre $q^0$ y $q^1$);
  \item multiproducto: economías de alcance \emph{necesarias}, no suficientes.
  \end{itemize}
\end{itemize}
\sub{III.2. Economías externas\label{rep:III.2}}
\begin{itemize}
\item Las vistas hasta aquí son \emph{internas}. \autor{Marshall}: \concepto{economías externas}: CMe $\downarrow$ al crecer la \emph{industria} $\Rightarrow$ desplazan la curva. Fuentes: trabajo especializado, proveedores, conocimiento (\emph{clusters}).
\item Compatibles con competencia perfecta; comercio entre países parecidos \vertema{3.B.6}.
\end{itemize}

\sub{III.3. Críticas desde la organización industrial\label{rep:III.3}}
\subsub{a) \autor{Stigler} y la reserva de capacidad}
\begin{itemize}
\item \concepto{\autor{Stigler} (1939)}: con demanda incierta, planta \emph{flexible}: factor «fijo» divisible $\Rightarrow$ CVMe = CMg constantes en la \concepto{reserva de capacidad}; horquilla de tamaños óptimos. Evidencia: costes en L. No discute la minimización, sino la forma.
\end{itemize}
\graficorepaso[0.45\linewidth]{stigler}{\autor{Stigler}: reserva de capacidad frente a la U}
\subsub{b) Ineficiencia X de \autor{Leibenstein} (1966)}
\begin{itemize}
\item Contratos incompletos, esfuerzo que responde a la presión $\Rightarrow$ $C^{\mathrm{obs}}=(1+\chi)\,C(\vect{w},q)$, $\chi\geq0$ $\downarrow$ con la competencia.
  \begin{itemize}
  \item Pérdida del monopolio $>$ triángulo de \autor{Harberger}.
  \item Evidencia: \autor{Nickell} (1996), \autor{Bloom}-\autor{Van Reenen} (2007).
  \item \autor{Stigler} (1976): es minimización con otros argumentos $\to$ agencia \vertema{3.A.15}.
  \end{itemize}
\end{itemize}
\sub{III.4. Otras extensiones\label{rep:III.4}}
\begin{itemize}
\item \concepto{Multiplanta}: $\mathrm{CMg}_1=\dots=\mathrm{CMg}_J$. \concepto{Multiproducto}: \concepto{economías de alcance} $C(q_1,q_2)<C(q_1,0)+C(0,q_2)$ (factores compartidos; \autor{Baumol}-\autor{Panzar}-\autor{Willig}, 1982); complementariedad de costes y economías de escala multiproducto. Diversificación y tamaño eficiente \vertema{3.A.15}.
\item \concepto{Curva de aprendizaje}: $c=c_1(Q^{\mathrm{ac}})^{-b}$ (\autor{Wright}, 1936); fotovoltaica $2^{-b}\approx0{,}73$ $\Rightarrow$ $b\approx0{,}45$. La escala mueve \emph{a lo largo} de la curva; el aprendizaje la \emph{desplaza}: ventaja del primero, precios bajo coste para aprender, industria naciente \vertema{3.B.8}.
\end{itemize}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}:
  \begin{itemize}
  \item $C$ cóncava, homog.\ de grado 1, \autor{Shephard}, = tecnología; con $B$ (\autor{Hotelling}) y $p=\mathrm{CMg}$, el triángulo tecnología--costes--beneficio;
  \item forma según el plazo: LP, rendimientos a escala; CP, factor variable; envolvente;
  \item estimable: costes en L, ineficiencias persistentes; críticas de \autor{Stigler} y \autor{Leibenstein}.
  \end{itemize}
\item \textbf{Relevancia}: cuántas empresas caben; referencia de reguladores y de competencia; base de la medición de la productividad.
\item \textbf{Extensiones}: oferta \vertema{3.A.16} \vertema{3.A.17} \vertema{3.A.19}; empresa y regulación \vertema{3.A.15} \vertema{3.A.20}; consumidor \vertema{3.A.9}; trabajo \vertema{3.A.25}; comercio \vertema{3.B.6} \vertema{3.B.8}.
\item \textbf{Opinión}: lo frágil es la precio-aceptación en factores (monopsonio \vertema{3.A.17}, equilibrio general \vertema{3.A.21}); el coste mínimo lo alcanza la competencia.
\item \textbf{Idea final}: fotovoltaica $-89$\,\% por escala y, sobre todo, aprendizaje (\emph{la práctica hace al maestro}), que desplaza la curva; en 2025 equipos $-6$\,\% y el coste nivelado no bajó (coste del capital).
  \begin{itemize}
  \item $C(\vect{w},q)$ recoge las tres cosas y la dualidad las separa con datos.
  \end{itemize}
\end{itemize}

\begin{enumerate}
\item \textbf{¿MB y MC, duales estrictos?} No: MB lleva ingresos.
\item \textbf{¿Por qué $C$ con cualquier rendimiento y $B$ no?} El MC está acotado; $B$, 0~o infinito.

\item A CP no hay economías de escala.
\item $\varepsilon_{C,q}<1$: economías; $\varepsilon_{q,\lambda}>1$: rendimientos crecientes.
\end{itemize}
\end{trampa}
