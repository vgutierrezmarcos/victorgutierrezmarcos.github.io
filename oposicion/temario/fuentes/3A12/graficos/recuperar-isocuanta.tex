% Recuperación de la isocuanta a partir de la función de costes.
% Isocuanta q: z_K = 3/z_L. La isocoste mínima a precios con w_L/w_K = 3/t^2
% es tangente en (t, 3/t) y corta los ejes en (2t, 0) y (0, 6/t).
%   t = 0,95; 1,4; 2,1; 3,1 (de muy inclinada a muy plana).
% Cada flechita apunta al semiplano w·z >= C(w,q), el de las combinaciones
% que cuestan al menos el mínimo. Su intersección es V(q).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,scale=0.75]
  % 1: ejes
  \capa{1}{\ejes{$z_L$}{$z_K$}{7.6}{7}}
  % 2-5: isocostes mínimas w^(k)·z = C(w^(k),q)
  \capa{2}{\draw[curva2] (0,6.316) coordinate (A1) -- (1.9,0) coordinate (B1);}
  \capa{2}{\draw[flecha,tceeazul] ($(A1)!0.3!(B1)$) -- ($(A1)!0.3!(B1)!6mm!90:(B1)$);}
  \capa{2}{\node[etiqueta,below,text=tceeazul] at (B1) {$\vect{w}^{(1)}$};}
  \capa{2}{\node[etiqueta,anchor=west,text=tceeazul] at ($(A1)!0.3!(B1)!6mm!90:(B1)$) {$\vect{w}^{(1)}\cdot\vect{z}\geq C(\vect{w}^{(1)},q)$};}
  \capa{3}{\draw[curva3] (0,4.286) coordinate (A2) -- (2.8,0) coordinate (B2);}
  \capa{3}{\draw[flecha,tceeverde] ($(A2)!0.3!(B2)$) -- ($(A2)!0.3!(B2)!6mm!90:(B2)$);}
  \capa{3}{\node[etiqueta,below,text=tceeverde] at (B2) {$\vect{w}^{(2)}$};}
  \capa{4}{\draw[curva4] (0,2.857) coordinate (A3) -- (4.2,0) coordinate (B3);}
  \capa{4}{\draw[flecha,tceedorado] ($(A3)!0.75!(B3)$) -- ($(A3)!0.75!(B3)!6mm!90:(B3)$);}
  \capa{4}{\node[etiqueta,below,text=tceedorado] at (B3) {$\vect{w}^{(3)}$};}
  \capa{5}{\draw[curvanueva] (0,1.935) coordinate (A4) -- (6.2,0) coordinate (B4);}
  \capa{5}{\draw[flecha,tceerojo] ($(A4)!0.75!(B4)$) -- ($(A4)!0.75!(B4)!6mm!90:(B4)$);}
  \capa{5}{\node[etiqueta,below,text=tceerojo] at (B4) {$\vect{w}^{(4)}$};}
  % 6: la isocuanta como envolvente de las isocostes y V(q) sombreado
  \capa{6}{\fill[area] plot[domain=0.47:6.9,samples=100,smooth] (\x,{3/\x}) -- (6.9,6.6) -- (0.47,6.6) -- cycle;}
  \capa{6}{\draw[curva1] plot[domain=0.47:6.9,samples=100,smooth] (\x,{3/\x}) node[etiqueta,above] {$q$};}
  \capa{6}{\foreach \t in {0.95,1.4,2.1,3.1} {\node[punto,fill=tceemorado] at (\t,{3/\t}) {};}}
  \capa{6}{\node[etiqueta,text=tceemorado,align=center] at (4.5,3.1) {$V(q)=\{\vect{z}:\ \vect{w}\cdot\vect{z}\geq C(\vect{w},q)$\\ para todo $\vect{w}\gg\vect{0}\}$};}
\end{tikzpicture}
\end{document}
