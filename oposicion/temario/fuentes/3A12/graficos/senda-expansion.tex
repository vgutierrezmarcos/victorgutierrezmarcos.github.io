% Senda de expansión con un tramo de trabajo inferior.
% Precios constantes: w_L/w_K = 1, isocostes z_L + z_K = 2,9; 5,7; 6,8.
% Isocuantas de la forma z_K = y0 - d + d^2/(z_L - x0 + d), tangentes con
% pendiente -1 en A = (1,6; 1,3) (d = 1), B = (3,3; 2,4) (d = 2,2) y
% C = (2,7; 4,1) (d = 0,8); convexas, sin cortarse en el dibujo.
% Entre B y C la senda vuelve hacia atrás: z_L baja al subir q.
% Autor: Víctor Gutiérrez Marcos

  \capa{2}{\draw[curva1] plot[domain=0.79:6.4,samples=100,smooth] (\x,{0.3+1/(\x-0.6)}) node[etiqueta,right] {$q^1$};}
  \capa{2}{\draw[curva3] (0,2.9) -- (2.9,0);}
