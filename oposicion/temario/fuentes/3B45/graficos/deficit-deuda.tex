% Saldo de las Administraciones Públicas (panel izquierdo) y deuda PDE (panel
% derecho), % del PIB, España y zona del euro (EA20), 1999-2025 (Eurostat,
% notificación de octubre de 2026) y previsión de la deuda 2026-2027 (FMI).
% Datos: ../datos/saldos-espana-zona-euro.csv y ../datos/deuda-pde-espana-zona-euro.csv.
% Cifras de control: saldo de España -11,2 (2009), -2,4 (2025); zona del euro
% -3,0 (2025); deuda de España 100,5 y de la zona del euro 87,7 (2025).
% Capas: valores de referencia -> zona del euro -> España -> rótulos.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
\pgfplotstableread[col sep=comma,comment chars={\#},empty cells with={nan}]{../datos/saldos-espana-zona-euro.csv}\sal
\pgfplotstableread[col sep=comma,comment chars={\#},empty cells with={nan}]{../datos/deuda-pde-espana-zona-euro.csv}\deu
\pgfplotsset{panel/.style={width=7.6cm,height=6.4cm,xmin=1998.5,xmax=2027.5,
    axis x line*=bottom,axis y line*=left,xtick={2000,2010,2020},
    x tick label style={font=\footnotesize,/pgf/number format/.cd,set thousands separator={}},
    yticklabel style={font=\footnotesize},ymajorgrids,grid style={tceegris!25},clip=false,
    unbounded coords=jump,title style={font=\footnotesize\bfseries},dicapas}}
\begin{tikzpicture}[tcee]
\begin{axis}[panel,ymin=-12,ymax=3,title={Saldo público (\% del PIB)},
    legend style={at={(0.5,-0.13)},anchor=north,draw=none,font=\footnotesize},legend columns=2]
  \capa{1}{\dicraya{(1998.5,0) (2027.5,0)}
    \dicraya[dicumbral,thick]{(1998.5,-3) (2027.5,-3)}
    \dicrotulo[font=\scriptsize,text=dicreferencia,anchor=north]{(2015.3,-3.1)}{$-3$~\%}}
  \capa{2}{\addplot[diclinea=dicze,thick,restrict x to domain=1999:2025] table[x=anio,y=EA20_saldo] {\sal}; \addlegendentry{Zona del euro}}
  \capa{3}{\addplot[diclinea=dicespana,ultra thick,restrict x to domain=1999:2025] table[x=anio,y=ES_saldo] {\sal}; \addlegendentry{España}}
  \capa{4}{\dicrotulo[font=\scriptsize,text=dicespana,anchor=east]{(2008.8,-11.2)}{$-11{,}2$}
    \dicrotulo[font=\scriptsize,text=dicespana,anchor=south]{(2025.4,-2.2)}{$-2{,}4$}}
\end{axis}
\begin{axis}[panel,at={(8.4cm,0)},ymin=0,ymax=130,title={Deuda pública (\% del PIB)}]
  \capa{1}{\dicraya[dicumbral,thick]{(1998.5,60) (2027.5,60)}
    \dicrotulo[font=\scriptsize,text=dicreferencia,anchor=north east]{(2027.5,60)}{60~\%}}
  \capa{2}{\addplot[diclinea=dicze,thick,restrict x to domain=1999:2025] table[x=anio,y=EA20_deuda] {\deu};
    \addplot[diclinea=dicze,thick,dashed] coordinates {(2025,87.7) (2026,88.0) (2027,88.5)};}
  \capa{3}{\addplot[diclinea=dicespana,ultra thick,restrict x to domain=1999:2025] table[x=anio,y=ES_deuda] {\deu};
    \addplot[diclinea=dicespana,ultra thick,dashed] coordinates {(2025,100.5) (2026,97.6) (2027,95.2)};}
  \capa{4}{\dicrotulo[font=\scriptsize,text=dicespana,anchor=south]{(2020,121)}{119,3}
    \dicrotulo[font=\scriptsize,text=dicespana,anchor=south west]{(2024.6,102)}{100,5}
    \dicrotulo[font=\scriptsize,text=dicze,anchor=north west]{(2024.6,86)}{87,7}}
\end{axis}
\end{tikzpicture}
\end{document}
