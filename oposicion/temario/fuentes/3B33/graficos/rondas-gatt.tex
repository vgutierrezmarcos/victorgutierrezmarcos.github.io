% Las ocho rondas del GATT: recorte arancelario medio de cada ronda (%) y número
% de participantes. Fuentes en la cabecera de ../datos/rondas-gatt.csv:
% participantes, OMC («Entender la OMC: los años del GATT»); recorte, Baldwin
% (2012), sec. 1.5 (rondas 1-7) y OMC, «Entender la OMC: aranceles» (Uruguay).
% Barras moradas; encima, el recorte y el número de participantes; arriba, las
% dos técnicas de negociación (producto a producto, rondas 1-5; fórmulas
% lineales y armonizadoras, rondas 6-8).
% Capas: barras -> ejes -> recortes -> participantes -> técnicas de negociación.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
\colorlet{gattrecorte}{tceemorado!80}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/rondas-gatt.csv}\rondas
\pgfplotsset{ejesgatt/.style={}}
\ifnum2>\tceepaso\relax
  \pgfplotsset{ejesgatt/.style={axis line style={draw=none},
    major tick style={draw=none}, ticklabel style={opacity=0},
    label style={opacity=0}, ymajorgrids=false}}
\fi
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=14.5cm, height=7.2cm,
      xmin=-0.6, xmax=7.6, ymin=0, ymax=50,
      xtick={0,...,7},
      xticklabels={{Ginebra\\(1947)},{Annecy\\(1949)},{Torquay\\(1951)},
        {Ginebra\\(1956)},{Dillon\\(1960-61)},{Kennedy\\(1964-67)},
        {Tokio\\(1973-79)},{Uruguay\\(1986-94)}},
      xticklabel style={font=\footnotesize, align=center},
      ytick={0,10,...,40}, yticklabel style={font=\footnotesize},
      ylabel={Recorte arancelario medio (\%)},
      axis x line*=bottom, axis y line*=left, tick align=outside,
      major x tick style={draw=none},
      ymajorgrids, grid style={tceegris!25},
      clip=false, ybar, bar width=0.62, bar shift=0pt,
      ejesgatt, dicapas,
    ]
    % 1: barras del recorte
    \capa{1}{
      \addplot[on layer=dicbarras, forget plot, dicbarra=gattrecorte]
        table[x expr=\coordindex, y=recorte] {\rondas};
    }
    % 2: ejes (ver ejesgatt)
    % 3: recorte de cada ronda
    \capa{3}{
      \addplot[on layer=axis descriptions, dicsuelto, only marks, mark=none,
        nodes near coords={\pgfmathprintnumber[fixed,precision=1,use comma]{\pgfplotspointmeta}~\%},
        nodes near coords style={font=\footnotesize\bfseries, text=tceemorado,
          anchor=south, yshift=0pt},
        point meta=explicit]
        table[x expr=\coordindex, y=recorte, meta=recorte] {\rondas};
    }
    % 4: participantes
    \capa{4}{
      \addplot[on layer=axis descriptions, dicsuelto, only marks, mark=none,
        nodes near coords={\pgfmathprintnumber{\pgfplotspointmeta}~países},
        nodes near coords style={font=\footnotesize, text=tceegris,
          anchor=south, yshift=11pt},
        point meta=explicit]
        table[x expr=\coordindex, y=recorte, meta=participantes] {\rondas};
    }
    % 5: técnicas de negociación
    \capa{5}{
      \addplot[on layer=axis descriptions, dicsuelto, draw=tceedorado, thick]
        coordinates {(-0.4,51) (-0.4,53) (4.4,53) (4.4,51)};
      \dicrotulo[font=\footnotesize,text=tceedorado,anchor=south]{(2,53)}{Producto a producto (rondas 1-5)}
      \addplot[on layer=axis descriptions, dicsuelto, draw=tceedorado, thick]
        coordinates {(4.6,51) (4.6,53) (7.4,53) (7.4,51)};
      \dicrotulo[font=\footnotesize,text=tceedorado,anchor=south]{(6,53)}{Fórmulas lineales y armonizadoras (6-8)}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
