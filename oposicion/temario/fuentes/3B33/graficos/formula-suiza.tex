% Métodos de reducción arancelaria: arancel final t_1 frente al inicial t_0.
%   Sin rebaja: t_1 = t_0 (bisectriz).
%   Reducción lineal (Ronda Uruguay, agricultura de los países desarrollados):
%     t_1 = 0,64 t_0 (recorte del 36 %).
%   Fórmula suiza t_1 = a t_0/(a + t_0) con a = 25 y a = 14 (Ronda de Tokio,
%   coeficiente de EE. UU.; la CE usó 16): el coeficiente a es el arancel
%   máximo final (asíntota) y la rebaja proporcional, t_0/(a + t_0), crece con
%   t_0 (armoniza).
% Capas: ejes -> bisectriz -> lineal -> suiza a = 25 -> suiza a = 14 y asíntotas.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.13cm,y=0.13cm]
  % 1: ejes
  \capa{1}{\draw[eje] (0,0) -- (104,0) node[right] {$t_0$ (\%)};}
  \capa{1}{\draw[eje] (0,0) -- (0,68) node[left] {$t_1$ (\%)};}
  \capa{1}{\node[below left] at (0,0) {$0$};}
  \capa{1}{\foreach \x in {20,40,60,80,100} {\draw (\x,0) -- (\x,-1) node[below,font=\footnotesize] {\x};}}
  \capa{1}{\foreach \y in {20,40,60} {\draw (0,\y) -- (-1,\y) node[left,font=\footnotesize] {\y};}}
  % 2: sin rebaja
  \capa{2}{\draw[guia] (0,0) -- (66,66) node[etiqueta,above,font=\footnotesize,text=tceegris] {sin rebaja};}
  % 3: reducción lineal del 36 %
  \capa{3}{\draw[curva4] (0,0) -- (100,64) node[etiqueta,right,text=tceedorado,font=\footnotesize,align=left] {lineal:\\$t_1=0{,}64\,t_0$};}
  % 4: fórmula suiza con a = 25
  \capa{4}{\draw[curva1] plot[domain=0:100,samples=80] (\x,{25*\x/(25+\x)}) node[etiqueta,right,text=tceemorado,font=\footnotesize] {suiza, $a=25$};}
  \capa{4}{\draw[guia,tceemorado] (0,25) -- (100,25);}
  % 5: fórmula suiza con a = 14 y su asíntota
  \capa{5}{\draw[curva2] plot[domain=0:100,samples=80] (\x,{14*\x/(14+\x)}) node[etiqueta,right,text=tceeazul,font=\footnotesize] {suiza, $a=14$};}
  \capa{5}{\draw[guia,tceeazul] (0,14) -- (100,14);}
  \capa{5}{\node[etiqueta,font=\footnotesize,anchor=south west,align=left] at (62,27) {$t_1=\dfrac{a\,t_0}{a+t_0}<a$};}
\end{tikzpicture}
\end{document}
