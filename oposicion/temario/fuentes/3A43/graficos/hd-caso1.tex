  \path (-1.0,-0.55) rectangle (9.9,4.0);
  % 1: ejes
  \capa{1}{\ejes{$k$}{$\gamma_k$}{8.2}{3.9}}

  \capa{2}{\draw[curva1] (0,2.6) node[left,text=black] {$s/v$} -- (2.4,2.6)
