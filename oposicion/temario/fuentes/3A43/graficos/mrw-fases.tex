% Modelo de Mankiw, Romer y Weil (1992) en el plano (k, h), alfa = beta = 1/3.
% Lugares geométricos (con s_K = s_H = 0,2 y n+g+delta = 0,2 * 3^(-1/3)):
%   dk/dt = 0:  h = [(n+g+delta)/s_K]^(1/beta) k^((1-alfa)/beta) = k^2/3     (convexo)
%   dh/dt = 0:  h = [s_H/(n+g+delta)]^(1/(1-beta)) k^(alfa/(1-beta)) = (3k)^(1/2)  (cóncavo)
%   corte único en k* = h* = 3.
% Dinámica: por encima de dk/dt = 0, k sube; por debajo de dh/dt = 0, h sube.
% Trayectoria: integración numérica de las 2 ecuaciones desde (0,6; 5,2).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,dir/.style={->,thick,tceegris}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-0.9,-0.75) rectangle (8.0,6.6);
  % 1: ejes
  \capa{1}{\ejes{$\tilde k$}{$\tilde h$}{6.8}{6.4}}
  % 2: lugar dk/dt = 0 (convexo)
  \capa{2}{\draw[curva1] plot[domain=0:4.2,samples=60] (\x,{\x*\x/3})
      node[etiqueta,above,text=tceemorado] {$\dot{\tilde k}=0$};}
  % 3: lugar dh/dt = 0 (cóncavo)
  \capa{3}{\draw[curva2] plot[domain=0:6.3,samples=120] (\x,{sqrt(3*\x)})
      node[etiqueta,right,text=tceeazul] {$\dot{\tilde h}=0$};}
  % 4: estado estacionario y proyecciones
  \capa{4}{\node[punto] (E) at (3,3) {};
    \node[etiqueta,below right] at (E) {$E$};
    \proyeccion{(E)}{$\tilde k^*$}{$\tilde h^*$}}
  % 5: flechas de la dinámica en las 4 regiones
  \capa{5}{
    % encima de los 2 lugares: k sube, h baja
    \draw[dir] (1.3,4.6) -- ++(0.55,0); \draw[dir] (1.3,4.6) -- ++(0,-0.55);
    % entre los 2, a la izquierda de E: k y h suben
    \draw[dir] (0.9,1.0) -- ++(0.45,0); \draw[dir] (0.9,1.0) -- ++(0,0.45);
    % entre los 2, a la derecha de E: k y h bajan
    \draw[dir] (4.5,4.9) -- ++(-0.45,0); \draw[dir] (4.5,4.9) -- ++(0,-0.45);
    % debajo de los 2 lugares: k baja, h sube
    \draw[dir] (5.4,1.6) -- ++(-0.55,0); \draw[dir] (5.4,1.6) -- ++(0,0.55);}
  % 6: trayectoria de convergencia desde un punto inicial
  \capa{6}{\node[punto,fill=tceerojo] at (0.6,5.2) {};
    \node[etiqueta,right,text=tceerojo] at (0.6,5.2) {$(\tilde k_0,\tilde h_0)$};
    \draw[->,thick,tceerojo] plot[smooth] coordinates {(0.600,5.200) (0.683,5.035) (0.766,4.883) (0.847,4.741) (0.926,4.611) (1.022,4.460) (1.116,4.323) (1.206,4.198) (1.309,4.062) (1.407,3.941) (1.514,3.814) (1.615,3.702) (1.722,3.590) (1.832,3.481) (1.943,3.378) (2.061,3.276) (2.181,3.181) (2.303,3.094) (2.431,3.014) (2.566,2.945) (2.709,2.898) (2.859,2.901)};}
\end{tikzpicture}
\end{document}
