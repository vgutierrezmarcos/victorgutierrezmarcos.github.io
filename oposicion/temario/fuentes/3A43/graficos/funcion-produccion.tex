% Función de producción neoclásica Y = F(K, AL) = K^(1/3) (AL)^(2/3), en la
% perspectiva del gráfico de cuasiconcavidad del 3.A.8 (vista por defecto de
% pgfplots, acimut 30° y elevación 28°): origen a la izquierda, Y vertical, K
% a la derecha algo hacia abajo y AL en profundidad hacia arriba a la derecha.
% Adaptado de fuentes/3A11/graficos/superficie-produccion.tex (ejes z_L, z_K -> K, AL).
%   corte con AL = 3: Y = 3^(2/3) K^(1/3) (cóncava: producto marginal decreciente)
%   rayo AL = K/2: Y = 2^(-2/3) K, recta (rendimientos constantes a escala);
%     llega a (4; 2; 2,5198)
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\pgfplotsset{
  superficie/.style={surf,shader=faceted,fill opacity=0.45,z buffer=sort,draw=tceemorado!40,line width=0.15pt,
    colormap={tcee}{rgb255=(226,239,217) rgb255=(167,137,189)}},
}
\begin{document}
\begin{tikzpicture}[tcee]
  \begin{axis}[x={(1.2cm,-0.33cm)},y={(0.69cm,0.56cm)},z={(0cm,1.22cm)},axis lines=none,
      xmin=0,xmax=4.4,ymin=0,ymax=4.4,zmin=0,zmax=4.4,clip=false]
    % 1: ejes
    \capa{1}{\draw[eje] (axis cs:0,0,0) -- (axis cs:4.8,0,0) node[below] {$K$};}
    \capa{1}{\draw[eje] (axis cs:0,0,0) -- (axis cs:0,5.0,0) coordinate (eAL);}
    \capa{1}{\draw[eje] (axis cs:0,0,0) -- (axis cs:0,0,4.6) node[above] {$Y$};}
    \capa{1}{\node[left] at (axis cs:0,0,0) {$0$};}
    % 2: superficie de producción
    \capa{2}{\addplot3[superficie,domain=0:4,y domain=0:4,samples=11,samples y=11] {x^(1/3)*y^(2/3)};}
    \capa{2}{\node[etiqueta,anchor=west] at (axis cs:4.05,4,4) {$Y=K^{1/3}(AL)^{2/3}$};}
    % 3: corte con AL fijo: producto marginal del capital decreciente
    \capa{3}{\addplot3[curva3,line width=1.4pt,domain=0:4,samples=60,samples y=0] (x,3,{x^(1/3)*3^(2/3)});}
    \capa{3}{\addplot3[guia,samples y=0] coordinates {(0,3,0) (4,3,0) (4,3,3.3019)};}
    \capa{3}{\node[etiqueta,text=tceeverde,anchor=south east,align=right] at (axis cs:0.9,3,2.5) {$AL$ fijo:\\ $F_{KK}<0$};}
    % 4: rayo desde el origen: rendimientos constantes a escala
    \capa{4}{\addplot3[thick,dashed,tceedorado,samples y=0] coordinates {(0,0,0) (4,2,0)};}
    \capa{4}{\addplot3[fill=tceedorado,fill opacity=0.15,draw=none,samples y=0] coordinates {(0,0,0) (4,2,0) (4,2,2.5198) (0,0,0)};}
    \capa{4}{\addplot3[curva4,line width=1.4pt,samples y=0] coordinates {(0,0,0) (4,2,2.5198)};}
    \capa{4}{\node[etiqueta,text=tceedorado,anchor=west,align=left] at (axis cs:4.05,2,2.3) {rayo $AL/K$ fijo:\\ una recta};}
    % rótulo del eje AL, al final para que quede encima de la superficie
    \capa{1}{\node[etiqueta,fill=white,fill opacity=0.85,text opacity=1,inner sep=1pt,above left=1pt and -2pt] at (eAL) {$AL$};}
  \end{axis}
\end{tikzpicture}
\end{document}
