\begin{tikzpicture}[tcee,y=1.4cm,tasa/.style={<->,thick,tceegris}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-1.0,-0.55) rectangle (9.9,4.0);
  % 1: ejes
  \capa{1}{\ejes{$k$}{$\gamma_k$}{8.2}{3.9}}
  % 2: s f(k)/k: horizontal en s/v hasta el codo y decreciente después
  \capa{2}{\draw[curva1] (0,0.8) node[left,text=black] {$s/v$} -- (2.4,0.8)
      plot[domain=2.4:7.6,samples=80] (\x,{1.92/\x})
      node[etiqueta,right,text=tceemorado] {$s\,f(k)/k$};

  \capa{3}{\draw[curva3] (0,1.3) node[left,text=black] {$n+\delta$} -- (7.9,1.3);}
  % 4: dinámica: gamma_k < 0 en todo k; el capital por trabajador cae hacia 0
  \capa{4}{\draw[tasa] (4.5,1.3) -- node[etiqueta,right,text=black] {$\gamma_k<0$} (4.5,0.4267);
