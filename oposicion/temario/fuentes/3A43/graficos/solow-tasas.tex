% Modelo de Solow en tasas de crecimiento: gamma_k = s f(k)/k - (n+g+delta).
% Curvas calculadas (Cobb-Douglas con alfa = 1/2, en unidades del dibujo):
%   s f(k)/k = a k^(-1/2); n+g+delta = 1,3 (horizontal)
%   s_0: a = 2,3255 -> k*_0 = (a/1,3)^2 = 3,2
%   s_1 > s_0: a = 3,0210 -> k*_1 = 5,4
%   s' < s_0 (país que ahorra menos): a = 1,2671 -> k*' = 0,95
%   convergencia: k_P = 0,5 (s f/k = 3,2888, gamma_P = 1,989);
%                 k_R = 2,2 (s f/k = 1,5679, gamma_R = 0,268)
% Cada curva empieza donde vale 3,6: k = (a/3,6)^2. Escala vertical 1,4 cm.
%   gamma_k a cada lado de k*_0 (dibujo): k = 1,6 (1,8385) y k = 7,0 (0,8790)
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,y=1.4cm,
    declare function={g0(\k)=2.3255/sqrt(\k); g1(\k)=3.021/sqrt(\k); gc(\k)=1.2671/sqrt(\k);},
    tasa/.style={<->,thick,tceegris},
    tasar/.style={->,thick,tceerojo}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-1.0,-0.55) rectangle (9.9,4.0);
  % 1: ejes
  \capa{1}{\ejes{$k$}{$\gamma_k$}{8.2}{3.9}}
  % 2: s f(k)/k, decreciente y convexa
  \capa{2}{\draw[curva1] plot[domain=0.4173:7.6,samples=100] (\x,{g0(\x)})
      node[etiqueta,right,text=tceemorado] {$s_0\,f(k)/k$};}
  % 3: n+g+delta
  \capa{3}{\draw[curva3] (0,1.3) node[left,text=black] {$n+g+\delta$} -- (7.9,1.3);}
  % 4: estado estacionario k*_0 y gamma_k a cada lado, con la dinámica en el eje
  \capa{4}{\node[punto] (E0) at (3.2,1.3) {};
    \draw[guia] (E0) -- (3.2,0) node[below,text=black] {$k^*_0$};
    \draw[tasa] (1.6,1.3) -- (1.6,1.8385);
    \node[etiqueta,left,text=black,inner sep=1pt] at (1.58,1.7) {$\gamma_k>0$};
    \draw[tasa] (7.0,1.3) -- (7.0,0.879);
    \node[etiqueta,below,text=black,inner sep=1pt] at (7.0,0.86) {$\gamma_k<0$};
    \draw[flecha] (1.8,0) -- (2.8,0);
    \draw[flecha] (4.4,0) -- (3.6,0);}
  % 5: subida de la tasa de ahorro: s_1 f(k)/k y k*_1
  \capa{5}{\draw[curvanueva] plot[domain=0.7042:7.6,samples=100] (\x,{g1(\x)})
      node[etiqueta,right,text=tceerojo] {$s_1\,f(k)/k$};
    \node[punto] (E1) at (5.4,1.3) {};
    \draw[guia] (E1) -- (5.4,0) node[below,text=black] {$k^*_1$};
    \draw[->,thick,tceerojo] (3.3,0.3) -- (5.3,0.3);}
  % 6: convergencia absoluta: una economía pobre (k_P) y otra rica (k_R)
  \capa{6}{\draw[guia] (0.5,3.2888) -- (0.5,0) node[below,text=black] {$k_P$};
    \draw[tasar] (0.5,1.3) -- node[etiqueta,right] {$\gamma_P$} (0.5,3.2888);
    \draw[guia] (2.2,1.5679) -- (2.2,0) node[below,text=black] {$k_R$};
    \draw[tasar] (2.2,1.3) -- (2.2,1.5679);
    \node[etiqueta,text=tceerojo,anchor=north west,inner sep=2pt] at (2.2,1.28) {$\gamma_R$};}
  % 7: convergencia condicional: el país que ahorra menos converge a un k*' menor
  \capa{7}{\draw[curva4] plot[domain=0.1239:7.6,samples=100] (\x,{gc(\x)})
      node[etiqueta,right,text=tceedorado] {$s'\,f(k)/k$};
    \node[punto,fill=tceedorado] (Ec) at (0.95,1.3) {};
    \draw[guia] (Ec) -- (0.95,0) node[below,text=black] {$k^{*\prime}$};}
\end{tikzpicture}
\end{document}
