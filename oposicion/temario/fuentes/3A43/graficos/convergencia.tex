\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
% Convergencia, 1960-2023: PIB por habitante en 1960 (PPA, dólares de 2021,
% escala logarítmica) frente al crecimiento medio anual del PIB por habitante
% (%). PWT 11.0, ../datos/convergencia-pwt.csv: filas 0-110 (periodo
% 1960-2023), sin exportadores de petróleo (petrolero=0): 103 países, 30 de
% ellos de la OCDE (ocde=1). En la copia de datos/ se quitaron las comillas de
% «China, Hong Kong SAR» (la coma partía la fila en pgfplots).
% Rectas de MCO de crecimiento sobre ln(PIB por habitante inicial):
%   todos (103):  g = 2,757 - 0,0947 ln y0   (casi plana: sin convergencia absoluta)
%   OCDE (30):    g = 10,910 - 0,9383 ln y0  (convergencia clara)
% (con los petroleros, la pendiente de todos es -0,117; desde 2000, -0,47).
% Capas (CRITERIOS §7): ejes y 0 → resto de países → su recta → OCDE → su
% recta → códigos y leyenda.
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/convergencia-pwt.csv}\conv
% Selección de filas con un solo índice: 1000·petrolero + 500·ocde + fila:
% 0-110 = resto de países de 1960-2023 sin petroleros; 500-610 = OCDE.
\pgfplotsset{
  resto/.style={only marks, mark=*, mark size=1.5pt, forget plot,
    mark options={fill=dicotroclaro, draw=dicotro, line width=0.3pt},
    restrict expr to domain={1000*\thisrow{petrolero}+500*\thisrow{ocde}+\coordindex}{0:110}},
  ocde/.style={only marks, mark=*, mark size=2.2pt, forget plot,
    mark options={fill=tceemorado!70, draw=tceemorado, line width=0.4pt},
    restrict expr to domain={1000*\thisrow{petrolero}+500*\thisrow{ocde}+\coordindex}{500:610}},
}
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=13cm, height=7.5cm, xmode=log,
      xmin=450, xmax=40000, ymin=-2, ymax=7,
      xtick={1000,3000,10000,30000}, xticklabels={1.000,3.000,10.000,30.000},
      ytick={-2,-1,...,7},
      xticklabel style={font=\footnotesize}, yticklabel style={font=\footnotesize},
      xlabel={PIB por habitante en 1960 (dólares de 2021 en PPA, escala log.)},
      ylabel={crecimiento medio anual, 1960-2023 (\%)},
      xlabel style={font=\footnotesize}, ylabel style={font=\footnotesize},
      axis x line*=bottom, axis y line*=left, tick align=outside,
      ymajorgrids, grid style={tceegris!25}, clip=false,
      unbounded coords=discard,
      legend style={at={(0.98,0.98)}, anchor=north east, draw=none,
        font=\footnotesize, cells={anchor=west}},
      legend cell align=left,
      dicapas,
    ]
    % 1: línea del 0
    \capa{1}{\dicraya{(450,0) (40000,0)}}
    % 2: países que no son de la OCDE
    \capa{2}{\addplot[on layer=dicpuntos, resto] table[x=pib_pc_inicial_ppa_2021usd, y=crec_pib_pc_medio_pct] {\conv};}
    % 3: recta de todos los países
    \capa{3}{
      \addplot[very thick, dicotro, domain=520:26000, samples=2, forget plot] {2.757-0.0947*ln(x)};
      \dicrotulo[font=\footnotesize,text=dicotro,anchor=south west,inner sep=2pt]{(26500,1.85)}{todos: $\beta\approx-0{,}09$}
    }
    % 4: países de la OCDE
    \capa{4}{\addplot[on layer=dicpuntos, ocde] table[x=pib_pc_inicial_ppa_2021usd, y=crec_pib_pc_medio_pct] {\conv};}
    % 5: recta de la OCDE
    \capa{5}{
      \addplot[very thick, tceemorado, domain=1300:26000, samples=2, forget plot] {10.910-0.9383*ln(x)};
      \dicrotulo[font=\footnotesize,text=tceemorado,anchor=north west,inner sep=2pt]{(26500,1.45)}{OCDE: $\beta\approx-0{,}94$}
    }
    % 6: códigos de algunos países y leyenda
    \capa{6}{
      \dicrotulo[font=\scriptsize,text=tceemorado,anchor=west,inner sep=3pt]{(1363,5.38)}{KOR}
      \dicrotulo[font=\scriptsize,text=tceemorado,anchor=south,inner sep=3pt]{(7658,3.84)}{IRL}
      \dicrotulo[font=\scriptsize,text=tceemorado,anchor=south west,inner sep=1pt]{(6472,2.95)}{JPN}
      \dicrotulo[font=\scriptsize,text=tceemorado,anchor=south,inner sep=3pt]{(5328,2.9)}{TUR}
      \dicrotulo[font=\scriptsize,text=tceemorado,anchor=north,inner sep=3pt]{(6965,2.47)}{ESP}
      \dicrotulo[font=\scriptsize,text=tceemorado,anchor=north,inner sep=3pt]{(7603,1.44)}{MEX}
      \dicrotulo[font=\scriptsize,text=tceemorado,anchor=south east,inner sep=1pt]{(21010,2.05)}{USA}
      \dicrotulo[font=\scriptsize,text=tceemorado,anchor=north,inner sep=3pt]{(24231,1.3)}{CHE}
      \addlegendimage{only marks, mark=*, mark size=2.2pt, mark options={fill=tceemorado!70, draw=tceemorado, line width=0.4pt}}
      \addlegendentry{países de la OCDE}
      \addlegendimage{only marks, mark=*, mark size=1.5pt, mark options={fill=dicotroclaro, draw=dicotro, line width=0.3pt}}
      \addlegendentry{resto de países}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
