% Autoselección en el seguro (Rothschild y Stiglitz, 1976): de la heterogeneidad al
% equilibrio separador, en un solo panel que crece por capas.
% Riqueza W = 10, pérdida L = 6. Temerarios pi_A = 0,5; prudentes pi_B = 0,25. Utilidad u = ln.
%   Dotación A = (10, 4). Rectas de isobeneficio nulo desde A:
%     (1 - pi_i) x_1 + pi_i x_2 = W - pi_i L, de pendiente -(1 - pi_i)/pi_i:
%     temerarios -1 (x_2 = 14 - x_1), prudentes -3 (x_2 = 34 - 3 x_1, más inclinada).
%   First best: E_A = (7, 7) y E_B = (8,5; 8,5) en la línea de certeza, con las curvas
%     tangentes: temerario x_2 = 49/x_1 (pendiente -1 en E_A); prudente
%     x_2 = 8,5 (8,5/x_1)^3 (pendiente -3 en E_B).
%   Los temerarios prefieren E_B: su curva por E_B, x_2 = 72,25/x_1, está por encima.
%   Separador: la curva del temerario por E_A corta la recta de los prudentes en
%     S_B = (9,639; 5,084): x_1 > x_2, seguro parcial. Curva del prudente por S_B:
%     x_2 = 5,0836 (9,6388/x_1)^3, por debajo de la del first best (5,08 < 5,83 en x_1 = 9,64).
% Construcción paso a paso en el PDF: rothschild-stiglitz-paso1.tex (capas 1-5,
% heterogeneidad), rothschild-stiglitz-paso2.tex (1-7, first best),
% rothschild-stiglitz-paso3.tex (1-8, la aseguradora no puede distinguir) y este
% (todas, equilibrio separador). El agrupador está en rs-agrupador.tex y la
% inexistencia en rs-inexistencia.tex, con el mismo marco y la misma escala.
% Elaboración propia a partir de Rothschild y Stiglitz (1976), Fernández de Castro
% y Tugores (1992, cap. 8) y Mas-Colell, Whinston y Green (1995, cap. 13).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.7cm,y=0.7cm,
    declare function={
      ra(\x)=14-\x;            % isobeneficio nulo con temerarios
      rb(\x)=34-3*\x;          % isobeneficio nulo con prudentes
      teE(\x)=49/\x;           % temerario por E_A
      prE(\x)=8.5*(8.5/\x)^3;  % prudente por E_B
      teEB(\x)=72.25/\x;       % temerario por E_B
      prS(\x)=5.0836*(9.6388/\x)^3;}] % prudente por S_B
  % Marco fijo (el mismo en rs-agrupador.tex y rs-inexistencia.tex), para que todos
  % los pasos salgan del mismo tamaño y a la misma escala
  \useasboundingbox (-2.4,-1.6) rectangle (14.2,12.4);
  % 1: ejes: riqueza sin accidente y con accidente
  \capa{1}{\ejes{$x_1$}{$x_2$}{13.5}{12}}
  % 2: línea de certeza
  \capa{2}{\draw[thick,tceegris] (0,0) -- (11,11) node[etiqueta,above,font=\footnotesize,text=black] {línea de certeza};}
  % 3: dotación A, sin seguro
  \capa{3}{\node[punto] (A) at (10,4) {};}
  \capa{3}{\node[etiqueta,above right] at (A) {$A$};}
  \capa{3}{\draw[guia] (10,0) node[below right=0pt and -3pt,text=black] {$\overline{W}$} |- (0,4) node[left,text=black] {$\overline{W}-L$};}
  % 4: isobeneficio nulo con los temerarios (prima justa pi_A)
  \capa{4}{\draw[curva4] plot[domain=10:4.8,samples=2] (\x,{ra(\x)}) node[etiqueta,above,text=tceedorado] {$\pi_{\mathrm{A}}$};}
  % 5: isobeneficio nulo con los prudentes (prima justa pi_B), más inclinada
  \capa{5}{\draw[curva4] plot[domain=10:8,samples=2] (\x,{rb(\x)}) node[etiqueta,above left=0pt and -4pt,text=tceedorado] {$\pi_{\mathrm{B}}$};}
  % 6: first best de los temerarios: seguro completo en E_A, tangencia con su curva
  \capa{6}{\draw[curva1] plot[domain=4.7:12.2,samples=50] (\x,{teE(\x)}) node[etiqueta,right] {$U_{\mathrm{A}}^{*}$};}
  \capa{6}{\node[punto] (EA) at (7,7) {};}
  \capa{6}{\node[etiqueta,left=3pt] at (EA) {$E_{\mathrm{A}}$};}
  % 7: first best de los prudentes: seguro completo en E_B, tangencia con su curva
  \capa{7}{\draw[curva2] plot[domain=8.1:11.2,samples=50] (\x,{prE(\x)}) node[etiqueta,right] {$U_{\mathrm{B}}^{*}$};}
  \capa{7}{\node[punto] (EB) at (8.5,8.5) {};}
  \capa{7}{\node[etiqueta,left=3pt] at (EB) {$E_{\mathrm{B}}$};}
  % 8: la aseguradora no distingue: los temerarios prefieren E_B (curva más alta)
  \capa{8}{\draw[curvanueva,desplazada] plot[domain=6.3:12.2,samples=50] (\x,{teEB(\x)}) node[etiqueta,right] {$U_{\mathrm{A}}'$};}
  \capa{8}{\draw[flecha,shorten >=2pt] (EA) to[bend right=50] (EB);}
  \capa{8}{\draw[thin,tceegris] (EB) -- (10.5,9.35);}
  \capa{8}{\node[etiqueta,anchor=west,align=left,font=\footnotesize,text=tceerojo] at (10.5,9.35) {los temerarios\\ prefieren $E_{\mathrm{B}}$};}
  % 9: equilibrio separador: S_B, corte de la curva del temerario por E_A con la
  % recta de los prudentes (seguro parcial)
  \capa{9}{\node[punto] (SB) at (9.639,5.084) {};}
  \capa{9}{\node[etiqueta,above right] at (SB) {$S_{\mathrm{B}}$};}
  \capa{9}{\draw[guia] (9.639,0) node[below left=0pt and -3pt,text=black] {$x_1^{\mathrm{B}}$} |- (0,5.084) node[left,text=black] {$x_2^{\mathrm{B}}$};}
  % 10: curva de indiferencia del prudente por S_B, por debajo de la del first best
  \capa{10}{\draw[curva2,desplazada] plot[domain=8.0:11.8,samples=50] (\x,{prS(\x)}) node[etiqueta,right] {$U_{\mathrm{B}}$};}
\end{tikzpicture}
\end{document}
