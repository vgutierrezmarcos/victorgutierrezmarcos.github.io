% Equilibrio de una empresa en el modelo de Krugman con elasticidad constante (anexo B):
% a) pocas variedades, beneficios; b) la entrada desplaza la demanda de cada variedad,
% con el precio fijo en el margen; c) tangencia con el coste medio, B = 0.
% Mismo gráfico y escalas que 3A18/graficos/dixit-stiglitz.tex, con los rótulos del
% comercio: precio relativo al salario p/w y producción de la empresa q.
% Parámetros: sigma = 2 (sigma/(sigma-1) = 2), c = 1, F = 1.
%   CMg = c = 1; CMe(q) = c + F/q = 1 + 1/q.
%   Demanda isoelástica de la variedad: p/w = 2 (q/q_0)^(-1/2); IMg = (q/q_0)^(-1/2).
%   IMg = c en q = q_0, con p*/w = sigma/(sigma-1) c = 2 sea cual sea q_0.
%   a) q_0 = 3: CMe(3) = 1,333 < 2: beneficio (2 - 1,333) 3 = 2.
%   b) q_0 = 1,8: CMe(1,8) = 1,556 < 2 (aún hay beneficios: sigue la entrada).
%   c) q_0 = 1 = q* = (sigma-1) F / c: 2/sqrt(q) es tangente a 1 + 1/q en (1, 2).
% Elaboración propia a partir de Krugman (1980) y Gandolfo (2014).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.75cm,y=1.1cm,
    declare function={cme(\q)=1+1/\q; dem(\q,\a)=2*sqrt(\a/\q); img(\q,\a)=sqrt(\a/\q);},
    titulo/.style={font=\small,anchor=south},
    dmd/.style={curva1}, imgs/.style={curva2}, cme/.style={curva3}, cmg/.style={curva4}]
  \useasboundingbox (-2.4,-0.5) rectangle (27.1,4.45);
  % 1: ejes de los 3 paneles
  \foreach \dx/\t in {0/{a) Pocas variedades: $B>0$},9.6/{b) Entrada},19.2/{c) Equilibrio: $B=0$}} {
    \capa{1}{\begin{scope}[xshift=\dx*0.75cm]
      \ejes{$q$}{$p/w$}{6.6}{3.85}
      \node[titulo] at (3.3,3.85) {\t};
    \end{scope}}
  }
  % 2: los mismos costes en los 3 paneles: CMg = c y CMe = c + F/q
  \foreach \dx in {0,9.6,19.2} {
    \capa{2}{\begin{scope}[xshift=\dx*0.75cm]
      \draw[cmg] (0,1) node[left,text=black] {$c$} -- (6.2,1);
      \draw[cme] plot[domain=0.4:6.2,samples=80,smooth] (\x,{cme(\x)});
      \node[etiqueta,text=tceeverde,anchor=west] at (0.45,3.5) {CMe};
    \end{scope}}
  }
  \capa{2}{\node[etiqueta,text=tceedorado,anchor=west] at (6.2,0.92) {$\mathrm{CMg}=c$};}
  % ---------------- 3: a) pocas variedades ----------------
  \capa{3}{\draw[dmd] plot[domain=0.98:6.2,samples=80,smooth] (\x,{dem(\x,3)})
    node[etiqueta,right] {$d$};}
  \capa{3}{\draw[imgs] plot[domain=0.245:5,samples=80,smooth] (\x,{img(\x,3)})
    node[etiqueta,right] {IMg};}
  \capa{3}{\node[punto] at (3,1) {};}
  \capa{3}{\draw[guia] (3,0) node[below,text=black] {$q_a$} -- (3,2);}
  \capa{3}{\node[punto] at (3,2) {};}
  \capa{3}{\draw[guia] (3,2) -- (0,2) node[left,text=black] {$p^*\!/w$};}
  \capa{3}{\fill[area] (0,{cme(3)}) rectangle (3,2);}
  \capa{3}{\node[punto,fill=tceeverde] at (3,{cme(3)}) {};}
  \capa{3}{\node[etiqueta,text=tceemorado,font=\footnotesize] at (2.2,1.8) {$B>0$};}
  % ---------------- 4: b) entrada ----------------
  \begin{scope}[xshift=9.6*0.75cm]
    \capa{4}{\draw[dmd,desplazada,opacity=0.4] plot[domain=0.98:6.2,samples=80,smooth] (\x,{dem(\x,3)});}
    \capa{4}{\draw[imgs,desplazada,opacity=0.4] plot[domain=0.245:5,samples=80,smooth] (\x,{img(\x,3)});}
    \capa{4}{\draw[curvanueva] plot[domain=0.588:6.2,samples=80,smooth] (\x,{dem(\x,1.8)})
      node[etiqueta,right] {$d'$};}
    \capa{4}{\draw[curvanueva,thick] plot[domain=0.147:6.2,samples=80,smooth] (\x,{img(\x,1.8)})
      node[etiqueta,right] {IMg$'$};}
    \capa{4}{\draw[flecha] (2.6,{dem(2.6,3)+0.08}) -- (1.75,{dem(1.75,1.8)+0.08});}
    \capa{4}{\node[punto] at (1.8,1) {};}
    \capa{4}{\draw[guia] (6.2,2) -- (0,2) node[left,text=black] {$p^*\!/w$};}
    \capa{4}{\node[punto] at (1.8,2) {};}
    \capa{4}{\draw[guia] (1.8,0) node[below,text=black] {$q_b$} -- (1.8,2);}
  \end{scope}
  % ---------------- 5: c) equilibrio con libre entrada ----------------
  \begin{scope}[xshift=19.2*0.75cm]
    \capa{5}{\draw[dmd] plot[domain=0.3265:6.2,samples=80,smooth] (\x,{dem(\x,1)})
      node[etiqueta,right] {$d$};}
    \capa{5}{\draw[imgs] plot[domain=0.0816:6.2,samples=80,smooth] (\x,{img(\x,1)})
      node[etiqueta,right] {IMg};}
    \capa{5}{\node[punto] at (1,1) {};}
    \capa{5}{\node[punto] (E) at (1,2) {};}
    \capa{5}{\node[etiqueta,above right] at (E) {$E^*$};}
    \capa{5}{\draw[guia] (1,0) node[below,text=black] {$q^*$} -- (E);}
    \capa{5}{\draw[guia] (6.2,2) -- (E) -- (0,2) node[left,text=black] {$p^*\!/w$};}
    \capa{5}{\node[etiqueta,font=\footnotesize,anchor=west] at (2.6,2.25)
      {$\frac{p^*}{w}=\frac{\sigma}{\sigma-1}\,c=\mathrm{CMe}(q^*)$};}
  \end{scope}
\end{tikzpicture}
\end{document}
