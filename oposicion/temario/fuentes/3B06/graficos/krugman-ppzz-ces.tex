% Krugman con elasticidad de sustitución constante (CES): |e| = sigma, de modo que la
% curva PP es horizontal, p/w = c sigma/(sigma-1). Con c = 1 y sigma = 3, p/w = 1,5.
%   ZZ (beneficio nulo): p/w = c + F/(L x); aquí 1 + 3/x, y con 2L, ZZ': 1 + 1,5/x.
%   E0 = (6; 1,5) y E1 = (3; 1,5): p/w no cambia; x se reduce a la mitad y q = L x
%   no cambia (3/0,5 = 6 en unidades de F/c); solo crece el número de variedades.
% Elaboración propia a partir de Krugman (1980) y Dixit y Stiglitz (1977).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.9cm,y=1.6cm,
    declare function={zz(\x,\k)=1+\k/\x;}]
  % 1: ejes y asíntota p/w = c
  \capa{1}{\ejes{$x$}{$p/w$}{8.6}{3.4}}
  \capa{1}{\draw[guia] (0,1) node[left,text=black] {$c$} -- (8.2,1);}
  % 2: PP horizontal: margen constante
  \capa{2}{\draw[curva1] (0,1.5) node[left,text=black] {$\frac{\sigma}{\sigma-1}c$} -- (8.2,1.5);}
  \capa{2}{\node[etiqueta,above right,text=tceemorado] at (0.1,1.5) {PP};}
  % 3: ZZ de autarquía
  \capa{3}{\draw[curva2] plot[domain=1.25:8,samples=80,smooth] (\x,{zz(\x,3)})
    node[etiqueta,right] {ZZ};}
  % 4: equilibrio de autarquía E0
  \capa{4}{\node[punto] (E0) at (6,1.5) {};}
  \capa{4}{\node[etiqueta,above right] at (E0) {$E_0$};}
  \capa{4}{\draw[guia] (E0) -- (6,0) node[below,text=black] {$x_0$};}
  % 5: la apertura (L -> 2L) baja ZZ
  \capa{5}{\draw[curvanueva] plot[domain=0.65:8,samples=80,smooth] (\x,{zz(\x,1.5)})
    node[etiqueta,right] {ZZ$'$};}
  \capa{5}{\draw[flecha] (2.2,{zz(2.2,3)-0.04}) -- (2.2,{zz(2.2,1.5)+0.04});}
  % 6: nuevo equilibrio E1 sobre la misma PP: mismo p/w, x a la mitad
  \capa{6}{\node[punto] (E1) at (3,1.5) {};}
  \capa{6}{\node[etiqueta,above right] at (E1) {$E_1$};}
  \capa{6}{\draw[guia] (E1) -- (3,0) node[below,text=black] {$x_1=x_0/2$};}
  \capa{6}{\draw[flecha,shorten >=4pt,shorten <=4pt] (E0) -- (E1);}
\end{tikzpicture}
\end{document}
