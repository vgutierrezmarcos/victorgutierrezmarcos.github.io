% Las 2 últimas recesiones de España (3.A.42, III.1 y conclusión): PIB real en
% índice (pico = 100) frente a los trimestres desde el pico.
%   índice = 100 exp((ES_ln_pib_x100(t) - ES_ln_pib_x100(pico))/100)
% Datos: ../datos/hp-pib-espana-zona-euro.csv (Eurostat namq_10_gdp, hasta
% 2026T3). Fila 53 (desde 0) = 2008T2, 100 ln PIB = 1254,603; fila 99 = 2019T4,
% 1261,488.
%   desde 2008T2: fondo en 2013T3 (21 trimestres), -8,9 %; recupera el pico en
%     2016T4 (34 trimestres después);
%   desde 2019T4: fondo en 2020T2 (2 trimestres), -22,0 %; recupera el pico en
%     2022T2 (10 trimestres después).
% Comprobado a mano con el CSV. Último dato: 2026T3, 112,6 en la de 2020. Picos de la AEE (fechado-recesiones.csv).
% Colores: la recesión que empezó en el crédito, dorado (el del ciclo
% financiero); la de la pandemia, azul.
% Autor: Víctor Gutiérrez Marcos

        restrict x to domain=0:40] {\hpdatos};

        restrict x to domain=0:40] {\hpdatos};
      \addlegendentry{desde 2019T4 (pandemia)}}
    \capa{3}{\addplot[dicpuntos=tceeazul, forget plot] coordinates {(2,77.95) (10,100.34)};}
    \capa{3}{\dicrotulo[anchor=west, font=\footnotesize, text=tceeazul]{(2.5,77.95)}{2020T2: $-22{,}0$~\%}
