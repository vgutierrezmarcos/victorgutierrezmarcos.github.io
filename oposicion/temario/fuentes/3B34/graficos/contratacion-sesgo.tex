% Preferencia nacional en la contratación pública (Baldwin, 1970; Evenett y
% Hoekman, 2005): país pequeño, bien homogéneo, oferta extranjera
% perfectamente elástica a p* y oferta nacional S^d(p) = (p - 0,5)/0,6.
%   p* = 2 -> S^d(p*) = 2,5
%   G_1 = 1,5 < S^d(p*): la preferencia es neutral (solo reetiqueta)
%   G_2 = 4   > S^d(p*): p_g = 0,5 + 0,6 * 4 = 2,9 > p*; coste para el
%   contribuyente (p_g - p*) G_2 (rectángulo sombreado).
% Capas: ejes -> oferta nacional -> precio mundial -> caso 1 (G_1) ->
% caso 2 (G_2, p_g y el sobrecoste).
% Autor: Víctor Gutiérrez Marcos

  \capa{2}{\draw[curva1] (0,0.5) -- (5.8,3.98) node[etiqueta,right,text=tceemorado] {$S^d$};}

  \capa{3}{\draw[curva2] (0,2) node[left,text=black] {$p^*$} -- (6.1,2) node[etiqueta,right,text=tceeazul,align=left,font=\footnotesize] {oferta\\extranjera};}
