% Composición sectorial de las exportaciones mundiales de servicios
% comerciales, 2005 y 2025 (% del total). Fuente en la cabecera de
% ../datos/servicios-composicion.csv (OMC, WTO-UNCTAD-ITC).
% Capas: barras de 2005 -> barras de 2025 -> ejes -> rótulos de 2025.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
\colorlet{comp2005}{tceegris!40}
\colorlet{comp2025}{tceemorado!80}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/servicios-composicion.csv}\composicion
\pgfplotsset{ejescomp/.style={}}
\ifnum3>\tceepaso\relax
  \pgfplotsset{ejescomp/.style={axis line style={draw=none},
    major tick style={draw=none}, ticklabel style={opacity=0},
    label style={opacity=0}, xmajorgrids=false}}
\fi
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=13.5cm, height=9cm,
      xmin=0, xmax=30, ymin=-0.6, ymax=9.6,
      y dir=reverse,
      ytick={0,...,9}, yticklabels from table={\composicion}{sector},
      yticklabel style={font=\footnotesize},
      major y tick style={draw=none},
      xtick={0,5,...,30}, xticklabel style={font=\footnotesize},
      xlabel={\% de las exportaciones mundiales de servicios comerciales},
      axis x line*=bottom, axis y line*=left, tick align=outside,
      xmajorgrids, grid style={tceegris!25},
      clip=false,
      legend style={at={(0.98,0.03)}, anchor=south east, draw=none, fill=none,
        font=\footnotesize, row sep=1pt},
      legend cell align=left,
      ejescomp, dicapas,
    ]
    % 1: 2005
    \capa{1}{
      \addplot[on layer=dicbarras, xbar, bar width=6pt, bar shift=3.4pt, dicbarra=comp2005, area legend]
        table[x=peso_2005, y expr=\coordindex] {\composicion};
      \addlegendentry{2005}
    }
    % 2: 2025
    \capa{2}{
      \addplot[on layer=dicbarras, xbar, bar width=6pt, bar shift=-3.4pt, dicbarra=comp2025, area legend]
        table[x=peso_2025, y expr=\coordindex] {\composicion};
      \addlegendentry{2025}
    }
    % 3: ejes (ver ejescomp)
    % 4: rótulos de 2025
    \capa{4}{
      \addplot[dicsuelto, only marks, mark=none,
        nodes near coords={\pgfmathprintnumber[fixed,precision=1,use comma]{\pgfplotspointmeta}},
        nodes near coords style={font=\scriptsize, anchor=west, text=tceemorado!90!black, yshift=-3.4pt},
        point meta=explicit]
        table[x=peso_2025, y expr=\coordindex, meta=peso_2025] {\composicion};
    }
  \end{axis}
\end{tikzpicture}
\end{document}
