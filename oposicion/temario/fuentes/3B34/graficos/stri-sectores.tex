% Índice de restricción del comercio de servicios (STRI) de la OCDE por sector,
% 2025 (0 = abierto; 1 = cerrado): media de los 51 países (barras), recorrido
% entre el país más abierto y el más cerrado (raya gris) y España (punto).
% 18 sectores (los que publica el dataflow SDMX). Fuente en la cabecera de
% ../datos/stri-sectores.csv.
% Capas: barras de la media -> ejes -> rayas del recorrido -> puntos de
% España -> etiquetas de la media.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
\colorlet{strimedia}{tceemorado!60}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/stri-sectores.csv}\stri
\pgfplotsset{ejesstri/.style={}}
\ifnum2>\tceepaso\relax
  \pgfplotsset{ejesstri/.style={axis line style={draw=none},
    major tick style={draw=none}, ticklabel style={opacity=0},
    label style={opacity=0}, xmajorgrids=false}}
\fi
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=13.5cm, height=10.5cm,
      xmin=0, xmax=1.02, ymin=-0.6, ymax=17.6,
      y dir=reverse,
      ytick={0,...,17}, yticklabels from table={\stri}{sector},
      yticklabel style={font=\footnotesize},
      major y tick style={draw=none},
      xtick={0,0.2,...,1}, xticklabel style={font=\footnotesize,
        /pgf/number format/.cd, fixed, precision=1, use comma},
      xlabel={STRI, 2025 (0 = abierto; 1 = cerrado)},
      axis x line*=bottom, axis y line*=left, tick align=outside,
      xmajorgrids, grid style={tceegris!25},
      clip=false,
      legend style={at={(0.98,0.02)}, anchor=south east, draw=none,
        font=\footnotesize, row sep=1pt},
      legend cell align=left,
      ejesstri, dicapas,
    ]
    % 1: media de los 51 países
    \capa{1}{
      \addplot[on layer=dicbarras, xbar, bar width=7pt, area legend, dicbarra=strimedia]
        table[x=media, y expr=\coordindex] {\stri};
      \addlegendentry{Media de 51 países}
    }
    % 2: ejes (ver ejesstri)
    % 3: recorrido entre el mínimo y el máximo
    \capa{3}{
      \addplot[dicsuelto, draw=none, mark=none, tceegris,
        error bars/.cd, x dir=both, x explicit, error mark=|,
        error bar style={thick, tceegris}]
        table[x expr=(\thisrow{minimo}+\thisrow{maximo})/2, y expr=\coordindex,
              x error expr=(\thisrow{maximo}-\thisrow{minimo})/2] {\stri};
      \addlegendimage{thick, tceegris}
      \addlegendentry{Del país más abierto al más cerrado}
    }
    % 4: España
    \capa{4}{
      \addplot[on layer=dicpuntos, only marks, mark=*, mark size=2.2pt,
        draw=tceerojo!70!black, fill=tceerojo]
        table[x=espana, y expr=\coordindex] {\stri};
      \addlegendentry{España}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
