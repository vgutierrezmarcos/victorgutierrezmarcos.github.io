% Componentes del problema dinámico del consumidor (rehace la tabla-esquema
% «Componentes» del tema antiguo, en analogía con esquema-ingredientes del 3.A.8):
% preferencias intertemporales -> función de utilidad intertemporal; con la
% restricción intertemporal de recursos -> problema de maximización -> senda óptima.
% Pasos: componentes-consumidor-paso1 (capas 1 y 2) y este gráfico completo.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
% En los pasos (\def\tceefantasma), lo que aún no se ha dibujado ocupa su sitio
% sin verse, para que todos los pasos tengan el mismo tamaño
\ifdefined\tceefantasma
  \renewcommand{\capa}[2]{\ifnum#1>\tceepaso\relax\begin{scope}[opacity=0]#2\end{scope}\else#2\fi}
\fi
\begin{document}
\begin{tikzpicture}[tcee,node distance=0.7cm and 0.9cm,
    caja/.style={draw=tceemorado,thick,fill=tceesalvia,rounded corners=3pt,align=center,
                 minimum height=1.3cm,inner sep=5pt},
    clave/.style={caja,fill=tceemorado!12,very thick},
    une/.style={->,thick,tceemorado}]
  % 1: las preferencias intertemporales, sobre sendas de consumo
  \capa{1}{\node[caja] (P) at (0,0) {Preferencias\\ intertemporales\\ sobre sendas $(C_0,\dots,C_T)$};}
  % 2: la función de utilidad intertemporal que las representa
  \capa{2}{\node[caja] (U) at (5.6,0) {Función de utilidad intertemporal\\ $U_0=\sum_{t=0}^{T}\beta^t\,u(C_t)$};}
  \capa{2}{\draw[une] (P) -- (U);}
  % 3: la restricción intertemporal de recursos
  \capa{3}{\node[caja] (R) at (0,-2.6) {Restricción intertemporal\\ de recursos\\ $b_{t+1}=(1+r)\,b_t+Y_t-C_t$\\ $b_0$ dado, $b_{T+1}\geq0$};}
  % 4: el problema de maximización de la utilidad, que se resuelve con un método
  \capa{4}{\node[clave] (M) at (5.6,-2.6) {Problema de maximización\\ $\max\ U_0$ s.a. las restricciones\\ {\footnotesize(\textsc{Lagrange}, control óptimo, \textsc{Bellman})}};}
  \capa{4}{\draw[une] (U) -- (M);}
  \capa{4}{\draw[une] (R) -- (M);}
  % 5: el resultado, la senda óptima de consumo
  \capa{5}{\node[caja] (S) at (5.6,-5.0) {Senda óptima de consumo\\ $\{C_t^*\}_{t=0}^{T}$ y de activos $\{b_{t+1}^*\}_{t=0}^{T}$};}
  \capa{5}{\draw[une] (M) -- (S);}
\end{tikzpicture}
\end{document}
