% La ESI y la curvatura de las curvas de indiferencia intertemporales.
% U = u(C_1) + u(C_2) con la CRRA (beta = 1 para que las 3 sean simétricas),
% todas por el punto (4, 4) de la bisectriz:
%   theta = 1/2 (ESI = 2):  raíz(C_1) + raíz(C_2) = 4      -> C_2 = (4 - raíz(C_1))^2
%   theta = 1   (ESI = 1):  C_1 C_2 = 16                    -> C_2 = 16/C_1
%   theta = 4   (ESI = 1/4): C_1^-3 + C_2^-3 = 2/64         -> C_2 = (0,03125 - C_1^-3)^(-1/3),
%     asíntota en C_1 = 4·2^(-1/3) = 3,175; C_2 = 10 en C_1 = 3,21.
% Cuanto mayor es la ESI, menos curvada es la curva (más cerca de una recta);
% cuanto menor, más se acerca a un ángulo recto.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,scale=0.55]
  % 1: ejes: consumo de hoy y de mañana
  \capa{1}{\ejes{$C_1$}{$C_2$}{10.8}{10.8}}
  % 2: bisectriz y punto común (4, 4)
  \capa{2}{\draw[guia] (0,0) -- (10,10) node[etiqueta,above right,text=tceegris] {$C_1=C_2$};}
  \capa{2}{\node[punto] (X) at (4,4) {};}
  % 3: ESI = 1 (logarítmica)
  \capa{3}{\draw[curva1] plot[domain=1.6:10,samples=80] (\x,{16/\x}) node[etiqueta,right] {$\mathrm{ESI}=1$};}
  % 4: ESI alta: menos curvada
  \capa{4}{\draw[curva2] plot[domain=0.7:10,samples=100] (\x,{(4-sqrt(\x))^2}) node[etiqueta,below right=-2pt and 0pt] {$\mathrm{ESI}=2$};}
  % 5: ESI baja: casi un ángulo recto
  \capa{5}{\draw[curva3] plot[domain=3.21:10,samples=150] (\x,{(0.03125-\x^(-3))^(-1/3)}) node[etiqueta,above right=-2pt and 0pt] {$\mathrm{ESI}=1/4$};}
\end{tikzpicture}
\end{document}
