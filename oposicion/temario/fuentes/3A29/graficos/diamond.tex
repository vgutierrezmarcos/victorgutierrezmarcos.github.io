% Modelo de Diamond (1965) con Cobb-Douglas, utilidad logarítmica y delta = 1:
% ley de movimiento k_{t+1} = beta(1-alfa)/[(1+beta)(1+n)(1+g)] k_t^alfa
% (no depende de delta: los viejos venden el capital no depreciado) y bisectriz.
% Gráfico de referencia de la ficha olg-samuelson-diamond del canon.
% Elaboración propia a partir de Romer (2019, cap. 2) y Acemoglu (2009, cap. 9).
% Cualitativo: alfa = 0,3, beta = 0,9 y el coeficiente escalado a
%   a = beta(1-alfa)A/[(1+beta)(1+n)(1+g)] = 1,6 (A absorbe la escala):
%   k_{t+1} = 1,6 k_t^0,3, cóncava, con pendiente infinita en el origen;
%   k* = a^(1/(1-alfa)) = 1,6^(1/0,7) = 1,957.
%   Regla de oro con delta = 1: alfa k^(alfa-1) = (1+n)(1+g) = 0,9·0,7·A/(1,9·1,6),
%   con A = 1: k_oro = [alfa/((1+n)(1+g))]^(1/(1-alfa)) = 1,696 < k*:
%   ineficiencia dinámica posible porque beta = 0,9 > alfa/(1-2alfa) = 0,75.
%   Senda desde k_0 = 0,3: k_1 = 1,115; k_2 = 1,653; k_3 = 1,860 (converge a k*);
%   la escalera se dibuja hasta k_3.
% Autor: Víctor Gutiérrez Marcos

  % 1: ejes: capital por unidad de trabajo eficiente hoy y mañana
  \capa{1}{\ejes{$k_t$}{$k_{t+1}$}{3.3}{3.0}}
  % 2: bisectriz: k_{t+1} = k_t
  \capa{2}{\draw[thick,tceegris] (0,0) -- (2.85,2.85) node[etiqueta,above,text=black] {$k_{t+1}=k_t$};}
  \capa{2}{\draw[thin,tceegris] (0.45,0) arc[start angle=0,end angle=45,radius=0.45];}
  \capa{2}{\node[etiqueta,font=\scriptsize] at (0.62,0.2) {$45^{\circ}$};}
  % 3: ley de movimiento, cóncava, con pendiente infinita en el origen
  \capa{3}{\draw[curva1] plot[domain=0:3.1,samples=150] (\x,{ley(\x)}) node[etiqueta,right,text=tceemorado] {ley de movimiento};}
  \capa{3}{\node[etiqueta,font=\footnotesize,text=tceemorado,anchor=west] at (2.2,1.72) {$k_{t+1}=\frac{\beta(1-\alpha)}{(1+\beta)(1+n)(1+g)}\,k_t^{\alpha}$};}
  % 4: estado estacionario: corte con la bisectriz
  \capa{4}{\node[punto] (E) at (1.957,1.957) {};}
  \capa{4}{\draw[guia] (0,1.957) node[left,text=black] {$k^*$} -| (1.957,0) node[below,text=black] {$k^*$};}
  \capa{4}{\node[etiqueta,above left] at (E) {$E$};}
  % 5: senda desde k_0: escalera entre la curva y la bisectriz
  \capa{5}{\node[etiqueta,below] at (0.3,0) {$k_0$};}
  \capa{5}{\draw[flecha] (0.3,0) -- (0.3,1.115) -- (1.115,1.115) -- (1.115,1.653) -- (1.653,1.653) -- (1.653,1.860) -- (1.860,1.860);}
  \capa{5}{\draw[guia] (1.115,0) -- (1.115,1.115);}
  \capa{5}{\node[etiqueta,below] at (1.115,0) {$k_1$};}
  % 6: regla de oro a la izquierda de k*: ineficiencia dinámica
  \capa{6}{\draw[thick,tceerojo] (1.696,-0.05) -- (1.696,0.05);}
  \capa{6}{\node[etiqueta,below,text=tceerojo] at (1.6,0) {$k_{\mathrm{oro}}$};}
  \capa{6}{\node[etiqueta,font=\footnotesize,text=tceerojo,align=left,anchor=north west] at (2.2,0.95) {ineficiencia dinámica\\si $k^*>k_{\mathrm{oro}}$};}
