% r - g en España, la zona del euro y EE. UU., 1965-2025: tipo de interés real a
% largo plazo (AMECO ILRV) menos crecimiento del PIB real (variación de OVGD),
% en medias móviles de 5 años hacia atrás (el dato de 2025 es la media de
% 2021-2025). Datos: ../datos/r-menos-g-espana-ea-eeuu.csv, columnas ESP_mm5,
% ZE_mm5 (EA12 hasta 2001 enlazada con EA20 desde 2002) y USA_mm5 (cálculo
% propio, 01/11/2026). AMECO, previsión de primavera de 2026.
% Cifras de control (media móvil): España 2,4 (1985), -3,5 (2007), 6,4 (2013),
%   -5,7 (2025); zona del euro 3,8 (1996), 2,9 (2012), -4,1 (2025); EE. UU.
%   -3,5 (1981), 3,2 (1986), -1,4 (2006), -4,0 (2025).
% Paleta del tema: España dicespana, zona del euro dicze, EE. UU. tceedorado
% (los mismos que criterio-abel).
% Capas (CRITERIOS §7): ejes y rejilla → r = g → España → zona del euro →
% EE. UU. → etiquetas de las etapas y leyenda.
% Autor: Víctor Gutiérrez Marcos

    xtick={1965,1975,...,2025},ytick={-6,-4,...,8},

    \dicrotulo[font=\footnotesize,align=center]{(2019.6,-5.2)}{2022-2025:\\$r<g$}
    \addlegendimage{diclinea=dicespana}\addlegendentry{España}
    \addlegendimage{diclinea=dicze}\addlegendentry{zona del euro}
    \addlegendimage{diclinea=tceedorado}\addlegendentry{EE. UU.}
