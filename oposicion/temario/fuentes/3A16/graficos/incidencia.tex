% Incidencia de un impuesto unitario t en equilibrio parcial.
% Demanda inversa p^D(x) = 4 - 0,5 x; oferta inversa p^S(q) = 0,4 + q.
% Sin impuesto: x* = 2,4 y p* = 2,8. Con t = 1,2: p^D(x) - p^S(x) = t da
% x*(t) = 1,6, p_c = p^D(1,6) = 3,2 y p_v = p^S(1,6) = 2,0 (p_c = p_v + t).
% La oferta es más inclinada que la demanda (menos elástica en el equilibrio):
% los consumidores soportan 0,4 (1/3) y los vendedores 0,8 (2/3), como dice la ley de Dalton.
% Recaudación T = t x*(t) partida por p*; exceso de gravamen: triángulo entre x*(t) y x*.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=1.5cm,y=1.25cm,
    declare function={pd(\x)=4-0.5*\x; ps(\x)=0.4+\x;},
    perdida/.style={fill=tceerojo,fill opacity=0.3,draw=none}]
  % 1: ejes
  \capa{1}{\ejes{$x,q$}{$p$}{5.4}{4.8}}
  % 2: demanda
  \capa{2}{\draw[curva1] (0,{pd(0)}) -- (5.2,{pd(5.2)}) node[etiqueta,right] {$p^D(x)$};}
  % 3: oferta
  \capa{3}{\draw[curva3] (0,{ps(0)}) -- (3.9,{ps(3.9)}) node[etiqueta,above] {$p^S(q)$};}
  % 4: equilibrio sin impuesto
  \capa{4}{\node[punto] (E) at (2.4,2.8) {};}
  \capa{4}{\draw[guia] (E) -- (2.4,0) node[below,text=black] {$x^*$};}
  \capa{4}{\draw[guia] (E) -- (0,2.8) node[left,text=black] {$p^*$};}
  % 5: el impuesto desplaza la oferta hacia arriba en t
  \capa{5}{\draw[curvanueva,desplazada] (0,{ps(0)+1.2}) -- (2.6,{ps(2.6)+1.2}) node[etiqueta,above] {$p^S(q)+t$};}
  % 6: nueva cantidad x*(t) y precio del consumidor p_c
  \capa{6}{\node[punto] (C) at (1.6,3.2) {};}
  \capa{6}{\draw[guia] (C) -- (0,3.2) node[left,text=black] {$p_c$};}
  \capa{6}{\draw[guia] (C) -- (1.6,0) node[below,text=black] {$x^*(t)$};}
  % 7: precio del vendedor p_v y cuña fiscal t (llave a la izquierda del eje, para que no la cruce la oferta desplazada)
  \capa{7}{\node[punto] (V) at (1.6,2.0) {};}
  \capa{7}{\draw[guia] (V) -- (0,2.0) node[left,text=black] {$p_v$};}
  \capa{7}{\draw[decorate,decoration={brace,amplitude=4pt},thick] (-0.42,2.0) -- (-0.42,3.2) node[midway,left=4pt] {$t$};}
  % 8: recaudación que soportan los consumidores (entre p* y p_c)
  \capa{8}{\fill[area] (0,2.8) rectangle (1.6,3.2);}
  \capa{8}{\node[etiqueta,text=tceemorado,font=\footnotesize] at (0.6,3.0) {consumidores};}
  % 9: recaudación que soportan los vendedores (entre p_v y p*)
  \capa{9}{\fill[area2] (0,2.0) rectangle (1.6,2.8);}
  \capa{9}{\node[etiqueta,text=tceedorado,font=\scriptsize] at (1.13,2.17) {vendedores};}
  % 10: exceso de gravamen
  \capa{10}{\fill[perdida] (1.6,3.2) -- (2.4,2.8) -- (1.6,2.0) -- cycle;}
  \capa{10}{\draw[guia,solid,->] (3.1,1.6) node[etiqueta,below,text=tceerojo] {EG} -- (2.0,2.62);}
\end{tikzpicture}
\end{document}
