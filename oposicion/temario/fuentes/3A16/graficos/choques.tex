% Estática comparativa del equilibrio parcial: dos paneles.
% Izquierda (preferencias): p^D(x;a0) = 3,8 - 0,7 x sube a p^D(x;a1) = 4,8 - 0,7 x;
%   oferta p^S(q) = 0,5 + 0,8 q. E0 = (2,2; 2,26), E1 = (2,867; 2,793).
%   Sombreado: ganancia segura de excedente del productor (entre p0* y p1*, sobre la oferta).
% Derecha (tecnología): p^S(q;a0) = 0,5 + 0,8 q baja a p^S(q;a1) = -0,4 + 0,8 q;
%   demanda p^D(x) = 3,8 - 0,7 x. E0 = (2,2; 2,26), E1 = (2,8; 1,84).
%   Sombreado: ganancia segura de excedente del consumidor (entre p1* y p0*, bajo la demanda).

    titulo/.style={font=\small\bfseries,align=center,anchor=south}]
  % ---------------- Izquierda: aumento de la demanda ----------------
