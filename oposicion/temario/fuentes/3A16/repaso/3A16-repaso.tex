\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: «a río revuelto, ganancia de pescadores».
  \begin{itemize}
  \item Aceite: sequías $\Rightarrow$ 1,49~Mt (21/22) $\to$ 0,67 (22/23); origen 4,1 $\to$ 9,0 €/kg (ene.\ 2024).
  \item Cosecha 24/25 de 1,42~Mt $\Rightarrow$ $<$3,6 €/kg (may.\ 2025).
  \item ¿Cómo se forma el precio si nadie puede fijarlo? ¿Llega el mercado solo? ¿Cuánto dura la ganancia?
  \end{itemize}
\item \textbf{Relevancia}:
  \begin{itemize}
  \item \emph{teórica}: 1TFEB más sencillo \vertema{3.A.22}; referencia de las demás estructuras \vertema{3.A.17}--\vertema{3.A.19};
  \item \emph{empírica}: agrarios, materias primas, lonjas, divisas; oferta y demanda cuantifican choques;
  \item \emph{normativa}: competencia \vertema{4.A.15}, incidencia \vertema{4.B.8}\vertema{4.B.9}.
  \end{itemize}
\item \textbf{Contextualización}: \concepto{individualismo metodológico}; 2~estructuras polares (competencia perfecta, monopolio) y 2~intermedias; \concepto{equilibrio parcial}: un mercado aislado.
  \begin{itemize}
  \item \autor{Smith} (1776, precios «naturales») $\to$ \autor{Cournot} (1838, demanda), \autor{Dupuit} (1844, excedente), \autor{Jenkin} (1870, curvas);
  \item \autor{Walras} (1874, todos los mercados), \autor{Marshall} (1890, parcial, \emph{ceteris paribus}, tijeras);
  \item telaraña (\autor{Kaldor}, \autor{Ezekiel}), correspondencia (\autor{Samuelson}) $\to$ MWG (1995).
  \end{itemize}
\item \textbf{Problemática}: ¿cómo se forma el precio? ¿Existe, es único, se alcanza? ¿Es eficiente? ¿Qué cambia con el plazo?
\item \textbf{Estructura}: del instante al tiempo: \hyperref[rep:I]{I}, modelo ($J$ fijo); \hyperref[rep:II]{II}, existencia, estabilidad, bienestar y estática comparativa; \hyperref[rep:III]{III}, plazos.
\end{itemize}

\bloque[8,5]{I. Construcción del modelo ($J$ fijo)\label{rep:I}}
\begin{ideaclave}
Con utilidad cuasilineal y CMg creciente: demanda = suma horizontal de las utilidades marginales; oferta = suma de los tramos crecientes de los CMg; equilibrio = su corte. Hasta \hyperref[rep:III]{III}, $J$ fijo.
\end{ideaclave}

\sub{I.1. Idea: supuestos y equilibrio parcial\label{rep:I.1}}
\subsub{a) Qué es: los 5~supuestos}
\begin{itemize}
\item Ningún agente tiene poder sobre el precio. 5~supuestos = \concepto{lista común de las estructuras de mercado} (las demás marcan cuál levantan \vertema{3.A.17}--\vertema{3.A.19}):
  \begin{enumerate}
  \item \textbf{agentes racionales}: maximizan utilidad y beneficio;
  \item \textbf{información perfecta y simétrica}: producto, precio, tecnología;
  \item \textbf{producto homogéneo}: solo importa el precio;
  \item \textbf{atomicidad}: muchos y pequeños: \concepto{precio-aceptantes};
  \item \textbf{libre entrada y salida} (a LP): los beneficios extraordinarios atraen; las pérdidas expulsan.
  \end{enumerate}
\item 2 + 3 $\Rightarrow$ \emph{un solo precio}. El 5~solo actúa a LP ($J$ fijo hasta \hyperref[rep:III]{III}).
\end{itemize}
\subsub{b) Equilibrio parcial}
\begin{itemize}
\item \concepto{Equilibrio parcial}: un mercado que no afecta a los demás ni es afectado (\emph{ceteris paribus}; ya en \autor{Molina}, 1597).
  \begin{itemize}
  \item Se justifica si el mercado es pequeño: efectos renta despreciables; efectos sustitución repartidos.
  \end{itemize}
\item \emph{Cómo se modeliza} (MWG §10.B): bien $x$ + \concepto{numerario} $n$ (gasto en todo lo demás, precio 1); preferencias \concepto{cuasilineales} (sin efecto renta); costes en numerario con $\vect{w}$ dado.
\item \emph{Alternativa}: \concepto{equilibrio general} (\autor{Walras}) \vertema{3.A.21}; con cuasilineal coincide con el parcial (ley de \autor{Walras}).
\end{itemize}

\sub{I.2. Modelo: demanda, oferta y equilibrio\label{rep:I.2}}
\subsub{a) Supuestos}
\begin{itemize}
\item 2~bienes ($x$ y $n$), $p$ igual para todos (2~y 3); precio-aceptantes (4) que optimizan (1); $I$ consumidores con dotación $\overline{W}_i$ (incluye beneficios); $J$ empresas con $C_j(\vect{w},q_j)$.
\item \emph{$J$ fijo y todos los factores variables} (= LP sin libre entrada) $\Rightarrow$ oferta no perfectamente elástica.
\end{itemize}
\subsub{b) Consumidores: la demanda}
\begin{itemize}
\item Utilidad cuasilineal, $\phi_i'>0$, $\phi_i''<0$, $\phi_i(0)=0$ \vertema{3.A.8}:
\begin{formula}
\centering $\optim{\max}{n_i,\,x_i}{n_i+\phi_i(x_i)}{n_i+p\,x_i\leq\overline{W}_i\\ x_i\geq0}$\quad$\Rightarrow$\quad $\boxed{\phi_i'(x_i)=p}$\ \ ($\lambda=1$)
\end{formula}
  \begin{itemize}
  \item La CPO no depende de la renta: \concepto{demanda marshalliana} $x_i(p)=(\phi_i')^{-1}(p)$ = hicksiana \vertema{3.A.9}.
  \item Demanda inversa = utilidad marginal; si $\phi_i'(0)\leq p$, $x_i=0$.
  \end{itemize}
\item \concepto{Demanda de mercado}: suma horizontal, $x(p)=\sum_ix_i(p)$, decreciente, $\varepsilon_D<0$; inversa $p^D(x)$ = \concepto{beneficio marginal social}.
\end{itemize}
\subsub{c) Empresas: la oferta}
\begin{itemize}
\item \concepto{2 etapas} \vertema{3.A.12}: mín.\ coste $\to C_j(q_j)$; después máx.\ beneficio; $\mathrm{CMg}_j$ creciente en el tramo relevante:
\begin{formula}
\centering $\optim{\max}{q_j}{p\,q_j-C_j(q_j)}{q_j\geq0}$\quad$\Rightarrow$\quad $\boxed{p=\mathrm{CMg}_j(q_j)}$,\ \ $C_j''\geq0$
\end{formula}
  \begin{itemize}
  \item IMg = precio (demanda horizontal, $\neq$ monopolio \vertema{3.A.17}); CSO: tramo \emph{creciente} del CMg.
  \end{itemize}
\item \emph{Inacción}: produce si $p$ cubre el coste medio \emph{evitable}.
  \begin{itemize}
  \item Sin hundidos, mín.\ CMe; con hundidos (CP), mín.\ CVMe; CMe en U $\Rightarrow$ la oferta salta de 0~a la \concepto{escala mínima eficiente} $\overline{q}_j$ en $\overline{c}_j=\min\mathrm{CMe}_j$.
  \end{itemize}
\item \concepto{Oferta de la empresa} $q_j(p)$: inversa del tramo creciente del CMg sobre el umbral; 0~debajo; no decreciente (ley de la oferta).
\item \concepto{Oferta de mercado}: $q(p)=\sum_jq_j(p)$, $\varepsilon_S\geq0$; inversa $p^S(q)$ = \concepto{coste marginal de la industria}.
\end{itemize}
\graficorepaso[0.5\linewidth]{oferta-empresa}{Oferta de la empresa: $p=\mathrm{CMg}_j$ sobre el mín.\ del CMe}
\subsub{d) El equilibrio competitivo}
\begin{itemize}
\item \concepto{Equilibrio competitivo}: $p^*$ y $(x_i^*;q_j^*)$ con:
  \begin{enumerate}
  \item máx.\ utilidad, $x_i^*=x_i(p^*)$;
  \item máx.\ beneficio, $q_j^*=q_j(p^*)$;
  \item vaciado, $x(p^*)=q(p^*)$. Interior:
  \end{enumerate}
\funcionclave{\phi_i'(x_i^*)=p^*\ \ \forall i,\qquad p^*=\mathrm{CMg}_j(q_j^*)\ \ \forall j,\qquad \textstyle\sum_ix_i^*=\sum_jq_j^*}
\item \emph{Interpretación}: decisiones óptimas y compatibles; solo en $p^*$ nadie quiere cambiar el precio.
  $p^D(x^*)=p^S(q^*)$: BMgS = CMgS $\Rightarrow$ anticipa la eficiencia.
\end{itemize}
\graficorepaso[0.66\linewidth]{equilibrio}{Oferta de mercado (suma horizontal) y equilibrio competitivo}

\bloque[12]{II. Estudio del equilibrio competitivo\label{rep:II}}
\begin{ideaclave}
Con curvas continuas y de pendiente normal existe y es único; la estabilidad depende del ajuste (la telaraña puede no converger aun con curvas normales). Si se alcanza, maximiza el ET (1TFEB); un impuesto se reparte según elasticidades y destruye bienestar.
\end{ideaclave}

\sub{II.1. Análisis positivo: existencia, unicidad, estabilidad\label{rep:II.1}}
\begin{itemize}
\item \concepto{Existencia} (si no, no describe nada), \concepto{unicidad} (si no, no predice), \concepto{estabilidad} (si no, curiosidad).
\end{itemize}
\subsub{a) Existencia}
\begin{itemize}
\item \emph{Idea}: \concepto{exceso de demanda} $z(p)=x(p)-q(p)$; equilibrio: $z(p^*)=0$.
\item \emph{Supuestos}: $\phi_i$ estr.\ cóncavas, costes estr.\ convexos; $\max_i\phi_i'(0)>\min_j\mathrm{CMg}_j(0)$.
\item \emph{Desarrollo}: $z$ continua, $>0$ a precios bajos, $<0$ a altos $\Rightarrow$ \concepto{teorema de \autor{Bolzano}}:
\begin{formula}
\centering $z(p')>0,\ z(p'')<0,\ p'<p''\ \Longrightarrow\ \exists\,p^*\in(p',p''):\ z(p^*)=0$
\end{formula}
\item \emph{Valoración}: existe con convexidad (en general, \autor{Brouwer} \vertema{3.A.21}).
  \begin{itemize}
  \item Falla con \emph{no convexidades}: CMe en U y una empresa $\Rightarrow$ la demanda pasa por el hueco $[0,\overline{q})$ (redes, electricidad \vertema{3.A.17}\vertema{3.A.20}).
  \end{itemize}
\end{itemize}
\subsub{b) Unicidad}
\begin{itemize}
\item Demanda estr.\ decreciente y oferta no decreciente $\Rightarrow$ $z$ estr.\ decreciente: \emph{asignación} única.
\item \emph{Multiplicidad}:
  \begin{itemize}
  \item \emph{demanda insuficiente} ($\max_i\phi_i'(0)<\min_j\mathrm{CMg}_j(0)$): cualquier precio intermedio, sin intercambio;
  \item \emph{CMg constantes}: $p^*=c$, reparto indeterminado;
  \item \emph{pendientes anómalas} (oferta decreciente por economías externas, o \autor{Giffen}): varios cortes.
  \end{itemize}
\end{itemize}
\subsub{c) Estabilidad: dinámicas de ajuste}
\begin{itemize}
\item \emph{Idea}: \concepto{estable} si fuera del equilibrio hay fuerzas que llevan a él; \concepto{local} o \concepto{global}.
\item \emph{Supuestos}:
  \begin{itemize}
  \item \concepto{ajuste walrasiano} (precios; sin intercambios fuera del equilibrio; \emph{tâtonnement}): $\dot p=f(z(p))$, $f'>0$;
  \item \concepto{ajuste marshalliano} (cantidades; precio de demanda frente a precio de oferta): $\dot q=g(p^D(q)-p^S(q))$, $g'>0$.
  \end{itemize}
\item \emph{Desarrollo}:
\begin{formula}
\centering \autor{Walras}: estable si $z'(p^*)=x'(p^*)-q'(p^*)<0$\\[2pt]
\autor{Marshall}: estable si $\dfrac{d}{dq}\big(p^D-p^S\big)\Big|_{q^*}<0$
\end{formula}
\item \emph{Implicaciones}: oferta creciente y demanda decreciente $\Rightarrow$ ambos estables.
  \begin{itemize}
  \item Ambas decrecientes ($a=dp^D/dq$, $b=dp^S/dq$): \autor{Walras} exige $a>b$ (oferta \emph{más inclinada}); \autor{Marshall}, $a<b$ (\emph{más plana}): incompatibles.
  \end{itemize}
\end{itemize}

\begin{itemize}

\subsubsub{Extensión: la telaraña}
\begin{itemize}
\item \autor{Kaldor} (1934, nombre), \autor{Ezekiel} (1938). \emph{Supuestos}: tiempo discreto; oferta según $p^e_t$; \emph{expectativas ingenuas} $p^e_t=p_{t-1}$; vaciado $x(p_t)=q(p^e_t)$.
\item \emph{Desarrollo}:
\begin{formula}
\centering $p_t-p^*\simeq-\dfrac{q'(p^*)}{\lvert x'(p^*)\rvert}\,(p_{t-1}-p^*)$
\end{formula}
  \begin{itemize}
  \item El signo $-$ $\Rightarrow$ el precio \emph{oscila}: $q'<\lvert x'\rvert$, \emph{convergente}; $=$, \emph{continua}; $>$, \emph{divergente}.
  \item Converge si $\varepsilon_S<\lvert\varepsilon_D\rvert$ $\Leftrightarrow$ en $(q,p)$, oferta \emph{más inclinada} que la demanda.
  \end{itemize}
\end{itemize}
\graficorepaso[0.74\linewidth]{telarana}{Telaraña convergente, continua y divergente}
\begin{itemize}
\item \emph{Evidencia}: ciclo del porcino (\autor{Ezekiel}); laboratorio (\autor{Hommes} et al., 2007): con equilibrio inestable, volatilidad persistente.
\item \emph{Valoración}:
  \begin{itemize}
  \item \autor{Muth} (1961): errores sistemáticos sin aprender $\to$ \emph{expectativas racionales} \vertema{3.A.6};
  \item adaptativas (\autor{Nerlove}, 1958) amortiguan más \vertema{3.A.5}; no lineal: ciclo límite;
  \item el walrasiano, irreal salvo en subastas.
  \end{itemize}
\end{itemize}

\sub{II.2. Análisis normativo: eficiencia y bienestar\label{rep:II.2}}
\subsub{a) Optimalidad de \autor{Pareto}: los excedentes}
\begin{itemize}
\item \concepto{Óptimo de \autor{Pareto}} (1906): no hay otra asignación factible en que alguien mejore y nadie empeore; ordinal e incompleto \vertema{3.A.24}.
\item \concepto{Excedente del consumidor} (EC; \autor{Dupuit}, 1844; \autor{Marshall}): bajo la demanda y sobre el precio; con cuasilineal, medida \emph{exacta} (= VC = VE \vertema{3.A.9}).
\item \concepto{Excedente del productor} (EP): sobre la oferta y bajo el precio; $\mathrm{EP}_j=B_j+\mathrm{CF}_j$. \concepto{Excedente total}: ET = EC + EP.
\begin{formula}
\centering $\mathrm{EC}=\displaystyle\int_0^{x^*}p^D(x)\,dx-p^*x^*$,\quad $\mathrm{EP}=p^*q^*-\displaystyle\int_0^{q^*}p^S(q)\,dq$
\end{formula}
\item \emph{Por qué basta el ET}: numerario uno a uno $\Rightarrow$ frontera de utilidades de pendiente $-1$: Pareto $\Leftrightarrow$ máx.\ ET. $\mathrm{ET}'(x)=p^D-p^S=0$ en $x^*$.
\end{itemize}
\graficorepaso[0.55\linewidth]{excedentes}{Excedentes: ET máximo en $x^*$}
\subsub{b) Teoremas fundamentales del bienestar}
\begin{itemize}
\item \concepto{1TFEB}: precio-aceptantes, un mercado para cada bien (sin externalidades ni bienes públicos) y un solo precio (sin impuestos) $\Rightarrow$ el equilibrio competitivo es Pareto-óptimo. Demostración:
\funcionclave{\optim{\max}{(x_i)_i,\,(q_j)_j}{\textstyle\sum_i\phi_i(x_i)-\sum_jC_j(q_j)}{\textstyle\sum_ix_i=\sum_jq_j\\ x_i\geq0,\ q_j\geq0}}
  \begin{itemize}
  \item CPO: $\phi_i'(x_i)=\lambda$, $\mathrm{CMg}_j(q_j)=\lambda$, vaciado: las del equilibrio con $\lambda^*=p^*$ (\emph{precio sombra}).
  \item No hace falta convexidad (libre entrada con CMe en U, también eficiente) ni cuasilineal (solo mide).
  \item Falla con poder de mercado ($p>\mathrm{CMg}$ \vertema{3.A.17}), externalidades y bienes públicos \vertema{3.A.23} e impuestos (\hyperref[rep:II.3]{II.3}).
  \item «Mano invisible» de \autor{Smith} \vertema{3.A.2}: precios como señales de escasez.
  \end{itemize}
\item \concepto{2TFEB}: todo óptimo es equilibrio tras transferencias de suma fija.
  \begin{itemize}
  \item En parcial, inmediato: mismas cantidades, solo cambia el reparto del numerario.
  \item En general exige convexidad (hiperplano separador \vertema{3.A.22}). Separa eficiencia y equidad.
  \end{itemize}
\end{itemize}

\sub{II.3. Estática comparativa\label{rep:II.3}}
\begin{itemize}
\item Comparar equilibrios antes y después de un cambio: \concepto{principio de correspondencia} (\autor{Samuelson}, 1941, 1947): la estabilidad da el signo:
\begin{formula}
\centering $\dfrac{dp^*}{d\alpha}=-\dfrac{\partial z/\partial\alpha}{z'(p^*)}$,\qquad $z'(p^*)<0$
\end{formula}
\end{itemize}
\subsub{a) Cambios en las preferencias y en la tecnología}
\begin{itemize}
\item \emph{Preferencias} ($\partial\phi_i'/\partial\alpha>0$, calor y helados): $p\uparrow$, $x\uparrow$; EP $\uparrow$, EC ambiguo, ET $\uparrow$.
\item \emph{Tecnología} ($\partial\mathrm{CMg}_j/\partial\alpha<0$): oferta $\downarrow$, $p\downarrow$, $x\uparrow$; EC $\uparrow$, EP ambiguo, ET $\uparrow$.
\end{itemize}
\graficorepaso[0.6\linewidth]{choques}{Aumento de la demanda (izq.) y mejora tecnológica (dcha.)}
\subsub{b) Introducción de un impuesto}
\begin{itemize}
\item \emph{Supuestos}: unitario $t$; \concepto{cuña fiscal} $p_c=p_v+t$.
\item \emph{Desarrollo}: da igual quién lo pague legalmente (\emph{incidencia legal} $\neq$ \emph{económica}):
\begin{formula}
\centering $x(p_v+t)=q(p_v)$;\quad $\dfrac{dp_c}{dt}=\dfrac{\varepsilon_S}{\varepsilon_S+\lvert\varepsilon_D\rvert}$,\quad $\dfrac{dp_v}{dt}=-\dfrac{\lvert\varepsilon_D\rvert}{\varepsilon_S+\lvert\varepsilon_D\rvert}$
\end{formula}
\item \concepto{Ley de \autor{Dalton}} (1922; ya \autor{Jenkin}, 1872): recae más sobre el lado \emph{más inelástico}.
  \begin{itemize}
  \item $D$ perf.\ inelástica u $S$ perf.\ elástica (LP con libre entrada): consumidores;
  \item $D$ perf.\ elástica u $S$ perf.\ inelástica (suelo): vendedores.
  \end{itemize}
\item \emph{Implicaciones}: recaudación $T=t\,x^*(t)$; se pierden transacciones que valían más de lo que costaban: \concepto{pérdida irrecuperable de eficiencia} o \concepto{exceso de gravamen} (triángulo de \autor{Harberger}, 1964):
\begin{formula}
\centering $\mathrm{EG}=-\Delta\mathrm{EC}-\Delta\mathrm{EP}-T\approx\tfrac12\,t^2\,\dfrac{x^*}{p^*}\,\dfrac{\varepsilon_S\lvert\varepsilon_D\rvert}{\varepsilon_S+\lvert\varepsilon_D\rvert}$
\end{formula}
  \begin{itemize}
  \item Crece con $t^2$ y con las elasticidades \vertema{4.B.9}; distorsiona aunque se devuelva lo recaudado.
  \end{itemize}
\end{itemize}
\graficorepaso[0.55\linewidth]{incidencia}{Impuesto: cuña, reparto y pérdida irrecuperable}

\bloque[6]{III. Análisis de plazos\label{rep:III}}
\begin{ideaclave}
A CP la oferta es rígida y caben beneficios o pérdidas; a LP con libre entrada $p=\min\mathrm{CMe}$ y beneficio extraordinario nulo; la oferta de la industria es horizontal, creciente o decreciente según las economías externas.
\end{ideaclave}
\begin{itemize}
\item \concepto{Plazo}: no es tiempo de calendario, sino lo que las empresas pueden ajustar (MWG §10.F): CP (algún factor fijo), LP ($J$ fijo), libre entrada ($J^*$).
  \begin{itemize}
  \item Aceite: jornales (CP), olivos y almazara (LP), olivares nuevos (libre entrada).
  \end{itemize}
\end{itemize}


\sub{III.1. Corto plazo\label{rep:III.1}}
\begin{itemize}
\item \emph{Supuestos}: $J$ fijo; factor fijo con coste \emph{hundido}: $C_j=\mathrm{CF}_j+\mathrm{CV}_j(q_j)$, $\mathrm{CV}_j(0)=0$ (S invertida: proporciones variables).
\item \emph{Desarrollo}: el CF no influye; $p=\mathrm{CMg}_j$ si $p$ cubre el CVMe:
  \begin{itemize}
  \item bajo el \concepto{mínimo de explotación} (mín.\ CVMe), cierra;
  \item entre él y el \concepto{óptimo de explotación} (mín.\ CMe), produce con pérdidas $<\mathrm{CF}$;
  \item encima, beneficios. Industria: $q^{\mathrm{CP}}(p)=\sum_jq_j^{\mathrm{CP}}(p)$.
  \end{itemize}
\item \emph{Implicaciones}: beneficios (atraen) o pérdidas (expulsan); oferta \emph{rígida}: la demanda se va al precio (ganancia de los pescadores).
\end{itemize}
\graficorepaso[0.6\linewidth]{oferta-corto-plazo}{Oferta a CP: CMg sobre el mínimo de explotación}

\sub{III.2. Largo plazo sin libre entrada\label{rep:III.2}}
\begin{itemize}
\item Todos los factores variables, $J$ dado (licencia, recurso escaso) = modelo de \hyperref[rep:I]{I}-\hyperref[rep:II]{II}: $p=\mathrm{CMg}_j$ si $p\geq\min\mathrm{CMe}$.
  \begin{itemize}
  \item \concepto{\autor{Le Chatelier}-\autor{Samuelson}} (1884, 1947): más restricciones, menos respuesta $\Rightarrow$ más elástica que a CP.
  \end{itemize}
\item \emph{Implicaciones}: beneficio $\geq0$ y duradero; solo lo elimina la entrada (\autor{Clark}, 1899). Con entrada cerrada: renta del recurso.
\end{itemize}

\sub{III.3. Largo plazo con libre entrada\label{rep:III.3}}
\subsub{a) Idea}
\begin{itemize}
\item Infinitas empresas potenciales con la misma tecnología entran con beneficios y salen con pérdidas (supuesto 5): $J$ endógeno.
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item \emph{Supuestos}: misma $C(q)$, $C(0)=0$ (sin hundidos); $x(p)$ decreciente; equilibrios simétricos ($Jq$).
\item \emph{Desarrollo}: \concepto{equilibrio a LP con libre entrada}: $p^*,q^*,J^*$ entero con
  \begin{enumerate}
  \item $q^*$ máx.\ $p^*q-C(q)$;
  \item $x(p^*)=J^*q^*$;
  \item \emph{libre entrada}: $p^*q^*-C(q^*)=0$ (nueva ecuación para la nueva incógnita).
  \end{enumerate}
  \begin{itemize}
  \item $\Rightarrow$ $p^*=\mathrm{CMg}(q^*)=\mathrm{CMe}(q^*)=\min\mathrm{CMe}$.
  \end{itemize}
\item \concepto{Oferta de la industria a LP}: $\infty$ si beneficio $>0$; 0~si $<0$; cualquiera si $=0$. Según los rendimientos:
  \begin{itemize}
  \item \emph{constantes} $C=cq$: $p^*=c$, oferta horizontal, $J^*$ y $q^*$ \emph{indeterminados} ($J^*q^*=x(c)$);
  \item \emph{decrecientes en todo el dominio}: ínfimo del CMe en $q\to0$ $\Rightarrow$ \emph{no hay equilibrio};
  \item \emph{crecientes en todo el dominio}: monopolio natural \vertema{3.A.17};
  \item \emph{CMe en U} (S invertida o coste fijo evitable: suficiente, no necesario):
  \end{itemize}
\funcionclave{p^*=\overline{c}=\min\mathrm{CMe},\qquad q^*=\overline{q},\qquad J^*=\frac{x(\overline{c})}{\overline{q}}}
\end{itemize}
\graficorepaso[0.64\linewidth]{libre-entrada}{Libre entrada: rendimientos constantes, decrecientes y CMe en U (arriba, empresa; abajo, mercado)}
\begin{itemize}
\item \emph{Implicaciones}:
  \begin{itemize}
  \item \emph{beneficio económico nulo}: cubre el coste de oportunidad; la ganancia dura lo que tarda la entrada;
  \item \emph{eficiencia productiva}: escala mínima eficiente, nunca en el tramo de economías de escala;
  \item \emph{eficiencia asignativa}: 1TFEB sin convexidad.
  \end{itemize}
\item \emph{Del CP al LP}: $D_0\to D_1$: CP, $p_1>\overline{c}$, $q_1>\overline{q}$, beneficios ($B$); entrada hasta $p=\overline{c}$ con $J_1>J_0$ ($C$). El precio reacciona más a CP; la cantidad, a LP.
\end{itemize}
\graficorepaso[0.6\linewidth]{corto-largo-industria}{Del CP ($B$) al LP ($C$) con libre entrada: empresa e industria}
\subsub{c) Extensiones: la oferta de la industria a LP}
\begin{itemize}
\item \emph{Supuestos}: \autor{Marshall}: \emph{economías internas} (tamaño de la empresa) frente a \concepto{economías externas} (de la industria), $C(q_j;q)$.
  \begin{itemize}
  \item \emph{Pecuniarias} (precios de factores) o \emph{tecnológicas} (distritos industriales) (\autor{Viner}, 1931; \autor{Scitovsky}, 1954).
  \end{itemize}
\item \emph{Desarrollo}:
  \begin{itemize}
  \item \concepto{costes constantes}: oferta horizontal;
  \item \concepto{crecientes} (deseconomías; suelo, trabajo especializado): creciente, más elástica que a CP;
  \item \concepto{decrecientes} (economías): decreciente; compatible con competencia: cada empresa con CMg creciente, baja toda su curva.
  \end{itemize}
\end{itemize}
\graficorepaso[0.78\linewidth]{oferta-largo-plazo}{Oferta de la industria a LP: costes constantes, crecientes y decrecientes}
\begin{itemize}
\item \emph{Implicaciones}: externas \emph{tecnológicas} $\Rightarrow$ ineficiencia (se produce poco \vertema{3.A.23}); \emph{pecuniarias}, solo trasladan rentas.
  \begin{itemize}
  \item Oferta decreciente: \autor{Walras} y \autor{Marshall} opuestos (\hyperref[rep:II.1]{II.1}). Concentración geográfica y comercio \vertema{3.B.6}.
  \end{itemize}
\end{itemize}
\subsub{d) Evidencia y e) valoración}
\begin{itemize}
\item CMe a LP en \textbf{L} (\autor{Johnston}, 1960; \autor{Stigler}, 1958): tamaño indeterminado sobre la escala mínima eficiente $\Rightarrow$ conviven tamaños distintos \vertema{3.A.12}.
\item Supone entrada sin costes y entrantes idénticos.
  \begin{itemize}
  \item Hundidos $\Rightarrow$ entrada lenta, barreras; mercados impugnables \vertema{3.A.15};
  \item $\overline{q}$ enorme $\Rightarrow$ monopolio natural \vertema{3.A.17}\vertema{3.A.20}.
  \end{itemize}
\end{itemize}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}:
  \begin{itemize}
  \item demanda = suma de utilidades marginales; oferta = suma de CMg; precio que las iguala (no lo fija nadie);
  \item existe y es único; estable por precios o cantidades; telaraña puede oscilar; maximiza el ET (1TFEB); impuesto según elasticidades;
  \item a LP con libre entrada, $p=\min\mathrm{CMe}$ y beneficio nulo.
  \end{itemize}
\item \textbf{Relevancia}: vara de medir (\autor{Lerner} nulo \vertema{3.A.17}), competencia \vertema{4.A.15}, impuestos \vertema{4.B.9}.
\item \textbf{Extensiones}: equilibrio general \vertema{3.A.21}; bienestar y fallos \vertema{3.A.22}\vertema{3.A.23}; levantar 4~o 3 $\to$ \vertema{3.A.17}--\vertema{3.A.19}; levantar 2 $\to$ \vertema{3.A.13}; impugnables \vertema{3.A.15}.
\item \textbf{Opinión}: «no es de la benevolencia del carnicero\ldots{}» (\autor{Smith}, 1776): describe pocos mercados y no lo pretende; aísla cómo los precios coordinan el interés propio.
  \begin{itemize}
  \item Débil en lo dinámico: estado de reposo, no proceso (\autor{Hayek}, 1948 \vertema{3.A.3}); sin beneficios duraderos, poco incentivo a innovar (\autor{Schumpeter}, 1942).
  \end{itemize}
\item \textbf{Idea final}: a río revuelto hay ganancia de pescadores, pero solo mientras la oferta no se ajusta: con entrada libre, el precio vuelve al coste; el mercado llega solo al equilibrio, y eficiente, pero no siempre ni por cualquier camino.
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Por qué utilidad cuasilineal? ¿Qué se pierde?} Sin efecto renta: demanda solo del precio, EC exacto, parcial = general. Se pierde el efecto renta.
\item \textbf{¿Por qué la oferta no es toda la curva de CMg?} Solo el tramo creciente sobre el mín.\ CVMe (CP) o CMe (LP): por debajo, cerrar es mejor.
\item \textbf{¿Economías de escala y competencia perfecta?} Con CMe en U, sí (se produce en $\overline{q}$); crecientes internas en todo el dominio, no; externas, sí.
\item \textbf{¿Crítica de \autor{Muth}?} Errores sistemáticos: con expectativas racionales, $p^e=p^*$: sin ciclo.
\item \textbf{¿Libre entrada y rendimientos decrecientes?} Sin equilibrio: el mín.\ del CMe está en $q\to0$.
\item \textbf{¿Por qué pueden discrepar \autor{Walras} y \autor{Marshall}?} Con ambas curvas decrecientes: uno ajusta precios y el otro cantidades.
\item \textbf{¿Quién paga si la demanda es perf.\ inelástica?} Los consumidores; si lo es la oferta, los vendedores.
\item \textbf{¿Economías externas pecuniarias o tecnológicas?} Solo las tecnológicas causan ineficiencia; las pecuniarias trasladan rentas.
\end{enumerate}
\begin{trampa}
\begin{itemize}
\item Mín.\ CVMe = \emph{mínimo} de explotación; mín.\ CMe = \emph{óptimo}.
\item Ambas $\downarrow$ y oferta más plana: \autor{Walras} inestable, \autor{Marshall} estable.
\item Telaraña converge si $\varepsilon_S<\lvert\varepsilon_D\rvert$; con curvas normales puede divergir.
\item \autor{Dalton}: más sobre el lado \emph{más inelástico}. EG crece con $t^2$.
\item Libre entrada: S invertida, suficiente, no necesaria; costes crecientes = \emph{deseconomías externas}.
\item 1TFEB sin convexidad; 2TFEB (en general) con ella. \autor{Dupuit} 1844, no 1884.
\end{itemize}
\end{trampa}
