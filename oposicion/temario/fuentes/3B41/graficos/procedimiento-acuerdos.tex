% Cómo se negocia y se celebra un acuerdo comercial de la UE (arts. 207 y
% 218 TFUE): recomendación de la Comisión, autorización y directrices del
% Consejo, negociación con el Comité de Política Comercial e información al
% Parlamento, firma (Consejo), aprobación del Parlamento, celebración
% (Consejo) y, si el acuerdo es mixto, ratificación de los Estados.
% Al margen, las mayorías en el Consejo y la aplicación provisional.
% Fuente: arts. 207 y 218 TFUE; Consejo de la UE.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\newcommand{\nocortar}{\hyphenpenalty=10000 \exhyphenpenalty=10000 }
\newcommand{\etapa}[2]{{\bfseries\color{tceemorado}#1}\hfill{\bfseries\color{tceedorado!70!black}#2}\par\vspace{1pt}}
\begin{tikzpicture}[tcee,every node/.append style={execute at begin node=\nocortar},
    paso/.style={draw=tceemorado,thick,fill=tceemorado!8,rounded corners=2pt,
      align=left,text width=7.2cm,inner sep=4pt,font=\footnotesize,anchor=north},
    mixto/.style={draw=tceedorado,thick,fill=tceedorado!10,rounded corners=2pt,
      align=left,text width=7.2cm,inner sep=4pt,font=\footnotesize,anchor=north},
    nota/.style={draw=tceegris,thick,dashed,fill=tceegris!6,rounded corners=2pt,
      align=left,text width=3.6cm,inner sep=4pt,font=\footnotesize,anchor=north west},
    sigue/.style={->,thick,tceegris}]
  \capa{1}{\node[paso] (p1) at (0,0) {\etapa{1. Recomendación}{Comisión}
      propone abrir negociaciones con un proyecto de directrices};}
  \capa{2}{\node[paso] (p2) at ($(p1.south)+(0,-0.45)$) {\etapa{2. Autorización y mandato}{Consejo}
      autoriza la apertura y aprueba las directrices de negociación};}
  \capa{2}{\draw[sigue] (p1) -- (p2);}
  \capa{3}{\node[paso] (p3) at ($(p2.south)+(0,-0.45)$) {\etapa{3. Negociación}{Comisión}
      negocia en consulta con el Comité de Política Comercial e informa por igual al Parlamento Europeo; al terminar, rubrica el texto};}
  \capa{3}{\draw[sigue] (p2) -- (p3);}
  \capa{4}{\node[paso] (p4) at ($(p3.south)+(0,-0.45)$) {\etapa{4. Firma}{Consejo}
      decide la firma y, si procede, la aplicación provisional};}
  \capa{4}{\draw[sigue] (p3) -- (p4);}
  \capa{5}{\node[paso] (p5) at ($(p4.south)+(0,-0.45)$) {\etapa{5. Aprobación}{Parlamento Europeo}
      lo aprueba o lo rechaza en bloque, sin enmiendas; puede pedir antes un dictamen al Tribunal de Justicia (art. 218.11)};}
  \capa{5}{\draw[sigue] (p4) -- (p5);}
  \capa{6}{\node[paso] (p6) at ($(p5.south)+(0,-0.45)$) {\etapa{6. Celebración}{Consejo}
      adopta la decisión de celebración: el acuerdo de competencia exclusiva («solo UE») entra en vigor};}
  \capa{6}{\draw[sigue] (p5) -- (p6);}
  \capa{7}{\node[mixto] (p7) at ($(p6.south)+(0,-0.45)$) {\etapa{7. Solo si es mixto}{27 Estados miembros}
      ratificación de cada Estado según su Constitución (en algunos, también parlamentos regionales); mientras tanto, se aplica de forma provisional la parte de competencia de la UE};}
  \capa{7}{\draw[sigue] (p6) -- (p7);}
  \capa{8}{\node[nota] (n1) at ($(p2.north east)+(0.4,0)$) {\textbf{Mayorías en el Consejo}: cualificada, salvo unanimidad en servicios, propiedad intelectual e inversión directa cuando la exijan las normas internas, y en servicios culturales, sociales, educativos y sanitarios (art. 207.4)};}
  \capa{8}{\node[nota] (n2) at ($(p5.north east)+(0.4,0)$) {\textbf{Desde 2018}: el acuerdo comercial va «solo UE» y la protección de inversiones, en un acuerdo aparte, mixto};}
\end{tikzpicture}
\end{document}
