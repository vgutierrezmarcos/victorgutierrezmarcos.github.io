% Composición del Parlamento Europeo por grupos políticos al constituirse la
% legislatura 2024-2029 (16/07/2024; 720 diputados), en un hemiciclo de
% izquierda a derecha. Datos en ../datos/grupos-pe.csv (Parlamento Europeo).
% Cada sector es proporcional a los escaños (180° = 720). Capas: un grupo por
% capa, de izquierda a derecha → línea de la mayoría absoluta (361) → mayoría
% de la Comisión (PPE + S&D + Renew).
% Autor: Víctor Gutiérrez Marcos

\def\Rext{5.2}\def\Rint{2.5}
% \sector{capa}{inicio (escaños acumulados)}{escaños}{color}{sigla}{escaños (texto)}
\newcommand{\sector}[6]{%
  \capa{#1}{%
    \pgfmathsetmacro{\aI}{180-#2/4}\pgfmathsetmacro{\aF}{180-(#2+#3)/4}%
    \pgfmathsetmacro{\aM}{(\aI+\aF)/2}%
    \fill[#4,draw=white,line width=0.8pt] (\aI:\Rext) arc[start angle=\aI,end angle=\aF,radius=\Rext]
      -- (\aF:\Rint) arc[start angle=\aF,end angle=\aI,radius=\Rint] -- cycle;
    \node[font=\footnotesize,align=center,anchor=\aM+180] at (\aM:\Rext+0.12) {\textbf{#5}\\#6};}}

  }
\end{tikzpicture}
