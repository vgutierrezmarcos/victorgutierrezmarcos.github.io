% La Europa de geometría variable: Consejo de Europa (46 Estados), Espacio
% Económico Europeo (UE + Islandia, Liechtenstein y Noruega), AELC/EFTA
% (Islandia, Liechtenstein, Noruega y Suiza), Unión Europea (27), zona del euro
% (21 desde el 1/1/2026, con Bulgaria), espacio Schengen (29: la UE salvo
% Irlanda y Chipre, más los cuatro de la AELC), unión aduanera con la UE
% (Turquía, Andorra y San Marino) y países que usan el euro con acuerdo
% monetario (Andorra, Mónaco, San Marino, Vaticano) o sin él (Montenegro,
% Kosovo). Vaticano y Kosovo no son miembros del Consejo de Europa.
% Cada país, con su bandera (dibujada en banderas.tikz) y su nombre corto,
% como en el diagrama de Euler del tema antiguo (Víctor, 16/11/2026). Datos a
% 16/11/2026. Sustituye al diagrama de Euler de Wikipedia del tema antiguo.
% Autor: Víctor Gutiérrez Marcos

\begin{document}
% \zona[estilo del borde]{color}{esquina}{esquina}: fondo translúcido en la
% capa de fondo (no tapa las banderas) y borde en la principal
\newcommand{\zona}[4][]{%
  \begin{pgfonlayer}{background}\fill[rounded corners=6pt,#2,fill opacity=0.07,#1] #3 rectangle #4;\end{pgfonlayer}
  \draw[thick,rounded corners=6pt,#2,#1] #3 rectangle #4;}
\newcommand{\nocortar}{\hyphenpenalty=10000 \exhyphenpenalty=10000 }
% \pais{x}{y}{código}{nombre}: bandera y, debajo, el nombre
\newcommand{\pais}[4]{\bandera{(#1,#2)}{#3}\node[nombrepais,anchor=base] at (#1,#2-0.42) {#4};}
\begin{tikzpicture}[tcee,every node/.append style={execute at begin node=\nocortar},
    zona/.style={thick,rounded corners=6pt},

    nota/.style={font=\footnotesize\itshape,text=tceegris,inner sep=2pt},
