% Poder de voto a priori de cada Estado en el Consejo de la UE con la doble
% mayoría (55 % de los Estados y 65 % de la población, minoría de bloqueo de
% al menos 4), frente a su cuota de población. Población de 2026 (Decisión
% 2025/2427 del Consejo). Índices de Shapley-Shubik y de Banzhaf normalizado:
% cálculo propio (../datos/poder-voto.py → ../datos/poder-voto.csv).
% Escalas logarítmicas para que se vean los 27. Capas: ejes → bisectriz
% (poder = población) → «un Estado, un voto» → Shapley-Shubik → Banzhaf → rótulos.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/poder-voto.csv}\poder
\pgfplotsset{ejespv/.style={}}
\ifnum1>\tceepaso\relax
  \pgfplotsset{ejespv/.style={axis line style={draw=none},
    major tick style={draw=none}, minor tick style={draw=none},
    ticklabel style={opacity=0}, label style={opacity=0}, xmajorgrids=false, ymajorgrids=false}}
\fi
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=11.5cm, height=9.2cm,
      xmode=log, ymode=log,
      xmin=0.1, xmax=25, ymin=0.45, ymax=25,
      xtick={0.1,0.2,0.5,1,2,5,10,20}, xticklabels={{0,1},{0,2},{0,5},1,2,5,10,20},
      ytick={1,2,5,10,20}, yticklabels={1,2,5,10,20},
      xticklabel style={font=\footnotesize}, yticklabel style={font=\footnotesize},
      xlabel={Cuota de la población de la UE (\%, escala logarítmica)},
      ylabel={Poder de voto a priori (\%, escala logarítmica)},
      label style={font=\footnotesize},
      axis x line*=bottom, axis y line*=left, tick align=outside,
      xmajorgrids, ymajorgrids, grid style={tceegris!20},
      clip=false, ejespv, dicapas,
    ]
    % 2: bisectriz: poder proporcional a la población
    \capa{2}{
      \dicraya[dicumbral]{(0.45,0.45) (25,25)}
      \dicrotulo[font=\footnotesize,text=tceegris,anchor=north,rotate=42,yshift=-2pt]{(1.6,1.6)}{poder = población}
    }
    % 3: un Estado, un voto (1/27 = 3,70 %)
    \capa{3}{
      \dicraya[thin,dotted,tceegris]{(0.1,3.70) (25,3.70)}
      \dicrotulo[font=\footnotesize,text=tceegris,anchor=south west]{(0.1,3.70)}{un Estado, un voto (3,7~\%)}
    }
    % 4: Shapley-Shubik
    \capa{4}{
      \addplot[on layer=dicpuntos, only marks, mark=*, mark size=2.2pt, color=tceemorado]
        table[x=poblacion, y=ss] {\poder};
    }
    % 5: Banzhaf normalizado
    \capa{5}{
      \addplot[on layer=dicpuntos, only marks, mark=diamond*, mark size=2.6pt, color=tceedorado]
        table[x=poblacion, y=bznorm] {\poder};
    }
    % 6: rótulos
    \capa{6}{
      \dicrotulo[font=\footnotesize,anchor=east,xshift=-3pt]{(18.47,17.54)}{Alemania}
      \dicrotulo[font=\footnotesize,anchor=east,xshift=-3pt]{(15.18,13.48)}{Francia}
      \dicrotulo[font=\footnotesize,anchor=east,xshift=-3pt]{(13.19,11.53)}{Italia}
      \dicrotulo[font=\footnotesize,anchor=east,xshift=-3pt,text=tceemorado]{(10.86,9.39)}{\textbf{España}}
      \dicrotulo[font=\footnotesize,anchor=east,xshift=-3pt]{(8.29,6.69)}{Polonia}
      \dicrotulo[font=\footnotesize,anchor=south,yshift=3pt]{(0.13,1.73)}{Malta}
      \dicrotulo[font=\footnotesize,anchor=north west,xshift=-2pt,yshift=-6pt,align=left]{(0.13,0.87)}{Malta y Luxemburgo: un poder\\6-7 veces su peso en población}
      \dicrotulo[font=\footnotesize,anchor=south east,align=right,text=tceemorado]{(24,1.2)}{$\bullet$ Shapley-Shubik}
      \dicrotulo[font=\footnotesize,anchor=south east,align=right,text=tceedorado]{(24,0.85)}{$\blacklozenge$ Banzhaf normalizado}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
