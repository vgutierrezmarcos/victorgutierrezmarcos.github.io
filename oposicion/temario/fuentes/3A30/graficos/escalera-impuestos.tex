% Colores: salvia para las rentas de los factores, salmón para los impuestos
% netos (como en valoracion-precios).
% Capas: barra 1 (coste de los factores) -> barra 2 (+ D.29 - D.39, precios

\begin{tikzpicture}[tcee, font=\footnotesize,

  fac/.style={tramo, fill=tceesalvia},
  cifra/.style={font=\scriptsize\bfseries},
  nota/.style={font=\scriptsize, align=left, anchor=west},
