% Cuota en las exportaciones mundiales de mercancías (%) en 1990, 2000 y 2025
% por regiones del Banco Mundial, y China y EE. UU. Banco Mundial, WDI
% (TX.VAL.MRCH.CD.WT, a partir de la OMC); ../datos/exportaciones-regiones.csv
% (consulta 14/11/2026). Asia oriental 22,1 -> 35,0; Europa y Asia central
% 51,0 -> 37,8; América del Norte 15,1 -> 10,4; China 1,8 -> 14,3.
% Barras horizontales: 1990 gris claro, 2000 gris, 2025 azul.
% Capas: barras 1990 -> 2000 -> 2025 -> ejes -> valores de 2025.
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/exportaciones-regiones.csv}\er
\begin{tikzpicture}[tcee]
  \begin{axis}[width=11cm, height=9cm, xbar, bar width=4pt, area legend,
      xmin=0, xmax=58, ymin=-0.6, ymax=8.6, y dir=reverse,
      ytick={0,...,8}, yticklabels from table={\er}{region},
      yticklabel style={font=\footnotesize}, major y tick style={draw=none},
      xticklabel style={font=\footnotesize}, xlabel={\% de las exportaciones mundiales}, xlabel style={font=\footnotesize},
      axis x line*=bottom, axis y line*=left, xmajorgrids, grid style={tceegris!25}, clip=false, dicapas,
      legend style={at={(0.5,-0.12)}, anchor=north, draw=none, font=\footnotesize, legend columns=3,
        /tikz/every even column/.append style={column sep=10pt}}, legend cell align=left]
    \capa{1}{\addplot[on layer=dicbarras, dicbarra=dicotromasclaro, bar shift=4.5pt] table[x=c1990, y expr=\coordindex] {\er}; \addlegendentry{1990}}
    \capa{2}{\addplot[on layer=dicbarras, dicbarra=dicotroclaro, bar shift=0pt] table[x=c2000, y expr=\coordindex] {\er}; \addlegendentry{2000}}
    \capa{3}{\addplot[on layer=dicbarras, dicbarra=dicexterna, bar shift=-4.5pt] table[x=c2025, y expr=\coordindex] {\er}; \addlegendentry{2025}}
    \capa{4}{\dicraya[dicumbral]{(0,6.5) (58,6.5)}}
    \capa{5}{\addplot[on layer=axis descriptions, dicsuelto, draw=none,
        nodes near coords={\pgfmathprintnumber[use comma, fixed, precision=1, fixed zerofill]{\pgfplotspointmeta}}, point meta=explicit,
        every node near coord/.append style={font=\scriptsize, anchor=west, text=dicexterna!60!black, inner sep=1pt}]
        table[x expr={max(\thisrow{c2025},\thisrow{c2000},\thisrow{c1990})+0.5}, y expr=\coordindex, meta=c2025] {\er};}
  \end{axis}
\end{tikzpicture}
