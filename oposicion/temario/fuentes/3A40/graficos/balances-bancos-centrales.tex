% Activo total de los 4 grandes bancos centrales en % del PIB, fin de
% trimestre, 2007T1-2026T2 (3.A.40, I.2). Datos:
% ../datos/balance-bancos-centrales-pib.csv (BPI, WS_CBTA; consulta 5/11/2026).
% Trimestral, sin huecos: x = 2007 + (fila + 1)/4 (fin del trimestre); el
% Eurosistema llega a 2026T1 (la última celda está vacía).
% Cifras de control: máximos Fed 36,9 (2021T4), Eurosistema 67,6 (2021T4),
%   BoE 48,2 (2021T4), BoJ 128,9 (2021T1); último dato Fed 21,2, Eurosistema
%   39,0 (2026T1), BoE 25,8, BoJ 94,4 (2026T2).
% Paleta de las 4 economías del tema (como tipo-natural y tipos-oficiales):
% EE. UU. tceedorado, zona del euro dicze, Reino Unido tceeverde, Japón tceerojo.
% Capas (CRITERIOS §7): ejes → las 4 series → puntos → etiquetas.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\pgfplotstableread[col sep=comma,comment chars={\#},empty cells with={nan}]{../datos/balance-bancos-centrales-pib.csv}\balances
\newcommand{\balance}[2]{\addplot[diclinea=#1] table[x expr={2007+(\coordindex+1)/4},y=#2] {\balances};}
\begin{document}
\begin{tikzpicture}[tcee]
\begin{axis}[width=12.5cm,height=6.8cm,
    xmin=2006.8,xmax=2027.2,ymin=0,ymax=140,
    axis x line*=bottom,axis y line*=left,tick align=outside,
    xtick={2007,2010,...,2025},ytick={0,20,...,140},
    x tick label style={/pgf/number format/.cd,set thousands separator={}},
    ticklabel style={font=\footnotesize},
    /pgf/number format/use comma,
    ylabel={\% del PIB},ylabel style={font=\footnotesize},
    ymajorgrids,grid style={tceegris!25},clip=false,
    legend style={at={(0.5,-0.11)},anchor=north,draw=none,font=\footnotesize,legend columns=4,
      /tikz/every even column/.append style={column sep=0.5cm}},
    legend cell align=left,unbounded coords=jump,dicapas]
  % 1: ejes (los dibuja el axis)
  % 2: las 4 series
  \capa{2}{
    \balance{tceedorado}{fed}\addlegendentry{Reserva Federal}
    \balance{dicze}{eurosistema}\addlegendentry{Eurosistema}
    \balance{tceeverde}{boe}\addlegendentry{Banco de Inglaterra}
    \balance{tceerojo}{boj}\addlegendentry{Banco de Japón}
  }
  % 3: puntos del máximo y del último dato
  \capa{3}{
    \addplot[dicpuntos=tceedorado,forget plot] coordinates {(2022,36.9) (2026.5,21.2)};
    \addplot[dicpuntos=dicze,forget plot] coordinates {(2022,67.6) (2026.25,39.0)};
    \addplot[dicpuntos=tceeverde,forget plot] coordinates {(2022,48.2) (2026.5,25.8)};
    \addplot[dicpuntos=tceerojo,forget plot] coordinates {(2021.25,128.9) (2026.5,94.4)};
  }
  % 4: etiquetas (máximo, con el trimestre, y último dato)
  \capa{4}{
    \dicrotulo[font=\footnotesize,text=tceerojo,anchor=east]{(2020.9,130)}{128,9 (2021T1)}
    \dicrotulo[font=\footnotesize,text=dicze,anchor=south]{(2022,69.5)}{67,6}
    \dicrotulo[font=\footnotesize,text=tceeverde,anchor=south]{(2022,50)}{48,2}
    \dicrotulo[font=\footnotesize,text=tceedorado,anchor=north]{(2022,34.5)}{36,9}
    \dicrotulo[font=\footnotesize,text=tceerojo,anchor=west]{(2026.8,94.4)}{94,4}
    \dicrotulo[font=\footnotesize,text=dicze,anchor=west]{(2026.8,39.0)}{39,0}
    \dicrotulo[font=\footnotesize,text=tceeverde,anchor=west]{(2026.8,27.6)}{25,8}
    \dicrotulo[font=\footnotesize,text=tceedorado,anchor=west]{(2026.8,19.6)}{21,2}
  }
\end{axis}
\end{tikzpicture}
\end{document}
