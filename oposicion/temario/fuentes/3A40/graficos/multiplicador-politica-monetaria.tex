% El multiplicador del gasto en el modelo neokeynesiano según la política
% monetaria (3.A.40, III.2 y anexo A). beta = 0,99, theta = varphi = 1
% (Gamma = 0,5), kappa = 0,02, phi_pi = 1,5. Eje horizontal: persistencia
% (rho_G con la regla de Taylor; rho_L en el límite inferior).
%   límite: m = [(1-x)(1-0,99x) - 0,01x] / [(1-x)(1-0,99x) - 0,02x]
%           x = 0: 1; 0,5: 1,021; 0,8: 1,313; 0,85: 2,255; 0,86: 3,386
%           (el denominador se anula en x = 0,8722)
%   Taylor: W = 0,02(1,5-x)/[(1-x)(1-0,99x)]; m = (1 + 0,5W)/(1 + W)
%           x = 0: 0,985; 0,5: 0,963; 0,8: 0,874; 0,85: 0,823; 0,86: 0,810
% Autor: Víctor Gutiérrez Marcos
