% Tipos de interés oficiales de la Reserva Federal (punto medio del intervalo
% objetivo), el BCE (facilidad de depósito), el Banco de Inglaterra (Bank Rate)
% y el Banco de Japón, último día de cada mes, ene. 2007-oct. 2026, en escalón
% (3.A.40, I.2). Datos: ../datos/tipos-oficiales-4-bancos-centrales.csv (Fed de
% Nueva York, BCE, Banco de Inglaterra y BPI; consulta 5/11/2026). Mensual, sin
% huecos: cada fila es el tipo del mes (fila), que acaba en x = 2007 + (fila + 1)/12.
% Último dato (oct. 2026): Fed 3,875 (intervalo 3,75-4,00), BCE 2,50, BoE 3,75,
%   BoJ 1,25. El Banco de Inglaterra decidía el 5/11/2026: si cambió el tipo,
%   añadir la fila de nov. 2026 al CSV y cambiar la etiqueta.
% Banco de Japón: sin dato de may. 2013 a ene. 2016 (objetivo de base monetaria;
%   ver la cabecera del CSV, filas 76-108): la línea se corta (2 tramos).

% Paleta de las 4 economías del tema (como tipo-natural y
% balances-bancos-centrales): EE. UU. tceedorado, zona del euro dicze, Reino
% Unido tceeverde, Japón tceerojo. Método de 1D01/graficos/tipos-balance-bce.tex.
% Capas (CRITERIOS §7): ejes y 0 → Fed → BCE → BoE → BoJ → etiquetas.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\pgfplotstableread[col sep=comma,comment chars={\#},empty cells with={nan}]{../datos/tipos-oficiales-4-bancos-centrales.csv}\tiposdatos
\newcommand{\tipo}[3][]{\addplot[const plot mark right,#2,#1] table[x expr={2007+(\coordindex+1)/12},y=#3] {\tiposdatos};}
\begin{document}
\begin{tikzpicture}[tcee]
\begin{axis}[width=12.5cm,height=6.6cm,
    xmin=2006.5,xmax=2027.2,ymin=-0.75,ymax=6,
    axis x line*=bottom,axis y line*=left,tick align=outside,
    xtick={2007,2010,...,2025},ytick={0,1,...,6},
    x tick label style={/pgf/number format/.cd,set thousands separator={}},
    ticklabel style={font=\footnotesize},
    /pgf/number format/use comma,
    ylabel={\% anual},ylabel style={font=\footnotesize},
    ymajorgrids,grid style={tceegris!25},clip=false,
    legend style={at={(0.5,-0.11)},anchor=north,draw=none,font=\footnotesize,legend columns=2,
      /tikz/every even column/.append style={column sep=0.6cm}},
    legend cell align=left,unbounded coords=jump,dicapas]
  % 1: ejes y línea del 0
  \capa{1}{\dicraya{(2006.5,0) (2027.2,0)}}
  % 2: Reserva Federal

    \addlegendentry{Reserva Federal (punto medio del intervalo)}}
  % 3: BCE
  \capa{3}{\tipo{diclinea=dicze}{bce_facilidad_deposito}
    \addlegendentry{BCE (facilidad de depósito)}}
  % 4: Banco de Inglaterra
  \capa{4}{\tipo{diclinea=tceeverde}{boe_bank_rate}
    \addlegendentry{Banco de Inglaterra (Bank Rate)}}
  % 5: Banco de Japón
  \capa{5}{\tipo[skip coords between index={76}{238}]{diclinea=tceerojo}{boj}
    \tipo[skip coords between index={0}{109},forget plot]{diclinea=tceerojo}{boj}
    \addlegendentry{Banco de Japón}}
  % 6: etiquetas del último dato (oct. 2026)
  \capa{6}{
    \dicrotulo[font=\footnotesize,text=tceedorado,anchor=west]{(2026.9,4.05)}{3,875}
    \dicrotulo[font=\footnotesize,text=tceeverde,anchor=west]{(2026.9,3.6)}{3,75}

  }
\end{axis}

\end{document}
