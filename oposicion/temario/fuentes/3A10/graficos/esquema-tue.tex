% De las loterías a la decisión óptima con riesgo: etapas de la teoría de la
% utilidad esperada, con el mismo diseño que el esquema de ingredientes del 3.A.8
% (conjunto de elección, preferencias, axiomas, función de utilidad, problema de
% maximización y decisión óptima).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,node distance=0.7cm and 0.7cm,
    caja/.style={draw=tceemorado,thick,fill=tceesalvia,rounded corners=3pt,align=center,
                 minimum height=1.3cm,inner sep=5pt},
    clave/.style={caja,fill=tceemorado!12,very thick},
    une/.style={->,thick,tceemorado}]
  % 1: conjunto de elección (loterías) y preferencias sobre él
  \capa{1}{
    \node[caja] (S) {Conjunto de elección:\\ loterías $\tilde{x}\in\mathcal{L}$};
    \node[caja,right=of S] (P) {Preferencias\\ $\succcurlyeq$ sobre $\mathcal{L}$};
    \draw[une] (S) -- (P);
  }
  % 2: axiomas y función de utilidad esperada (los nodos existen siempre, invisibles
  % antes del paso 2, porque el problema de maximización se sitúa respecto de U)
  \ifnum\tceepaso<2 \tikzset{c2/.style={opacity=0}}\else\tikzset{c2/.style={}}\fi
  \node[caja,right=of P,c2] (A) {Axiomas 1-5\\ (\textsc{von Neumann}-\textsc{Morgenstern})};
  \node[caja,right=of A,c2] (U) {Utilidad esperada\\ $U(\tilde{x})=\sum_s \pi_s\,u(x_s)$};
  \capa{2}{
    \draw[une] (P) -- (A);
    \draw[une] (A) -- (U);
  }
  % posición del problema de maximización (coordenada sin dibujo)
  \path (U.south) ++(0,-1.9) coordinate (cMAX);
  % 3: loterías a su alcance (lo que puede elegir: cobertura, cartera, producción)
  \capa{3}{
    \node[caja,anchor=east] (B) at ($(cMAX)+(-2.2,0)$) {Loterías alcanzables\\ $\tilde{x}\in\mathcal{L}_B\subseteq\mathcal{L}$\\ (seguro, cartera, producción)};
  }
  % 4: maximización de la utilidad esperada
  \capa{4}{
    \node[clave] (MAX) at (cMAX) {Maximización\\ $\max\limits_{\tilde{x}}\ \mathbb{E}\big[u(\tilde{x})\big]$\\ s.a.\ $\tilde{x}\in\mathcal{L}_B$};
    \draw[une] (U) -- (MAX);
    \draw[une] (B) -- (MAX);
  }
  % 5: decisión óptima con riesgo: la lotería alcanzable de mayor utilidad esperada
  \capa{5}{
    \node[caja,right=0.9cm of MAX] (X) {Decisión óptima\\ $\tilde{x}^*\in\mathcal{L}_B$};
    \draw[une] (MAX) -- (X);
  }
  % 6: aplicaciones: la decisión óptima en cada problema concreto
  \capa{6}{
    \node[caja,below=0.9cm of X] (M) {Aplicaciones: demanda de seguros $S^*$,\\ de activos con riesgo $\alpha^*$ y producción $q^*$};
    \draw[une] (X) -- (M);
  }
\end{tikzpicture}
\end{document}
