% Utilidad de Friedman y Savage (izquierda) y de Markowitz (derecha).
% Izquierda: U(W) = 1,5 (1 - e^{-1,5 W}) + 2/(1 + e^{-3(W - 3,5)}), W en [0; 6],
%   dibujada con escala vertical 0,9. U'' cambia de signo en W = 1,97 y W = 3,50
%   (puntos de inflexión, calculados numéricamente): cóncava, convexa, cóncava.
%   U(1,97) = 1,44 y U(3,50) = 2,49.
% Derecha: z = cambio respecto a la riqueza habitual (origen).
%   s(t) = 1/(1 + e^{-t}), k = 1,6.
%   Ganancias: U(z) = 2 [s(k(z - 1,5)) - s(-1,5k)]: convexa hasta z = 1,5 y cóncava después.
%   Pérdidas: U(z) = -3,03 [s(k(-z - 1,8)) - s(-1,8k)]: cóncava entre -1,8 y 0 y
%   convexa antes de -1,8. La amplitud 3,03 iguala la pendiente a ambos lados del
%   origen (sin quiebro): el origen es el tercer punto de inflexión.
%   Con z = 3,5: U = 1,76; con z = -3,5: U = -2,68 (las pérdidas pesan más).
% Elaboración propia a partir de Friedman y Savage (1948) y Markowitz (1952).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,
    declare function={
      fs(\x)=0.9*(1.5*(1-exp(-1.5*\x))+2*exp(3*(\x-3.5))/(1+exp(3*(\x-3.5))));
      sg(\t)=1/(1+exp(-\t));
      mg(\z)=2*(sg(1.6*(\z-1.5))-sg(-2.4));
      mp(\z)=-3.0304*(sg(1.6*(-\z-1.8))-sg(-2.88));}]
  % ---------- Panel izquierdo: Friedman y Savage ----------
  % 1: ejes
  \capa{1}{\ejes{$W$}{$U(W)$}{6.6}{3.6}}
  % 2: función cóncava-convexa-cóncava
  \capa{2}{\draw[curva1] plot[domain=0:6,samples=120] (\x,{fs(\x)});}
  % 3: puntos de inflexión
  \capa{3}{\node[punto] (I1) at (1.97,{0.9*1.44205}) {};}
  \capa{3}{\node[punto] (I2) at (3.497,{0.9*2.48815}) {};}
  \capa{3}{\draw[guia] (I1) -- (1.97,0);}
  \capa{3}{\draw[guia] (I2) -- (3.497,0);}
  % 4: tramos
  \capa{4}{\node[etiqueta,font=\footnotesize,align=center] at (1,0.35) {aversión};}
  \capa{4}{\node[etiqueta,font=\footnotesize,align=center] at (2.73,0.4) {amor por\\el riesgo};}
  \capa{4}{\node[etiqueta,font=\footnotesize,align=center] at (4.9,0.35) {aversión};}
  % ---------- Panel derecho: Markowitz ----------
  \begin{scope}[xshift=11.3cm,yshift=1.6cm]
  % 5: ejes centrados en la riqueza habitual
  \capa{5}{\draw[eje] (-3.9,0) -- (3.9,0) node[below left=2pt and 0pt] {$\Delta W$};}
  \capa{5}{\draw[eje] (0,-2.9) -- (0,2.2) node[left] {$U$};}
  \capa{5}{\node[etiqueta,font=\footnotesize,align=center] (rh) at (1.6,-1.2) {riqueza\\habitual};}
  \capa{5}{\draw[->,thin] (rh.north west) -- (0.1,-0.1);}
  % 6: tramo de ganancias: convexo y después cóncavo
  \capa{6}{\draw[curva1] plot[domain=0:3.6,samples=80] (\x,{mg(\x)});}
  % 7: tramo de pérdidas: cóncavo y después convexo, más inclinado
  \capa{7}{\draw[curva1] plot[domain=-3.6:0,samples=80] (\x,{mp(\x)});}
  % 8: puntos de inflexión
  \capa{8}{\node[punto] at (0,0) {};}
  \capa{8}{\node[punto] (M1) at (1.5,{mg(1.5)}) {};}
  \capa{8}{\node[punto] (M2) at (-1.8,{mp(-1.8)}) {};}
  \capa{8}{\draw[guia] (M1) -- (1.5,0);}
  \capa{8}{\draw[guia] (M2) -- (-1.8,0);}
  % 9: tramos
  \capa{9}{\node[etiqueta,font=\scriptsize,align=center] at (0.8,0.75) {amor};}
  \capa{9}{\node[etiqueta,font=\scriptsize,align=center] at (2.75,0.75) {aversión};}
  \capa{9}{\node[etiqueta,font=\scriptsize,align=center] at (-0.9,0.3) {aversión};}
  \capa{9}{\node[etiqueta,font=\scriptsize,align=center] at (-2.7,0.3) {amor};}
  \end{scope}
\end{tikzpicture}
