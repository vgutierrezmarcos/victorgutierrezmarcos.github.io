% Dominancia estocástica de primer orden (izquierda) y de segundo orden (derecha).
% Funciones de distribución logísticas, L(x; m, k) = 1/(1 + exp(-k(x - m))),
% dibujadas con altura 3 (probabilidad 1 = 3 unidades).
% Izquierda: G = L(x; 2,2; 2) y F = L(x; 3,3; 2): F es G desplazada a la
%   derecha, F(x) <= G(x) para todo x; el área entre ambas es E_F - E_G = 1,1.
% Derecha: F = L(x; 2,7; 2,6) y G = L(x; 2,7; 1,3), con la misma media 2,7.
%   Se cortan en la media; G > F a la izquierda y G < F a la derecha, con áreas
%   iguales por simetría: G es una dispersión de F que conserva la media
%   (la integral de F - G hasta x es <= 0 para todo x).
% Elaboración propia a partir de Mas-Colell, Whinston y Green (1995, §6.D).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,
    declare function={L(\x,\m,\k)=3/(1+exp(-\k*(\x-\m)));}]
  % ---------- Panel izquierdo: primer orden ----------
  % 1: ejes y probabilidad 1
  \capa{1}{\ejes{$x$}{$F,\ G$}{5.8}{3.8}}
  \capa{1}{\draw[guia] (0,3) node[left,text=black] {$1$} -- (5.4,3);}
  \capa{1}{\node[etiqueta,font=\footnotesize] at (2.9,4.3) {Primer orden};}
  % 2: distribución G
  \capa{2}{\draw[curva2] plot[domain=0.15:5.3,samples=80] (\x,{L(\x,2.2,2)});}
  \capa{2}{\node[etiqueta,text=tceeazul] at (1.35,2.2) {$G$};}
  % 3: distribución F, siempre por debajo de G
  \capa{3}{\draw[curva1] plot[domain=0.15:5.3,samples=80] (\x,{L(\x,3.3,2)});}
  \capa{3}{\node[etiqueta,text=tceemorado] at (4.1,0.9) {$F$};}
  % 4: diferencia de medias: área entre G y F
  \capa{4}{\fill[area] plot[domain=0.15:5.3,samples=80] (\x,{L(\x,2.2,2)}) -- plot[domain=5.3:0.15,samples=80] (\x,{L(\x,3.3,2)}) -- cycle;}
  \capa{4}{\node[etiqueta,font=\footnotesize,align=center] at (1.3,3.45) {$F(x)\leq G(x)$};}
  % ---------- Panel derecho: segundo orden ----------
  \begin{scope}[xshift=7.3cm]
  % 5: ejes y probabilidad 1
  \capa{5}{\ejes{$x$}{$F,\ G$}{5.8}{3.8}}
  \capa{5}{\draw[guia] (0,3) node[left,text=black] {$1$} -- (5.4,3);}
  \capa{5}{\node[etiqueta,font=\footnotesize] at (2.9,4.3) {Segundo orden};}
  % 6: distribución F, concentrada en la media
  \capa{6}{\draw[curva1] plot[domain=0.15:5.3,samples=80] (\x,{L(\x,2.7,2.6)});}
  \capa{6}{\node[etiqueta,text=tceemorado] at (2.65,2.45) {$F$};}
  % 7: G, misma media y más dispersa
  \capa{7}{\draw[curva2] plot[domain=0.15:5.3,samples=80] (\x,{L(\x,2.7,1.3)});}
  \capa{7}{\node[etiqueta,text=tceeazul] at (4.6,2.3) {$G$};}
  \capa{7}{\draw[guia] (2.7,1.5) -- (2.7,0) node[below,text=black] {$\mu_F=\mu_G$};}
  % 8: áreas iguales a cada lado de la media
  \capa{8}{\fill[area2] plot[domain=0.15:2.7,samples=60] (\x,{L(\x,2.7,1.3)}) -- plot[domain=2.7:0.15,samples=60] (\x,{L(\x,2.7,2.6)}) -- cycle;}
  \capa{8}{\fill[area2] plot[domain=2.7:5.3,samples=60] (\x,{L(\x,2.7,2.6)}) -- plot[domain=5.3:2.7,samples=60] (\x,{L(\x,2.7,1.3)}) -- cycle;}
  \capa{8}{\node[etiqueta,font=\footnotesize] (a1) at (0.9,1.3) {$A_1$};}
  \capa{8}{\draw[thin] (a1.south east) -- (1.9,0.55);}
  \capa{8}{\node[etiqueta,font=\footnotesize] (a2) at (4.5,1.6) {$A_2$};}
  \capa{8}{\draw[thin] (a2.north west) -- (3.5,2.45);}
  \capa{8}{\node[etiqueta,font=\footnotesize] at (1.2,3.45) {$A_1=A_2$};}
  \end{scope}
\end{tikzpicture}
\end{document}
