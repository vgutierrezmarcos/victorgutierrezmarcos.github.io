% Subvención a la exportación en un país pequeño (3.B.7, II.2).
% Mismas curvas y escala que arancel-pequeno (x = 100 - p, q = p), con un
% precio mundial por encima de la autarquía (50), así que el país exporta.
% Cifras orientativas: p^W = 60; p^W(1+s) = 70 (s = 1/6).
%   Libre comercio: x^0 = 40, q^0 = 60, X^0 = 20.
%   Con la subvención: x^1 = 30, q^1 = 70, X^1 = 40.
%   a = 300; b = 50; c = 400; d = 50. Consumidores: -(a+b) = -350;
%   productores: a+b+c = 750; coste de la subvención: s p^W X^1 = 10 x 40 =
%   400 = b+c+d (rectángulo de x^1 a q^1); PIE: b+d = 100.
% Colores de arancel-pequeno: a, transferencia a los productores (dorado);
% c, lo que paga el Estado (verde, como la recaudación); b y d, PIE (rojo).
% Elaboración propia a partir de Feenstra y Taylor (2021, cap. 10).
% Autor: Víctor Gutiérrez Marcos

  % 3: precio mundial y exportaciones de libre comercio
  \capa{3}{\draw[guia] (0,60) node[etiqueta,left,text=black] {$p^W$} -- (102,60);}

  \capa{3}{\draw[llave] (40,-8) -- (60,-8) node[midway,below=4pt] {$X^0=q^0-x^0$};}
  % 4: precio con la subvención y exportaciones nuevas
  \capa{4}{\draw[curvanueva] (0,70) node[etiqueta,left,text=black] {$p^W(1+s)$} -- (102,70);}
