% Determinantes de la NAIRU: desplazamientos de la WS y de la PS (Layard,
% Nickell y Jackman, 1991). Misma función y escala que ws-ps.tex:
%   WS:  g(U) = 1 + 5,2 e^{-0,3 U};  PS: PAT/(1+mu_0) = g(4) = 2,5662, U^n_0 = 4
%   a) sube Z: WS' = g(U) + 0,5  =>  U^n_1 = ln(5,2/1,0662)/0,3 = 5,2818
%   b) sube mu: PS' = 2,5662 - 0,6 = 1,9662  =>  U^n_1 = ln(5,2/0,9662)/0,3 = 5,6098
% El panel a leído al revés es la reducción del subsidio de desempleo
% (imagen 11 del original).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.72cm,y=0.8cm,
    declare function={g(\u)=1+5.2*exp(-0.3*\u);}]
  \def\h{2.5662}\def\hb{1.9662}
  \def\ua{5.2818}\def\ub{5.6098}
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-3.4,-1.3) rectangle (23.4,7.2);
  % 1: ejes de los 2 paneles
  \capa{1}{\ejes{$U$}{$W/P$}{9}{6.1}}
  \capa{1}{\node[etiqueta, anchor=south] at (4.5,6.5) {a) Más prestaciones o poder sindical ($\uparrow Z$)};}
  \begin{scope}[shift={(12.8,0)}]
    \capa{1}{\ejes{$U$}{$W/P$}{9}{6.1}}
    \capa{1}{\node[etiqueta, anchor=south] at (4.5,6.5) {b) Más poder de mercado ($\uparrow\mu$)};}
  \end{scope}
  % 2: panel a, WS, PS y U^n_0
  \capa{2}{\draw[curva1] plot[domain=0.55:8.2, samples=80, smooth] (\x,{g(\x)})
      node[etiqueta, right, text=tceemorado] {WS};}
  \capa{2}{\draw[curva2] (0,\h) node[left, text=black] {$\mathrm{PAT}/(1+\mu)$} -- (8.2,\h)
      node[etiqueta, right, text=tceeazul] {PS};}
  \capa{2}{\node[punto] (Ea) at (4,\h) {};}
  \capa{2}{\draw[guia] (Ea) -- (4,0) node[below, text=black] {$U^n_0$};}
  % 3: panel a, la WS sube a WS' y la NAIRU sube a U^n_1
  \capa{3}{\draw[curvanueva, desplazada] plot[domain=1.2:8.2, samples=80, smooth] (\x,{g(\x)+0.5})
      node[etiqueta, right, text=tceerojo] {WS$'$};}
  \capa{3}{\draw[flecha] (6.3,{g(6.3)+0.08}) -- (6.3,{g(6.3)+0.42});}
  \capa{3}{\node[punto] (Ea1) at (\ua,\h) {};}
  \capa{3}{\draw[guia] (Ea1) -- (\ua,0) node[below, text=black] {$U^n_1$};}
  \capa{3}{\draw[flecha, shorten <=2pt, shorten >=2pt] (4,-0.95) -- (\ua,-0.95);}
  % 4: panel b, WS, PS y U^n_0
  \begin{scope}[shift={(12.8,0)}]
    \capa{4}{\draw[curva1] plot[domain=0.55:8.2, samples=80, smooth] (\x,{g(\x)})
        node[etiqueta, right, text=tceemorado] {WS};}
    \capa{4}{\draw[curva2] (0,\h) node[left, text=black] {$\mathrm{PAT}/(1+\mu_0)$} -- (8.2,\h)
        node[etiqueta, right, text=tceeazul] {PS};}
    \capa{4}{\node[punto] (Eb) at (4,\h) {};}
    \capa{4}{\draw[guia] (Eb) -- (4,0) node[below, text=black] {$U^n_0$};}
    % 5: panel b, la PS baja a PS' y la NAIRU sube a U^n_1
    \capa{5}{\draw[curvanueva, desplazada] (0,\hb) node[left, text=black] {$\mathrm{PAT}/(1+\mu_1)$} -- (8.2,\hb)
        node[etiqueta, right, text=tceerojo] {PS$'$};}
    \capa{5}{\draw[flecha] (7,{\h-0.06}) -- (7,{\hb+0.06});}
    \capa{5}{\node[punto] (Eb1) at (\ub,\hb) {};}
    \capa{5}{\draw[guia] (Eb1) -- (\ub,0) node[below, text=black] {$U^n_1$};}
    \capa{5}{\draw[flecha, shorten <=2pt, shorten >=2pt] (4,-0.95) -- (\ub,-0.95);}
  \end{scope}
\end{tikzpicture}
