% Expansión fiscal en A con 2 países grandes, tipo de cambio flexible y
% movilidad perfecta del capital (Mundell-Fleming; efecto locomotora). Misma
% familia, escala y marco que mundell-fleming-dos-paises.tex. Plano (Y, i).
% Paso a paso, con marcadores numerados (los mismos números que el texto):
%   1. G_A sube: la IS de A se desplaza a la derecha (IS').
%   2. Sube el tipo en A, entra capital y A se aprecia: la IS vuelve en parte (IS'').
%   3. Como A es grande, el tipo mundial sube: i* = 2,5 -> i*' = 3,0 (menos que el
%      tipo de A si fuera cerrada, 3,433: el gasto de A se reparte en un mercado
%      de dinero mundial mayor).
%   4. B exporta más a A y se deprecia: la IS de B a la derecha.
%   5. Nuevos equilibrios: suben las 2 rentas (locomotora).
% Rectas; equilibrio inicial (3; 2,5) en los 2 países:
%   A: LM  i = 2,5 + 0,8 (Y - 3) (no cambia)
%      IS  Y = 5,5 - i;  IS' (gasto) Y = 7,6 - i;  IS'' (neta de la apreciación) Y = 6,625 - i
%      nuevo equilibrio (3,625; 3,0): LM en 3,625 da 3,0; IS'' en 3,0 da 3,625
%      (economía cerrada: IS' y LM en (4,167; 3,433); la renta de A sube menos)
%      Flecha 1 a i = 1,7: de IS (3,8) a IS' (5,9). Flecha 2 a i = 3,9: de IS' (3,7) a IS'' (2,725).
%   B: LM  i = 2,5 + 1,0 (Y - 3) (no cambia)
%      IS  Y = 5,5 - i;  IS' Y = 6,5 - i; nuevo equilibrio (3,5; 3,0)
%      Flecha 4 a i = 1,3: de IS (4,2) a IS' (5,6).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.95cm,y=0.95cm,
    declare function={lmA(\y)=2.5+0.8*(\y-3); lmB(\y)=2.5+1.0*(\y-3);
      is(\y)=5.5-\y; isG(\y)=7.6-\y; isA(\y)=6.625-\y; isB(\y)=6.5-\y;},
    paso/.style={circle,draw=tceerojo,text=tceerojo,font=\scriptsize\bfseries,inner sep=1pt,minimum size=11pt},
    leyenda/.style={font=\footnotesize,anchor=west,inner sep=1pt}]
  \newcommand{\pasoley}[4]{\node[paso] at (#1,#2) {#3}; \node[leyenda] at ({#1+0.25},#2) {#4};}
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-0.9,-2.95) rectangle (14.6,5.4);
  % ---------------- País A ----------------
  % 1: ejes de A
  \capa{1}{\node[font=\small\bfseries, anchor=base west] at (0,5.0) {a) País A (expansión fiscal)};}
  \capa{1}{\ejes{$Y_A$}{$i$}{6.4}{4.7}}
  % 2: IS, LM y tipo mundial en A (las IS se dibujan dando i: Y = c - i)
  \capa{2}{\draw[curva2] ({is(4.5)},4.5) -- ({is(0.4)},0.4) node[etiqueta,right,text=tceeazul] {IS};}
  \capa{2}{\draw[curva1] (1.5,{lmA(1.5)}) -- (5.0,{lmA(5.0)}) node[etiqueta,right,text=tceemorado] {LM};}
  \capa{2}{\draw[curva3] (0,2.5) node[left,text=black] {$i^*$} -- (6.2,2.5);}
  \capa{2}{\node[punto] (A0) at (3,2.5) {};}
  \capa{2}{\draw[guia] (A0) -- (3,0) node[below,text=black] {$Y_{A0}$};}
  % 3: paso 1, sube el gasto: IS a la derecha
  \capa{3}{\draw[curva2,desplazada] ({isG(4.5)},4.5) node[etiqueta,above,text=tceeazul] {IS$'$} -- ({isG(1.2)},1.2);}
  \capa{3}{\draw[flecha] (3.9,1.7) -- (5.8,1.7);}
  \capa{3}{\node[paso] at (4.85,1.4) {1};}
  \capa{3}{\pasoley{0.15}{-0.95}{1}{$G_A\uparrow$: la IS de A, a la derecha}}
  % 4: paso 2, entra capital y A se aprecia: la IS vuelve en parte
  \capa{4}{\draw[curva2,desplazada,tceeazul!65] ({isA(4.5)},4.5) node[etiqueta,above left,text=tceeazul] {IS$''$} -- ({isA(0.9)},0.9);}
  \capa{4}{\draw[flecha] (3.65,3.9) -- (2.8,3.9);}
  \capa{4}{\node[paso] at (4.05,3.9) {2};}
  \capa{4}{\pasoley{0.15}{-1.4}{2}{sube $i$, entra capital, A se aprecia: IS$''$}}
  % 5: paso 3, A es grande: sube el tipo mundial
  \capa{5}{\draw[curva3,desplazada] (0,3.0) node[left,text=black] {$i^{*\prime}$} -- (6.2,3.0);}
  \capa{5}{\draw[flecha] (0.55,2.55) -- (0.55,2.95);}
  \capa{5}{\node[paso] at (0.95,2.75) {3};}
  \capa{5}{\pasoley{0.15}{-1.85}{3}{A es grande: el tipo mundial sube}}
  % ---------------- País B ----------------
  \begin{scope}[xshift=7.9*0.95cm]
  % 1: ejes de B
  \capa{1}{\node[font=\small\bfseries, anchor=base west] at (0,5.0) {b) País B};}
  \capa{1}{\ejes{$Y_B$}{$i$}{6.4}{4.7}}
  % 2: IS, LM y tipo mundial en B
  \capa{2}{\draw[curva2] ({is(4.5)},4.5) -- ({is(0.4)},0.4) node[etiqueta,right,text=tceeazul] {IS};}
  \capa{2}{\draw[curva1] (1.2,{lmB(1.2)}) -- (4.6,{lmB(4.6)}) node[etiqueta,right,text=tceemorado] {LM};}
  \capa{2}{\draw[curva3] (0,2.5) node[left,text=black] {$i^*$} -- (6.2,2.5);}
  \capa{2}{\node[punto] (B0) at (3,2.5) {};}
  \capa{2}{\draw[guia] (B0) -- (3,0) node[below,text=black] {$Y_{B0}$};}
  % 5: paso 3 en B: sube el tipo mundial
  \capa{5}{\draw[curva3,desplazada] (0,3.0) node[left,text=black] {$i^{*\prime}$} -- (6.2,3.0);}
  \capa{5}{\draw[flecha] (0.55,2.55) -- (0.55,2.95);}
  \capa{5}{\node[paso] at (0.95,2.75) {3};}
  % 6: paso 4, B exporta más a A y se deprecia: IS a la derecha
  \capa{6}{\draw[curva2,desplazada] ({isB(4.3)},4.3) -- ({isB(1.0)},1.0) node[etiqueta,right,text=tceeazul] {IS$'$};}
  \capa{6}{\draw[flecha] (4.3,1.3) -- (5.1,1.3);}
  \capa{6}{\node[paso] at (5.05,0.9) {4};}
  \capa{6}{\pasoley{0.15}{-0.95}{4}{B exporta más a A y se deprecia:}}
  \capa{6}{\node[leyenda] at (0.4,-1.35) {la IS de B, a la derecha};}
  \end{scope}
  % 7: paso 5, nuevos equilibrios
  \capa{7}{\node[punto,fill=tceerojo] (A1) at (3.625,3.0) {};}
  \capa{7}{\draw[guia,tceerojo] (A1) -- (3.625,0) node[below right=0pt and -4pt,text=tceerojo] {$Y_{A1}$};}
  \capa{7}{\pasoley{0.15}{-2.3}{5}{\textcolor{tceerojo}{sube la renta de A (menos que si fuera cerrada)}}}
  \begin{scope}[xshift=7.9*0.95cm]
  \capa{7}{\node[punto,fill=tceerojo] (B1) at (3.5,3.0) {};}
  \capa{7}{\draw[guia,tceerojo] (B1) -- (3.5,0) node[below right=0pt and -2pt,text=tceerojo] {$Y_{B1}$};}
  \capa{7}{\pasoley{0.15}{-1.85}{5}{\textcolor{tceerojo}{con $M_B$ igual e $i^*$ más alto,}}}
  \capa{7}{\node[leyenda,text=tceerojo] at (0.4,-2.25) {la LM de B fija una renta mayor:};}
  \capa{7}{\node[leyenda,text=tceerojo] at (0.4,-2.65) {sube la renta de B (locomotora)};}
  \end{scope}
\end{tikzpicture}
