% Saldo por cuenta corriente y saldo público (% del PIB) de Alemania, Países
% Bajos y Corea en 2019 y 2025 (FMI, WEO de octubre de 2026; fuente y
% comprobación en la cabecera de ../datos/saldos-corrientes-fiscales-superavit.csv).
% Paleta: cuenta corriente en azules (2019 claro, 2025 oscuro) y saldo público
% en dorados (2019 claro, 2025 oscuro). Cada barra con su desplazamiento fijo
% (bar shift), para que no se muevan al aparecer las capas.
% Capas: barras → ejes y cero → etiquetas y leyenda.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
\colorlet{cc19}{tceeazul!35}
\colorlet{cc25}{tceeazul!85!black}
\colorlet{sp19}{tceedorado!40}
\colorlet{sp25}{tceedorado!75!black}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/saldos-corrientes-fiscales-superavit.csv}\saldos
% Los ejes salen en la capa 3 (después de las barras): antes, invisibles.
\pgfplotsset{ejessaldos/.style={}}
\ifnum3>\tceepaso\relax
  \pgfplotsset{ejessaldos/.style={axis line style={draw=none},
    major tick style={draw=none}, ticklabel style={opacity=0},
    label style={opacity=0}, ymajorgrids=false}}
\fi
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=13cm, height=7.5cm,
      xmin=0.45, xmax=3.55, xtick=\empty,
      ymin=-4, ymax=10, ytick={-4,-2,...,10},
      yticklabel={\pgfmathprintnumber[use comma]{\tick}},
      ticklabel style={font=\footnotesize},
      ylabel={\% del PIB},
      axis y line*=left, axis x line=none, tick align=outside,
      ymajorgrids, grid style={tceegris!25},
      clip=false,
      legend style={at={(0.5,-0.08)}, anchor=north, legend columns=2,
        draw=none, font=\footnotesize,
        /tikz/every even column/.append style={column sep=10pt}},
      legend cell align=left,
      ejessaldos, dicapas,
    ]
    % 1: saldo por cuenta corriente, 2019 y 2025
    \capa{1}{
      \addplot[on layer=dicbarras, forget plot, ybar, bar width=0.17, bar shift=-0.255,
        dicbarra=cc19] table[x=x, y=cc_2019] {\saldos};
      \addplot[on layer=dicbarras, forget plot, ybar, bar width=0.17, bar shift=-0.085,
        dicbarra=cc25] table[x=x, y=cc_2025] {\saldos};
    }
    % 2: saldo público, 2019 y 2025
    \capa{2}{
      \addplot[on layer=dicbarras, forget plot, ybar, bar width=0.17, bar shift=0.085,
        dicbarra=sp19] table[x=x, y=sp_2019] {\saldos};
      \addplot[on layer=dicbarras, forget plot, ybar, bar width=0.17, bar shift=0.255,
        dicbarra=sp25] table[x=x, y=sp_2025] {\saldos};
    }
    % 3: ejes y línea del cero (con los nombres de los países debajo)
    \capa{3}{
      \dicraya[thick]{(0.45,0) (3.55,0)}
      \dicrotulo[font=\small,anchor=north]{(1,-4)}{Alemania}
      \dicrotulo[font=\small,anchor=north]{(2,-4)}{Países Bajos}
      \dicrotulo[font=\small,anchor=north]{(3,-4)}{Corea}
    }
    % 4: etiquetas de valor (datos del CSV) y leyenda
    \capa{4}{
      \dicrotulo[font=\scriptsize,anchor=south,text=cc25]{(0.745,7.8)}{7,8}
      \dicrotulo[font=\scriptsize,anchor=south,text=cc25]{(0.915,4.5)}{4,5}
      \dicrotulo[font=\scriptsize,anchor=south,text=sp25]{(1.085,1.3)}{1,3}
      \dicrotulo[font=\scriptsize,anchor=north,text=sp25]{(1.255,-2.9)}{$-$2,9}
      \dicrotulo[font=\scriptsize,anchor=south,text=cc25]{(1.745,6.9)}{6,9}
      \dicrotulo[font=\scriptsize,anchor=south,text=cc25]{(1.915,7.6)}{7,6}
      \dicrotulo[font=\scriptsize,anchor=south,text=sp25]{(2.085,1.9)}{1,9}
      \dicrotulo[font=\scriptsize,anchor=north,text=sp25]{(2.255,-1.6)}{$-$1,6}
      \dicrotulo[font=\scriptsize,anchor=south,text=cc25]{(2.745,3.4)}{3,4}
      \dicrotulo[font=\scriptsize,anchor=south,text=cc25]{(2.915,6.5)}{6,5}
      \dicrotulo[font=\scriptsize,anchor=south,text=sp25]{(3.085,0.8)}{0,8}
      \dicrotulo[font=\scriptsize,anchor=north,text=sp25]{(3.255,-2.3)}{$-$2,3}
      \addlegendimage{area legend, dicbarra=cc19}
      \addlegendentry{Saldo por cuenta corriente, 2019}
      \addlegendimage{area legend, dicbarra=sp19}
      \addlegendentry{Saldo público, 2019}
      \addlegendimage{area legend, dicbarra=cc25}
      \addlegendentry{Saldo por cuenta corriente, 2025}
      \addlegendimage{area legend, dicbarra=sp25}
      \addlegendentry{Saldo público, 2025}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
