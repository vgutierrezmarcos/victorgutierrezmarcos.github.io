\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
% Precio de la vivienda en Estados Unidos, enero de 2000 a diciembre de 2016
% (enero de 2000 = 100). S&P Cotality Case-Shiller: índice nacional
% (CSUSHPINSA) y compuesto de 20 ciudades (SPCS20RNSA), vía FRED; nacional en
% términos reales con el IPC urbano (CPIAUCNS), cálculo propio (fuente y
% comprobación en la cabecera de ../datos/vivienda-eeuu.csv).
% Eje x: 2000 + n.º de mes/12 (fila del CSV). Recesión del NBER, de pico a
% valle: diciembre de 2007 (2007,917) a junio de 2009 (2009,417), como en
% fuentes/3A42/datos/fechado-recesiones.csv.
% Máximo nacional: julio de 2006 (184,6); mínimo: febrero de 2012 (134,0): -27,4 %.
% Capas «dic» (tikz-tcee.tex): sombreado → ejes → líneas → puntos → rótulos.
% Autor: Víctor Gutiérrez Marcos
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/vivienda-eeuu.csv}\vivienda
\pgfplotsset{
  leyenda linea corta/.style={legend image code/.code={%
    \draw[##1] (0cm,0cm) -- (0.6cm,0cm);}},
}
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=14cm, height=7.5cm,
      xmin=2000, xmax=2017.2,
      xtick={2000,2002,...,2016}, minor xtick={2001,2003,...,2017},
      x tick label style={/pgf/number format/1000 sep={}},
      ymin=80, ymax=240, ytick={80,100,...,240},
      ticklabel style={font=\footnotesize},
      ylabel={Índice (enero de 2000 $=100$)}, ylabel style={font=\footnotesize},
      axis lines*=left, tick align=outside, clip=false,
      ymajorgrids, grid style={tceegris!25},
      legend style={at={(0.5,-0.13)}, anchor=north, legend columns=3, draw=none, font=\footnotesize,
        /tikz/every even column/.append style={column sep=10pt}},
      legend cell align=left, dicapas,
    ]
    % 1: recesión del NBER, diciembre de 2007 a junio de 2009
    \capa{1}{
      \dicsombra[fill=tceegris!22]{2007.917}{80}{2009.417}{240}
    }
    % 2: ejes (los dibuja el entorno axis)
    % 3: índice nacional
    \capa{3}{
      \addplot[line width=1.6pt, tceemorado, leyenda linea corta] table[x expr=2000+\coordindex/12, y=nacional] {\vivienda};
      \addlegendentry{nacional}
    }
    % 4: 20 ciudades
    \capa{4}{
      \addplot[thick, tceedorado, leyenda linea corta] table[x expr=2000+\coordindex/12, y=veinte_ciudades] {\vivienda};
      \addlegendentry{20 ciudades}
    }
    % 5: nacional en términos reales
    \capa{5}{
      \addplot[thick, dashed, tceegris!55!black, leyenda linea corta] table[x expr=2000+\coordindex/12, y=nacional_real] {\vivienda};
      \addlegendentry{nacional, en términos reales}
    }
    % 6: máximo (julio de 2006) y mínimo (febrero de 2012) del índice nacional
    \capa{6}{
      \addplot[dicpuntos=tceemorado, forget plot] coordinates {(2006.5,184.6) (2012.083,134.0)};
    }
    % 7: rótulos de los puntos y de la recesión
    \capa{7}{
      \dicrotulo[font=\footnotesize, anchor=north, align=center, text=tceemorado]{(2006.6,178)}{jul.\ 2006\\184,6}
      \dicrotulo[font=\footnotesize, anchor=north, align=center, text=tceemorado]{(2012.083,130)}{feb.\ 2012\\134,0 ($-27$~\%)}
      \dicrotulo[font=\footnotesize, anchor=north, align=center, text=tceegris!60!black]{(2008.667,236)}{recesión\\(NBER)}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
