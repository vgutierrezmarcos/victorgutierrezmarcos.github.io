%   b) Caso 2, superávit primario: d_t = -1 + 0,5 d_{t-1}; d* = -2, estable.

% \escalera{color}{d_0, d_1, d_2, ...}: de d_0 en el eje va a la recta y
% en horizontal a la bisectriz, una y otra vez
\newcommand{\escalera}[2]{%
  \foreach \d [count=\k] in {#2}{%
    \ifnum\k=1\relax\xdef\prev{\d}\xdef\prevy{0}\else
      \draw[paso=#1] (\prev,\prevy) -- (\prev,\d);
      \draw[paso=#1] (\prev,\d) -- (\d,\d);
      \xdef\prev{\d}\xdef\prevy{\d}\fi}}
