% Estabilidad del modelo de Cagan (1956) con expectativas adaptativas.
% Demanda ln(M/P) = ln zeta - alpha pi^e; adaptativas dot pi^e = gamma (pi - pi^e).
% Derivando: dot pi^e = [gamma/(1 - alpha gamma)] (gamma_M - pi^e).
%   a) alpha gamma < 1: pendiente negativa, pi^e converge a gamma_M (estable).
%   b) alpha gamma > 1: pendiente positiva, pi^e se aleja de gamma_M (inestable).
% Rectas: a) dot pi^e = -0,7 (pi^e - 3,5);  b) dot pi^e = 0,7 (pi^e - 3,5).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}

\begin{document}

  \path (-1.6,-4.2) rectangle (21.2,5.2);
  % ---------------- a) estable ----------------
  % 1: ejes de los 2 paneles
  \capa{1}{\draw[eje] (0,0) -- (7.6,0) node[below left=2pt and 0pt] {$\pi^e$};}
  \capa{1}{\draw[eje] (0,-3.2) -- (0,3.6) node[left] {$\dot\pi^e$};}
  \capa{1}{\node[below left] at (0,0) {$0$};}
  \capa{1}{\node[font=\normalsize\bfseries, anchor=base] at (3.8,4.4) {a) $\boldsymbol{\alpha\gamma<1}$: estable};}

  \capa{1}{\draw[eje] (0,0) -- (7.6,0) node[below left=2pt and 0pt] {$\pi^e$};}
  \capa{1}{\draw[eje] (0,-3.2) -- (0,3.6) node[left] {$\dot\pi^e$};}
  \capa{1}{\node[below left] at (0,0) {$0$};}
  \capa{1}{\node[font=\normalsize\bfseries, anchor=base] at (3.8,4.4) {b) $\boldsymbol{\alpha\gamma>1}$: inestable};}
