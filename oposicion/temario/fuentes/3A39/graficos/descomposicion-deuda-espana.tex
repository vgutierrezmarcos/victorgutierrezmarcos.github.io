% Descomposición de la variación de la deuda pública de España sobre el PIB,
% 1996-2025, en puntos del PIB:
%   Delta d_t = (g_t - tau_t) + [(i_t - gamma^N_t)/(1 + gamma^N_t)] d_{t-1} + ADD_t,
% con el ajuste déficit-deuda (ADD) calculado de forma exacta (residuo 0).
% Datos: ../datos/descomposicion-deuda-espana-zona-euro.csv (cálculo propio con
% Eurostat gov_10dd_edpt1 y nama_10_gdp, octubre de 2026; consulta 04/11/2026).
% Cifras de control: 2012: 8,47 + 5,22 + 6,34 = 20,03; 2020: 7,68 + 12,99 +
% 0,94 = 21,62; 2025: 0,01 - 3,09 + 2,20 = -0,88.
% Paleta: déficit primario dicconspriv (naranja, política fiscal), bola de
% nieve dicopf (azul, el tipo de interés; la misma que en bola-nieve-espana),
% ajuste déficit-deuda gris; Delta d en negro.
% Capas (CRITERIOS §7): barras → ejes y cero → línea de Delta d → puntos → etiquetas.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/descomposicion-deuda-espana-zona-euro.csv}\desc
\pgfplotsset{leyenda linea corta/.style={legend image code/.code={%
  \draw[##1,mark repeat=2,mark phase=2] plot coordinates {(0cm,0cm) (0.3cm,0cm) (0.6cm,0cm)};}}}
\begin{tikzpicture}[tcee]
\begin{axis}[width=14cm,height=8cm,
    ybar stacked,stack negative=separate,bar width=0.62,
    xmin=1995.3,xmax=2025.7,ymin=-12,ymax=24,
    xtick={1996,2000,...,2024},ytick={-10,-5,...,20},
    x tick label style={/pgf/number format/.cd,set thousands separator={}},
    ticklabel style={font=\footnotesize},
    ylabel={puntos del PIB},ylabel style={font=\footnotesize},
    axis x line*=bottom,axis y line*=left,tick align=outside,clip=false,
    ymajorgrids,grid style={tceegris!25},
    legend style={at={(0.5,-0.1)},anchor=north,legend columns=2,draw=none,font=\footnotesize,
      /tikz/every even column/.append style={column sep=10pt}},
    legend cell align=left,area legend,dicapas]
  % 1: barras apiladas de las 3 contribuciones
  \capa{1}{
    \addplot[on layer=dicbarras,dicbarra=dicconspriv] table[x=anio,y=ES_saldo_primario_contrib] {\desc};
    \addlegendentry{Déficit primario, $g_t-\tau_t$}
    \addplot[on layer=dicbarras,dicbarra=dicopf] table[x=anio,y=ES_bola_nieve] {\desc};
    \addlegendentry{Efecto bola de nieve}
    \addplot[on layer=dicbarras,dicbarra=dicotroclaro] table[x=anio,y=ES_ajuste_stock_flujo] {\desc};
    \addlegendentry{Ajuste déficit-deuda, $\mathrm{ADD}_t$}
  }
  % 2: ejes y línea del cero
  \capa{2}{\dicraya{(1995.3,0) (2025.7,0)}}
  % 3: línea de la variación de la deuda
  \capa{3}{
    \addplot[stack plots=false,sharp plot,diclineahalo=dicinflacion,leyenda linea corta,
      legend image post style={mark=*,mark size=1.8pt,mark options={fill=white,line width=0.9pt}}]
      table[x=anio,y=ES_variacion_deuda] {\desc};
    \addlegendentry{Variación de la deuda, $\Delta d_t$}
  }
  % 4: puntos de la variación de la deuda
  \capa{4}{
    \addplot[on layer=dicpuntos,forget plot,stack plots=false,dicpuntos=dicinflacion]
      table[x=anio,y=ES_variacion_deuda] {\desc};
  }
  % 5: etiquetas de 2012, 2020 y 2025
  \capa{5}{
    \dicrotulo[font=\footnotesize,anchor=east,align=right]{(2011.5,17.5)}{2012: ajuste\\déficit-deuda $+6{,}3$}
    \dicrotulo[font=\footnotesize,anchor=east,align=right]{(2019.5,17.5)}{2020: bola\\de nieve $+13{,}0$}
    \dicrotulo[font=\footnotesize,anchor=south,align=center]{(2025.1,2.5)}{2025:\\$\Delta d_t=-0{,}9$}
  }
\end{axis}
\end{tikzpicture}
\end{document}
