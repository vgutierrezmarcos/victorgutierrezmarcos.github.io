% Cooper y Massell (1965): saldo de bienestar de la unión aduanera frente a la
% rebaja no discriminatoria del arancel, con las áreas del ejemplo numérico de
% viner-lipsey.tex (b = 0,225; d = 0,3; e = 1,425; f = 0,45; g = 0,6).
% Unión: b + d - e = -0,9. Rebaja: b + d + f + g = +1,575. Escala: 1 = 1,6 cm.
% Elaboración propia a partir de Lipsey (1957) y Cooper y Massell (1965).
% Autor: Víctor Gutiérrez Marcos
\begin{tikzpicture}[tcee,x=1cm,y=1.6cm,
    ganancia/.style={fill=tceedorado,fill opacity=0.45,draw=tceedorado},
    recaudacion/.style={fill=tceeverde,fill opacity=0.45,draw=tceeverde},
    perdida/.style={fill=tceerojo,fill opacity=0.35,draw=tceerojo},
    saldo/.style={very thick,draw=black}]
  % 1: eje horizontal (bienestar cero) y rótulos de las 2 políticas
  \capa{1}{\draw[thin,black] (-0.3,0) -- (8.6,0);
    \node[etiqueta,left] at (-0.3,0) {$\Delta W=0$};
    \node[etiqueta,font=\small\bfseries] at (2,2.05) {Unión aduanera};
    \node[etiqueta,font=\small\bfseries] at (6.6,2.05) {Rebaja no discriminatoria};}
  % 2: unión aduanera: b + d sube, e baja, saldo negativo
  \capa{2}{\fill[ganancia] (0.4,0) rectangle (1.4,0.525);
    \node[etiqueta] at (0.9,0.26) {$b+d$};
    \fill[perdida] (1.6,0.525) rectangle (2.6,-0.9);
    \node[etiqueta] at (2.1,-0.2) {$-e$};
    \draw[guia] (1.4,0.525) -- (1.6,0.525);}
  \capa{3}{\filldraw[saldo,fill=tceerojo!12] (2.8,0) rectangle (3.6,-0.9);
    \node[etiqueta,below,text=tceerojo] at (3.2,-0.9) {$b+d-e<0$};}
  % 4: rebaja no discriminatoria: b + d igual, f + g de recaudación, saldo positivo
  \capa{4}{\fill[ganancia] (5.0,0) rectangle (6.0,0.525);
    \node[etiqueta] at (5.5,0.26) {$b+d$};
    \fill[recaudacion] (6.2,0.525) rectangle (7.2,1.575);
    \node[etiqueta] at (6.7,1.05) {$f+g$};
    \draw[guia] (6.0,0.525) -- (6.2,0.525);}
  \capa{5}{\filldraw[saldo,fill=tceeverde!12] (7.4,0) rectangle (8.2,1.575);
    \node[etiqueta,below,text=tceeverde] at (7.8,0) {$b+d+f+g>0$};}
  % 6: la diferencia entre las 2: e + f + g
  \capa{6}{\draw[<->,thick,tceemorado] (4.3,-0.9) -- (4.3,1.575)
    node[midway,fill=white,inner sep=1.5pt,text=tceemorado] {$e+f+g$};
    \draw[guia] (3.6,-0.9) -- (4.3,-0.9);
    \draw[guia] (4.3,1.575) -- (7.4,1.575);}
\end{tikzpicture}
\end{document}
