% Economías de escala con coste marginal creciente (nota al pie de Corden), 3.B.10, I.3.3.
% C(q) = 4 + 5q - 1,2q^2 + 0,12q^3 (unidades del gráfico):
%   CMg = 5 - 2,4q + 0,36q^2, mínimo en q = 3,33 (CMg = 1);
%   CMe = 4/q + 5 - 1,2q + 0,12q^2, mínimo (= CMg) en q = 5,543 (CMe = 2,757).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.75cm,y=0.75cm]
  \pgfmathsetmacro{\qg}{10/3}
  \pgfmathsetmacro{\cg}{5-2.4*\qg+0.36*\qg*\qg}
  \pgfmathsetmacro{\qm}{5.5425}
  \pgfmathsetmacro{\cm}{4/\qm+5-1.2*\qm+0.12*\qm*\qm}
  % 1: ejes
  \capa{1}{\ejes{$q$}{$CMe$, $CMg$}{8.6}{7.5}}
  % 2: coste medio, en U
  \capa{2}{\draw[curva3] plot[domain=1.3:8,samples=60,smooth] (\x,{4/\x+5-1.2*\x+0.12*\x*\x})
    node[etiqueta,right,tceeverde] {$CMe$};}
  % 3: coste marginal, que corta al medio en su mínimo
  \capa{3}{\draw[curva1] plot[domain=0.8:7.2,samples=60,smooth] (\x,{5-2.4*\x+0.36*\x*\x})
    node[etiqueta,above,tceemorado] {$CMg$};
    \node[punto] at (\qm,\cm) {};}
  % 4: tramo con economías de escala y coste marginal creciente
  \capa{4}{\fill[area2,on background layer] (\qg,0) rectangle (\qm,7);
    \draw[guia] (\qg,\cg) -- (\qg,0);
    \draw[guia] (\qm,\cm) -- (\qm,0);
    \draw[<->,thin,tceemorado] (0,-0.45) -- node[below,font=\scriptsize] {$CMg$ decreciente} (\qg,-0.45);
    \draw[<->,thin,tceemorado] (\qg,-0.45) -- node[below,font=\scriptsize] {$CMg$ creciente} (8,-0.45);
    \draw[<->,thin,tceeverde] (0,-1.25) -- node[below,font=\scriptsize] {economías de escala} (\qm,-1.25);
    \draw[<->,thin,tceeverde] (\qm,-1.25) -- node[below,font=\scriptsize] {deseconomías} (8,-1.25);}
\end{tikzpicture}
\end{document}
