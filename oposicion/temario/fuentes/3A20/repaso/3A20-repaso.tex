\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: «\emph{ni tanto ni tan calvo}».
  \begin{itemize}
  \item Monopolio natural: ni precio de monopolio (rentas) ni $p=\mathrm{CMg}$ (pérdidas); término medio: tarifa en dos partes (potencia y energía), \autor{Ramsey}.
  \item CNMC (dic. 2025): retribución de las redes eléctricas del 5,58~al 6,58\,\%: ¿mucho (rentas) o poco (sin inversión)? Lo sabe la empresa.
  \end{itemize}
\item \textbf{Relevancia}:
  \begin{itemize}
  \item \emph{teórica}: economía de la información y diseño de mecanismos; Nobel a \autor{Tirole} (2014);
  \item \emph{empírica}: margen sobre CMg en EE.\,UU. 21\,\% (1980) $\to$ 61\,\% (2016);
  \item \emph{política}: redes y, desde 2023, plataformas (DMA), reguladas \emph{ex ante} \vertema{3.B.39} \vertema{4.A.15}.
  \end{itemize}
\item \textbf{Contextualización}:
  \begin{itemize}
  \item \concepto{Poder de mercado}: cobrar de forma duradera $p>\mathrm{CMg}$ sin perder a todos los clientes.
  \item \concepto{Regulación económica}: intervención \emph{ex ante} sobre precios, entrada, calidad o acceso; \concepto{óptima}: máx. bienestar s.a. viabilidad de la empresa e información del regulador.
  \item Rompe el supuesto precio-aceptante del 1TFEB \vertema{3.A.22}.
  \item Teoría tradicional (\autor{Dupuit} $\to$ \autor{Boiteux}) $\to$ captura (\autor{Stigler}, 1971) $\to$ \concepto{nueva economía de la regulación} (información asimétrica, años 80); SSNIP (EE.\,UU., 1982).
  \end{itemize}
\item \textbf{Problemática}: ¿por qué regular? ¿Qué es el poder de mercado, cómo se mide y en qué mercado? ¿Regulación óptima con información completa? ¿Y si la empresa sabe más?
\item \textbf{Estructura}: de menos a más realismo: \hyperref[rep:I]{I}, poder de mercado y mercado relevante; \hyperref[rep:II]{II}, información completa; \hyperref[rep:III]{III}, asimétrica y aplicaciones. Hilo: Redes del Valle, distribuidora eléctrica de una comarca.
\end{itemize}

\bloque[7,5]{I. Regulación, poder de mercado y mercado relevante\label{rep:I}}
\begin{ideaclave}
De los motivos para regular, el poder de mercado: aleja al mercado del óptimo, se agrava con conductas anticompetitivas y se mide en un mercado que hay que definir antes; la respuesta depende de su origen y de lo que sepa el regulador.
\end{ideaclave}

\sub{I.1. ¿Por qué regular?\label{rep:I.1}}
\begin{itemize}
\item \concepto{Regulación}: normas de conducta del sector público; completa el mercado. Motivos (\autor{Musgrave}, 1959):
  \begin{itemize}
  \item \emph{eficiencia}: fallos de mercado (poder de mercado, información asimétrica, externalidades, bienes públicos; continuidad, coordinación);
  \item \emph{equidad} (bienes preferentes; tarifas uniformes) y \emph{estabilidad} (financiera; topes en crisis).
  \end{itemize}
\item Varios a la vez (\autor{Baldwin}-\autor{Cave}-\autor{Lodge}, 2012); frente a ellos, los fallos del sector público \vertema{3.A.23}.
\end{itemize}

\sub{I.2. Poder de mercado: concepto, medida, coste y conductas\label{rep:I.2}}
\begin{itemize}
\item \concepto{Poder de mercado} (CE, 2004): subir precios con beneficio o recortar producción, calidad o innovación.
  \begin{itemize}
  \item Precio-aceptación rota + \concepto{barreras de entrada}: legales, estratégicas, tecnológicas (monopolio natural) \vertema{3.A.15}.
  \item \concepto{Posición dominante}: abusar de ella es ilegal (art.~102~TFUE).
  \end{itemize}
\end{itemize}
\subsub{Medida}
\begin{itemize}
\item \emph{Índice de \autor{Lerner}} (1934), en el óptimo del monopolista \vertema{3.A.17}:
\funcionclave{\mathcal{L}\equiv\frac{p-\mathrm{CMg}}{p}=\frac{1}{|\varepsilon_{q,p}|}}
  \begin{itemize}
  \item \autor{Cournot}: $\mathcal{L}_i=s_i/|\varepsilon|$ \vertema{3.A.19}.
  \item Problemas: CMg no observable; margen alto para cubrir fijos; bajo con ineficiencia X.
  \end{itemize}
\item \emph{Estructura}: cuotas, $\mathrm{CR}_m$ e \concepto{IHH} $\equiv\sum_i(100\,s_i)^2\in(0;10.000]$.
  \begin{itemize}
  \item En \autor{Cournot}, $\sum_is_i\mathcal{L}_i=\mathrm{IHH}/(10.000|\varepsilon|)$.
  \item Cuota $\geq50$\,\% presume dominancia; concentraciones sin problema: IHH $<1.000$; 1.000-2.000 y $\Delta<250$; $>2.000$ y $\Delta<150$.
  \end{itemize}
\end{itemize}
\subsub{Por qué preocupa: las 3~ineficiencias}
\begin{itemize}
\item \emph{Asignativa}: \autor{Harberger} (1954), $\tfrac12\mathcal{L}p^Mq^M$, $<0{,}1$\,\% de la renta; + \concepto{búsqueda de rentas} (\autor{Tullock}, 1967).
\item \emph{Productiva}: \concepto{ineficiencia X} (\autor{Leibenstein}, 1966).
\item \emph{Dinámica}, ambigua: \autor{Schumpeter} vs \autor{Arrow}; U invertida (\autor{Aghion} et al., 2005). Y efecto \emph{distributivo}.
\end{itemize}
\subsub{Conductas anticompetitivas}
\begin{itemize}
\item Cárteles (art.~101~TFUE), abusos (art.~102: explotativos y excluyentes), concentraciones.
\item \concepto{Precios predatorios}:
  \begin{itemize}
  \item irracionales (\autor{McGee}, 1958; \autor{Selten}, 1978) salvo reputación (\autor{Kreps}-\autor{Wilson}, \autor{Milgrom}-\autor{Roberts}, 1982);
  \item \autor{Areeda}-\autor{Turner} (1975): $p<$ CVMe; AKZO (1991): $<$ CVMe abusivo; entre CVMe y CTMe, si hay plan de exclusión.
  \end{itemize}
\item \emph{Estrechamiento de márgenes}: Telefónica, 151,9~M€ (CE 2007; TJUE 2014).
\end{itemize}
\begin{itemize}
\item \emph{Respuesta}: conductas $\to$ \concepto{defensa de la competencia}, \emph{ex post}; subaditividad $\to$ \concepto{regulación económica}, \emph{ex ante} (CNMC, ambas desde 2013).
\end{itemize}

\sub{I.3. Definición del mercado relevante\label{rep:I.3}}
\subsub{a) Idea}
\begin{itemize}
\item \concepto{Mercado relevante}: productos y zonas que se disciplinan mutuamente (CE, 2024); de \emph{producto} y \emph{geográfico}.
\end{itemize}
\subsub{b) Modelo: el test del monopolista hipotético}
\begin{itemize}
\item \concepto{SSNIP} (EE.\,UU., 1982): ¿ganaría un monopolista del candidato subiendo un 5-10\,\%?
  \begin{itemize}
  \item Si no, se añade el sustitutivo más próximo.
  \item Mercado = \emph{menor} conjunto en que la subida es rentable.
  \end{itemize}
\item \concepto{Pérdida crítica} (\autor{Harris}-\autor{Simons}, 1989):
\begin{formula}
\centering $\ell^{c}=\dfrac{\tau}{\tau+\mathcal{L}}$\quad ($\tau=5$\,\%, $\mathcal{L}=20$\,\% $\Rightarrow$ 20\,\%)
\end{formula}
\end{itemize}
\graficorepaso[0.8\linewidth]{ssnip}{SSNIP: el mercado candidato se amplía hasta que la subida es rentable}
\begin{itemize}
\item Redes del Valle: casi nadie se desconecta $\Rightarrow$ mercado = \emph{distribución en la comarca}; la \emph{comercialización}, otro (nacional): se regula la red.
\end{itemize}
\subsub{c) Extensiones}
\begin{itemize}
\item \concepto{Falacia del celofán} (du Pont, 1956): a precio de monopolio $|\varepsilon|>1$, ninguna subida es rentable $\Rightarrow$ mercado demasiado amplio.
\item \emph{Herramientas}: elasticidades y \emph{ratios de desvío}; correlación de precios (solo necesaria); flujos de comercio.
\item \emph{Precio cero}: \concepto{SSNDQ} (calidad; Google Android). \emph{2 lados}: un mercado o uno por lado.
\item \emph{Valoración}: binaria; medidas directas de presión al alza (\autor{Farrell}-\autor{Shapiro}, 2010). Primer paso, no diagnóstico.
\end{itemize}

\sub{I.4. Hoja de ruta: algoritmo de \autor{Braeutigam} ampliado\label{rep:I.4}}
\begin{itemize}
\item ¿Origen? Conductas $\to$ competencia. ¿Sostenible? No: regular la entrada. ¿Conoce costes y demanda? Sí: \hyperref[rep:II]{II}; no: \hyperref[rep:III]{III}.
\end{itemize}

\bloque[8]{II. Regulación con información completa\label{rep:II}}
\begin{ideaclave}
Un regulador que lo sabe todo sabe qué precios poner, aunque, si los fondos públicos cuestan, no los de competencia perfecta; a veces le basta con subastar el mercado o liberalizar los tramos competitivos.
\end{ideaclave}

\sub{II.1. Monopolio natural\label{rep:II.1}}
\begin{itemize}
\item \concepto{Monopolio natural}: costes \concepto{subaditivos}, $C(\sum_jq_j)<\sum_jC(q_j)$ (\autor{Baumol}, 1977).
  \begin{itemize}
  \item Escala: suficiente, no necesaria; \emph{sostenible} (\autor{Panzar}-\autor{Willig}, 1977).
  \end{itemize}
\item \emph{Dilema}: ni desmantelar, ni dejar libre, ni pasarse: eficiencia asignativa vs \emph{viabilidad}.
\end{itemize}

\sub{II.2. Primer y segundos óptimos\label{rep:II.2}}
\subsub{a) Primer óptimo}
\begin{itemize}
\item $p=\mathrm{CMg}$ (\autor{Hotelling}, 1938); si $\mathrm{CMg}<\mathrm{CMe}$, pérdida $\Rightarrow$ subvención a coste $1+\lambda$ (\concepto{coste sombra de los fondos públicos}).
\item Redes del Valle: 90~M€ casi fijos, CMg $\approx0$ $\Rightarrow$ con $p=\mathrm{CMg}$ pierde 90~M€.
\end{itemize}
\pareja{subaditividad-sostenibilidad}{Subaditividad entre $q^0$ y $q^1$}{regulacion-monopolio-natural}{$p=\mathrm{CMg}$: pérdidas; $p=\mathrm{CMe}$: beneficio nulo}
\subsub{b) Segundos óptimos}
\begin{itemize}
\item $p=\mathrm{CMe}$: beneficio nulo, con PIE.
\item \emph{Dos partes} $T=A+p\,q$ con $p=\mathrm{CMg}$ (\autor{Coase}, 1946; \autor{Oi}, 1971): factura con término de potencia + de energía.
\item \emph{Punta y valle} (\autor{Steiner}, 1957): $p_v=b$, $p_p=b+\beta$; peajes de punta.
\end{itemize}
\begin{itemize}
\item \autor{Ramsey}-\autor{Boiteux} (1927, 1956):
\begin{formula}
\centering $\dfrac{p_i-\mathrm{CMg}_i}{p_i}=\dfrac{\lambda}{1+\lambda}\cdot\dfrac{1}{|\varepsilon_{q_i,p_i}|}$
\end{formula}
  \begin{itemize}
  \item \autor{Lerner} \emph{amortiguado}: $\lambda=0$ $\Rightarrow$ $p=\mathrm{CMg}$; $\lambda\to\infty$ $\Rightarrow$ monopolio (cánones de ADIF).
  \end{itemize}
\end{itemize}
\subsub{c) Valoración}
\begin{itemize}
\item Exige conocer costes y elasticidades, y benevolencia; la empresa declara costes altos.
\item Con otras distorsiones, \emph{second best} (\autor{Lipsey}-\autor{Lancaster}, 1956).
\end{itemize}

\sub{II.3. Competencia por el mercado e impugnabilidad\label{rep:II.3}}
\begin{itemize}
\item \autor{Demsetz} (1968): subastar el derecho exclusivo al menor precio $\Rightarrow$ $p=\mathrm{CMe}$ sin conocer costes.
  \begin{itemize}
  \item La subasta revela la información \vertema{3.A.14}; subasta + incentivos (\autor{Laffont}-\autor{Tirole}, 1987).
  \end{itemize}
\item Límites (\autor{Williamson}, 1976):
  \begin{itemize}
  \item colusión; activos específicos y contratos incompletos; ventaja del titular al renovar;
  \item duración (corta: no invierte; larga: sin competencia); renegociación (\autor{Guasch} et al., 2008).
  \end{itemize}
\item \concepto{Mercados perfectamente impugnables} (\autor{Baumol}-\autor{Panzar}-\autor{Willig}, 1982) \vertema{3.A.15}:
  \begin{itemize}
  \item misma tecnología, sin \emph{costes hundidos}, reacción lenta $\Rightarrow$ \emph{hit and run} $\Rightarrow$ $p=\mathrm{CMe}$;
  \item las redes tienen costes hundidos: no son impugnables.
  \end{itemize}
\end{itemize}

\sub{II.4. Liberalización e industrias de red\label{rep:II.4}}
\subsub{Liberalización y re-regulación}
\begin{itemize}
\item \concepto{Liberalizar} (quitar barreras legales) $\neq$ \concepto{desregular}: \emph{re-regulación}.
\item \concepto{Industria de red}: activo esencial no competitivo + actividades competitivas.
\item 4~problemas:
  \begin{itemize}
  \item \concepto{costes de transición a la competencia} (España: Ley 54/1997, $\approx$11.951~M€; suprimidos en 2006);
  \item \emph{precio de acceso} (cuello de botella);
  \item inversión en la red; \emph{servicio universal} \vertema{4.A.12}.
  \end{itemize}
\end{itemize}
\subsub{Regla del precio eficiente de los componentes}
\begin{itemize}
\item \concepto{ECPR} (\autor{Willig}, 1979; \autor{Baumol}-\autor{Sidak}, 1994): peaje = coste del acceso + margen perdido:
\begin{formula}
\centering $A=c_a+\big[p-(c_a+c_I)\big]=p-c_I\ \Rightarrow\ \text{entra si } c_E\leq c_I$
\end{formula}
  \begin{itemize}
  \item Como techo: \emph{estrechamiento de márgenes} (Telefónica).
  \item Perpetúa un $p$ de monopolio (\autor{Armstrong}-\autor{Doyle}-\autor{Vickers}, 1996).
  \end{itemize}
\end{itemize}
\subsub{Acceso regulado y separación vertical}
\begin{itemize}
\item 6~opciones (OCDE, 2001):
  \begin{itemize}
  \item \concepto{acceso regulado} (Telefónica, MARCo);
  \item \concepto{separación de la propiedad} (ADIF/Renfe, Aena);
  \item \concepto{propiedad compartida} del activo \emph{no} competitivo (CLH, hoy Exolum);
  \item \concepto{separación operativa} (EE.\,UU.); \concepto{partes recíprocas} (móviles); \concepto{partes más pequeñas} (distribuidoras).
  \end{itemize}
\end{itemize}
\graficorepaso[0.72\linewidth]{separacion-vertical}{Acceso regulado y formas de separación vertical (ANC: activo no competitivo; AC: actividad competitiva)}
\begin{itemize}
\item \concepto{Plataforma de 2~lados} (\autor{Rochet}-\autor{Tirole}, 2003): precio nulo en un lado sin depredación; el \autor{Lerner} de un lado dice poco.
\end{itemize}

\bloque[11]{III. Información asimétrica y aplicaciones prácticas\label{rep:III}}
\begin{ideaclave}
Sin información no hay primer óptimo: para que la eficiente no se haga pasar por ineficiente hay que dejarle una renta informativa; la regulación óptima equilibra incentivos y extracción de rentas.
\end{ideaclave}

\sub{III.1. Problema de información y tasa de rentabilidad\label{rep:III.1}}
\begin{itemize}
\item Agencia: regulador principal, empresa agente \vertema{3.A.13}.
  \begin{itemize}
  \item \concepto{Selección adversa} (tipo de coste) y \concepto{riesgo moral} (esfuerzo).
  \item \concepto{Renta informativa}; principio de revelación, $\mathrm{PC}_i$ e $\mathrm{IC}_i$ \vertema{3.A.14}; \emph{terceros óptimos}.
  \end{itemize}
\end{itemize}
\subsub{Tasa de rentabilidad: efecto \autor{Averch}-\autor{Johnson}}
\begin{itemize}
\item \concepto{Tasa de rentabilidad}: costes + rentabilidad «justa» sobre la base de activos.
\item \autor{Averch}-\autor{Johnson} (1962): con $s>w_K$, $B\leq(s-w_K)\,z_K$: \emph{el beneficio permitido crece con el capital}; con $0<\mu<1$:
\begin{formula}
\centering $\mathrm{RMST}_{LK}=\dfrac{w_L}{w_K-\frac{\mu}{1-\mu}(s-w_K)}>\dfrac{w_L}{w_K}$
\end{formula}
\end{itemize}
\graficorepaso[0.42\linewidth]{averch-johnson}{\autor{Averch}-\autor{Johnson}: $z_K'>z_K^*$}
\begin{itemize}
\item Sobrecapitaliza: \emph{ineficiencia productiva}, baja potencia.
  \begin{itemize}
  \item Redes del Valle: cambiar el transformador antes que mantenerlo.
  \item \emph{Desfase regulatorio}: algo de incentivo; evidencia, mixta (\autor{Spann}, 1974).
  \end{itemize}
\item \emph{Valoración}:
  \begin{itemize}
  \item (+) poca información (datos contables), seguridad para invertir;
  \item ($-$) sobrecapitaliza, beneficio contable $\neq$ económico, sin incentivo a abaratar, exagera costes en cada revisión;
  \item corrección: TOTEX (CNMC, 8/2025).
  \end{itemize}
\end{itemize}

\sub{III.2. \autor{Loeb}-\autor{Magat} y \autor{Baron}-\autor{Myerson}\label{rep:III.2}}
\begin{itemize}
\item \autor{Loeb}-\autor{Magat} (1979): subvención = excedente del consumidor $\Rightarrow$ beneficio $=S(q)-C(q)$ $\Rightarrow$ $p=\mathrm{CMg}$ sin revelar costes (\autor{Groves}, estrategia dominante) \vertema{3.A.14}.
  \begin{itemize}
  \item Pero conoce la demanda y la empresa se queda todo el excedente: el problema son las rentas.
  \end{itemize}
\item \autor{Baron}-\autor{Myerson} (1982): coste $\underline{\beta}$ (prob. $\nu$) u $\overline{\beta}$; peso del beneficio $\alpha<1$. Activas $\mathrm{IC}_1$ y $\mathrm{PC}_2$:
\begin{formula}
\centering $\underline{U}=\Delta\beta\,\overline{q};\ p(\underline{q})=\underline{\beta},\ p(\overline{q})=\overline{\beta}+(1-\alpha)\dfrac{\nu}{1-\nu}\Delta\beta$
\end{formula}
  \begin{itemize}
  \item \emph{No hay distorsión arriba}; la ineficiente produce menos para recortar la renta.
  \item Distorsión ↑ con $\nu$ y $\Delta\beta$; $\alpha=1$ la elimina: eficiencia asignativa a cambio de rentas \vertema{3.A.17}.
  \item Redes del Valle ($\beta=20/40$, $\nu=\tfrac12$, $p=100-q$, $\alpha=0$): 80~y 40 (no 60); renta 800 (no 1.200).
  \item Cierre de la ineficiente si $\nu\geq3/4$; continuo: \emph{coste virtual}.
  \end{itemize}
\end{itemize}
\graficorepaso[0.45\linewidth]{baron-myerson}{\autor{Baron}-\autor{Myerson}: sin distorsión arriba}

\sub{III.3. Menú de \autor{Laffont}-\autor{Tirole}\label{rep:III.3}}
\subsub{a) Idea y b) modelo}
\begin{itemize}
\item \autor{Laffont}-\autor{Tirole} (1986, 1993): $C=\beta-e$; $\beta$ oculto (selección adversa), $e$ oculto con desutilidad $\psi(e)$ (riesgo moral); observa $C$; fondos a $1+\lambda$.
\item Información completa: $\psi'(e^*)=1$. La eficiente imita esforzándose $\Delta\beta$ menos $\Rightarrow$ renta $\Phi(\overline{e})=\psi(\overline{e})-\psi(\overline{e}-\Delta\beta)$:
\funcionclave{\psi'(\underline{e})=1,\quad \psi'(\overline{e})=1-\frac{\lambda}{1+\lambda}\cdot\frac{\nu}{1-\nu}\cdot\Phi'(\overline{e})<1}
  \begin{itemize}
  \item Menos esfuerzo a la ineficiente: \concepto{incentivos vs extracción de rentas}; «\emph{el que algo quiere, algo le cuesta}».
  \item Con $\lambda=0$ desaparece.
  \end{itemize}
\item \concepto{Menú lineal} $t(C)=a-b\,(C-C^a)$ $\Rightarrow$ $\psi'(e)=b$:
  \begin{itemize}
  \item $b=1$: \concepto{precio fijo} (eficiente, alta potencia);
  \item $b\to0$: \concepto{coste incrementado} (baja potencia).
  \end{itemize}
\end{itemize}

\sub{III.4. Precios máximos y esquemas intermedios\label{rep:III.4}}
\subsub{a) Idea y b) modelo}
\begin{itemize}
\item \concepto{Price cap} (\autor{Littlechild}, 1983, privatización de BT; RPI$-$3~en 1984): desliga precio y costes declarados; alta potencia ($b=1$).
\item $X$ = mejora de productividad exigida en el periodo:
\funcionclave{p_{t+1}\leq p_t\,(1+\mathrm{IPC}_t-X)}
  \begin{itemize}
  \item Lo ahorrado se lo queda hasta la revisión; periodo más largo: más incentivo, menos traslado.
  \item Cesta $\Rightarrow$ tipo \autor{Ramsey} (\autor{Vogelsang}-\autor{Finsinger}, 1979); no distorsiona factores.
  \item Redes del Valle: 40 €/MWh, IPC 2,5\,\%, $X=1{,}5$ $\Rightarrow$ $\leq40{,}40$.
  \end{itemize}
\end{itemize}
\begin{itemize}
\item \concepto{Revenue cap} (RIIO de Ofgem); \concepto{reparto de beneficios} o escala móvil: $p=b\,F+(1-b)\,C$; incentivos de calidad.
\item España: cuota de abono de Telefónica ($\mathrm{IPC}-X$ hasta 2012); Aena, DORA ($X$ se \emph{suma}).
\item \emph{Valoración}:
  \begin{itemize}
  \item (+) incentivos, traslado \emph{a priori} y \emph{a posteriori}, sencillo.
  \item ($-$) información para $p_0$ y $X$, retrasar mejoras, calidad e inversión, más riesgo; recalibrado a costes = coste del servicio con desfase.
  \end{itemize}
\end{itemize}

\sub{III.5. Competencia por comparación\label{rep:III.5}}
\begin{itemize}
\item \concepto{Yardstick competition} (\autor{Shleifer}, 1985): precio según los costes de las \emph{demás}; «\emph{las comparaciones son odiosas}».
\item $N$ idénticas, inversión en reducir costes $R(c_i)$, $R'<0$:
\begin{formula}
\centering $p_i=\bar{c}_{-i}=\dfrac{1}{N-1}\sum_{j\neq i}c_j,\ T_i=\bar{R}_{-i}\Rightarrow -R'(c_i)=q(\bar{c}_{-i})$
\end{formula}
  \begin{itemize}
  \item Equilibrio simétrico: coste eficiente, $p=c$: $b=1$ sin rentas.
  \item Costes 20, 22, 24 $\Rightarrow$ precios 23, 22, 21.
  \end{itemize}
\item \emph{Extensiones}: heterogeneidad observable: regresión o fronteras (DEA, SFA) \vertema{3.A.12}; filtra choques comunes; Alemania.
\item \emph{Requisitos}: comparables; riesgo de \emph{colusión} \vertema{3.A.19}.
\end{itemize}

\sub{III.6. Compromiso y captura\label{rep:III.6}}
\subsub{Compromiso}
\begin{itemize}
\item \concepto{Efecto trinquete} (\autor{Weitzman}, 1980): coste bajo hoy = exigencia mayor mañana.
\item \emph{Inconsistencia temporal} (\autor{Kydland}-\autor{Prescott}, 1977): red hundida $\Rightarrow$ ↓inversión; antídoto: reglas previas, independencia.
\end{itemize}
\subsub{Captura: la teoría positiva}
\begin{itemize}
\item \autor{Bernstein} (1955), ciclo de vida; \autor{Stigler} (1971, Nobel 1982): la industria adquiere la regulación; «\emph{poner al zorro a cuidar de las gallinas}».
\item \autor{Olson} (1965): los grupos pequeños se organizan \vertema{4.B.2}.
\item \autor{Peltzman} (1976): máx. apoyo $\mathcal{A}(p,B(p))$:
\begin{formula}
\centering $\dfrac{\partial\mathcal{A}}{\partial p}+\dfrac{\partial\mathcal{A}}{\partial B}B'(p)=0\Rightarrow B'(p)>0\Rightarrow p^c<p^*<p^M$
\end{formula}
\item \autor{Becker} (1983): grupos de presión. \autor{Laffont}-\autor{Tirole} (1991): captura = colusión; con riesgo alto, \emph{baja potencia}; puertas giratorias (\autor{Dal Bó}, 2006).
\item \emph{Valoración}: comparar fallos de mercado y de regulación \vertema{3.A.23}; justifica independencia y transparencia; difícil de contrastar.
\end{itemize}

\sub{III.7. Aplicaciones prácticas\label{rep:III.7}}
\subsub{Redes eléctricas y gasistas en España}
\begin{itemize}
\item Tasa de retribución + incentivos TOTEX: la tasa asegura la inversión; los incentivos corrigen \autor{Averch}-\autor{Johnson}.
\item \emph{Excepción ibérica} (2022-2023): tope al gas de las centrales marginales, contra el contagio del coste marginal (no el poder de mercado); temporal.
\end{itemize}
\subsub{Telecomunicaciones}
\begin{itemize}
\item Código Europeo (2018): mercados, 3~criterios, \emph{peso significativo}; NEBA suprimida (2025): se regula mientras hay cuello de botella.
\end{itemize}
\subsub{Mercados digitales}
\begin{itemize}
\item \concepto{DMA} (2022, aplicable desde 2023): \emph{guardianes de acceso} por umbrales.
  \begin{itemize}
  \item No autopreferencia, portabilidad, interoperabilidad; multas hasta 10\,\% (20\,\% si reincide).
  \item Regular sin conocer costes; la plataforma sigue sabiendo más.
  \end{itemize}
\end{itemize}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}:
  \begin{itemize}
  \item poder de mercado: \autor{Lerner}/IHH en un mercado relevante (SSNIP); conductas, a la defensa de la competencia;
  \item información completa: $p=\mathrm{CMg}$ con subvención o segundos óptimos, subasta, acceso;
  \item asimétrica: renta informativa, incentivos vs rentas (tasa de rentabilidad y \emph{price cap}, extremos del menú); compromiso y captura deciden.
  \end{itemize}
\item \textbf{Relevancia}: descarbonizar exige ampliar las redes: reglas estables y creíbles; en lo digital, cuellos de botella de datos y plataformas.
\item \textbf{Extensiones}: \vertema{4.B.15} \vertema{3.B.39} \vertema{4.A.15} \vertema{4.A.5} \vertema{4.A.12} \vertema{3.A.13} \vertema{3.A.14} \vertema{4.B.2}.
\item \textbf{Opinión}: no hay instrumento óptimo general; más potencia cuanto más compromiso e independencia.
  \begin{itemize}
  \item \autor{Kahn}: «la mejor combinación posible de una regulación inevitablemente imperfecta y una competencia inevitablemente imperfecta».
  \end{itemize}
\item \textbf{Idea final}: ni tanto ni tan calvo, pero el término medio lo conoce la empresa: la regulación óptima le paga lo justo para que revele sus costes, la renta informativa.
\end{itemize}

\begin{enumerate}
\item \textbf{¿Por qué no basta \autor{Lerner}?} CMg no observable; márgenes que cubren fijos o esconden ineficiencia X.
\item \textbf{¿Falacia del celofán?} SSNIP a precios de monopolio: mercado demasiado amplio.

\item \textbf{¿Defensa de la competencia o regulación?} Conductas: competencia, \emph{ex post}; subaditividad: regulación \emph{ex ante}.
\item \textbf{¿Por qué la DMA designa por umbrales?} Para regular sin conocer costes ni demandas.
\end{enumerate}
