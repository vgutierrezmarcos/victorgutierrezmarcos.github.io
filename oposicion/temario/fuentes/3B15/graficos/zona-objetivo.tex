% Las zonas objetivo de Krugman (1991): luna de miel y divorcio.
% Fundamentos f = m + v en abscisas, tipo de cambio e en ordenadas; la paridad
% central, normalizada en el origen. Escala: 1 unidad = 1,4 cm en los 2 ejes
% (la recta de 45° se ve a 45°).
%   Banda creíble: g(f) = f + A(e^{λf} - e^{-λf}), λ = 1,5, f̄ = 2,
%   A = -1/(λ(e^{λf̄}+e^{-λf̄})) = -0,033109 (smooth pasting: g'(±f̄) = 0);
%   bordes e^max = g(f̄) = 1,3366 y e^min = -1,3366.
%   Sin credibilidad (Bertola y Caballero, 1992): h(f) = f + 0,05(e^{3f}-e^{-3f}),
%   pendiente 1,3 en el centro, toca los bordes en f = ±0,7962 (antes que ±f̄).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=1.4cm,y=1.4cm,
    declare function={g(\f)=\f-0.033109*(exp(1.5*\f)-exp(-1.5*\f));
      h(\f)=\f+0.05*(exp(3*\f)-exp(-3*\f));}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-3.3,-2.3) rectangle (4.0,2.3);
  % 1: ejes, con la paridad central en el origen
  \capa{1}{\draw[eje] (-2.75,0) -- (2.75,0) node[etiqueta,right,align=left] {fundamentos,\\$m+v$};}
  \capa{1}{\draw[eje] (0,-2.15) -- (0,2.25) node[etiqueta,left] {$e$};}
  \capa{1}{\node[punto] at (0,0) {};}
  \capa{1}{\node[etiqueta,font=\footnotesize,anchor=north west] at (0.02,-0.02) {paridad central};}
  % 2: sin banda, el tipo de cambio sigue a los fundamentos (45°)
  \capa{2}{\draw[curva2] (-1.75,-1.75) -- (1.75,1.75);}
  \capa{2}{\node[etiqueta,font=\footnotesize,text=tceeazul,anchor=west] at (1.78,1.78) {sin banda: $\tilde e=m+v$};}
  % 3: los bordes de la banda
  \capa{3}{\draw[guia,thick] (-2.6,1.3366) -- (2.6,1.3366) node[etiqueta,right,text=black] {$e^{\max}$};}
  \capa{3}{\draw[guia,thick] (-2.6,-1.3366) node[etiqueta,left,text=black] {$e^{\min}$} -- (2.6,-1.3366);}
  % 4: banda creíble: S aplanada, tangente a los bordes en ±f̄ (luna de miel)
  \capa{4}{\draw[curva1] plot[domain=-2:2,samples=100] (\x,{g(\x)});}
  \capa{4}{\node[etiqueta,font=\footnotesize,text=tceemorado,align=left,anchor=north west] at (1.05,0.95) {banda creíble:\\luna de miel};}
  % 5: los fundamentos en los que se interviene, ±f̄
  \capa{5}{\node[punto] at (2,1.3366) {};}
  \capa{5}{\node[punto] at (-2,-1.3366) {};}
  \capa{5}{\draw[guia] (2,1.3366) -- (2,0) node[etiqueta,below,text=black] {$\bar f$};}
  \capa{5}{\draw[guia] (-2,-1.3366) -- (-2,0) node[etiqueta,above,text=black] {$-\bar f$};}
  % 6: sin credibilidad: S invertida, más inclinada que la de 45° (divorcio)
  \capa{6}{\draw[curvanueva,dashed] plot[domain=-0.7962:0.7962,samples=80] (\x,{h(\x)});}
  \capa{6}{\node[etiqueta,font=\footnotesize,text=tceerojo,align=right,anchor=south east] at (-0.06,1.38) {sin credibilidad:\\divorcio};}
\end{tikzpicture}
\end{document}
