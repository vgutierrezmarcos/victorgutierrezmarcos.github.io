% Un impuesto sobre el endeudamiento exterior en el modelo intertemporal de
% 2 periodos con dotación (Obstfeld y Rogoff, 1996, cap. 1; Schmitt-Grohé,
% Uribe y Woodford, 2022). Mismos ejes, escala y estilos que
% fuentes/3B12/graficos/cc-intertemporal.tex, sin producción.
%   Utilidad ln C_t + β ln C_{t+1}, β = 0,5: curvas de indiferencia
%   C_{t+1} = K / C_t^2; RMS = C_{t+1}/(β C_t).
%   Dotación (y autarquía) A = (3; 5); U_0: K = 45.
%   Recta nacional: C_{t+1} = 8,15 - 1,05 C_t (1 + r* = 1,05).
%   Libertad de capitales: C = (5,175; 2,717), RMS = 1,05; U_1: K = 72,77.
%   Impuesto: 1 + r* + τ = 1,876; D = (4,1; 3,845) sobre la recta nacional,
%   RMS = 1,876; U_2: K = 64,63. U_0 < U_2 < U_1.
%   Déficit corriente en t (sin impuesto): C_t^C - Y_t = 2,175.
% Capas: 1 ejes; 2 autarquía; 3-5 libertad de capitales; 6-8 impuesto.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,scale=0.85,
    declare function={i0(\x)=45/(\x*\x); i1(\x)=72.77/(\x*\x);
      i2(\x)=64.63/(\x*\x); nacional(\x)=8.15-1.05*\x;
      privada(\x)=3.845-1.876*(\x-4.1);}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-1.6,-1.9) rectangle (10.2,9.1);
  % 1: ejes: consumo de hoy y de mañana
  \capa{1}{\ejes{$C_t$}{$C_{t+1}$}{9.2}{9}}
  % 2: la dotación A, que en autarquía se consume, y su curva de indiferencia
  \capa{2}{\draw[thick,tceegris!70] plot[domain=2.75:6.2,samples=80] (\x,{i0(\x)}) node[etiqueta,right,text=tceegris] {$U_0$};}
  \capa{2}{\node[punto] (A) at (3,5) {};}
  \capa{2}{\node[etiqueta,below left] at (A) {$A$};}
  \capa{2}{\draw[guia] (0,5) node[left,text=black] {$Y_{t+1}$} -| (3,0) node[below,text=black] {$Y_t$};}
  % 3: libertad de capitales: la recta nacional por A, pendiente -(1+r*)
  \capa{3}{\draw[curva2] (0,8.15) -- (7.762,0);}
  \capa{3}{\node[etiqueta,font=\footnotesize,text=tceeazul,anchor=west] at (0.15,8.55) {pendiente $-(1+r^*)$};}
  % 4: el consumo con libertad de capitales: tangencia de U_1 en C
  \capa{4}{\draw[curva3] plot[domain=3.35:9.0,samples=80] (\x,{i1(\x)}) node[etiqueta,right] {$U_1$};}
  \capa{4}{\node[punto] (C) at (5.175,2.717) {};}
  \capa{4}{\node[etiqueta,above right] at (C) {$C$};}
  \capa{4}{\draw[guia] (0,2.717) node[left,text=black] {$C_{t+1}^C$} -| (5.175,0) node[below,text=black] {$C_t^C$};}
  % 5: el déficit corriente de t
  \capa{5}{\draw[decorate,decoration={brace,mirror,amplitude=4pt},thick,tceerojo] (3,-0.75) -- (5.175,-0.75) node[midway,below=4pt,etiqueta,text=tceerojo,font=\footnotesize] {déficit corriente en $t$};}
  % 6: con el impuesto, el país sigue sobre la recta nacional: punto D
  \capa{6}{\node[punto] (D) at (4.1,3.845) {};}
  \capa{6}{\node[etiqueta,above right] at (D) {$D$};}
  \capa{6}{\draw[guia] (4.1,3.845) -- (4.1,0) node[below,text=black] {$C_t^D$};}
  % 7: la recta privada por D, pendiente -(1+r*+τ), tangente a U_2 en D
  \capa{7}{\draw[curvanueva,dashed] (2.6,{privada(2.6)}) -- (5.55,{privada(5.55)});}
  \capa{7}{\node[etiqueta,font=\footnotesize,text=tceerojo,anchor=south,align=center] at (2.6,{privada(2.6)+0.05}) {pendiente\\$-(1+r^*+\tau)$};}
  \capa{7}{\draw[curva4] plot[domain=3.15:7.6,samples=80] (\x,{i2(\x)}) node[etiqueta,anchor=north west,inner sep=1pt] {$U_2$};}
  % 8: el bienestar
  \capa{8}{\node[etiqueta,anchor=north east] at (9.6,8.6) {$U_0<U_2<U_1$};}
