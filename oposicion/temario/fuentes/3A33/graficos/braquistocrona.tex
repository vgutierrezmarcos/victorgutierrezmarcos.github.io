% La braquistócrona: la curva de descenso más rápido entre A y B es un arco de
% cicloide, no la recta. Elaboración propia (rehace un GIF sin fuente).
%   A = (0, 2); cicloide x = a (th - sen th), y = 2 - a (1 - cos th), th en [0; 4,2]
%   con a = 0,7887 llega a B = (4; 0,825) y baja hasta y = 0,423 (th = pi,
%   x = 2,478), por debajo de B, antes de subir a él
%   bolitas en el mismo instante: la de la cicloide, más adelantada
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,
    declare function={cx(\t)=0.7887*(\t-sin(\t r)); cy(\t)=2-0.7887*(1-cos(\t r));}]
  % 1: los puntos A y B
  \capa{1}{\node[punto] (A) at (0,2) {};}
  \capa{1}{\node[etiqueta,left] at (A) {$A$};}
  \capa{1}{\node[punto] (B) at (4,0.825) {};}
  \capa{1}{\node[etiqueta,right] at (B) {$B$};}
  % 2: la recta entre A y B
  \capa{2}{\draw[guia,thick] (A) -- (B);}
  \capa{2}{\node[etiqueta,font=\footnotesize,text=tceegris,above right] at (2.1,1.45) {recta};}
  % 3: la cicloide, la braquistócrona
  \capa{3}{\draw[curva1] plot[domain=0:4.2,samples=80] ({cx(\x)},{cy(\x)});}
  \capa{3}{\node[etiqueta,font=\footnotesize,text=tceemorado,below] at (2.48,0.42) {braquistócrona (cicloide)};}
  % 4: bolitas en el mismo instante y la gravedad
  \capa{4}{\fill[tceemorado] ({cx(2.1)},{cy(2.1)}) circle (0.09);}
  \capa{4}{\fill[tceegris] (1.15,{2-1.175*1.15/4}) circle (0.09);}
  \capa{4}{\draw[->,thick] (4.6,2.0) -- (4.6,1.3) node[right,midway] {$g$};}
\end{tikzpicture}
\end{document}
