% Crecimiento semiendógeno (Jones, 1995): tasa de crecimiento de las ideas,
% gamma_N, frente al número de ideas, N.
% Romer (phi = 1): horizontal gamma_N = eta_I L_I = 2,6; con más investigadores,
%   eta_I L_I' = 3,2 (sube la tasa: efecto escala).
% Jones (phi < 1; aquí phi = 1/2): gamma_N = eta_I L_I^lambda N^(phi-1) = a/sqrt(N),
%   a = 1,2 sqrt(3) = 2,0785, que corta a la senda gamma_N* = lambda n/(1-phi) = 1,2
%   en N* = 3; con más investigadores, a' = 1,3 a = 2,7020 y N*' = (2,702/1,2)^2 = 5,070.
%   La curva empieza por debajo de las horizontales de Romer (N = 0,9 -> 2,19;
%   N = 1,3 -> 2,37) para no cruzarlas.
% Escala: x = 0,8 cm, y = 1,2 cm.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.8cm,y=1.2cm,
    declare function={ja(\n)=2.0785/sqrt(\n); jb(\n)=2.7020/sqrt(\n);},
    dinamica/.style={->,thick,black}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-2.6,-0.55) rectangle (12.3,3.95);
  % 1: ejes
  \capa{1}{\ejes{$N$}{$\gamma_N$}{10.2}{3.8}}
  % 2: Romer (phi = 1): horizontal eta_I L_I
  \capa{2}{\draw[curva2] (0,2.6) node[left,text=black] {$\eta_I L_I$} -- (9.6,2.6)
    node[etiqueta,above left,text=tceeazul] {Romer ($\phi=1$)};}
  % 3: más investigadores: en Romer sube la tasa (efecto escala)
  \capa{3}{\draw[curvanueva] (0,3.2) node[left,text=black] {$\eta_I L_I'$} -- (9.6,3.2)
    node[etiqueta,above left,text=tceerojo] {$L_I'>L_I$: sube la tasa (efecto escala)};}
  % 4: Jones (phi < 1): curva decreciente eta_I L_I^lambda N^(phi-1)
  \capa{4}{\draw[curva1] plot[domain=0.9:9.2,samples=100] (\x,{ja(\x)})
    node[etiqueta,anchor=north west,text=tceemorado] {Jones ($\phi<1$)};}
  % 5: la senda gamma_N* = lambda n/(1-phi), su corte N* y la dinámica hacia ella
  \capa{5}{\draw[curva3] (0,1.2) node[left,text=black] {$\dfrac{\lambda n}{1-\phi}$} -- (9.6,1.2);
    \node[punto] (E) at (3,1.2) {};
    \draw[guia] (E) -- (3,0) node[below,text=black] {$N^*$};
    \draw[dinamica] plot[domain=1.15:2.0,samples=20] (\x,{ja(\x)-0.2});
    \draw[dinamica] plot[domain=6.2:4.0,samples=20] (\x,{ja(\x)-0.2});}
  % 6: al crecer L_I la curva se desplaza a la derecha; misma tasa, más ideas
  \capa{6}{\draw[curvanueva] plot[domain=1.3:9.2,samples=100] (\x,{jb(\x)})
      node[etiqueta,anchor=west,text=tceerojo] {$L_I'$};
    \draw[flecha] (1.3,1.9) -- (1.95,1.9);
    \node[punto] (E2) at (5.07,1.2) {};
    \draw[guia] (E2) -- (5.07,0) node[below,text=black] {$N^{*\prime}$};
    \draw[flecha] (3.15,0.18) -- (4.9,0.18);}
  % 7: rótulos: la senda y el efecto nivel
  \capa{7}{\node[etiqueta,anchor=south west,text=tceeverde] at (5.6,1.2)
      {senda: $\gamma_N^*=\lambda n/(1-\phi)$};
    \node[etiqueta,anchor=south west,text=tceerojo] at (5.25,0.02)
      {efecto nivel};}
\end{tikzpicture}
\end{document}
