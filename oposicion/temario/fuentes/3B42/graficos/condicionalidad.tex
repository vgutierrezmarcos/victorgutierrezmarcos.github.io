% La condicionalidad de la adhesión: un candidato reforma si el coste interno
% de la reforma C es menor que el beneficio descontado de la adhesión,
% C < delta^T B p, con p la credibilidad de la promesa de adhesión y T el plazo
% hasta adherirse. Escala: pendiente delta^{T_1} B = 4 (adhesión cercana),
% delta^{T_2} B = 2 (adhesión lejana, T_2 > T_1) y recompensas intermedias
% g = 1,2. Candidato X: p_X = 0,6 y C_X = 1,8, que reforma con T_1
% (1,8 < 2,4), no con T_2 (1,8 > 1,2) y otra vez con T_2 y g (1,8 < 2,4).
% Capas: ejes → rayo con T_1 y zona de reforma → rayo con T_2 → rayo
% desplazado por g → candidato X.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=6cm,y=1cm]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-0.12,-0.7) rectangle (1.95,5.5);
  % 1: ejes, el 0 y la marca de p = 1
  \capa{1}{\draw[eje] (0,0) -- (1.25,0) node[right] {$p$ (credibilidad)};}
  \capa{1}{\draw[eje] (0,0) -- (0,5.3) node[left] {$C$ (coste interno)};}
  \capa{1}{\node[below left] at (0,0) {$0$};}
  \capa{1}{\draw (1,0.08) -- (1,-0.08) node[below] {$1$};}
  % 2: rayo con adhesión cercana y zona en que el candidato reforma
  \capa{2}{\fill[area] (0,0) -- (1,0) -- (1,4) -- cycle;}
  \capa{2}{\draw[curva1] (0,0) -- (1,4) node[etiqueta,right,text=tceemorado] {$C=\delta^{T_1}B\,p$};}
  \capa{2}{\node[etiqueta,text=tceemorado,align=center] at (0.8,0.75) {reforma\\[-1pt]{\footnotesize adhesión cercana ($T_1$ bajo)}};}
  % 3: adhesión lejana (fatiga de la ampliación): el rayo gira hacia abajo
  \capa{3}{\draw[desplazada,tceegris] (0,0) -- (1,2) node[etiqueta,right,text=tceegris,align=left] {$\delta^{T_2}B\,p$: fatiga\\de la ampliación};}
  % 4: recompensas intermedias: el rayo con T_2 sube en g
  \capa{4}{\draw[curvanueva] (0,1.2) -- (1,3.2) node[etiqueta,right,text=tceerojo,align=left] {$g+\delta^{T_2}B\,p$: recompensas\\intermedias};}
  % 5: el candidato X
  \capa{5}{\proyeccion{(0.6,1.8)}{$p_X$}{$C_X$}}
  \capa{5}{\node[punto] at (0.6,1.8) {};}
  \capa{5}{\node[etiqueta,above left] at (0.6,1.8) {$X$};}
\end{tikzpicture}
\end{document}
