% Isocuantas CES con pesos iguales que pasan por (1,1):
% z_L^rho + z_K^rho = 2, con rho = (sigma-1)/sigma (adaptado de 3A08/ces-curvas-indiferencia).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,scale=1.7]
  % 1: ejes, punto común y rayo z_K/z_L = a/b
  \capa{1}{\ejes{$z_L$}{$z_K$}{3.6}{3.4}}
  \capa{1}{\foreach \t in {1,2,3} {
    \draw (\t,0) -- (\t,-0.05) node[below] {$\t$};
    \draw (0,\t) -- (-0.05,\t) node[left] {$\t$};
  }}
  \capa{1}{\draw[guia] (0,0) -- (3.1,3.1) node[etiqueta,above,text=tceegris] {$z_K/z_L=a/b$};}
  \capa{1}{\node[punto] at (1,1) {};}
  % 2: sigma = 0 (Leontief), vértice en el rayo
  \capa{2}{\draw[curva1] (1,3) -- (1,1) -- (3,1) node[etiqueta,right] {$\sigma=0$};}
  % 3: sigma intermedios: 0,5 (rho = -1) y 5 (rho = 0,8)
  \capa{3}{\draw[curva1,desplazada] plot[domain=0.6:3,samples=80,smooth] (\x,{\x/(2*\x-1)})
      node[etiqueta,right] {$\sigma=0{,}5$};}
  \capa{3}{\draw[curva2,desplazada] plot[domain=0.001:2.377,samples=80,smooth] (\x,{(2-\x^0.8)^1.25});}
  \capa{3}{\node[etiqueta,anchor=west] at (2.4,0.1) {$\sigma=5$};}
  % 4: sigma = 1 (Cobb-Douglas): z_L z_K = 1
  \capa{4}{\draw[curva3] plot[domain=0.3333:3,samples=80,smooth] (\x,{1/\x})
      node[etiqueta,right] {$\sigma=1$};}
  % 5: sigma = infinito (lineal)
  \capa{5}{\draw[curva2] (0,2) -- (2,0);}
  \capa{5}{\node[etiqueta,text=tceeazul] at (0.38,1.22) {$\sigma=\infty$};}
\end{tikzpicture}
\end{document}
