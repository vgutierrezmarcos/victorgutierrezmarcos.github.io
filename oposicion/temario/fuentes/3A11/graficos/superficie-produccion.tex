% Superficie de producción q = z_K^(1/2) z_L^(1/2) en la vista por defecto de pgfplots
% (acimut 30°, elevación 28°; origen a la izquierda, z_L a la derecha, z_K en profundidad):
% cortes horizontales (isocuantas), corte con z_K = 3 (corto plazo) y corte por
% el rayo z_K = z_L/2 (largo plazo: recta, rendimientos constantes).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\pgfplotsset{
  superficie/.style={surf,shader=faceted,fill opacity=0.45,z buffer=sort,draw=tceemorado!40,line width=0.15pt,
    colormap={tcee}{rgb255=(226,239,217) rgb255=(167,137,189)}},
  corte/.style={curvanueva,line width=1.1pt},
  proyeccion/.style={curva1,line width=1.1pt},
}
\begin{document}
\begin{tikzpicture}[tcee]
  % vista por defecto de pgfplots (acimut 30°, elevación 28°), escalada a 0,85
  \begin{axis}[x={(1.02cm,-0.28cm)},y={(0.587cm,0.476cm)},z={(0cm,1.037cm)},axis lines=none,
      xmin=0,xmax=4.4,ymin=0,ymax=4.4,zmin=0,zmax=4.4,clip=false]
    % 1: ejes
    \capa{1}{\draw[eje] (axis cs:0,0,0) -- (axis cs:4.7,0,0) node[below] {$z_L$};}
    \capa{1}{\draw[eje] (axis cs:0,0,0) -- (axis cs:0,5.2,0) coordinate (zK);}
    \capa{1}{\draw[eje] (axis cs:0,0,0) -- (axis cs:0,0,4.4) node[above] {$q$};}
    % 2: superficie de producción
    \capa{2}{\addplot3[superficie,domain=0:4,y domain=0:4,samples=9,samples y=9] {sqrt(x*y)};}
    \capa{2}{\node[etiqueta,anchor=west] at (axis cs:4.05,4,4) {$q=z_K^{1/2}z_L^{1/2}$};}
    % 3: cortes a q = 1, 2, 3 y su proyección en la base: el mapa de isocuantas
    \capa{3}{\addplot3[corte,domain=0.25:4,samples=40,samples y=0] (x,{1/x},1);}
    \capa{3}{\addplot3[proyeccion,domain=0.25:4,samples=40,samples y=0] (x,{1/x},0);}
    \capa{3}{\addplot3[corte,domain=1:4,samples=40,samples y=0] (x,{4/x},2);}
    \capa{3}{\addplot3[proyeccion,domain=1:4,samples=40,samples y=0] (x,{4/x},0);}
    \capa{3}{\addplot3[corte,domain=2.25:4,samples=40,samples y=0] (x,{9/x},3);}
    \capa{3}{\addplot3[proyeccion,domain=2.25:4,samples=40,samples y=0] (x,{9/x},0);}
    \capa{3}{\node[etiqueta,text=tceerojo,anchor=west] at (axis cs:4.05,0.25,1) {$q=1$};}
    \capa{3}{\node[etiqueta,text=tceerojo,anchor=west] at (axis cs:4.05,1,2) {$q=2$};}
    \capa{3}{\node[etiqueta,text=tceerojo,anchor=west] at (axis cs:4.05,2.25,3) {$q=3$};}
    \capa{3}{\node[etiqueta,text=tceemorado,anchor=west] at (axis cs:4.3,1.5,0) {isocuantas};}
    % 4: plano z_K = 3 y función de producción a corto plazo
    \capa{4}{\addplot3[area2,draw=tceedorado,thin,domain=0:4,samples=60,samples y=0] (x,3,{sqrt(3*x)}) -- (axis cs:4,3,0) -- (axis cs:0,3,0) -- cycle;}
    \capa{4}{\addplot3[curva4,line width=1.4pt,domain=0:4,samples=60,samples y=0] (x,3,{sqrt(3*x)});}
    \capa{4}{\node[etiqueta,fill=white,fill opacity=0.85,text opacity=1,inner sep=1pt,left=3pt] at (axis cs:0,3,0) {$\bar z_K$};}
    \capa{4}{\node[etiqueta,text=tceedorado,anchor=south east] at (axis cs:1.2,3,2.2) {corto plazo};}
    % 5: rayo z_K = z_L/2 en la base, plano vertical y producción a lo largo del rayo
    \capa{5}{\addplot3[curva3,thick,dashed] coordinates {(0,0,0) (4,2,0)};}
    \capa{5}{\addplot3[fill=tceeverde,fill opacity=0.18,draw=none] coordinates {(0,0,0) (4,2,0) (4,2,2.828) (0,0,0)};}
    \capa{5}{\addplot3[curva3,line width=1.4pt] coordinates {(0,0,0) (4,2,2.828)};}
    \capa{5}{\node[etiqueta,text=tceeverde,anchor=west,align=left] at (axis cs:4.05,2,2.6) {largo plazo:\\ rayo $z_K/z_L$ fijo};}
    % rótulo del eje z_K, al final para que quede encima de la superficie y de los planos
    \capa{1}{\node[etiqueta,fill=white,fill opacity=0.85,text opacity=1,inner sep=1pt,above left=1pt and -2pt] at (zK) {$z_K$};}
  \end{axis}
\end{tikzpicture}
\end{document}
