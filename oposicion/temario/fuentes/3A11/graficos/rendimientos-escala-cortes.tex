\newcommand{\corte}[6]{%

  \capa{#1}{\draw[eje] (0,0) -- (0,3.6) node[left] {$q$};}
  \capa{#1}{\node[below left] at (0,0) {$0$};}
  \capa{#1}{\draw[guia,thick] (0,0) -- (2.25,2.25) node[etiqueta,right,text=tceegris] {$\lambda F(\overline{\vect{z}})$};}
  \capa{#1}{\draw[#3] plot[domain=0:2.1,samples=60] (\x,{\x^#2});}
  \capa{#1}{#5}
  \capa{#1}{\node[punto] at (1,1) {};}
  \capa{#1}{\draw[guia] (1,0) node[below] {$1$} -- (1,1);}
  \capa{#1}{\node[etiqueta,anchor=north,text=#4] at (1.25,-0.55) {#6};}}
