\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: ¿puede una sola función resumir lo que una empresa sabe hacer? Fábrica de alfileres de \autor{Smith} (1776): 10 operarios, 18 operaciones, $>$48.000 alfileres/día ($\approx$4.800 por persona) frente a $<$20 cada uno solo $\Rightarrow$ organización y escala multiplican la producción.
\item \textbf{Contexto}: individualismo metodológico; empresa con 3 restricciones: \emph{técnicas} (este tema), \emph{económicas} (costes \vertema{3.A.12}), \emph{de mercado y organizativas} \vertema{3.A.15}\vertema{3.A.16-19}.
\item \textbf{Relevancia}: micro (costes, oferta, demanda de factores, distribución \vertema{3.A.25}); macro (crecimiento, PTF, RBC); comercio (\autor{Ricardo}, H-O, intraindustrial); política ($\sigma$ y participación del trabajo; rendim.\ y monopolio natural).
\item \textbf{Historia}: \autor{Hesíodo}, \autor{Jenofonte}; \autor{Turgot} (1767) proporciones variables $\to$ 1815 \autor{West}, \autor{Malthus}, \autor{Ricardo}, \autor{Torrens}; \autor{Walras} (1874), \autor{Pareto}; \autor{Wicksteed} (1894), \autor{Clark}, \autor{Wicksell}; \autor{Cournot}, \autor{Marshall} (plazos), \autor{Sraffa} (1926); \autor{Cobb}-\autor{Douglas} (1928), \autor{Hicks} (1932), \autor{Robinson} (1933), ACMS (1961); \autor{Koopmans} (1951), \autor{Debreu} (1959); dualidad \autor{Shephard} (1953), \autor{Uzawa} (1964).
\item \textbf{Problemática}: ¿cómo representar lo que la empresa \emph{puede} producir? ¿Cómo responde la producción en cada plazo? ¿Cómo medir sustitución y escala? ¿Qué cambia con varios productos?
\item \textbf{Estructura}: I. tecnología; II. plazos de \autor{Marshall}; III. multiproducto. Orden distinto del título, deliberado: plazos como hilo; $\sigma$ junto a las isocuantas (mide su curvatura).
\end{itemize}

\bloque[10]{I. La tecnología}
\begin{itemize}
\item Método del consumidor \vertema{3.A.8} en 3 pasos: conjunto $\mathcal{Y}$ y axiomas $\to$ función de producción (un producto) $\to$ isocuantas y sustitución. Con precios: costes y dualidad \vertema{3.A.12}.
\end{itemize}

\sub{I.1. Conjunto de posibilidades de producción}
\begin{itemize}
\item \concepto{Conjunto tecnológico} $\mathcal{Y}\subset\mathbb{R}^S$: planes posibles; \concepto{netput} $\vect{y}$: negativos = factores, positivos = productos (MWG cap.~5). Forma más general: no presupone cuántos bienes ni cuáles son factores.
\item \concepto{Requerimientos de factores} $V(q)=\{\vect{z}\geq\vect{0}:(q,-\vect{z})\in\mathcal{Y}\}$; frontera = isocuanta.
\item \concepto{Axiomas básicos}: (1) no vacío; (2) cerrado (existe máximo); (3) eliminación gratuita ($\vect{y}'\leq\vect{y}\Rightarrow\vect{y}'\in\mathcal{Y}$; todo el «sudoeste» es posible, frontera no creciente); (4) imposib.\ de producción gratuita $\mathcal{Y}\cap\mathbb{R}^S_+\subseteq\{\vect{0}\}$.
\item + \concepto{inacción} $\vect{0}\in\mathcal{Y}$ (no con costes hundidos: propiedad del LP) $\Rightarrow$ $\mathcal{Y}\cap\mathbb{R}^S_+=\{\vect{0}\}$: frontera por el origen.
\item Frontera = planes \concepto{técnicamente eficientes}: no se obtiene más de un producto sin más de algún factor o menos de otro producto.
\end{itemize}
\graficorepaso[0.7\linewidth]{conjunto-produccion}{$\mathcal{Y}$ en el plano de los \emph{netputs}: cada rasgo, un axioma}

\sub{I.2. Función de producción}
\begin{formula}
\centering $q=F(\vect{z})\equiv\optim{\max}{q\geq 0}{q}{(q,-\vect{z})\in\mathcal{Y}}$
\end{formula}
\begin{itemize}
\item Solo métodos técnicamente eficientes: máx.\ $q$ para cada $\vect{z}$ = mín.\ factores para cada $q$. Notación general: \autor{Wicksteed} (1894); antes, von~\autor{Thünen}.
\item \concepto{Propiedades}: \emph{cardinal}; $F(\vect{0})=0$; no decreciente ($\mathrm{PMg}_i\geq0$); continua y $C^2$ (instrumental: divisibilidad).
\item \concepto{Propiedades de escala} de $\mathcal{Y}$: \emph{divisibilidad} ($\lambda\vect{y}$, $\lambda\in[0,1]$: rendim.\ no crecientes); \emph{aditividad} ($\vect{y}+\vect{y}'$: no decrecientes); \emph{rendim.\ constantes} ($\mathcal{Y}$ cono); \emph{convexidad} ($\tau\vect{y}+(1-\tau)\vect{y}'\in\mathcal{Y}$); \emph{irreversibilidad}. Aditivo + divisible $\Leftrightarrow$ cono convexo (MWG 5.B.2).
\item \concepto{Cuasiconcavidad} estricta $\Leftrightarrow$ isocuantas estr.\ convexas (complementariedad: mezclas rinden más que extremos); no impone escala.
\item \concepto{Concavidad} $\Leftrightarrow$ $\mathcal{Y}$ convexo; con $F(\vect{0})=0$ $\Rightarrow$ rendim.\ no crecientes. Cóncava $\Rightarrow$ cuasicóncava, no al revés (CD con $\alpha+\beta>1$).
\item \concepto{Neoclásica de buen comportamiento}: RCE + $\mathrm{PMg}>0$ decrecientes + \autor{Inada} (1963). Cóncava, no estrictamente (lineal en cada rayo).
\end{itemize}
\begin{formula}
\centering \autor{Inada}: $\displaystyle\lim_{z_i\to0}\mathrm{PMg}_i=+\infty,\qquad \lim_{z_i\to\infty}\mathrm{PMg}_i=0$
\end{formula}
\begin{itemize}
\item Beneficio: con RCE o crecientes, problema no acotado; solo concavidad estricta da oferta función. Minimizar costes solo exige cuasiconcavidad $\Rightarrow$ costes con cualquier rendimiento.
\end{itemize}

\sub{I.3. Isocuantas}
\begin{itemize}
\item \concepto{Isocuanta}: $\{\vect{z}:F(\vect{z})=\overline{q}\}$, curva de nivel de la superficie. Infinitas, más lejos = más $q$, no se cortan ($F$ función), continuas, pendiente $\leq0$, convexas si $F$ cuasicóncava, cardinales.
\end{itemize}
\begin{formula}
\centering $\mathrm{RMST}_{LK}\equiv-\left.\dfrac{dz_K}{dz_L}\right|_{\overline{q}}=\dfrac{\mathrm{PMg}_L}{\mathrm{PMg}_K}$\\[4pt]
$\sigma\equiv\dfrac{d\ln(z_K/z_L)}{d\ln \mathrm{RMST}_{LK}}\geq0$
\end{formula}
\begin{itemize}
\item RMST $\downarrow$ al $\uparrow z_L$; depende de unidades. \concepto{Isoclinas}: igual RMST (rectas por el origen si homotética; líneas de contorno y senda de expansión lo son).
\item $\sigma$: número puro, \emph{curvatura}; \autor{Hicks} (1932) y \autor{Robinson} (1933). $\sigma\uparrow$ $\Rightarrow$ isocuanta más plana. Con costes mínimos, $\sigma=d\ln(z_K/z_L)/d\ln(w_L/w_K)$.
\item \concepto{Distribución}: $\dfrac{w_Lz_L}{w_Kz_K}=\dfrac{w_L/w_K}{z_K/z_L}$; si $w_L/w_K\uparrow$, participación del trabajo $\uparrow$ si $\sigma<1$, $\downarrow$ si $\sigma>1$, igual si $\sigma=1$.
\end{itemize}
\begin{formula}
\raggedright CD: $q=A\,z_K^{\alpha}z_L^{\beta}$;\quad $\mathrm{PMg}_K=\alpha\dfrac{q}{z_K}$;\quad $\mathrm{RMST}_{LK}=\dfrac{\beta}{\alpha}\dfrac{z_K}{z_L}$;\quad $\sigma=1$\\[4pt]
CES: $q=A\Big[\sum_i a_iz_i^{\rho}\Big]^{\nu/\rho}$,\quad $\rho=\dfrac{\sigma-1}{\sigma}$,\quad $\sigma=\dfrac{1}{1-\rho}$\\[4pt]
\autor{Leontief}: $q=\min\Big\{\dfrac{z_K}{a};\,\dfrac{z_L}{b}\Big\}$,\quad $\sigma=0$
\end{formula}
\begin{itemize}
\item \concepto{Cobb-Douglas}: $\varepsilon_{q,z_K}=\alpha$, $\varepsilon_{q,z_L}=\beta$; homogénea de grado $\alpha+\beta$; cumple \autor{Inada} (toda en la región económica); con RCE, participaciones constantes ($\alpha$, $1-\alpha$, \autor{Euler}); neutralidades coinciden. Estimación 1928: $q\approx1{,}01\,z_K^{1/4}z_L^{3/4}$ (EE.\,UU., 1899-1922). Precio: impone $\sigma=1$.
\item \concepto{CES} (\autor{Arrow}-\autor{Chenery}-\autor{Minhas}-\autor{Solow}, 1961): homogénea de grado $\nu$; \emph{límites} (no casos: en $\rho=0$ no definida): $\sigma\to\infty$ lineal, $\sigma\to1$ CD, $\sigma\to0$ \autor{Leontief}. $\sigma>1$: ningún factor imprescindible; agregador de \autor{Dixit}-\autor{Stiglitz} \vertema{3.A.18}. Evidencia: $\sigma<1$.
\item \concepto{\autor{Leontief}} (1936; \autor{Cassel} 1918): ángulo recto en $z_K/z_L=a/b$, RCE; $C=(aw_K+bw_L)q$. Varias técnicas $\Rightarrow$ isocuantas poligonales (\autor{Koopmans}, 1951: análisis de actividades).
\end{itemize}
\graficorepaso[0.6\linewidth]{ces-isocuantas}{CES: de $\sigma=0$ (\autor{Leontief}) a $\sigma=\infty$ (lineal)}
\begin{ideaclave}
Pocos axiomas sobre $\mathcal{Y}$ dan $F$; sus isocuantas miden la sustitución con la RMST y, sin unidades, con $\sigma$: de 0 (\autor{Leontief}) a $\infty$ (lineal).
\end{ideaclave}

\bloque[11,5]{II. Leyes temporales de la producción}
\begin{itemize}
\item \autor{Marshall} (1890): plazos = grados de libertad, no calendario. Cortes de la superficie: \emph{corto} ($z_K=\overline{z}_K$, productividad); \emph{largo} (rayo, rendim.\ a escala); \emph{muy largo} (superficie $\uparrow$, progreso técnico).
\end{itemize}
\graficorepaso[0.8\linewidth]{plazos-3d}{Los 3 plazos sobre la superficie de producción}

\sub{II.1. Corto plazo: productividad}
\begin{itemize}
\item $\mathcal{Y}(\overline{z}_K)\subseteq\mathcal{Y}$ $\Rightarrow$ $q^{LP}\geq q^{CP}$; coste LP = envolvente; sin inacción si hay coste hundido.
\item Función con \concepto{rendim.\ marginales mixtos} (ley de proporciones variables, \autor{Turgot}): \emph{incumple eliminación gratuita} (PMg $<0$). El problema no es PMg decreciente, sino negativa. Se mantiene por motivo \emph{histórico} (\autor{Cassels} es anterior a la axiomática) y \emph{didáctico} (delimita la región económica).
\item \concepto{Líneas de contorno} ($\mathrm{PMg}_K=0$ arriba, $\mathrm{PMg}_L=0$ abajo): isoclinas con RMST $\infty$ y 0. Entre ellas, \concepto{región económica}: ambas $\mathrm{PMg}>0$, zona de sustitución. Con \autor{Inada}, toda la función.
\end{itemize}
\graficorepaso[0.55\linewidth]{region-economica}{Región económica entre las líneas de contorno}
\begin{formula}
\centering $\mathrm{PMe}_L=\dfrac{q}{z_L}$,\qquad $\mathrm{PMg}_L=\dfrac{\partial F}{\partial z_L}$,\qquad $\varepsilon_{q,z_L}=\dfrac{\mathrm{PMg}_L}{\mathrm{PMe}_L}$\\[4pt]
$\dfrac{d\mathrm{PMe}_L}{dz_L}=\dfrac{\mathrm{PMg}_L-\mathrm{PMe}_L}{z_L}$\ \ $\Rightarrow$\ \ $\mathrm{PMg}=\mathrm{PMe}$ en máx.\ PMe
\end{formula}
\begin{itemize}
\item PMe = pendiente del rayo desde el origen; PMg = pendiente de la tangente: máx.\ en la inflexión del PT, 0 en el máx.\ del PT.
\item \concepto{Etapas} (\autor{Cassels}, 1936): \textbf{I} demasiado factor fijo (con RCE, \autor{Euler}: $\mathrm{PMe}_L=\mathrm{PMg}_L+\mathrm{PMg}_K\,z_K/z_L$ $\Rightarrow$ $\mathrm{PMg}_K<0$); \textbf{II} aquí produce la empresa (punto según precios); \textbf{III} demasiado factor variable.
\end{itemize}
{\centering\renewcommand{\arraystretch}{1.15}\setlength{\tabcolsep}{4pt}%
\begin{tabular}{@{}lccl@{}}
\toprule
\textit{Tramo} & \textit{Productividades} & $\varepsilon_{q,z_L}$ & \textit{Producción}\\
\midrule
Etapa I & $\mathrm{PMg}_L>\mathrm{PMe}_L$ & $>1$ & extensiva\\
Margen extensivo & $\mathrm{PMg}_L=\mathrm{PMe}_L$ máx. & $=1$ & óptimo técnico\\
Etapa II & $0<\mathrm{PMg}_L<\mathrm{PMe}_L$ & $(0,1)$ & región económica\\
Margen intensivo & $\mathrm{PMg}_L=0$, máx.\ $q$ & $=0$ & máximo técnico\\
Etapa III & $\mathrm{PMg}_L<0$ & $<0$ & intensiva\\
\bottomrule
\end{tabular}\par}
\graficorepaso[0.68\linewidth]{corto-plazo}{PT, PMe, PMg y las 3 etapas}

\sub{II.2. Largo plazo: rendimientos a escala}
\begin{itemize}
\item $\vect{z}\to\lambda\vect{z}$, $\lambda$ = \concepto{factor de escala}. Para $\lambda>1$: $F(\lambda\vect{z})\gtreqless\lambda F(\vect{z})$ $\Rightarrow$ rendim.\ crecientes / constantes / decrecientes. Isocuantas $q,2q,3q$ sobre un rayo: más juntas, equidistantes, más separadas.
\item \concepto{Causas}: crecientes: especialización (\autor{Smith}), indivisibilidades, relaciones físicas (tubería: diámetro $\times2$ $\Rightarrow$ sección $\times4$); decrecientes: coordinación, factores no ampliables.
\end{itemize}
\begin{formula}
\centering $\varepsilon_{q,\lambda}(\vect{z})\equiv\left.\dfrac{d\ln F(\lambda\vect{z})}{d\ln\lambda}\right|_{\lambda=1}=\dfrac{\sum_i\mathrm{PMg}_iz_i}{F(\vect{z})}=\sum_i\varepsilon_{q,z_i}$\\[5pt]
Homogénea: $F(\lambda\vect{z})=\lambda^kF(\vect{z})$;\qquad \autor{Euler}: $\sum_i\mathrm{PMg}_iz_i=kF$
\end{formula}
\begin{itemize}
\item \concepto{Globales}: para \emph{todo} $\vect{z}$ (propiedad de $\mathcal{Y}$: cono, divisibilidad, aditividad); homogéneas de grado $k$: $\varepsilon_{q,\lambda}=k$ siempre. $\mathrm{PMg}$ de grado $k-1$; RMST de grado 0. Con $k=1$, pagar la PMg agota el producto (\autor{Wicksteed}).
\item \concepto{Locales}: en un punto, $\varepsilon_{q,\lambda}(\vect{z})$. CD: $\alpha+\beta$. Une CP (eje) y LP (rayo).
\end{itemize}
\graficorepaso[0.62\linewidth]{rendimientos-escala-isocuantas}{Isocuantas sobre un rayo: crecientes, constantes, decrecientes y variables}
\begin{itemize}
\item \concepto{Rendim.\ variables} (\emph{regular ultra passum law}, \autor{Frisch} 1965; \autor{Wicksell}): crecientes de pequeña, constantes, decrecientes de grande.
\end{itemize}
\begin{formula}
\centering $q=\dfrac{\kappa}{1+z_L^{-a}z_K^{-b}}$\quad$\Rightarrow$\quad $\varepsilon_{q,\lambda}=(a+b)\Big(1-\dfrac{q}{\kappa}\Big)$
\end{formula}
\begin{itemize}
\item Crecientes si $q<\kappa\frac{a+b-1}{a+b}$. Homotética, no homogénea: tecnología «en S» $\to$ CMe en U \vertema{3.A.12}.
\end{itemize}
\graficorepaso[\linewidth]{rendimientos-locales}{Tecnología «en S»: $\varepsilon_{q,\lambda}>1$ a escala baja, $<1$ a escala alta}
\begin{itemize}
\item \concepto{Homotética} $F=g(h(\vect{z}))$, $h$ homog.\ grado 1: isoclinas rectas; $\varepsilon_{q,\lambda}$ depende solo de $q$ (igual en cada isocuanta). Homogénea $\Rightarrow$ homotética, no al revés: $\ln(1+z_K^{1/2}z_L^{1/2})$.
\item Rendim.\ a escala $\neq$ \concepto{economías de escala} (costes); en la senda de mínimo coste $\varepsilon_{q,\lambda}=\mathrm{CMe}/\mathrm{CMg}$ (\autor{Hanoch}, 1975); coinciden punto a punto si homotética.
\item RCE $\Rightarrow$ tamaño de la empresa indeterminado (lo deciden demanda o entrada).
\item \emph{Evidencia}: industria de EE.\,UU.\ con rendim.\ constantes o algo decrecientes (\autor{Basu}-\autor{Fernald}, 1997) $\Rightarrow$ avala RCE agregados.
\end{itemize}

\sub{II.3. Muy largo plazo: progreso técnico}
\begin{itemize}
\item Innovaciones de \emph{proceso} (más con lo mismo) y de \emph{producto} (calidad, bienes nuevos). $F\uparrow$, isocuantas hacia el origen.
\end{itemize}
\graficorepaso[0.95\linewidth]{progreso-tecnico}{$F$ se desplaza hacia arriba y la isocuanta hacia el origen}
\begin{formula}
\centering \autor{Hicks}: $A_tF(z_K,z_L)$;\quad \autor{Solow}: $F(A_tz_K,z_L)$;\\[2pt] \autor{Harrod}: $F(z_K,A_tz_L)$\\[4pt]
CD: $A_tz_K^{\alpha}z_L^{\beta}=\big(A_t^{1/\alpha}z_K\big)^{\alpha}z_L^{\beta}=z_K^{\alpha}\big(A_t^{1/\beta}z_L\big)^{\beta}$
\end{formula}
\begin{itemize}
\item Clasificación por el reparto (\autor{Hicks}, 1932): \emph{neutral} si no altera una relación entre PMg; \emph{sesgado} (ahorrador de trabajo o de capital) si la altera.
\item \concepto{\autor{Hicks}} (aumentador del producto): RMST constante para $z_K/z_L$ dado; isocuantas hacia el origen sin cambiar de pendiente en cada rayo; base de la PTF \vertema{3.A.45}. Si la RMST sube, ahorrador de capital.
\item \concepto{\autor{Solow}} (aumentador del capital): participaciones constantes para $z_L/q$. \concepto{\autor{Harrod}} (aumentador del trabajo, 1948): para $z_K/q$. No jerarquía: 3 criterios.
\end{itemize}
\graficorepaso[\linewidth]{progreso-tecnico-neutral}{Neutralidad de \autor{Hicks} (A), \autor{Solow} (B) y \autor{Harrod} (C)}
\begin{itemize}
\item \concepto{Teorema de \autor{Uzawa}} (1961): crecimiento equilibrado exige progreso aumentador del trabajo $\Rightarrow$ \autor{Solow} lo supone \vertema{3.A.43}.
\item Aumentar $\neq$ sesgar: con $\sigma<1$, aumentador del capital sesgado hacia el trabajo (\autor{Acemoglu}). Sesgo sectorial $\to$ comercio y RRI \vertema{3.B.9}.
\end{itemize}
\begin{ideaclave}
Cada plazo es un corte de la superficie: CP, rendim.\ decrecientes y etapas; LP, $\varepsilon_{q,\lambda}$ global o local; muy LP, progreso técnico neutral o sesgado.
\end{ideaclave}

\bloque[4,5]{III. Empresa multiproducto}
\sub{III.1. Caja de \autor{Edgeworth}}
\begin{itemize}
\item Procesos separables: $X=F_X(L_X,K_X)$, $Y=F_Y(L_Y,K_Y)$, $L_X+L_Y=\overline{L}$, $K_X+K_Y=\overline{K}$. Caja: mapa de $Y$ girado 180° sobre el de $X$ \vertema{3.A.21}.
\end{itemize}
\begin{itemize}
\item Isocuantas que se cortan ($A$) $\Rightarrow$ ineficiente: reasignando, más $X$ con igual $Y$ ($B$) o más $Y$ con igual $X$ ($C$). \concepto{Eficiencia técnica} (\autor{Pareto} en la producción): $\mathrm{RMST}^X_{LK}=\mathrm{RMST}^Y_{LK}$; tangencias = \concepto{curva de contrato} \vertema{3.A.22} $\to$ \concepto{FPP}.
\end{itemize}
\graficorepaso[\linewidth]{caja-edgeworth-fpp}{Caja, curva de contrato y FPP}
\begin{formula}
\centering $\mathrm{RMT}_{XY}\equiv-\dfrac{dY}{dX}=\dfrac{\mathrm{PMg}^Y_L}{\mathrm{PMg}^X_L}=\dfrac{\mathrm{PMg}^Y_K}{\mathrm{PMg}^X_K}=\dfrac{\mathrm{CMg}_X}{\mathrm{CMg}_Y}$
\end{formula}
\begin{itemize}
\item RMST: pendiente de la isocuanta (2 factores, 1 producto); RMT: de la frontera de transformación (2 productos, factores dados).
\item FPP decreciente: rendim.\ decrecientes $\Rightarrow$ cóncava; RCE: cóncava si intensidades distintas, lineal si iguales; crecientes: puede ser convexa. Progreso técnico y dotaciones la desplazan hacia fuera.
\item Competencia perfecta: $\mathrm{RMST}^X=\mathrm{RMST}^Y=w_L/w_K$ (en la curva de contrato), $\mathrm{RMT}=p_X/p_Y$.
\end{itemize}

\sub{III.2. Producción conjunta}
\begin{itemize}
\item \concepto{Función de transformación}: $\mathcal{Y}=\{\vect{y}:T(\vect{y})\leq0\}$, $T=0$ eficientes; un producto: $T=q-F(\vect{z})$. $\mathcal{Y}$ lo admite sin cambios; lo que no sirve es $F$.
\end{itemize}
\begin{formula}
\centering $\mathrm{RMT}_{12}\equiv-\left.\dfrac{dq_2}{dq_1}\right|_{T=0}=\dfrac{\partial T/\partial q_1}{\partial T/\partial q_2}$;\qquad óptimo: $\mathrm{RMT}_{12}=\dfrac{p_1}{p_2}$
\end{formula}
\graficorepaso[0.6\linewidth]{transformacion-empresa}{Curva de transformación de la empresa y RMT}
\begin{itemize}
\item $\mathcal{Y}$ convexo $\Rightarrow$ curva cóncava, RMT $\uparrow$ al especializarse; isobeneficio tangente.
\item \concepto{Producción conjunta} estricta: \emph{proporciones fijas} (carne y cuero, trigo y paja: \autor{Marshall}; frontera = punto; sin imputación técnica del coste) o \emph{variables} (refino del petróleo, lana o carne).
\item \concepto{Economías de alcance} (\autor{Panzar}-\autor{Willig}, 1981): $C(q_1,q_2)<C(q_1,0)+C(0,q_2)$, por factores compartidos (instalaciones, marca, conocimiento) $\Rightarrow$ subaditividad, monopolio natural multiproducto \vertema{3.A.17}, límites de la empresa \vertema{3.A.15}.
\end{itemize}
\begin{ideaclave}
Con varios productos, estar en cada isocuanta no basta: hay que igualar las RMST (curva de contrato); la RMT es el coste de oportunidad.
\end{ideaclave}

\bloque[1,5]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}: axiomas $\to F$ e isocuantas ($\sigma$; CD, CES, \autor{Leontief}); plazos de \autor{Marshall}; multiproducto (contrato, FPP, $T$).
\item \textbf{Relevancia}: la forma de $F$ decide resultados: con RCE el producto se agota; participaciones según $\sigma$; caída de la participación del trabajo desde los 80 (\autor{Karabarbounis}-\autor{Neiman}, 2014) difícil con $\sigma=1$; \autor{Harrod} y crecimiento equilibrado; rendim.\ crecientes $\to$ crecimiento endógeno y nuevo comercio.
\item \textbf{Extensiones}: costes y dualidad \vertema{3.A.12}; oferta \vertema{3.A.16-19}; bienestar \vertema{3.A.22}; controversia del capital \vertema{3.A.4}; crecimiento \vertema{3.A.43}\vertema{3.A.44}; comercio \vertema{3.B.5}\vertema{3.B.6}.
\item \textbf{Opinión}: idealización poderosa, pero la empresa como \emph{caja negra}; lo de dentro, organización industrial \vertema{3.A.15}.
\item \textbf{Idea final}: ¿una función resume lo que sabe hacer la empresa? Lo que \emph{puede} producir, no lo que producirá: el mapa de lo técnicamente posible.
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Por qué usa una función con PMg negativa si incumple la eliminación gratuita?} Por historia (\autor{Cassels} es anterior a la axiomática) y didáctica: delimita la región económica.
\item \textbf{¿Cuasiconcavidad o concavidad?} Cuasiconcavidad: isocuantas convexas, sin escala; concavidad: rendim.\ no crecientes. Para maximizar el beneficio no basta la primera.
\item \textbf{¿Productividad marginal decreciente y rendimientos crecientes a la vez?} Sí: CD con $\alpha,\beta<1$ y $\alpha+\beta>1$.
\item \textbf{¿Rendimientos globales y locales?} Globales: todo $\vect{z}$ (homogéneas, $\varepsilon_{q,\lambda}=k$); locales: $\varepsilon_{q,\lambda}(\vect{z})=\sum\varepsilon_{q,z_i}$, pueden cambiar con la escala.
\item \textbf{¿Rendimientos y economías de escala?} Tecnología frente a costes; $\varepsilon_{q,\lambda}=\mathrm{CMe}/\mathrm{CMg}$ en la senda de mínimo coste; coinciden con homotéticas.
\item \textbf{¿Por qué \autor{Leontief} y CD son límites de la CES?} En $\rho=0$ la CES no está definida (\autor{L'Hôpital}); $\rho\to-\infty$ da el mínimo.
\item \textbf{¿Por qué el modelo de \autor{Solow} supone progreso de \autor{Harrod}?} Teorema de \autor{Uzawa}: lo exige el crecimiento equilibrado.
\item \textbf{¿Qué determina la forma de la FPP?} Rendimientos e intensidades factoriales.
\item \textbf{¿Qué pasa con el beneficio con RCE?} Homogéneo de grado 1 $\Rightarrow$ no acotado o máximo 0; escala indeterminada, oferta correspondencia.
\item \textbf{¿Qué mide $\sigma$ y qué dice de la distribución?} La curvatura de la isocuanta; si $\sigma<1$, al subir $w_L/w_K$ sube la participación del trabajo (\autor{Hicks}, 1932).
\item \textbf{¿Qué es la región económica?} Zona entre las líneas de contorno ($\mathrm{PMg}_K=0$, $\mathrm{PMg}_L=0$) con ambas $\mathrm{PMg}>0$; coincide con la etapa II. Con \autor{Inada} (CD), toda la función.
\item \textbf{¿Qué diferencia la caja de \autor{Edgeworth} de la producción conjunta?} La caja supone procesos separables; en la conjunta un mismo proceso da varios productos (función de transformación, economías de alcance).
\item \textbf{¿Por qué existe la función de costes con cualquier rendimiento?} Minimizar el coste de un $q$ dado es un problema acotado: basta la cuasiconcavidad.
\item \textbf{¿Qué aporta una función homotética?} RMST independiente de la escala (isoclinas rectas, senda de expansión = rayo) y, a la vez, rendim.\ que cambian con la escala.
\end{enumerate}

\begin{trampa}
\begin{itemize}
\item Imposib.\ de producción gratuita: $\mathcal{Y}\cap\mathbb{R}^S_+\subseteq\{\vect{0}\}$; con inacción, $=\{\vect{0}\}$.
\item Etapas: \autor{Cassels} (1936), no \autor{Cassel} (1918, coeficientes fijos).
\item Margen extensivo = máx.\ PMe ($\varepsilon_{q,z_L}=1$); intensivo = máx.\ PT ($\varepsilon_{q,z_L}=0$).
\item Isocuantas convexas $\Leftarrow$ cuasiconcavidad; rendim.\ no crecientes $\Leftarrow$ concavidad.
\item $\sigma=0$ \autor{Leontief}, 1 CD, $\infty$ lineal; CES $\sigma=1/(1-\rho)$, $\rho=(\sigma-1)/\sigma$.
\item Homogénea $\Rightarrow$ homotética, no al revés; RMST de grado 0.
\item \autor{Hicks}: $z_K/z_L$; \autor{Solow}: $z_L/q$; \autor{Harrod}: $z_K/q$. Coinciden en la CD.
\item Con $\sigma<1$, $w_L/w_K\uparrow$ $\Rightarrow$ participación del trabajo $\uparrow$.
\item Convexidad de $\mathcal{Y}$ + inacción $\Rightarrow$ divisibilidad (rendim.\ no crecientes).
\item Neoclásica con \autor{Inada}: cóncava, pero \emph{no estrictamente} (lineal en cada rayo).
\item Productividad: un factor (paralelo a un eje, CP); rendimientos: todos en la misma proporción (rayo, LP).
\item Rendim.\ a escala (tecnología) $\neq$ economías de escala (costes): coinciden punto a punto solo si homotética.
\item CD con $\alpha,\beta<1$ y $\alpha+\beta\geq1$: PMg decrecientes y rendim.\ constantes o crecientes.
\end{itemize}
\end{trampa}
