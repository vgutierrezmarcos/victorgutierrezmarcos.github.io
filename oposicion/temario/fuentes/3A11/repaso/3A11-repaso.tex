\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: ¿puede una sola función resumir lo que una empresa sabe hacer? Fábrica de alfileres de \autor{Smith} (1776): 10 operarios, 18 operaciones, $>$48.000 alfileres/día ($\approx$4.800 por persona) frente a $<$20 cada uno solo $\Rightarrow$ organización y escala multiplican la producción. ¿Qué puede producir la empresa y cómo cambia con sus factores, su tamaño o su tecnología?
\item \textbf{Relevancia}: \emph{micro} (costes, oferta, demanda y remuneración de factores \vertema{3.A.12}\vertema{3.A.25}); \emph{macro} (crecimiento \vertema{3.A.43}, productividad total de los factores (PTF) \vertema{3.A.45}, ciclo real); \emph{internacional} (\autor{Ricardo}, \autor{Heckscher}-\autor{Ohlin}, intraindustrial \vertema{3.B.5}\vertema{3.B.6}); \emph{política} ($\sigma$ y automatización frente a participación del trabajo; rendimientos y monopolio natural \vertema{3.A.17}).
\item \textbf{Contextualización}: \autor{Hesíodo}, \autor{Jenofonte} $\to$ \autor{Turgot} (1767), proporciones variables $\to$ rendim.\ decrecientes de la tierra (1815) $\to$ \autor{Walras}, \autor{Pareto}: producción = intercambio indirecto; \autor{Wicksteed}, \autor{Clark}, \autor{Wicksell}: función de producción $\to$ \autor{Marshall} (plazos); \autor{Sraffa} (1926) $\to$ competencia imperfecta \vertema{3.A.18} $\to$ \autor{Cobb}-\autor{Douglas} (1928), $\sigma$ (\autor{Hicks}, \autor{Robinson}), CES (1961) $\to$ \autor{Koopmans} (1951), \autor{Debreu} (1959): conjunto de producción; dualidad (\autor{Shephard}, \autor{Uzawa}).
\item \textbf{Problemática}: ¿cómo representar lo que la empresa \emph{puede} producir y qué supuestos llevan a una función manejable? ¿Cómo responde la producción si cambia un factor, todos o la tecnología? ¿Cómo medir sustitución y escala; por qué rendimientos distintos según el tamaño? ¿Qué cambia con varios productos?
\item \textbf{Estructura}: I. tecnología ($\mathcal{Y}$, $F$, isocuantas); II. leyes temporales (corto, largo, muy largo plazo); III. empresa multiproducto. Orden distinto del título, deliberado: los plazos son el hilo y $\sigma$ va con las isocuantas (mide su curvatura).
\end{itemize}
\begin{itemize}
\item \concepto{Individualismo metodológico}: lo agregado (oferta de un mercado, crecimiento) desde agentes individuales que optimizan \vertema{3.A.8}.
\item \concepto{Teoría de la empresa}: transforma factores (\emph{inputs}) en productos (\emph{outputs}) con 3 restricciones: \emph{técnicas} (\concepto{teoría de la producción}, este tema), \emph{económicas} (costes \vertema{3.A.12}) y \emph{de mercado y organizativas} \vertema{3.A.15}\vertema{3.A.16-19}.
\end{itemize}

\bloque[10]{I. La tecnología: definición y propiedades de la función de producción}
\begin{itemize}
\item \emph{Nexo}: planteadas las preguntas, empezamos por la tecnología en sí: cómo se representa lo que la empresa puede producir.
\item \emph{Conclusión anticipada}: pocos axiomas sobre los planes posibles dan una función de producción; sus isocuantas miden con un solo número, $\sigma$, la facilidad para sustituir factores.
\item \emph{Método}: el del consumidor \vertema{3.A.8} en 3 pasos: $\mathcal{Y}$ y axiomas $\to$ $F$ (un producto) $\to$ isocuantas y sustitución. Con precios de los factores: costes y dualidad \vertema{3.A.12}.
\end{itemize}

\sub{I.1. Conjunto tecnológico: el conjunto de posibilidades de producción}
\begin{itemize}
\item \emph{Definición}: \concepto{conjunto de posibilidades de producción} (o \concepto{tecnológico}) $\mathcal{Y}\subset\mathbb{R}^S$: todos los planes que permite la tecnología (\autor{Mas-Colell}-\autor{Whinston}-\autor{Green}, 1995, cap.~5). Forma más general: no presupone cuántos bienes ni cuáles son factores.
\item \emph{Plan}: \concepto{vector de producción neta} (\emph{netput}) $\vect{y}=(y_1,\dots,y_S)$: negativos = factores; positivos = productos. Un bien puede ser factor en un plan y producto en otro.
\item \emph{Requerimientos}: \concepto{conjunto de requerimientos de factores} $V(q)=\{\vect{z}\geq\vect{0}:(q,-\vect{z})\in\mathcal{Y}\}$; su frontera, la isocuanta (I.3).
\end{itemize}
\begin{itemize}
\item \emph{Axiomas básicos} (necesarios para $F$): (1) \concepto{no vacío}; (2) \concepto{cerrado}: incluye su frontera $\Rightarrow$ \emph{máximo} de $q$ para cada $\vect{z}$; (3) \concepto{eliminación gratuita}: $\vect{y}'\leq\vect{y}\in\mathcal{Y}$ $\Rightarrow$ $\vect{y}'\in\mathcal{Y}$: todo el «sudoeste» es posible, frontera no creciente; (4) \concepto{imposibilidad de producción gratuita}: $\mathcal{Y}\cap\mathbb{R}^S_+\subseteq\{\vect{0}\}$.
\item \emph{Deseables}: \concepto{posibilidad de inacción} $\vect{0}\in\mathcal{Y}$ (no con costes hundidos: propiedad del LP) $\Rightarrow$ $\mathcal{Y}\cap\mathbb{R}^S_+=\{\vect{0}\}$, frontera por el origen; y convexidad (I.2).
\item \emph{Eficiencia}: la frontera = planes \concepto{técnicamente eficientes}: no se obtiene más de un producto sin más de algún factor o menos de otro producto.
\end{itemize}
\graficorepaso[0.62\linewidth]{conjunto-produccion}{$\mathcal{Y}$ en el plano de los \emph{netputs}: cada rasgo de su forma, un axioma}

\sub{I.2. La función de producción}
\begin{itemize}
\item \emph{Definición}: definido $\mathcal{Y}$, con un producto, relación técnica que asigna a cada $\vect{z}\geq\vect{0}$ la \emph{máxima} producción posible (existe porque $\mathcal{Y}$ es cerrado):
\end{itemize}
\funcionclave{q=F(\vect{z})\equiv\optim{\max}{q\geq 0}{q}{(q,-\vect{z})\in\mathcal{Y}}}
\begin{itemize}
\item \emph{Métodos}: solo los \concepto{técnicamente eficientes}: máx.\ $q$ para cada $\vect{z}$ = mín.\ factores para cada $q$. Con $L$ y $K$, \concepto{superficie de producción}; debajo, planes posibles; encima, imposibles.
\item \emph{Propiedades generales}: (1) \emph{cardinal} (unidades físicas, $\neq$ utilidad); (2) $F(\vect{0})=0$; (3) no decreciente, $\mathrm{PMg}_i\equiv\partial F/\partial z_i\geq0$ (productividad marginal); (4) continua y dos veces diferenciable (supuesto instrumental: factores sustituibles, infinitas técnicas, producto divisible).
\end{itemize}
\pareja{funcion-produccion-3d}{$F(z_L,z_K)$: superficie sobre el plano de los factores}{convexidad-conjunto}{$\mathcal{Y}$ convexo: si $\vect{a}$, $\vect{b}$ posibles, también $\vect{c}$}
\begin{itemize}
\item \emph{Propiedades de escala} de $\mathcal{Y}$ (deciden la curvatura y los rendim.\ \emph{globales}): \concepto{divisibilidad} ($\lambda\vect{y}$, $\lambda\in[0,1]$: rendim.\ no crecientes); \concepto{aditividad} ($\vect{y}+\vect{y}'$, libre entrada: no decrecientes); \concepto{rendimientos constantes} ($\lambda\geq0$: $\mathcal{Y}$ \emph{cono}); \concepto{convexidad} ($\tau\vect{y}+(1-\tau)\vect{y}'\in\mathcal{Y}$; con inacción $\Rightarrow$ divisibilidad); \concepto{irreversibilidad} ($-\vect{y}\notin\mathcal{Y}$). Aditivo y divisible $\Leftrightarrow$ cono convexo (MWG, 5.B.2).
\item \emph{Complementariedad}: $F$ \concepto{estrictamente cuasicóncava} $\Leftrightarrow$ isocuantas estr.\ convexas (mezclas rinden más que extremos); \emph{no impone escala}. $\neq$ complementariedad de \autor{Edgeworth}, $\partial^2F/\partial z_L\partial z_K>0$.
\item \emph{Escala}: con un producto, $\mathcal{Y}$ convexo $\Leftrightarrow$ $F$ \concepto{cóncava}; con $F(\vect{0})=0$ $\Rightarrow$ rendim.\ no crecientes. Cóncava $\Rightarrow$ cuasicóncava, no al revés (CD con $\alpha+\beta>1$).
\item \emph{Buen comportamiento}: \concepto{función neoclásica}: rendimientos constantes a escala (RCE), $\mathrm{PMg}>0$ y decrecientes y condiciones de \autor{Inada} (1963). Cóncava, no estrictamente (lineal en cada rayo).
\end{itemize}
\begin{formula}
\centering \autor{Inada}:\ \ $\displaystyle\lim_{z_i\to0}\mathrm{PMg}_i=+\infty,\qquad \lim_{z_i\to\infty}\mathrm{PMg}_i=0$
\end{formula}
\begin{itemize}
\item \emph{Por qué importa}: el beneficio no tiene renta que lo acote. Con RCE, $B(\lambda\vect{z})=\lambda B(\vect{z})$: no acotado, no producir o máximo 0 con escala indeterminada (oferta = correspondencia); crecientes: no acotado; solo concavidad estricta da oferta función. Minimizar costes solo exige cuasiconcavidad $\Rightarrow$ $C$ existe con cualquier rendimiento \vertema{3.A.12}.
\end{itemize}

\sub{I.3. Las curvas isocuantas}
\begin{itemize}
\item \emph{Definición}: caracterizada $F$, sus curvas de nivel: \concepto{isocuanta} $\{\vect{z}:F(\vect{z})=\overline{q}\}$: combinaciones que, usadas eficientemente, dan igual $q$; frontera de $V(\overline{q})$; corte horizontal de la superficie proyectado en la base.
\item \emph{Propiedades} (como las curvas de indiferencia \vertema{3.A.8}): infinitas; más lejos = más $q$ (monotonía); no se cortan ($F$ es función); continuas; pendiente $\leq0$ ($<0$ si $\mathrm{PMg}>0$); convexas si $F$ cuasicóncava; pero \emph{cardinales}.
\item \emph{Pendiente}: \concepto{relación marginal de sustitución técnica} (RMST): capital que se libera al usar una unidad más de trabajo, con $q$ constante. Decrece al $\uparrow z_L$ (isocuanta convexa); depende de las unidades.
\end{itemize}
\begin{formula}
\centering $\mathrm{RMST}_{LK}\equiv-\left.\dfrac{dz_K}{dz_L}\right|_{q=\overline{q}}=\dfrac{\mathrm{PMg}_L}{\mathrm{PMg}_K}$
\end{formula}
\begin{itemize}
\item \emph{Isoclinas}: \concepto{isoclinas}: unen puntos de igual RMST; rectas por el origen si $F$ homotética; lo son las líneas de contorno (II.1) y la senda de expansión \vertema{3.A.12}.
\item \emph{Elasticidad de sustitución}: \concepto{elasticidad de sustitución}: número puro que mide la \emph{curvatura} de la isocuanta (\autor{Hicks} 1932, \autor{Robinson} 1933, por separado):
\end{itemize}
\funcionclave{\sigma\equiv\frac{d\ln(z_K/z_L)}{d\ln \mathrm{RMST}_{LK}}\geq 0}
\begin{itemize}
\item \emph{Curvatura}: $\sigma\uparrow$ $\Rightarrow$ isocuanta más plana, sustitución más fácil (PMg poco sensibles a la proporción). Sustitutivos perfectos: $\sigma=\infty$; coeficientes fijos: $\sigma=0$.
\item \emph{Interpretación económica}: minimizando costes, $\mathrm{RMST}_{LK}=w_L/w_K$ $\Rightarrow$ $\sigma=d\ln(z_K/z_L)/d\ln(w_L/w_K)$: \% de sustitución de $L$ por $K$ si el salario relativo sube un 1\,\%.
\item \emph{Distribución}: $\dfrac{w_Lz_L}{w_Kz_K}=\dfrac{w_L/w_K}{z_K/z_L}$; si $w_L/w_K\uparrow$: participación del trabajo $\uparrow$ si $\sigma<1$, $\downarrow$ si $\sigma>1$, igual si $\sigma=1$ (\autor{Hicks}).
\end{itemize}
\pareja{isocuantas-rmst}{Isocuanta convexa: la RMST decrece de $A$ a $B$}{ces-isocuantas}{CES: de $\sigma=0$ (\autor{Leontief}) a $\sigma=\infty$ (lineal)}

\textbf{a) Cobb-Douglas} (la más usada del temario)
\begin{itemize}
\item \emph{Idea}: forma más sencilla con buen comportamiento; nombre por \autor{Cobb}-\autor{Douglas} (1928), que la estiman; la forma, ya en \autor{Wicksell} (1901) y von~\autor{Thünen}.
\item \emph{Supuestos}: $q=A\,z_K^{\alpha}z_L^{\beta}$, $A,\alpha,\beta>0$; $A$ nivel tecnológico; $\alpha,\beta$ elasticidades-producto.
\item \emph{Desarrollo}: $\mathrm{PMg}_K=\alpha\,q/z_K$, $\mathrm{PMg}_L=\beta\,q/z_L$ (exponente $\times$ PMe); cumple \autor{Inada} $\Rightarrow$ toda en la región económica; $\mathrm{RMST}_{LK}=\frac{\beta}{\alpha}\frac{z_K}{z_L}$ $\Rightarrow$ $\sigma=1$; homogénea de grado $\alpha+\beta$.
\item \emph{Implicaciones}: con RCE y pago según PMg (\autor{Euler}), participaciones constantes $\alpha$ y $1-\alpha$ $\Rightarrow$ $Y=AK^{\alpha}L^{1-\alpha}$ \vertema{3.A.43}\vertema{3.A.45}; las 3 neutralidades del progreso técnico coinciden (II.3).
\item \emph{Evidencia}: industria de EE.\,UU.\ 1899-1922: $q\approx1{,}01\,z_K^{1/4}z_L^{3/4}$: RCE y exponentes $\approx$ participaciones.
\item \emph{Valoración}: impone $\sigma=1$ y participaciones que no responden a los precios $\Rightarrow$ hace falta la CES.
\end{itemize}

\textbf{b) CES} (elasticidad de sustitución constante)
\begin{itemize}
\item \emph{Idea}: generaliza la CD a cualquier $\sigma$ constante; \autor{Arrow}-\autor{Chenery}-\autor{Minhas}-\autor{Solow} (ACMS, 1961); \autor{Solow} (1956) ya la usa de ejemplo.
\item \emph{Supuestos}: $q=A\big[\sum_ia_iz_i^{\rho}\big]^{\nu/\rho}$, $\rho\equiv\frac{\sigma-1}{\sigma}$, $\sum_ia_i=1$.
\item \emph{Desarrollo}: homogénea de grado $\nu$ (habitual $\nu=1$); $\mathrm{RMST}_{LK}=\frac{a_L}{a_K}\big(\frac{z_K}{z_L}\big)^{1-\rho}$ $\Rightarrow$ $\sigma=\frac{1}{1-\rho}$. \emph{Límites}, no casos (en $\rho=0$ no está definida; \autor{L'Hôpital}): $\rho\to1$ lineal ($\sigma\to\infty$); $\rho\to0$ CD; $\rho\to-\infty$ \autor{Leontief} ($\sigma\to0$).
\item \emph{Implicaciones}: participaciones que varían con los precios $\Rightarrow$ forma para estimar $\sigma$. $\sigma>1$: ningún factor imprescindible \vertema{3.A.44}; agregador de \autor{Dixit}-\autor{Stiglitz} \vertema{3.A.18}.
\item \emph{Evidencia}: $\sigma_{KL}<1$ (datos clave); agregada $\neq$ de planta (reasignación). CD: comodidad, no hecho empírico.
\item \emph{Valoración}: $\sigma$ igual en todos los puntos y pares $\Rightarrow$ formas flexibles (translog, anexo).
\end{itemize}

\textbf{c) Coeficientes fijos (\autor{Leontief})}
\begin{itemize}
\item \emph{Idea}: límite $\sigma\to0$: sin sustitución; \autor{Cassel} (1918); \autor{Leontief} (1936): tablas \emph{input-output} \vertema{3.A.31}.
\item \emph{Supuestos}: $q=\min\{z_K/a;\,z_L/b\}$, coeficientes técnicos $a,b>0$.
\item \emph{Desarrollo}: isocuantas en ángulo recto, vértice en $z_K/z_L=a/b$; el factor sobrante tiene PMg nula; RMST $0$ o $\infty$, $\sigma=0$; RCE.
\item \emph{Implicaciones}: proporción independiente de los precios; $C=(a\,w_K+b\,w_L)\,q$ \vertema{3.A.12}.
\item \emph{Extensiones}: varias técnicas $\Rightarrow$ isocuantas poligonales: \emph{análisis de actividades} (\autor{Koopmans}, 1951), enlace con $\mathcal{Y}$.
\item \emph{Valoración}: útil si la sustitución es despreciable (CP, intersectorial); a LP sí se sustituye.
\end{itemize}
\begin{ideaclave}
Pocos axiomas sobre $\mathcal{Y}$ dan $F$; sus isocuantas miden la sustitución con la RMST y, sin unidades, con $\sigma$: de 0 (\autor{Leontief}) a $\infty$ (lineal), con la CD en $\sigma=1$. La superficie de producción es el mapa del bloque II: cada plazo, un corte.
\end{ideaclave}

\bloque[11,5]{II. Análisis de la función de producción: las leyes temporales}
\begin{itemize}
\item \emph{Nexo}: caracterizada la tecnología, pasamos a cómo varía lo posible según el plazo.
\item \emph{Conclusión anticipada}: los 3 plazos de \autor{Marshall} son 3 cortes de la superficie: CP, rendim.\ decrecientes; LP, rendim.\ a escala globales o locales; muy LP, progreso técnico, cuya neutralidad depende de qué se mantenga constante.
\item \emph{Plazos} (\autor{Marshall}, 1890, libro V): grados de libertad, no calendario. \emph{Corto}: algún factor fijo ($z_K=\overline{z}_K$), corte por un plano: \concepto{productividad}. \emph{Largo}: todos variables, corte por un rayo: \concepto{rendimientos a escala}. \emph{Muy largo}: la superficie se desplaza: \concepto{progreso técnico}.
\end{itemize}
\graficorepaso[0.52\linewidth]{plazos-3d}{Los 3 plazos sobre la superficie de producción}

\sub{II.1. Corto plazo: productividad}
\begin{itemize}
\item \emph{Conjunto}: $\mathcal{Y}(\overline{z}_K)=\{\vect{y}\in\mathcal{Y}:z_K=\overline{z}_K\}\subseteq\mathcal{Y}$ $\Rightarrow$ $q^{\mathrm{LP}}\geq q^{\mathrm{CP}}$; coste LP = envolvente de los de CP \vertema{3.A.12}; sin inacción si hay coste hundido.
\item \emph{Función}: \concepto{rendimientos marginales mixtos}: $q$ crece a ritmo creciente, luego decreciente, máximo y cae (\concepto{ley de las proporciones variables}, \autor{Turgot}).
\item \emph{Advertencia}: \emph{incumple la eliminación gratuita} (PMg $<0$); no es neoclásica. Lo malo no es PMg decreciente, sino negativa. Se mantiene por motivo \emph{histórico} (\autor{Cassels}, 1936, anterior a la axiomática) y \emph{didáctico} (delimita la región económica).
\end{itemize}
\begin{itemize}
\item \concepto{Líneas de contorno}: $\mathrm{PMg}_K=0$ (arriba) y $\mathrm{PMg}_L=0$ (abajo); isoclinas con RMST $\infty$ y 0.
\item \concepto{Región económica}: entre ellas, ambas PMg $>0$: zona de sustitución. Fuera, lo mismo con más de algún factor. Con \autor{Inada} (CD), toda la función.
\end{itemize}
\begin{itemize}
\item \concepto{Productividad media} (PMe): pendiente del rayo desde el origen al producto total (PT).
\item \concepto{Productividad marginal} (PMg): pendiente de la tangente al PT; máx.\ en su inflexión, 0 en el máx.\ del PT.
\item \concepto{Elasticidad-producto}: \% de $q$ ante un 1\,\% más de factor variable.
\end{itemize}
\begin{formula}
\centering $\mathrm{PMe}_L=\dfrac{q}{z_L}$,\quad $\mathrm{PMg}_L=\dfrac{\partial F(z_L,\overline{z}_K)}{\partial z_L}$,\quad $\varepsilon_{q,z_L}=\dfrac{\mathrm{PMg}_L}{\mathrm{PMe}_L}$\\[4pt]
$\dfrac{d\mathrm{PMe}_L}{dz_L}=\dfrac{\mathrm{PMg}_L-\mathrm{PMe}_L}{z_L}$\ \ $\Rightarrow$\ \ $\mathrm{PMg}_L=\mathrm{PMe}_L$ en el máx.\ de la PMe
\end{formula}
\begin{itemize}
\item \emph{Etapas} (\autor{Cassels}, 1936): \emph{I}, demasiado factor fijo, producción extensiva; con RCE, \autor{Euler}: $\mathrm{PMe}_L=\mathrm{PMg}_L+\mathrm{PMg}_K\,z_K/z_L$ $\Rightarrow$ $\mathrm{PMg}_K<0$. \emph{II}: región económica; aquí produce la empresa (punto según precios \vertema{3.A.16}). \emph{III}: demasiado factor variable; reducir $z_L$ aumenta $q$.
\end{itemize}

\sub{II.2. Largo plazo: rendimientos a escala}
\begin{itemize}
\item \emph{Nexo}: estudiado el CP, pasamos al LP: todos los factores variables; corte por un rayo desde el origen.
\item \emph{Definición}: $\vect{z}\to\lambda\vect{z}$, $\lambda$ = \concepto{factor de escala}. Para $\lambda>1$, $F(\lambda\vect{z})\gtreqless\lambda F(\vect{z})$: rendim.\ \emph{crecientes}, \emph{constantes}, \emph{decrecientes}. Isocuantas $q,2q,3q$ en un rayo: cada vez más juntas, equidistantes, más separadas.
\item \emph{Causas} (\autor{Varian}): crecientes: especialización (\autor{Smith}), indivisibilidades, relaciones físicas (tubería: diámetro $\times2$ $\Rightarrow$ sección $\times4$); decrecientes: coordinación, factores no ampliables (en rigor, proporciones variables).
\item \emph{Medida}: \concepto{elasticidad de escala} (coeficiente de la función): \% de $q$ ante un 1\,\% más de todos los factores; $>$, $=$, $<1$.
\end{itemize}
\begin{formula}
\centering $\varepsilon_{q,\lambda}(\vect{z})\equiv\left.\dfrac{d\ln F(\lambda\vect{z})}{d\ln\lambda}\right|_{\lambda=1}=\dfrac{\sum_i\mathrm{PMg}_iz_i}{F(\vect{z})}=\sum_i\varepsilon_{q,z_i}$\\[5pt]
Homogénea: $F(\lambda\vect{z})=\lambda^kF(\vect{z})$;\qquad \autor{Euler}: $\sum_i\mathrm{PMg}_iz_i=kF(\vect{z})$
\end{formula}
\graficorepaso[0.42\linewidth]{rendimientos-escala-isocuantas}{Isocuantas sobre un rayo: crecientes, constantes, decrecientes y variables}
\begin{itemize}
\item \concepto{Rendimientos globales}: propiedad de \emph{toda} la tecnología ($\forall\vect{z}$, $\lambda>1$) = propiedad de escala de $\mathcal{Y}$ (cono, divisibilidad, aditividad). Caso: \concepto{homogénea de grado $k$}: $\varepsilon_{q,\lambda}=k$ siempre; PMg de grado $k-1$; RMST de grado 0. Con $k=1$, pagar la PMg agota el producto (\autor{Wicksteed}).
\item \concepto{Rendimientos locales}: en un punto, con $\varepsilon_{q,\lambda}(\vect{z})$ = suma de elasticidades-producto: une CP (eje) y LP (rayo). CD: $\alpha+\beta$.
\item \concepto{Rendimientos variables}: \emph{regular ultra passum law} (\autor{Frisch}, 1965) o «wicksellianos»: crecientes de pequeña (indivisibilidades), constantes, decrecientes de grande. Ejemplo (\autor{Jehle}-\autor{Reny}):
\end{itemize}
\begin{formula}
\centering $q=\dfrac{\kappa}{1+z_L^{-a}z_K^{-b}}$\quad$\Rightarrow$\quad $\varepsilon_{q,\lambda}=(a+b)\Big(1-\dfrac{q}{\kappa}\Big)$
\end{formula}
\begin{itemize}
\item Crecientes si $q<\kappa\frac{a+b-1}{a+b}$ ($a+b>1$). Homotética, no homogénea: tecnología «en S» $\to$ CMe en U \vertema{3.A.12}.
\end{itemize}
\begin{itemize}
\item \emph{Homotéticas}: $F=g(h(\vect{z}))$, $h$ homogénea de grado 1, $g$ creciente: isoclinas rectas; $\varepsilon_{q,\lambda}$ depende solo de $q$. Homogénea $\Rightarrow$ homotética, no al revés: $\ln(1+z_K^{1/2}z_L^{1/2})$.
\item \emph{Economías de escala}: $\neq$ rendimientos: propiedad de los costes. En la senda de mínimo coste, $\varepsilon_{q,\lambda}=\mathrm{CMe}/\mathrm{CMg}$ (coste medio / coste marginal; \autor{Hanoch}, 1975); coinciden punto a punto si homotética \vertema{3.A.12}.
\item \emph{RCE}: tamaño de la empresa indeterminado; lo deciden demanda o entrada \vertema{3.A.16}.
\item \emph{Evidencia}: industria de EE.\,UU.\ (2 dígitos): rendim.\ constantes o algo decrecientes, mayores al agregar (\autor{Basu}-\autor{Fernald}, 1997) $\Rightarrow$ avala RCE agregados.
\end{itemize}

\sub{II.3. Muy largo plazo: progreso técnico}
\begin{itemize}
\item \emph{Nexo}: con la tecnología dada en CP y LP; ahora cambia la propia tecnología.
\item \emph{Concepto}: \concepto{progreso técnico}: cambios en los métodos; innovaciones de \emph{proceso} (más con lo mismo) y de \emph{producto} (calidad, bienes nuevos). Amplía $\mathcal{Y}$: $F\uparrow$, isocuantas hacia el origen.
\item \emph{Clasificación} (\autor{Hicks}, 1932, por el reparto): \concepto{neutral} si no altera una relación concreta entre PMg; \concepto{sesgado} (ahorrador de trabajo o de capital) si la altera.
\end{itemize}
\begin{formula}
\centering \autor{Hicks}: $A_tF(z_K,z_L)$;\quad \autor{Solow}: $F(A_tz_K,z_L)$;\quad \autor{Harrod}: $F(z_K,A_tz_L)$\\[4pt]
CD: $A_tz_K^{\alpha}z_L^{\beta}=\big(A_t^{1/\alpha}z_K\big)^{\alpha}z_L^{\beta}=z_K^{\alpha}\big(A_t^{1/\beta}z_L\big)^{\beta}$
\end{formula}
\begin{itemize}
\item \concepto{Neutral de \autor{Hicks}} (aumentador del producto): RMST constante para $z_K/z_L$ dado; isocuantas hacia el origen sin cambiar de pendiente en cada rayo. RMST $\uparrow$ $\Rightarrow$ ahorrador de capital; $\downarrow$, de trabajo. Base de la PTF \vertema{3.A.45}.
\item \concepto{Neutral de \autor{Solow}} (aumentador del capital): participaciones constantes para $z_L/q$ dado.
\item \concepto{Neutral de \autor{Harrod}} (aumentador del trabajo, 1948): participaciones constantes para $z_K/q$ dado.
\end{itemize}
\graficorepaso[0.75\linewidth]{progreso-tecnico-neutral}{Neutralidad de \autor{Hicks} (A), \autor{Solow} (B) y \autor{Harrod} (C)}
\begin{itemize}
\item \emph{Relación}: no jerarquía, 3 criterios con 3 proporciones constantes; coinciden en la CD.
\item \emph{Teorema de \autor{Uzawa}} (1961): crecimiento equilibrado con RCE exige progreso aumentador del trabajo $\Rightarrow$ el modelo de \autor{Solow} lo supone \vertema{3.A.43}.
\item \emph{Aumentar $\neq$ sesgar}: con $\sigma<1$, el aumentador del capital eleva más la PMg del trabajo: sesgado hacia el trabajo (\autor{Acemoglu}). Sesgo sectorial $\to$ comercio y relación real de intercambio \vertema{3.B.9}.
\end{itemize}
\begin{ideaclave}
Cada plazo es un corte de la superficie: a CP, la ley de rendimientos decrecientes delimita la región económica y las etapas; a LP, la elasticidad de escala, global o local, mide la respuesta al tamaño; a muy LP, el progreso técnico desplaza la tecnología, neutral o sesgado.
\end{ideaclave}

\bloque[4,5]{III. Empresa multiproducto: curva de contrato}
\begin{itemize}
\item \emph{Nexo}: vista la producción de un bien en cada plazo, pasamos a la empresa que obtiene varios.
\item \emph{Conclusión anticipada}: estar en la isocuanta de cada producto ya no basta: hay que asignar bien los factores (curva de contrato); la frontera es una función de transformación, y su pendiente, la RMT, el coste de oportunidad.
\end{itemize}

\sub{III.1. Caja de \autor{Edgeworth}: curva de contrato y FPP}
\begin{itemize}
\item \emph{Idea}: varios productos con procesos \emph{separables} (una planta cada uno, o empresas de una economía): la frontera sale de \emph{asignar} los factores. Construcción del equilibrio general \vertema{3.A.21}, resumida.
\item \emph{Supuestos}: $X=F_X(L_X,K_X)$, $Y=F_Y(L_Y,K_Y)$; $L_X+L_Y=\overline{L}$, $K_X+K_Y=\overline{K}$.
\item \emph{Desarrollo}: \concepto{caja de \autor{Edgeworth}}: mapa de $Y$ girado 180° sobre el de $X$, lados $\overline{L}$ y $\overline{K}$. Si las isocuantas se cortan ($A$), reasignando: más $X$ con igual $Y$ ($B$) o más $Y$ con igual $X$ ($C$). \concepto{Eficiencia técnica} (de \autor{Pareto} en la producción): tangencia, $\mathrm{RMST}^X_{LK}=\mathrm{RMST}^Y_{LK}$; lugar de tangencias = \concepto{curva de contrato} \vertema{3.A.22} $\to$ \concepto{frontera de posibilidades de producción} (FPP).
\end{itemize}
\graficorepaso[0.64\linewidth]{caja-edgeworth-fpp}{Caja de \autor{Edgeworth}, curva de contrato y FPP}
\begin{formula}
\centering $\mathrm{RMT}_{XY}\equiv-\dfrac{dY}{dX}=\dfrac{\mathrm{PMg}^Y_L}{\mathrm{PMg}^X_L}=\dfrac{\mathrm{PMg}^Y_K}{\mathrm{PMg}^X_K}=\dfrac{\mathrm{CMg}_X}{\mathrm{CMg}_Y}$
\end{formula}
\begin{itemize}
\item \emph{Implicaciones}: \concepto{relación marginal de transformación} (RMT) = coste de oportunidad de una unidad más de $X$. RMST: pendiente de la isocuanta (2 factores, 1 producto); RMT: de la frontera (2 productos, factores dados).
\item \emph{Forma de la FPP}: decreciente; rendim.\ decrecientes $\Rightarrow$ estr.\ cóncava; RCE: cóncava si intensidades distintas, \emph{lineal} si iguales; crecientes: puede ser convexa \vertema{3.B.6}. Progreso técnico y dotaciones: hacia fuera.
\item \emph{Extensiones}: competencia perfecta: $p_j\mathrm{PMg}^j_i=w_i$ $\Rightarrow$ $\mathrm{RMST}^X=\mathrm{RMST}^Y=w_L/w_K$ (en la curva de contrato) y $\mathrm{RMT}=p_X/p_Y$.
\item \emph{Valoración}: procesos separables: no recoge un proceso con varios productos.
\end{itemize}

\sub{III.2. La empresa multiproducto y la producción conjunta}
\begin{itemize}
\item \emph{Idea}: \concepto{empresa multiproducto}: varios productos con los mismos factores. $\mathcal{Y}$ la admite sin cambios (más de un componente positivo); lo que no sirve es $F$.
\item \emph{Supuestos}: \concepto{función de transformación}: $\mathcal{Y}=\{\vect{y}:T(\vect{y})\leq0\}$, $T=0$ en los planes eficientes; un producto: $T(q,-\vect{z})=q-F(\vect{z})$.
\item \emph{Desarrollo}: con 2 productos y $\overline{\vect{z}}$, \concepto{curva de transformación} $T(q_1,q_2,-\overline{\vect{z}})=0$; su pendiente, la RMT de la empresa. $\mathcal{Y}$ convexo $\Rightarrow$ cóncava, RMT $\uparrow$ al especializarse; más factores o progreso: hacia fuera.
\end{itemize}
\begin{formula}
\centering $\mathrm{RMT}_{12}\equiv-\left.\dfrac{dq_2}{dq_1}\right|_{T=0}=\dfrac{\partial T/\partial q_1}{\partial T/\partial q_2}$;\qquad óptimo: $\mathrm{RMT}_{12}=\dfrac{p_1}{p_2}$
\end{formula}
\graficorepaso[0.42\linewidth]{transformacion-empresa}{Curva de transformación de la empresa y RMT}
\begin{itemize}
\item \emph{Implicaciones}: \concepto{producción conjunta} estricta (un proceso, varios productos): \emph{proporciones fijas} (carne y cuero, trigo y paja: oferta conjunta de \autor{Marshall}; frontera = punto o ángulo recto; sin imputación técnica del coste) o \emph{variables} (refino del petróleo; ganadería de lana o carne). Con precios, isobeneficio tangente: $\mathrm{RMT}_{12}=p_1/p_2$.
\item \emph{Extensiones}: \concepto{economías de alcance} (\autor{Panzar}-\autor{Willig}, 1981): $C(q_1,q_2)<C(q_1,0)+C(0,q_2)$, por factores compartidos (instalaciones, marca, conocimiento) $\Rightarrow$ con las de escala, subaditividad y monopolio natural multiproducto \vertema{3.A.17}; límites de la empresa \vertema{3.A.15}.
\item \emph{Valoración}: $T$ generaliza $F$, pero solo dice qué es posible; qué se elige depende de los precios.
\end{itemize}
\begin{ideaclave}
Con varios productos, la eficiencia exige además igualar las RMST (curva de contrato); la frontera es $T(\vect{y})=0$ y su pendiente, la RMT, el coste de oportunidad.
\end{ideaclave}

\bloque[1,5]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}: (1) axiomas $\to$ $F$ e isocuantas, RMST y $\sigma$; CD, CES y \autor{Leontief}; (2) plazos de \autor{Marshall}: rendim.\ decrecientes (CP), a escala globales o locales (LP), progreso técnico (muy LP); (3) varios productos: curva de contrato, FPP y función de transformación.
\item \textbf{Relevancia}: la forma de $F$ decide resultados: con RCE el producto se agota y las participaciones dependen de $\sigma$; caída de la del trabajo desde los 80 (\autor{Karabarbounis}-\autor{Neiman}, 2014), difícil con $\sigma=1$; \autor{Harrod} y crecimiento equilibrado; rendim.\ crecientes $\to$ crecimiento endógeno y nuevo comercio.
\item \textbf{Extensiones}: costes y dualidad \vertema{3.A.12}; oferta \vertema{3.A.16-19}; bienestar \vertema{3.A.22}; controversia de los dos Cambridge \vertema{3.A.4}; comercio \vertema{3.B.5}\vertema{3.B.6}.
\item \textbf{Opinión}: idealización poderosa, pero la empresa como \emph{caja negra}; lo de dentro, organización industrial \vertema{3.A.15}.
\item \textbf{Idea final}: ¿una función resume lo que sabe hacer la empresa? Lo que \emph{puede} producir, no lo que producirá ni cómo se organiza: el mapa de lo técnicamente posible sobre el que precios, costes y mercados trazan la elección.
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Por qué una función con PMg $<0$?} Historia (\autor{Cassels}, anterior a la axiomática) y didáctica: delimita la región económica.
\item \textbf{¿Cuasiconcavidad o concavidad?} Cuasiconcavidad: isocuantas convexas, nada de escala; concavidad: rendim.\ no crecientes. Para maximizar el beneficio no basta la primera.
\item \textbf{¿PMg decrecientes y rendimientos crecientes a la vez?} Sí: CD con $\alpha,\beta<1$ y $\alpha+\beta>1$.
\item \textbf{¿Rendimientos globales y locales?} Globales: todo $\vect{z}$ (homogéneas, $\varepsilon_{q,\lambda}=k$); locales: $\varepsilon_{q,\lambda}(\vect{z})=\sum\varepsilon_{q,z_i}$, cambian con la escala.
\item \textbf{¿Rendimientos y economías de escala?} Tecnología frente a costes; $\varepsilon_{q,\lambda}=\mathrm{CMe}/\mathrm{CMg}$ en la senda de mínimo coste.
\item \textbf{¿Qué pasa con el beneficio con RCE?} Homogéneo de grado 1 $\Rightarrow$ no acotado, no producir o máximo 0; escala indeterminada, oferta correspondencia.
\end{enumerate}

\begin{trampa}
\begin{itemize}
\item Imposib.\ de producción gratuita: $\mathcal{Y}\cap\mathbb{R}^S_+\subseteq\{\vect{0}\}$; con inacción, $=\{\vect{0}\}$.
\item Etapas: \autor{Cassels} (1936), no \autor{Cassel} (1918). Margen extensivo = máx.\ PMe ($\varepsilon_{q,z_L}=1$); intensivo = máx.\ PT ($\varepsilon_{q,z_L}=0$).
\item Isocuantas convexas $\Leftarrow$ cuasiconcavidad; rendim.\ no crecientes $\Leftarrow$ concavidad. Neoclásica con \autor{Inada}: cóncava, \emph{no estrictamente}.
\item $\sigma=0$ \autor{Leontief}, 1 CD, $\infty$ lineal: \emph{límites} de la CES, $\sigma=1/(1-\rho)$.
\item \autor{Hicks}: $z_K/z_L$; \autor{Solow}: $z_L/q$; \autor{Harrod}: $z_K/q$; coinciden en la CD. Con $\sigma<1$, $w_L/w_K\uparrow$ $\Rightarrow$ participación del trabajo $\uparrow$.
\end{itemize}
\end{trampa}
