% Flujos de financiación exterior hacia los países de renta baja y media
% (agregado LMY del Banco Mundial), 2000-2025, miles de millones de dólares
% corrientes: flujos netos de deuda externa (barras), IED, remesas y AOD neta
% recibida (hasta 2023). Datos: ../datos/flujos-renta-baja-media.csv (WDI e
% IDS, IDR 2026). No se pinta la columna cartera_acciones.
% Valores de control: remesas 703,6 (2025); IED 494,3 (2025); AOD 251,9 (2023);
% deuda neta 423,5 (2025) y -283,3 (2015); en 2022 remesas 591,4 > IED 588,0.
% Orden de capas (CRITERIOS §7): barras → ejes y cero → líneas → puntos → rótulos.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/flujos-renta-baja-media.csv}\flujos
\tikzset{
  barradeuda/.style={fill=tceegris!30, draw=tceegris!80, line width=0.3pt},
  lineaied/.style={very thick, tceedorado},
  linearemesas/.style={line width=1.8pt, tceemorado},
  lineaaod/.style={very thick, tceeverde, dash pattern=on 5pt off 2.5pt},
}
\begin{document}
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=14cm, height=8cm,
      xmin=1999.4, xmax=2025.6, ymin=-400, ymax=800,
      xtick={2000,2005,...,2025},
      xticklabel style={/pgf/number format/1000 sep=, font=\footnotesize},
      ytick={-400,-200,...,800},
      yticklabel style={font=\footnotesize, /pgf/number format/.cd,1000 sep=, use comma},
      ylabel={miles de millones de dólares},
      axis lines*=left, tick align=outside, clip=false,
      ymajorgrids, grid style={tceegris!25},
      unbounded coords=discard,
      legend style={at={(0.02,0.98)}, anchor=north west, draw=none,
        font=\footnotesize},
      legend cell align=left, dicapas,
    ]
    % 1: barras de los flujos netos de deuda externa (la serie más volátil)
    \capa{1}{
      \addplot[on layer=dicbarras, ybar, bar width=0.3cm, barradeuda, forget plot]
        table[x=anio, y=deuda_neta] {\flujos};
    }
    % 2: ejes (los dibuja el axis) y la línea del 0, encima de las barras
    \capa{2}{
      \dicraya{(1999.4,0) (2025.6,0)}
    }
    % 3: IED
    \capa{3}{
      \addplot[lineaied, forget plot] table[x=anio, y=ied] {\flujos};
    }
    % 4: remesas (la protagonista)
    \capa{4}{
      \addplot[linearemesas, forget plot] table[x=anio, y=remesas] {\flujos};
    }
    % 5: AOD neta recibida (hasta 2023)
    \capa{5}{
      \addplot[lineaaod, forget plot] table[x=anio, y=aod] {\flujos};
    }
    % 6: puntos del último dato de cada serie
    \capa{6}{
      \addplot[on layer=dicpuntos, forget plot, dicpuntos=tceemorado] coordinates {(2025,703.6)};
      \addplot[on layer=dicpuntos, forget plot, dicpuntos=tceedorado] coordinates {(2025,494.3)};
      \addplot[on layer=dicpuntos, forget plot, dicpuntos=tceeverde] coordinates {(2023,251.9)};
      \addplot[on layer=dicpuntos, forget plot, dicpuntos=tceegris] coordinates {(2022,591.4)};
    }
    % 7: rótulos del último dato, del cruce remesas-IED y leyenda
    \capa{7}{
      \dicrotulo[font=\footnotesize, text=tceemorado, anchor=west]{(2025.4,703.6)}{remesas 704}
      \dicrotulo[font=\footnotesize, text=tceedorado!85!black, anchor=west]{(2025.4,500)}{IED 494}
      \dicrotulo[font=\footnotesize, text=tceegris!80!black, anchor=west]{(2025.4,420)}{deuda neta 424}
      \dicrotulo[font=\footnotesize, text=tceeverde, anchor=west]{(2025.4,251.9)}{AOD 252 (2023)}
      \addplot[on layer=axis descriptions, dicsuelto, thin, dotted, tceeverde]
        coordinates {(2023.3,251.9) (2025.4,251.9)};
      \dicrotulo[font=\scriptsize, anchor=south, align=center]{(2022,770)}{desde 2022, las remesas\\superan a la IED}
      \addplot[on layer=axis descriptions, dicsuelto, thin, -{Stealth[length=1.5mm]}]
        coordinates {(2022,768) (2022,612)};
      \addlegendimage{area legend, barradeuda}\addlegendentry{flujos netos de deuda externa}
      \addlegendimage{linearemesas}\addlegendentry{remesas}
      \addlegendimage{lineaied}\addlegendentry{inversión extranjera directa}
      \addlegendimage{lineaaod}\addlegendentry{AOD neta recibida}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
