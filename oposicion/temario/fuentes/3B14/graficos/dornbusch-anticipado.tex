% Dornbusch con una expansión monetaria anunciada en t_0 y aplicada en t_1
% (Wilson, 1979): e salta en t_0 (menos que si no se anticipa), se sigue
% depreciando hasta t_1 (senda inestable) y desde t_1 se aprecia hacia ē_1 por
% la senda de silla nueva.
%   t_0 = 1,5; t_1 = 4; ē_0 = 0,6; ē_1 = 1,6; e(t_0+) = 1,1; e(t_1) = 2,0.
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee]
  % 1: ejes, t_0, t_1, ē_0 y ē_1
  \capa{1}{%
    \draw[eje] (0,0) -- (7,0) node[below left=2pt and 0pt] {$t$};
    \draw[eje] (0,0) -- (0,2.6) node[left] {$e$};
    \draw[guia] (1.5,0) node[below,text=black,align=center] {$t_0$\\[-1pt]\footnotesize anuncio} -- (1.5,2.4);
    \draw[guia] (4,0) node[below,text=black,align=center] {$t_1$\\[-1pt]\footnotesize aplicación} -- (4,2.4);
    \node[left] at (0,0.6) {$\bar e_0$};
    \draw[guia] (0,1.6) node[left,text=black] {$\bar e_1$} -- (6.8,1.6);}
  % 2: trayectoria de e
  \capa{2}{%
    \draw[curva1] (0,0.6) -- (1.5,0.6);
    \draw[guia,thin,dotted] (1.5,0.6) -- (1.5,1.1);
    \draw[curva1] plot[domain=1.5:4,samples=40] (\x,{1.1+0.9*(exp(0.6*(\x-1.5))-1)/(exp(1.5)-1)})
      plot[domain=4:6.8,samples=40] (\x,{1.6+0.4*exp(-1.1*(\x-4))});}
\end{tikzpicture}
\end{document}
