% Las 3 dimensiones de De Grauwe (2022): simetría, flexibilidad e integración.
% Arriba, la superficie AMO en 3D: las combinaciones mínimas de las 3 con las
% que los costes y los beneficios de la unión se igualan; por encima, la unión
% compensa. Debajo, sus 3 cortes en 2D, cada uno con la tercera dimensión
% dada (son las líneas AMO de De Grauwe en cada plano).
%   Superficie: S(F,I) = 8 / ((1 + 0,2 F)(1 + 0,2 I)), F, I en [0, 7]
%   Zona del euro (ilustrativa): (F, I, S) = (2; 4; 2,4); la superficie pasa
%   por S(2,4) = 3,175, así que queda por debajo.
%   Estados Unidos (ilustrativo): (5,5; 4; 4); S(5,5; 4) = 2,116, por encima.
%   a) Corte con I = 4 (azul): S = 4,444 / (1 + 0,2 F)
%   b) Corte con F = 2 (verde): S = 5,714 / (1 + 0,2 I)
%   c) Corte con S = 2,4 (dorado): I = 5 (3,333 / (1 + 0,2 F) - 1), de
%      (1,944; 7) a (7; 1,944); en F = 2 hace falta I = 6,9 > 4.
% Perspectiva del gráfico de cuasiconcavidad del 3.A.8 (acimut 30°,
% elevación 28°), a escala: flexibilidad a la derecha y algo hacia abajo,
% integración en profundidad, simetría vertical.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\pgfplotsset{
  superficie/.style={surf,z buffer=sort,shader=faceted,fill opacity=0.40,draw=tceemorado!40,line width=0.15pt,
    colormap={tcee}{rgb255=(226,239,217) rgb255=(167,137,189)}},
}
\tikzset{
  fondo/.style={fill=white,fill opacity=0.75,text opacity=1,inner sep=1.5pt},
  titulopanel/.style={anchor=base west,font=\small\bfseries},
  zonabuena/.style={fill=tceesalvia},
  rotulo/.style={etiqueta,align=center,font=\footnotesize,text=tceegris},
}
\begin{document}
\begin{tikzpicture}[tcee,
    declare function={amoS(\f,\i)=8/((1+0.2*\f)*(1+0.2*\i));
                      amoI(\f)=5*(3.3333/(1+0.2*\f)-1);}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-0.6,-6.6) rectangle (17.4,8.6);
  % ==================================================== arriba: la superficie AMO en 3D
  \begin{scope}[xshift=4.2cm,yshift=1.2cm]
  \begin{axis}[x={(0.70cm,-0.19cm)},y={(0.40cm,0.32cm)},z={(0cm,0.55cm)},axis lines=none,
      xmin=0,xmax=7,ymin=0,ymax=7,zmin=0,zmax=8,clip=false]
    % 1: ejes
    \capa{1}{\draw[eje] (axis cs:0,0,0) -- (axis cs:8.2,0,0) node[below right=-2pt] {flexibilidad};}
    \capa{1}{\draw[eje] (axis cs:0,0,0) -- (axis cs:0,9.4,0);}
    \capa{1}{\node[etiqueta,anchor=south east,fondo] at (axis cs:0,9.4,0) {integración};}
    \capa{1}{\draw[eje] (axis cs:0,0,0) -- (axis cs:0,0,9.2) node[left] {simetría};}
    \capa{1}{\node[left] at (axis cs:0,0,0) {$0$};}
    % 2: la superficie AMO
    \capa{2}{\addplot3[superficie,domain=0:7,y domain=0:7,samples=15,samples y=15] {amoS(x,y)};}
    \capa{2}{\node[etiqueta,anchor=west,text=tceemorado] at (axis cs:7,7,{amoS(7,7)}) {superficie AMO};}
    % 3: por encima, la unión compensa; por debajo, mejor moneda propia
    \capa{3}{\node[rotulo] at (axis cs:4.6,5.2,7.2) {la unión compensa\\(beneficios $>$ costes)};}
    \capa{3}{\node[rotulo,fondo] at (axis cs:3.6,0.4,0.9) {mejor moneda propia};}
    % 4: zona del euro (por debajo) y Estados Unidos (por encima), ilustrativos
    \capa{4}{\draw[guia] (axis cs:2,4,0) -- (axis cs:2,4,2.4);}
    \capa{4}{\node[punto] (ze3) at (axis cs:2,4,2.4) {};}
    \capa{4}{\node[etiqueta,below left=0pt and 1pt,font=\footnotesize,fondo] at (ze3) {zona del euro};}
    \capa{4}{\draw[guia] (axis cs:5.5,4,0) -- (axis cs:5.5,4,4);}
    \capa{4}{\node[punto,fill=tceegris] (us3) at (axis cs:5.5,4,4) {};}
    \capa{4}{\node[etiqueta,right=1pt,font=\footnotesize,text=tceegris,fondo] at (us3) {Estados Unidos};}
    % 5: corte a, integración dada (I = 4)
    \capa{5}{\addplot3[curva2,domain=0:7,samples=30,samples y=1] ({x},{4},{amoS(x,4)});}
    \capa{5}{\node[etiqueta,text=tceeazul,anchor=west,font=\footnotesize\bfseries] at (axis cs:7,4,{amoS(7,4)}) {a};}
    % 6: corte b, flexibilidad dada (F = 2)
    \capa{6}{\addplot3[curva3,domain=0:7,samples=30,samples y=1] ({2},{x},{amoS(2,x)});}
    \capa{6}{\node[etiqueta,text=tceeverde,anchor=east,font=\footnotesize\bfseries] at (axis cs:2,0,{amoS(2,0)}) {b};}
    % 7: corte c, simetría dada (S = 2,4): la curva de nivel sobre la superficie
    \capa{7}{\addplot3[curva4,domain=1.944:7,samples=30,samples y=1] ({x},{amoI(x)},{2.4});}
    \capa{7}{\addplot3[curva4,thin,dashed,domain=1.944:7,samples=30,samples y=1] ({x},{amoI(x)},{0});}
    \capa{7}{\node[etiqueta,text=tceedorado,anchor=west,font=\footnotesize\bfseries] at (axis cs:7,1.944,2.4) {c};}
  \end{axis}
  \end{scope}
  % ==================================================== abajo: los 3 cortes en 2D
  % Escala común: 1 unidad = 0,6 cm en cada panel; F, I en [0, 7], S en [0, 6]
  % ---- a) simetría y flexibilidad, con la integración dada
  \begin{scope}[x=0.6cm,y=0.6cm,yshift=-6.0cm]
  \capa{5}{\node[titulopanel,text=tceeazul] at (-0.6,8.2) {a) Integración dada};}
  \capa{5}{\fill[zonabuena] plot[domain=0:7,samples=30] (\x,{amoS(\x,4)}) -- (7,6) -- (0,6) -- cycle;}
  \capa{5}{\draw[eje] (0,0) -- (7.6,0) node[below left=2pt and 0pt,font=\footnotesize] {flexibilidad};}
  \capa{5}{\draw[eje] (0,0) -- (0,6.4) node[right=1pt,font=\footnotesize] {simetría};}
  \capa{5}{\draw[curva2] plot[domain=0:7,samples=30] (\x,{amoS(\x,4)});}
  \capa{5}{\node[punto] at (2,2.4) {};}
  \capa{5}{\node[etiqueta,below left=0pt and 0pt,font=\scriptsize] at (2,2.4) {ZE};}
  \capa{5}{\node[punto,fill=tceegris] at (5.5,4) {};}
  \capa{5}{\node[etiqueta,above=1pt,font=\scriptsize,text=tceegris] at (5.5,4) {EE.\,UU.};}
  \end{scope}
  % ---- b) simetría e integración, con la flexibilidad dada
  \begin{scope}[x=0.6cm,y=0.6cm,xshift=6.0cm,yshift=-6.0cm]
  \capa{6}{\node[titulopanel,text=tceeverde] at (-0.6,8.2) {b) Flexibilidad dada};}
  \capa{6}{\fill[zonabuena] plot[domain=0:7,samples=30] (\x,{amoS(2,\x)}) -- (7,6) -- (0,6) -- cycle;}
  \capa{6}{\draw[eje] (0,0) -- (7.6,0) node[below left=2pt and 0pt,font=\footnotesize] {integración};}
  \capa{6}{\draw[eje] (0,0) -- (0,6.4) node[right=1pt,font=\footnotesize] {simetría};}
  \capa{6}{\draw[curva3] plot[domain=0:7,samples=30] (\x,{amoS(2,\x)});}
  \capa{6}{\node[punto] at (4,2.4) {};}
  \capa{6}{\node[etiqueta,below left=0pt and 0pt,font=\scriptsize] at (4,2.4) {ZE};}
  \end{scope}
  % ---- c) integración y flexibilidad, con la simetría dada
  \begin{scope}[x=0.6cm,y=0.6cm,xshift=12.0cm,yshift=-6.0cm]
  \capa{7}{\node[titulopanel,text=tceedorado] at (-0.6,8.2) {c) Simetría dada};}
  \capa{7}{\fill[zonabuena] plot[domain=1.944:7,samples=30] (\x,{amoI(\x)}) -- (7,7) -- cycle;}
  \capa{7}{\draw[eje] (0,0) -- (7.6,0) node[below left=2pt and 0pt,font=\footnotesize] {flexibilidad};}
  \capa{7}{\draw[eje] (0,0) -- (0,7.4) node[right=1pt,font=\footnotesize] {integración};}
  \capa{7}{\draw[curva4] plot[domain=1.944:7,samples=30] (\x,{amoI(\x)});}
  \capa{7}{\node[punto] at (2,4) {};}
  \capa{7}{\node[etiqueta,below left=0pt and 0pt,font=\scriptsize] at (2,4) {ZE};}
  \end{scope}
  % Leyenda común de los paneles: verde, la unión compensa
  \capa{5}{\node[rotulo,anchor=north] at (8.4,-6.5) {En cada panel, la curva es el corte de la superficie AMO; en la zona sombreada, la unión compensa.};}
\end{tikzpicture}
\end{document}
