% El apalancamiento financiero: ROE = [ROA + (V_D/V_E)(ROA - r_D)](1 - t_IS),
% con r_D = 5 % y t_IS = 25 %, para V_D/V_E = 0, 2/3 (la empresa del ejemplo)
% y 1,5. Las 3 rectas se cortan en ROA = r_D (ROE = 3,75 %): a su derecha, más
% deuda da más ROE; a su izquierda, menos. La pendiente, (1 + V_D/V_E)(1 - t_IS),
% crece con la deuda: más riesgo. Punto de la empresa del ejemplo: (10 %, 10 %).
% Capas: 1 ejes; 2 sin deuda; 3 deuda del ejemplo y su punto; 4 más deuda;
% 5 el corte en ROA = r_D.
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee]
  \begin{axis}[width=10.5cm,height=7.5cm,scale only axis=false,
      xmin=-2,xmax=16,ymin=-9,ymax=21,
      axis lines=middle,axis line style={->,thick},
      xlabel={ROA (\%)},ylabel={ROE (\%)},
      xlabel style={anchor=west,at={(axis description cs:1.0,0.3)}},
      ylabel style={anchor=south},
      xtick={5,10},ytick={-5,5,10,15,20},
      tick label style={font=\footnotesize},clip=false]
    % 2: sin deuda
    \capa{2}{\addplot[curva3,domain=-2:16] {0.75*x};
      \node[font=\footnotesize,text=tceeverde,anchor=west] at (axis cs:16,12) {$V_D/V_E=0$};}
    % 3: la deuda del ejemplo
    \capa{3}{\addplot[curva1,domain=-2:16] {0.75*(x+(2/3)*(x-5))};
      \node[font=\footnotesize,text=tceemorado,anchor=west] at (axis cs:16,17.5) {$V_D/V_E=2/3$};
      \addplot[guia] coordinates {(10,0) (10,10) (0,10)};
      \addplot[only marks,mark=*,mark size=1.8pt] coordinates {(10,10)};
      \node[font=\footnotesize,anchor=south east] at (axis cs:10,10.3) {ejemplo};}
    % 4: más deuda
    \capa{4}{\addplot[curva2,domain=-2:15.2] {0.75*(x+1.5*(x-5))};
      \node[font=\footnotesize,text=tceeazul,anchor=east] at (axis cs:12.6,19.5) {$V_D/V_E=1{,}5$};}
    % 5: el corte en ROA = r_D
    \capa{5}{\addplot[guia] coordinates {(5,-9) (5,3.75)};
      \addplot[only marks,mark=*,mark size=1.8pt] coordinates {(5,3.75)};
      \node[font=\footnotesize,anchor=north west,align=left] at (axis cs:5.2,-1.5) {$\mathrm{ROA}=r_D$:\\la deuda ni suma ni resta};
      \node[font=\footnotesize,text=tceegris,align=center] at (axis cs:13.8,-7.2) {a la derecha, la deuda\\sube el ROE};}
  \end{axis}
\end{tikzpicture}
\end{document}
