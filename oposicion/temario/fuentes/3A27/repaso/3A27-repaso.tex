\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: «\emph{lo barato sale caro}».
  \begin{itemize}
  \item \autor{Ford}, enero 1914: 5 \$/día (antes $\approx2{,}34$); rotación 370\,\% $\to$ 54\,\%, absentismo 10\,\% $\to$ 2,5\,\%.
  \item Colas en la puerta y no bajó el salario: ¿\emph{por qué no bajan los salarios si sobra gente dispuesta a trabajar?}
  \item \concepto{Rigidez salarial}: el $w$ real no se ajusta para vaciar el mercado. \concepto{Paro involuntario}: quien aceptaría el salario vigente (o menos) y no encuentra empleo.
  \end{itemize}
\item \textbf{Relevancia}:
  \begin{itemize}
  \item Principal renta de los hogares y coste de las empresas. Macro: paro, persistencia, respuesta a \emph{shocks}; microfundamentos de la WS \vertema{3.A.28} y de las rigideces reales de la NEK \vertema{3.A.7}.
  \item Política: cobertura de convenios del \dato{92\,\%}{OCDE, 2024} \vertema{4.A.16}.
  \end{itemize}
\item \textbf{Contextualización}: neoclásico, $w$ real vacía el mercado, solo paro voluntario \vertema{3.A.25}.
  \begin{itemize}
  \item Antecedentes: \autor{Hicks} (1932), \autor{Dunlop} (1944), \autor{Keynes} (1936, rigidez \emph{nominal}) \vertema{3.A.4}.
  \item Años 70-90: rigidez del $w$ \emph{real} con agentes que optimizan; cada teoría levanta un supuesto:
    \begin{itemize}
    \item atomicidad $\to$ negociación;
    \item información perfecta $\to$ eficiencia;
    \item flexibilidad período a período $\to$ contratos implícitos (y, con costes de rotación, \emph{insiders}).
    \end{itemize}
  \end{itemize}
\item \textbf{Problemática}: hechos estilizados (\autor{Romer}):
  \begin{itemize}
  \item Ajuste por cantidades; $w$ nominales casi nunca bajan (\autor{Dickens} et al., 2007); primas entre industrias (\autor{Krueger}-\autor{Summers}, 1988); curva salarial $\approx-0{,}1$.
  \item ¿Cómo se fija $w$ negociando? ¿Es eficiente? ¿Por qué pagar de más? ¿Rigidez, paro o ambos?
  \end{itemize}
\item \textbf{Estructura}: \hyperref[rep:I]{I} negociación (rigidez que imponen los trabajadores), \hyperref[rep:II]{II} eficiencia (la elige la empresa), \hyperref[rep:III]{III} contratos implícitos e \emph{insiders}.
\end{itemize}

\bloque[11,5]{I. Modelos de negociación\label{rep:I}}
\begin{ideaclave}
El resultado depende de \emph{qué se negocia} y del \emph{poder de cada parte}: si solo se negocia $w$, el sindicato lo sube a costa del empleo (paro involuntario, ineficiente); si entra también $L$, el poder solo reparte la renta. Se levanta la \emph{atomicidad}: «hablando se entiende la gente».
\end{ideaclave}

\sub{I.1. Teoría de la negociación\label{rep:I.1}}
\begin{itemize}
\item \concepto{Teoría de la negociación}: cómo 2~partes que ganan con un acuerdo se reparten la ganancia; aquí, la renta conjunta empresa-trabajadores, que $w$ reparte.
  \begin{itemize}
  \item \concepto{Monopolio bilateral}: resultado \emph{indeterminado} (\autor{Bowley}, 1928) \vertema{3.A.17}; decide el \emph{poder de negociación}.
  \end{itemize}
\end{itemize}
\subsub{a) Precursores: \autor{Edgeworth} (1881) y \autor{Hicks} (1932)}
\begin{itemize}
\item \autor{Edgeworth}: la curva de contrato no elige el reparto \vertema{3.A.21}.
\item \autor{Hicks}: curva de concesión (empresa) y de resistencia (sindicato) frente a la duración de la huelga.
  \begin{itemize}
  \item Con información completa, huelgas solo \emph{potenciales}.
  \end{itemize}
\end{itemize}
\subsub{b) Enfoque axiomático: \autor{Nash} (1950)}
\begin{itemize}
\item Axiomas: eficiencia, simetría, invariancia a transformaciones \emph{afines}, independencia de alternativas irrelevantes.
  \begin{itemize}
  \item Sin simetría: \autor{Nash} generalizada (\autor{Kalai}, 1977) \vertema{3.A.14}.
  \item $(U_S^0,B^0)$: \concepto{punto de desacuerdo}; $\gamma$: \concepto{poder de negociación}.
  \end{itemize}
\funcionclave{\max\ \big(U_S-U_S^0\big)^{\gamma}\big(B-B^0\big)^{1-\gamma}}
\item Utilidad transferible: desacuerdo + cuota $\gamma$ del excedente; simetría $\Rightarrow$ $\gamma=\tfrac12$ (resultado, no supuesto).
\end{itemize}
\subsub{c) Enfoque estratégico: \autor{Rubinstein} (1982)}
\begin{itemize}
\item Ofertas alternadas, \emph{sin fecha final}: único ENPS, acuerdo inmediato.
\item Límite (\autor{Binmore}-\autor{Rubinstein}-\autor{Wolinsky}, 1986): gana poder el que puede esperar más:
\begin{formula}
\centering $\gamma=\dfrac{\rho_B}{\rho_S+\rho_B}$
\end{formula}
\item \emph{Valoración}: herramienta de todos los modelos salariales.
  \begin{itemize}
  \item Huelgas reales: información privada (\autor{Kennan}-\autor{Wilson}, 1993).
  \item No dice nada del empleo $\Rightarrow$ importa \emph{qué} se negocia.
  \end{itemize}
\end{itemize}

\sub{I.2. Modelos de negociación colectiva\label{rep:I.2}}
\begin{itemize}
\item Qué entra en la mesa:
  \begin{itemize}
  \item \emph{sindicato monopolista} (\autor{Dunlop}, 1944): fija $w$;
  \item \emph{derecho a gestionar} (\autor{Leontief}, 1946; \autor{Nickell}-\autor{Andrews}, 1983): se negocia $w$, la empresa elige $L$;
  \item \emph{contratos eficientes} (\autor{McDonald}-\autor{Solow}, 1981): $w$ y $L$ juntos;
  \item \emph{fuertemente eficientes} (\autor{Booth}, 1995): $w$, $L$ y un seguro de paro.
  \end{itemize}
\end{itemize}
\subsub{a) El aparato gráfico: empresa y sindicato}
\begin{itemize}
\item \emph{Empresa}: $B=AF(L)-wL$; demanda $AF'(L)=w$.
  \begin{itemize}
  \item \concepto{Isobeneficio}: máximo sobre $L^D$; más baja, más beneficio:
\begin{formula}
\centering $\left.\dfrac{dw}{dL}\right|_{dB=0}=\dfrac{AF'(L)-w}{L}$
\end{formula}
  \end{itemize}
\item \emph{Sindicato} (\autor{Booth}, 1995): $\overline{L}$ afiliados idénticos, empleo al azar, $z$ = valor del desempleo:
\funcionclave{U_S=\frac{L}{\overline{L}}\,u(w)+\Big(1-\frac{L}{\overline{L}}\Big)u(z),\quad u'>0,\ u''\le0}
  \begin{itemize}
  \item \concepto{Salario de reserva}: $u(w^R)=u(z)$ $\Rightarrow$ $w^R=z$.
  \item Indiferencias decrecientes y convexas si $w>w^R$; horizontales en $w^R$ y desde $\overline{L}$.
  \item $u$ lineal: $U_S=wL-v(L)$ del monopolio bilateral \vertema{3.A.17}; lo nuevo, la \emph{aversión al riesgo}.
  \item Competencia: punto $C$, $AF'(L^c)=w^R$.
  \end{itemize}
\end{itemize}
\subsub{b) Sindicato monopolista (\autor{Dunlop}, 1944)}
\begin{itemize}
\item $\gamma=1$; 2~etapas, \autor{Stackelberg} con líder el sindicato $\Rightarrow$ ENPS \vertema{3.A.14}.
  \begin{itemize}
  \item $\max_w U_S(w,L^D(w))$ s.a. $L^D\leq\overline{L}$:
\begin{formula}
\centering $\dfrac{u(w)-u(w^R)}{w\,u'(w)}=\dfrac{1}{|\varepsilon_{L^D,w}|}$;\ neutral: $w^S\big(1-\tfrac{1}{|\varepsilon|}\big)=w^R$
\end{formula}
  \item Margen sobre $w^R$ (regla del monopolio). Cobb-Douglas $F=L^\alpha$: $w^S=w^R/\alpha$, \emph{sin $A$} $\Rightarrow$ el \emph{shock} lo absorbe el empleo.
  \end{itemize}
\item Punto $M$: tangencia con $L^D$; $w^S>w^R$, $L^S<L^c$ $\Rightarrow$ \emph{paro involuntario}.
  \begin{itemize}
  \item ENPS pero no \autor{Pareto}: \concepto{lente de mejoras paretianas}.
  \item Seminal, pero la empresa sin poder no es realista.
  \end{itemize}
\end{itemize}
\subsub{c) Derecho a gestionar (\autor{Nickell}-\autor{Andrews}, 1983)}
\begin{itemize}
\item 1.ª etapa: \autor{Nash} sobre $w$ (desacuerdo: cierre, $B^0=0$, $U_S^0=u(z)$); 2.ª: $L$ sobre $L^D$.
  \begin{itemize}
  \item \emph{No es ENPS} (negociación = concepto cooperativo).
  \item $\gamma=1\to M$, $\gamma=0\to C$; $R$ entre ambos. Cobb-Douglas y neutral:
\begin{formula}
\centering $w=(1-\gamma)\,w^R+\gamma\,w^S=w^R\Big[1+\gamma\dfrac{1-\alpha}{\alpha}\Big]$
\end{formula}
  \end{itemize}
\item ↑$\gamma$ $\Rightarrow$ ↑$w$, ↓$L$: \emph{la moderación salarial crea empleo}.
  \begin{itemize}
  \item Ineficiente (lente) salvo $\gamma=0$.
  \item Describe mejor los convenios; base de la WS \vertema{3.A.28}.
  \end{itemize}
\end{itemize}
\subsub{d) Contratos eficientes (\autor{McDonald}-\autor{Solow}, 1981)}
\begin{itemize}
\item Una etapa, \autor{Nash} sobre $(w,L)$. \concepto{Curva de contratos}: tangencias isobeneficio-indiferencia:
\begin{formula}
\centering $AF'(L)=w-\dfrac{u(w)-u(w^R)}{u'(w)}$;\ $\gamma=0\to C$, $\gamma=1\to F$ ($B=0$)
\end{formula}
  \begin{itemize}
  \item \emph{Neutral}: vertical en $L^c$: el poder solo reparte.
  \item \emph{Averso}: $AF'(L)<w^R$, pendiente $>0$, de $C$ a $D$ ($\overline{L}$) y vertical hasta $F$: \emph{sobreempleo} (el sindicato asegura a los afiliados).
  \item Fuera de $L^D$: la empresa querría despedir.
  \end{itemize}
\item ↑$\gamma$ $\Rightarrow$ ↑$w$ y $L$ igual o mayor: la moderación \emph{no} crea empleo (receta opuesta al derecho a gestionar).
  \begin{itemize}
  \item Solo \emph{débilmente} eficientes con aversión; poco creíbles (la empresa quiere volver a $L^D$) $\Rightarrow$ verificación o juego repetido (\autor{Espinosa}-\autor{Rhee}, 1989).
  \end{itemize}
\end{itemize}
\graficorepaso[0.8\linewidth]{negociacion-colectiva}{Los 3~modelos en un plano: $M$ ($\gamma=1$), $R$ y $C$ ($\gamma=0$) sobre $L^D$; curva de contratos $C$-$D$-$F$}
\subsub{e) Contratos fuertemente eficientes (\autor{Booth}, 1995)}
\begin{itemize}
\item Se negocia también un seguro de paro $b$ $\Rightarrow$ \concepto{principio de indiferencia}: $w=w^R+b$.
  \begin{itemize}
  \item Indiferencias horizontales; curva vertical en $L^c$ \emph{también con aversión}.
  \item No se observa: seguros públicos e incompletos; riesgo moral \vertema{3.A.13}.
  \end{itemize}
\end{itemize}
\subsub{f) Valoración y evidencia}
\begin{itemize}
\item \emph{¿Cuál describe la realidad?} Convenios: salario y condiciones, casi nunca plantilla; $w$ en el sector, $L$ en la empresa $\Rightarrow$ derecho a gestionar (\autor{Cahuc} et al., 2014).
  \begin{itemize}
  \item Contrastes directos entre los 2: sin veredicto claro.
  \item Coinciden: \emph{prima sindical}, menos dispersión, efecto ambiguo en el empleo.
  \end{itemize}
\item \emph{España}: cobertura \dato{92,1\,\%}{OCDE, 2024} con afiliación \dato{12,5\,\%}{2023}: eficacia general (art. 82.3~ET).
\item \emph{Nivel de negociación} (joroba de \autor{Calmfors}-\autor{Driffill}, coordinación, reformas de 2012 y 2021): en el anexo, no se canta \vertema{4.A.16}.
\end{itemize}

\sub{I.3. La negociación en los modelos macroeconómicos\label{rep:I.3}}
\begin{itemize}
\item \autor{Diamond}-\autor{Mortensen}-\autor{Pissarides} (Nobel 2010): \autor{Nash} reparte la renta de cada emparejamiento \vertema{3.A.26}:
\begin{formula}
\centering $w=(1-\gamma)\,z+\gamma\,(y+c\,\theta)$,\quad $\theta=V/U$
\end{formula}
\item \autor{Layard}-\autor{Nickell}-\autor{Jackman} (1991): derecho a gestionar $\Rightarrow$ curva WS; con la de precios, la NAIRU \vertema{3.A.28}:
\begin{formula}
\centering $\dfrac{W}{P^e}=F(U,Z)\cdot\mathrm{PAT}^e$,\quad $F_U<0$
\end{formula}
\item \concepto{Curva salarial} (\autor{Blanchflower}-\autor{Oswald}, 1994): elasticidad del $w$ regional al paro $\approx-0{,}1$.
\end{itemize}

\bloque[10]{II. Salarios de eficiencia\label{rep:II}}
\begin{ideaclave}
Si el rendimiento depende del salario, a la empresa le conviene pagar más que el salario que vacía el mercado y no bajarlo aunque haya parados: rigidez real y paro involuntario \emph{sin poder de mercado}. Se levanta la \emph{información perfecta}: «lo barato sale caro».
\end{ideaclave}

\sub{II.1. Idea\label{rep:II.1}}
\begin{itemize}
\item \concepto{Salario de eficiencia}: el que la empresa paga, por decisión propia, por encima del walrasiano porque la plantilla le devuelve en rendimiento más de lo que cuesta (\autor{Akerlof}-\autor{Yellen}, 1986).
\item 4~motivos (\autor{Romer}):
  \begin{enumerate}
  \item \emph{nutrición} (\autor{Leibenstein}, 1957);
  \item \emph{supervisión imperfecta}: escaqueo (\autor{Shapiro}-\autor{Stiglitz}, 1984) y rotación (\autor{Stiglitz}, 1974; \autor{Salop}, 1979);
  \item \emph{atraer a los más hábiles}: selección adversa (\autor{Weiss}, 1980) \vertema{3.A.13};
  \item \emph{lealtad}: intercambio de regalos (\autor{Akerlof}, 1982).
  \end{enumerate}
\end{itemize}

\sub{II.2. \autor{Solow} (1979): información perfecta\label{rep:II.2}}
\subsub{a) Supuestos y modelo}
\begin{itemize}
\item $J$ empresas competitivas; \concepto{trabajo efectivo} $eL$; $B=AF(eL)-wL$.
  \begin{itemize}
  \item Esfuerzo $e(w)$, $e'>0$, en forma de S; observable.
  \item $\overline{L}$ trabajadores con oferta inelástica.
  \end{itemize}
\item Con paro, la empresa elige $L$ y $w$: $\max_{L,w}\ AF(e(w)L)-wL$.
  \begin{itemize}
  \item CPO: $AF'e=w$ y $AF'e'L=L$; dividiendo, la \concepto{condición de \autor{Solow}}:
\funcionclave{\varepsilon_{e,w}\equiv\frac{w^*\,e'(w^*)}{e(w^*)}=1}
  \item $w^*$ \emph{minimiza el coste por unidad de trabajo efectivo} $w/e(w)$: rayo desde el origen tangente a $e(w)$.
  \item CSO: $e''(w^*)<0$; si $e$ cóncava desde $e(0)>0$, no hay salario de eficiencia.
  \end{itemize}
\item \emph{Empleo}: $AF'(e^*L)=w^*/e^*$ $\Rightarrow$ $L^*$; si $JL^*<\overline{L}$, paro $\overline{L}-JL^*$.
\end{itemize}
\graficorepaso[0.9\linewidth]{solow-esfuerzo}{\autor{Solow}: (a) rayo tangente a $e(w)$ en $T$, $\varepsilon_{e,w}=1$; (b) $e(w)$ cóncava: sin salario de eficiencia}
\subsub{b) Implicaciones}
\begin{itemize}
\item \emph{Paro involuntario}: bajar $w$ cuesta más en esfuerzo.
\item \emph{Rigidez real}: $w^*$ solo depende de $e(\cdot)$, no de $A$ ni de $L^D$ $\Rightarrow$ ajuste por empleo.
  \begin{itemize}
  \item Amplifica las rigideces nominales (\autor{Ball}-\autor{Romer}, 1990) \vertema{3.A.7}.
  \end{itemize}
\end{itemize}
\subsub{c) Extensión: el paro de equilibrio (\autor{Summers}, 1988)}
\begin{itemize}
\item El esfuerzo depende de lo que se paga sobre la alternativa (el salario de fuera, ponderado por la probabilidad de encontrar empleo).
  \begin{itemize}
  \item Cada empresa quiere pagar una prima, pero en equilibrio todas pagan lo mismo $\Rightarrow$ una tasa de paro que fijan los parámetros del esfuerzo, no $A$.
  \item El crecimiento ya no elimina el paro, acaba en salarios; otra WS \vertema{3.A.28}.
  \end{itemize}
\end{itemize}
\subsub{d) Valoración}
\begin{itemize}
\item + paro involuntario sencillo, sin poder de mercado.
\item $-$ rigidez \emph{demasiado} fuerte (el paro iría a 0~con el crecimiento); no dice \emph{por qué} $e$ depende de $w$ $\Rightarrow$ \autor{Shapiro}-\autor{Stiglitz}.
\end{itemize}

\sub{II.3. \autor{Shapiro}-\autor{Stiglitz} (1984): información asimétrica\label{rep:II.3}}
\subsub{a) Idea}
\begin{itemize}
\item La empresa no ve el esfuerzo; su única sanción es despedir.
  \begin{itemize}
  \item Sin paro el despido no duele $\Rightarrow$ tiene que haber paro: \emph{mecanismo de disciplina}.
  \end{itemize}
\end{itemize}
\subsub{b) Modelo}
\subsubsub{i) Supuestos}
\begin{itemize}
\item Tiempo continuo, estado estacionario; $\overline{L}$ trabajadores neutrales al riesgo, descuento $\rho$; $J$ empresas.
\item Esfuerzo $\overline{e}$ (cuesta $\overline{e}$) o $0$; utilidad $w-\overline{e}$ (E), $w$ (S, \emph{shirking}), 0 (U).
\item Transiciones de \autor{Poisson}: \concepto{separación} $s$ y \concepto{detección} $q$ exógenas; hallar empleo $f$, endógena.
\end{itemize}
\subsubsub{ii) Desarrollo}
\begin{itemize}
\item \emph{Valores} (como rendimiento de un activo):
\begin{formula}
\centering $\rho V_{\mathrm E}=(w-\overline{e})-s(V_{\mathrm E}-V_{\mathrm U})$\\
$\rho V_{\mathrm S}=w-(s+q)(V_{\mathrm S}-V_{\mathrm U})$,\quad $\rho V_{\mathrm U}=f(V_{\mathrm E}-V_{\mathrm U})$
\end{formula}
\item \concepto{Compatibilidad de incentivos} $V_{\mathrm E}\geq V_{\mathrm S}$ \vertema{3.A.13} $\Rightarrow$ $V_{\mathrm E}-V_{\mathrm U}=\overline{e}/q$: \emph{renta} del empleo.
\item Estado estacionario $sJL=f(\overline{L}-JL)$ $\Rightarrow$ $f+s=s/U$; \concepto{condición de no escaqueo} (CNE):
\funcionclave{w\geq\overline{e}+\Big(\rho+\frac{s}{U}\Big)\frac{\overline{e}}{q}}
  \begin{itemize}
  \item ↑ con $\overline{e}$, $\rho$, $s$; ↓ con $q$ y con el paro.
  \item $U\to0$ $\Rightarrow$ $w\to\infty$: con pleno empleo ningún salario disuade.
  \end{itemize}
\item \emph{Equilibrio} en $(JL,w)$:
  \begin{itemize}
  \item demanda $A\overline{e}F'(\overline{e}L)=w$; oferta horizontal en $\overline{e}$ hasta $\overline{L}$;
  \item CNE creciente, vertical en $\overline{L}$;
  \item walrasiano $E^W$ (pleno empleo); con información asimétrica, $E$: $w^*$ y \emph{paro}.
  \end{itemize}
\end{itemize}
\graficorepaso[0.62\linewidth]{shapiro-stiglitz}{CNE (sombreada: sin escaqueo), $E^W$ y $E$ con paro}
\subsubsub{iii) Estática comparativa}
\begin{itemize}
\item ↑$q$ baja la CNE: ↓$w$, ↑$L$ ($q\to\infty$: $E^W$).
\item $s=0$ $\Rightarrow$ CNE horizontal, $w=\overline{e}+\rho\overline{e}/q$.
\item ↓$A$: con supervisión perfecta solo ↓$w$; con imperfecta, a lo largo de la CNE: ↓$L$ y algo ↓$w$.
\item ↑$\overline{e}$, $\rho$, $s$ o subsidio de paro suben la CNE.
\end{itemize}
\subsubsub{iv) Implicaciones}
\begin{itemize}
\item Paro involuntario de equilibrio: los parados «no pueden prometer de forma creíble que no se escaquearán».
\item \emph{Ineficiencia}: imponer pleno empleo no sirve; sí \emph{subsidio salarial} con impuesto sobre beneficios.
  \begin{itemize}
  \item Subsidio de paro más generoso \emph{aumenta} el paro.
  \item Paro no keynesiano, pero la demanda mueve el empleo a corto plazo.
  \end{itemize}
\end{itemize}
\subsub{c) Extensiones}
\begin{itemize}
\item \emph{Mercados duales} (\autor{Bulow}-\autor{Summers}, 1986): sector primario con rentas y colas.
\item Otros costes del despido (capital específico) $\Rightarrow$ menos paro necesario.
\end{itemize}
\subsub{d) Evidencia empírica}
\begin{itemize}
\item \autor{Ford}: productividad +40-70\,\% (\autor{Raff}-\autor{Summers}, 1986, 1987).
\item Primas entre industrias, menos abandonos (\autor{Krueger}-\autor{Summers}, 1988); menos despidos disciplinarios con más prima y más paro (\autor{Cappelli}-\autor{Chauvin}, 1991).
\item Laboratorio (\autor{Fehr}-\autor{Kirchsteiger}-\autor{Riedl}, 1993); directivos temen la moral (\autor{Bewley}, 1999).
\end{itemize}
\subsub{e) Valoración}
\begin{itemize}
\item + fundamento con información asimétrica; pieza de la NEK \vertema{3.A.7}.
\item $-$ \emph{fianzas} (\autor{Carmichael}, 1985) o salarios crecientes con antigüedad (\autor{Lazear}, 1979) lo evitarían.
  \begin{itemize}
  \item Respuesta: sin liquidez, y la empresa acusaría en falso.
  \end{itemize}
\item $-$ CNE calibrada demasiado empinada; regalo que se diluye (\autor{Gneezy}-\autor{List}, 2006).
\end{itemize}

\bloque[5]{III. Contratos implícitos\label{rep:III}}
\begin{ideaclave}
Con relaciones largas e incertidumbre, la empresa (neutral al riesgo) asegura a sus trabajadores (aversos) con una renta estable: el salario no sigue a la productividad, pero el empleo pactado es el eficiente. Explican la \emph{rigidez}, no el paro; el paro, los \emph{insiders}: «a las duras y a las maduras».
\end{ideaclave}

\sub{III.1. Idea\label{rep:III.1}}
\begin{itemize}
\item ¿Por qué $w$ real apenas responde a los \emph{shocks}? Relaciones de décadas (\autor{Hall}, 1982).
\item \autor{Baily} (1974), \autor{Gordon} (1974), \autor{Azariadis} (1975): un \emph{seguro}.
  \begin{itemize}
  \item Trabajador averso sin seguro de su renta laboral; dueños diversificados, neutrales.
  \item \concepto{Implícito}: no escrito ni exigible ante un juez; lo sostiene la reputación.
  \end{itemize}
\end{itemize}

\sub{III.2. Modelo\label{rep:III.2}}
\begin{itemize}
\item \emph{Supuestos}: estados $j$ con productividad $A_j$ y probabilidad $\pi_j$, observables.
  \begin{itemize}
  \item Empresa neutral: produce $A_jF(L_j)$, paga renta $C_j$, pide $L_j$ horas.
  \item Trabajador: $u(C_j)-v(L_j)$, $u$ cóncava, $v$ convexa; \concepto{utilidad de reserva} $U^R$.
  \item Contrato \emph{antes} de conocer el estado:
  \end{itemize}
\funcionclave{\optim{\max}{\{C_j,L_j\}}{\textstyle\sum_j\pi_j\big[A_jF(L_j)-C_j\big]}{\textstyle\sum_j\pi_j\big[u(C_j)-v(L_j)\big]\geq U^R}}
\item CPO: $u'(C_j)=1/\lambda$ $\Rightarrow$ $C_j=\overline{C}$: \concepto{seguro completo} (el neutral carga el riesgo \vertema{3.A.10}); empleo \emph{eficiente}:
\begin{formula}
\centering $A_jF'(L_j)=\dfrac{v'(L_j)}{u'(\overline{C})}$\quad (más horas en los estados buenos)
\end{formula}
\item \emph{Implicaciones}:
  \begin{itemize}
  \item rigidez del $w$ real; salario medio menor que al contado: \emph{prima de riesgo};
  \item salario \emph{por hora} $\overline{C}/L_j$ incluso contracíclico;
  \item \emph{no hay paro involuntario}: $w$ deja de ser el precio que decide el empleo (\autor{Barro}, 1977).
  \end{itemize}
\end{itemize}

\sub{III.3. Extensiones e \emph{insiders}\label{rep:III.3}}
\subsub{a) Extensiones del modelo básico}
\begin{itemize}
\item \emph{Despidos}: trabajo indivisible; sin indemnización, paro involuntario \emph{a posteriori}.
\item \emph{Información asimétrica} (solo la empresa ve el estado): el contrato que le hace decir la verdad distorsiona el empleo (sub- o sobreempleo).
\item \emph{Esfuerzo no observable}: el seguro completo quita incentivos; el contrato deja parte del riesgo \vertema{3.A.13}.
\item \emph{Autoexigibilidad}: reputación, juego repetido (\autor{Bull}, 1987) \vertema{3.A.14}; $w$ sube si el mercado se tensa, no baja si se relaja.
\end{itemize}
\subsub{b) \emph{Insiders} y \emph{outsiders} (\autor{Lindbeck}-\autor{Snower}, 1988)}
\begin{itemize}
\item \emph{Insiders}: empleados que negocian; \emph{outsiders}: parados o precarios, fuera de la mesa.
\item Poder: \concepto{costes de rotación} (despido $c_D$, contratación $c_C$; formación específica \vertema{3.A.25}).
  \begin{itemize}
  \item \emph{Outsiders} cobran $w_O$; salario máximo de los \emph{insiders}:
\funcionclave{w_I=w_O+c_D+c_C}
  \end{itemize}
\item Casos:
  \begin{itemize}
  \item A, pocos \emph{insiders}: cobran el máximo y se contratan \emph{outsiders};
  \item B, intermedio: ni contrata ni despide, $w_I=AF'(\overline{L}_I)+c_D$ (empleo = historia);
  \item C, muchos: despide, $w^R$.
  \end{itemize}
\end{itemize}
\graficorepaso[0.62\linewidth]{insiders}{Salario máximo de los \emph{insiders} según cuántos sean (casos A, B, C)}
\begin{itemize}
\item \emph{Implicaciones}: paro involuntario de \emph{outsiders}; \emph{histéresis} (\autor{Blanchard}-\autor{Summers}, 1986) \vertema{3.A.28}.
  \begin{itemize}
  \item \emph{Dualidad}: en la Gran Recesión el ajuste recayó en los temporales (\autor{Bentolila} et al., 2012) \vertema{4.A.16}.
  \item Política: ↓costes de rotación, descuelgue, formación, subvención a contratar parados.
  \end{itemize}
\end{itemize}

\sub{III.4. Valoración\label{rep:III.4}}
\begin{itemize}
\item + estabilidad del $w$ real con ajuste por empleo, sin poder de mercado; rigidez $\neq$ ineficiencia.
\item \emph{Evidencia}: $w$ depende del \emph{paro más bajo} desde la entrada en la empresa (\autor{Beaudry}-\autor{DiNardo}, 1991); ERTE (2020): seguro explícito que completa al implícito.
\item $-$ sin extensiones no hay paro; la rigidez nominal tiene otras causas (menú, contratos escalonados) \vertema{3.A.7}.
\end{itemize}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}: el salario no solo vacía el mercado:
  \begin{itemize}
  \item negociación: poder; paro si solo se negocia $w$;
  \item eficiencia: la empresa paga más; el paro disciplina;
  \item contratos implícitos: seguro, rigidez sin paro; el paro, con \emph{insiders}.
  \end{itemize}
\item Los 3: microfundamentos de la WS \vertema{3.A.28}.
\end{itemize}

\item \textbf{Relevancia}: paro sobre todo \emph{de equilibrio} (instituciones); con $w$ real rígido la demanda mueve mucho el empleo (\autor{Ball}-\autor{Romer}, 1990).
\item \textbf{Extensiones}: \autor{Nash} en búsqueda (\autor{Hall}, 2005) \vertema{3.A.26}; WS, NAIRU, histéresis \vertema{3.A.28}; nivel de negociación \vertema{4.A.16}; unión monetaria \vertema{3.B.16}.
\item \textbf{Opinión}: ninguna basta sola; cada una encaja con un hecho; \autor{Ford}, leído con varias. Abaratar el trabajo no siempre crea empleo; en España, negociación sectorial y dualidad.
\item \textbf{Idea final}: \autor{Ford} no pagaba de más: pagaba lo que le salía más barato. Lo barato sale caro: que el salario más alto no lo paguen siempre los de fuera.

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Qué modelos generan paro involuntario?} Monopolista, derecho a gestionar, eficiencia, \emph{insiders}; no contratos eficientes ni implícitos.
\item \textbf{¿La moderación salarial crea empleo?} Con derecho a gestionar sí; en la curva de contratos, no.
\item \textbf{¿Por qué $\varepsilon_{e,w}=1$?} $w^*$ minimiza $w/e(w)$.
\item \textbf{¿Por qué el pleno empleo es incompatible con la CNE?} $U\to0$: el despedido se recoloca al instante.
\item \textbf{¿Qué hace un subsidio de paro mayor en \autor{Shapiro}-\autor{Stiglitz}?} Sube la CNE y el paro.
\item \textbf{¿Cómo cubren los convenios al 92\,\% con un 12,5\,\% de afiliación?} Eficacia general (art. 82.3~ET).
\end{enumerate}
\begin{trampa}
\begin{itemize}
\item \autor{Nash}: transformaciones \emph{afines} (no «lineales»).
\item \autor{Rubinstein}: horizonte \emph{infinito} (el finito, \autor{Ståhl}, 1972).
\item Monopolista: ENPS no \autor{Pareto}; derecho a gestionar: \emph{no} ENPS.
\item \autor{Solow}: minimiza $w/e(w)$, no $e/w$.
\item Escaqueo: \autor{Shapiro}-\autor{Stiglitz}, no \autor{Akerlof}; rotación: \autor{Salop} (1979), \autor{Stiglitz} (1974).
\item CNE con $s/U$, no $s$; $s=0$ $\Rightarrow$ CNE horizontal.
\item Curva de contratos: vertical si neutral; creciente si averso (sobreempleo).
\item Contratos implícitos: \emph{sin} paro involuntario; \emph{insiders}: $w_I\leq w_O+c_D+c_C$.
\end{itemize}
\end{trampa}
