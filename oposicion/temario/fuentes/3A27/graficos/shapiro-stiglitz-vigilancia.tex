% Shapiro y Stiglitz (1984), estática comparativa (i): más vigilancia (q sube).
% Código base de shapiro-stiglitz.tex: Lbar = 6, ebar = 1;
% demanda w = 0,4 + 6 e^{-0,25 JL}; CNE w = 2 + 1,5/(6 - JL) (rho/q = 1, s/q = 0,25).
% Con q × 1,5: rho/q = 2/3 y s/q = 1/6, CNE' w = 5/3 + 1/(6 - JL), que llega a
% w = 6,4 en JL = 6 - 1/4,7333 = 5,7887. Cortes con la demanda (calculados):
% E en JL = 3,8433; E' en JL = 4,5156.
% Sombreado: zona sin escaqueo con la CNE inicial (morado) y la que se gana con
% más vigilancia, entre CNE' y CNE (dorado).
% Elaboración propia a partir de Shapiro y Stiglitz (1984), p. 439, y Romer
% (2019), cap. 11. Autor: Víctor Gutiérrez Marcos
\begin{tikzpicture}[tcee, y=1.3cm,
    demanda/.style={very thick,tceerojo},
    declare function={LD(\x)=0.4+6*exp(-0.25*\x); CNE(\x)=2+1.5/(6-\x);
                      CNEq(\x)=5/3+1/(6-\x);}]
  % 1: ejes
  \capa{1}{\ejes{$J\,L$}{$w$}{7.2}{6.6}}
  % 2: demanda y oferta de trabajo, y equilibrio walrasiano
  \capa{2}{
    \draw[demanda, domain=0.25:6, samples=60] plot (\x,{LD(\x)});
    \node[etiqueta, text=tceerojo, anchor=west] at (0.55,5.85) {$L^D$};
    \draw[curva2] (0,1) -- (6,1) -- (6,6.4) node[etiqueta, above] {$L^S$};
    \node[left] at (0,1) {$\overline{e}$};
    \node[below] at (6,0) {$\overline{L}$};
    \draw[guia] (6,0) -- (6,1);
    \node[punto] at (6,{LD(6)}) {};
    \node[etiqueta, right] at (6,{LD(6)}) {$E^W$};
  }
  % 3: CNE inicial con su zona sin escaqueo y el equilibrio E
  \capa{3}{
    \fill[area] plot[domain=0:5.6591, samples=80] (\x,{CNE(\x)}) -- (0,6.4) -- cycle;
    \draw[curva1, domain=0:5.6591, samples=80] plot (\x,{CNE(\x)});
    \node[etiqueta, text=tceemorado, anchor=east] at (5.5,5.6) {CNE};
    \coordinate (E) at (3.8433,{CNE(3.8433)});
    \node[punto] at (E) {};
    \node[etiqueta, above left] at (E) {$E$};
    \draw[guia] (E) -- (3.8433,0) node[below,xshift=-6pt] {$J\,L^*$};
    \draw[guia] (E) -- (0,{CNE(3.8433)}) node[left] {$w^*$};
  }
  % 4: más vigilancia: la CNE baja y se amplía la zona sin escaqueo
  \capa{4}{
    \fill[area2] plot[domain=0:5.6591, samples=80] (\x,{CNE(\x)}) -- (5.7887,6.4)
      -- plot[domain=5.7887:0, samples=80] (\x,{CNEq(\x)}) -- cycle;
    \draw[curva1, desplazada, domain=0:5.7887, samples=80] plot (\x,{CNEq(\x)});
    \node[etiqueta, text=tceemorado, anchor=north west] at (0.1,{CNEq(0)-0.05}) {CNE$'$ ($q\uparrow$)};
    \draw[flecha, tceemorado] (4.95,{CNE(4.95)}) -- (5.25,{CNEq(5.25)-0.05});
  }
  % 5: nuevo equilibrio E', con más empleo, menos salario y menos paro
  \capa{5}{
    \coordinate (E2) at (4.5156,{CNEq(4.5156)});
    \node[punto] at (E2) {};
    \node[etiqueta, below left] at (E2) {$E'$};
    \draw[guia] (E2) -- (4.5156,0) node[below,xshift=6pt] {$J\,L^{*\prime}$};
    \draw[guia] (E2) -- (0,{CNEq(4.5156)}) node[left] {$w^{*\prime}$};
    \draw[decorate, decoration={brace, amplitude=4pt}, thick, tceerojo]
      (4.5156,0.2) -- (6,0.2);
    \node[etiqueta, text=tceerojo, anchor=west] at (6.05,0.17) {paro$'$};
  }
\end{tikzpicture}
\end{document}
