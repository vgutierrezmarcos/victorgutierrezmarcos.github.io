% Los modelos de negociación colectiva en un solo plano (gráfico conjunto de
% la pizarra): sindicato monopolista (M), derecho a gestionar (R), competencia
% (C) y curva de contratos (C-D-F). Elaboración propia a partir de McDonald y
% Solow (1981) y Heijdra (2017), pp. 244-253. Rehace la imagen 5 del original
% (image12.jpg). Es la suma de los paneles de aparato-grafico.tex con los puntos
% de sindicato-monopolista, derecho-gestionar y contratos-eficientes.
%
% CÓDIGO BASE COMÚN de los 5 gráficos del bloque I del 3.A.27 (aparato-grafico,
% sindicato-monopolista, derecho-gestionar, contratos-eficientes y
% negociacion-colectiva): mismas funciones, escala y puntos.
%   Empresa: F(L) = L^a con a = 0,4 (Cobb-Douglas: demanda isoelástica).
%     Unidades: w^R = z = 2,2 y L^c = 4,4 (A elegido para ello).
%     demanda   L^D:  w = A a L^{a-1} = 2,2 (L/4,4)^{-0,6}          dem(L)
%     producto medio  w = A L^{a-1}   = 5,5 (L/4,4)^{-0,6}          ap(L)
%     isobeneficio por (p,q): w = ap(L) - (ap(p) - q) p / L        iso(L,p,q)
%       (B = p (ap(p) - q); si (p,q) está sobre L^D, su máximo está en p; la
%        de B = 0 es el producto medio).
%   Sindicato: u(w) = ln w (aversión al riesgo), Lbar = 6,6 = 1,5 L^c.
%     indiferencia por (p,q): (L/Lbar) ln(w/z) = (p/Lbar) ln(q/z), es decir,
%     w = z (q/z)^{p/min(L,Lbar)}                                  ind(L,p,q)
%     (decreciente y convexa; horizontal desde Lbar). Las curvas se cortan arriba
%     en w = 5,6: los dominios de cada plot empiezan donde la curva vale 5,6.
%   C = (4,4; 2,2): A F'(L^c) = w^R.
%   M (sindicato monopolista, gamma = 1): ln(w^S/z) = 1 - a  =>  w^S = z e^{0,6}
%     = 4,0087; L^S = L^c e^{-1} = 1,6187 (tangencia con la indiferencia).
%   R (derecho a gestionar, gamma = 0,5, Nash sobre L^D): (2,1540; 3,3771).
%   Curva de contratos: A F'(L) = w - (u(w) - u(z))/u'(w) = w (1 - ln(w/z));
%     con t = w/z: L = 4,4 (t (1 - ln t))^{-5/3}, w = 2,2 t.
%     P (tangencia intermedia): t = 1,36289 -> (4,8700; 2,9984).
%     D (llega a Lbar): t = 1,72579 -> (6,6; 3,7967).
%     F (sobre B = 0 en Lbar): (6,6; ap(6,6) = 4,3123).
% Autor: Víctor Gutiérrez Marcos

\begin{tikzpicture}[tcee,x=0.85cm,y=1cm,
    declare function={
      dem(\x)=2.2*exp(-0.6*ln(\x/4.4));
      ap(\x)=5.5*exp(-0.6*ln(\x/4.4));
      iso(\x,\p,\q)=ap(\x)-(ap(\p)-\q)*\p/\x;
      ind(\x,\p,\q)=2.2*exp(\p*ln(\q/2.2)/min(\x,6.6));
    },
    demanda/.style={very thick,tceerojo},
    isob/.style={thick,tceenaranja},
    indif/.style={thick,tceeazul},
    rotulo/.style={etiqueta,font=\footnotesize},
    contratos/.style={line width=2.2pt,tceeverde}]
  \def\xM{1.6187}\def\yM{4.0087}
  \def\xR{2.1540}\def\yR{3.3771}
  \def\xP{4.8700}\def\yP{2.9984}
  \def\yD{3.7967}
  % 1: ejes
  \capa{1}{\ejes{$L$}{$w$}{9.8}{6.3}}
  % 2: demanda de trabajo
  \capa{2}{\draw[demanda] plot[domain=0.92:8.6,samples=60] (\x,{dem(\x)}) node[etiqueta,right,text=tceerojo] {$L^D$};}
  % 3: isobeneficio con su máximo sobre L^D (más beneficio cuanto más bajas)
  \capa{3}{\draw[isob] plot[domain=0.794:9.4,samples=80] (\x,{iso(\x,\xM,\yM)});}
  \capa{3}{\draw[isob] plot[domain=1.003:9.4,samples=80] (\x,{iso(\x,\xR,\yR)});}
  \capa{3}{\draw[isob] plot[domain=1.774:9.4,samples=80] (\x,{iso(\x,4.4,2.2)});}
  % 4: salario de reserva y afiliados
  \capa{4}{\draw[indif] (0,2.2) node[left,text=black] {$w^R$} -- (9.4,2.2);}
  \capa{4}{\draw[guia] (6.6,0) node[below,text=black] {$\overline{L}$} -- (6.6,6.0);}
  % 5: indiferencias del sindicato, horizontales desde Lbar
  \capa{5}{\draw[indif] plot[domain=1.0395:6.6,samples=80] (\x,{ind(\x,\xM,\yM)}) -- (9.4,{ind(9,\xM,\yM)});}
  \capa{5}{\draw[indif] plot[domain=1.614:6.6,samples=80] (\x,{ind(\x,\xP,\yP)}) -- (9.4,{ind(9,\xP,\yP)});}
  \capa{5}{\draw[indif] plot[domain=3.855:6.6,samples=60] (\x,{ind(\x,6.6,\yD)}) -- (9.4,\yD);}
  % 6: competencia, C (gamma = 0)
  \capa{6}{\node[punto] (C) at (4.4,2.2) {};}
  \capa{6}{\node[etiqueta,anchor=west] at (4.45,1.5) {$C;\ \gamma=0$};}
  % 7: sindicato monopolista, M (gamma = 1)
  \capa{7}{\node[punto] (M) at (\xM,\yM) {};}
  \capa{7}{\node[etiqueta,right=2pt] at (M) {$M$};}
  % 8: derecho a gestionar, R
  \capa{8}{\node[punto] (R) at (\xR,\yR) {};}
  \capa{8}{\node[etiqueta,right=2pt] at (R) {$R$};}
  % 9: curva de contratos C-D-F e isobeneficio de beneficio nulo
  \capa{9}{\draw[contratos] plot[variable=\t,domain=1:1.72579,samples=40] ({4.4*exp(-ln(\t*(1-ln(\t)))/0.6)},{2.2*\t});}
  \capa{9}{\node[punto] (D) at (6.6,\yD) {};}
  \capa{9}{\node[etiqueta,right] at (D) {$D$};}
  \capa{9}{\draw[isob] plot[domain=4.27:9.4,samples=60] (\x,{ap(\x)}) node[rotulo,right,text=tceenaranja] {$B=0$};}
  \capa{9}{\draw[contratos] (D) -- (6.6,4.3123);}
  \capa{9}{\node[punto] (F) at (6.6,4.3123) {};}
  \capa{9}{\node[etiqueta,above right] at (F) {$F;\ \gamma=1$};}
\end{tikzpicture}
