\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
% Los 3 movimientos de la curva (esquema): paralelo (nivel), giro (pendiente:
% aplanamiento y positivización) y mariposa (curvatura). En gris, la curva de
% partida; en color, la nueva. Curva de partida: 0,8 + 1,4 (1 - e^{-n/1,2}).
\begin{tikzpicture}[tcee]
  \newcommand{\panel}[1]{%
    \draw[eje] (0,0) -- (4.2,0) node[below left=1pt and 0pt,font=\footnotesize] {plazo};
    \draw[eje] (0,0) -- (0,3.3) node[left,font=\footnotesize] {$r_n$};
    \node[etiqueta] at (2.1,3.65) {#1};}
  \newcommand{\base}{\draw[thick,tceegris] plot[domain=0.1:3.9,samples=40,smooth] (\x,{0.8+1.4*(1-exp(-\x/1.2))});}
  % 1: paralelo
  \capa{1}{\panel{Paralelo (nivel)}\base}
  \capa{2}{\draw[curva1] plot[domain=0.1:3.9,samples=40,smooth] (\x,{1.4+1.4*(1-exp(-\x/1.2))});
    \draw[->,thick,tceemorado] (2.8,2.05) -- (2.8,2.6);}
  \begin{scope}[xshift=5.4cm]
    % 3: giro
    \capa{3}{\panel{Giro (pendiente)}\base}
    \capa{4}{\draw[curva2] plot[domain=0.1:3.9,samples=40,smooth] (\x,{1.2+0.75*(1-exp(-\x/1.2))});
      \node[etiqueta,font=\footnotesize,text=tceeazul,anchor=north east] at (4.0,1.85) {aplanamiento};}
    \capa{5}{\draw[curva3] plot[domain=0.1:3.9,samples=40,smooth] (\x,{0.5+2.1*(1-exp(-\x/1.2))});
      \node[etiqueta,font=\footnotesize,text=tceeverde,anchor=south east] at (4.0,2.6) {positivización};}
  \end{scope}
  \begin{scope}[xshift=10.8cm]
    % 6: mariposa
    \capa{6}{\panel{Mariposa (curvatura)}\base}
    \capa{7}{\draw[curva4] plot[domain=0.1:3.9,samples=60,smooth] (\x,{0.8+1.4*(1-exp(-\x/1.2))+0.7*exp(-(\x-1.7)*(\x-1.7)/0.45)});
      \node[etiqueta,font=\footnotesize,text=tceedorado,anchor=south] at (1.7,2.75) {plazos medios};}
  \end{scope}
\end{tikzpicture}
\end{document}
