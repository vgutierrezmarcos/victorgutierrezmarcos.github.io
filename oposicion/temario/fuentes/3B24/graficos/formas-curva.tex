\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
% Las 4 formas de la curva de tipos (esquema): creciente, plana, invertida y
% jorobada. Un panel por forma, mismos ejes.
\begin{tikzpicture}[tcee]
  \newcommand{\panel}[1]{%
    \draw[eje] (0,0) -- (3.6,0) node[below left=1pt and 0pt,font=\footnotesize] {plazo};
    \draw[eje] (0,0) -- (0,2.8) node[left,font=\footnotesize] {$r_n$};
    \node[etiqueta] at (1.8,3.15) {#1};}
  % 1: creciente
  \capa{1}{\panel{Creciente}}
  \capa{2}{\draw[curva1] plot[domain=0.1:3.3,samples=40,smooth] (\x,{0.6+1.5*(1-exp(-\x/1.1))});}
  \begin{scope}[xshift=4.6cm]
    % 3: plana
    \capa{3}{\panel{Plana}}
    \capa{4}{\draw[curva2] plot[domain=0.1:3.3,samples=40,smooth] (\x,{1.45+0.08*(1-exp(-\x/1.1))});}
  \end{scope}
  \begin{scope}[xshift=9.2cm]
    % 5: invertida
    \capa{5}{\panel{Invertida}}
    \capa{6}{\draw[curvanueva] plot[domain=0.1:3.3,samples=40,smooth] (\x,{2.3-1.3*(1-exp(-\x/1.1))});}
  \end{scope}
  \begin{scope}[xshift=13.8cm]
    % 7: jorobada
    \capa{7}{\panel{Jorobada}}
    \capa{8}{\draw[curva4] plot[domain=0.1:3.3,samples=40,smooth] (\x,{1.0+2.2*(\x/0.9)*exp(-\x/0.9)});}
  \end{scope}
\end{tikzpicture}
\end{document}
