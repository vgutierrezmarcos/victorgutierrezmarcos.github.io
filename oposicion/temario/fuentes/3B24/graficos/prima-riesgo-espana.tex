\begin{tikzpicture}[tcee]
\begin{axis}[width=14cm,height=6.2cm,xmin=1993,xmax=2027,ymin=-50,ymax=600,

  xtick={1995,2000,...,2025},ytick={0,100,...,600},
  x tick label style={/pgf/number format/1000 sep={}},ticklabel style={font=\footnotesize},

  % 3: la prima de riesgo de Francia
  \capa{3}{\addplot[very thick,tceenaranja] table[x expr=1993+\coordindex/12,y=prima_fr_pb] {\pr};}

    \dicrotulo[font=\footnotesize,anchor=west,text=tceenaranja]{(2001,410)}{Francia}
    \dicrotulo[font=\footnotesize,anchor=west,text=tceemorado]{(2012.8,560)}{555 (julio de 2012)}

    \dicrotulo[font=\footnotesize,anchor=west,text=tceemorado]{(2027.1,55)}{65}
    \dicrotulo[font=\footnotesize,anchor=west,text=tceenaranja]{(2027.1,150)}{147}}
\end{axis}
