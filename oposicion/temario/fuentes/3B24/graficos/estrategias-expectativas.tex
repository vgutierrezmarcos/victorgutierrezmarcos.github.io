\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
% Figura: 2 estrategias para invertir 100 € a 2 años (anexo de ejemplos).
% A: un bono a 2 años al 10 %: 100 -> 121. B: un bono a 1 año al 8 % (108) y
% reinvertir al tipo esperado; con E[i] = 12,04 % se llega también a 121.
\begin{tikzpicture}[tcee,caja/.style={draw,rounded corners=2pt,minimum width=1.5cm,minimum height=0.75cm,font=\small}]
  % 1: eje del tiempo
  \capa{1}{\foreach \x/\t in {0/0,4/1,8/2}{\node[font=\small] at (\x,3.3) {$t+\t$};}
    \draw[thin,tceegris] (-0.8,3.0) -- (8.8,3.0);}
  % 2: estrategia A
  \capa{2}{\node[caja] (a0) at (0,2) {100~€};
    \node[caja,draw=tceemorado,text=tceemorado] (a2) at (8,2) {121~€};
    \draw[->,very thick,tceemorado] (a0) -- node[above,font=\footnotesize] {bono a 2 años: $(1+i^{(2)}_t)^2=1{,}10^2$} (a2);
    \node[anchor=east,font=\footnotesize] at (-0.9,2) {A};}
  % 3: estrategia B, primer año
  \capa{3}{\node[caja] (b0) at (0,0) {100~€};
    \node[caja] (b1) at (4,0) {108~€};
    \draw[->,very thick,tceeazul] (b0) -- node[below=2pt,font=\footnotesize] {$1+i_t=1{,}08$} (b1);
    \node[anchor=east,font=\footnotesize] at (-0.9,0) {B};}
  % 4: estrategia B, segundo año al tipo esperado
  \capa{4}{\node[caja,draw=tceeazul,text=tceeazul] (b2) at (8,0) {121~€};
    \draw[->,very thick,tceeazul,dashed] (b1) -- node[below=2pt,font=\footnotesize] {$1+\mathbb{E}_t[i_{t+1}]=1{,}1204$} (b2);}
  % 5: igualdad en equilibrio
  \capa{5}{\draw[<->,thick,tceegris] (a2) -- node[right,font=\footnotesize,align=left] {iguales en\\equilibrio} (b2);}
\end{tikzpicture}
\end{document}
