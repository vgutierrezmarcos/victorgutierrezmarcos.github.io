\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: «\emph{unos por otros, la casa sin barrer}»: si lo que hago beneficia a todos y nadie me lo paga, espero a que lo haga otro.
  \begin{itemize}
  \item Clima: la UE se fija por ley $-90$\,\% de emisiones netas en 2040 (frente a 1990), pero retrasa el RCDE~2~de 2027 a 2028.

  \end{itemize}
\item \textbf{Relevancia}:
  \begin{itemize}
  \item \emph{Teórica}: los fallos justifican intervenir por eficiencia; sin ellos basta el 1TFEB \vertema{3.A.22}.
  \item \emph{Práctica}: el clima, «el mayor fallo de mercado» (\autor{Stern}) \vertema{3.B.31}; CO\textsubscript{2} a $\approx$85~€/t (oct. 2026).
  \item \emph{TCEE}: fiscalidad ambiental, defensa, I+D, regulación \vertema{3.A.20}: comparar siempre 2~imperfecciones.
  \end{itemize}
\item \textbf{Contextualización}:
  \begin{itemize}
  \item \autor{Pareto} (1906); \autor{Arrow}, \autor{Debreu} (1951): teoremas.
  \item Fallos: \autor{Pigou} (1920), \autor{Samuelson} (1954), \autor{Coase} (1960).
  \item Del Estado: \autor{Buchanan}-\autor{Tullock} (1962), \autor{Kydland}-\autor{Prescott} (1977).
  \end{itemize}

  \begin{itemize}

  \item ¿Falla también el Estado? ¿Cuándo intervenir?
  \end{itemize}
\item \textbf{Estructura}: ¿cuándo falla?, ¿cómo se corrige?, ¿quién corrige al que corrige?
  \begin{itemize}

\end{itemize}

\bloque[3]{I. Optimalidad del EGC y fallos de mercado\label{rep:I}}
\begin{ideaclave}
El equilibrio competitivo es eficiente solo si los precios recogen todos los costes y todas las valoraciones; cada fallo de mercado es un supuesto del 1TFEB que no se cumple.
\end{ideaclave}

\sub{I.1. Condiciones de optimalidad\label{rep:I.1}}
\begin{itemize}
\item \concepto{Óptimo de \autor{Pareto}}: nadie puede mejorar sin que otro empeore.
\item Economía 2×2×2×2 (\autor{Bator}, 1957), 3~condiciones:
  \begin{itemize}
  \item \emph{producción}: $\mathrm{RMST}^X_{LK}=\mathrm{RMST}^Y_{LK}$ (en la FPP);
  \item \emph{intercambio}: $\mathrm{RMS}^A_{XY}=\mathrm{RMS}^B_{XY}$ (curva de contrato);
  \item \emph{global}: $\mathrm{RMS}^A_{XY}=\mathrm{RMS}^B_{XY}=\mathrm{RMT}_{XY}$ $\Leftarrow$ la que rompen externalidades y bienes públicos.
  \end{itemize}
\end{itemize}

\sub{I.2. EGC y teoremas del bienestar\label{rep:I.2}}
\begin{itemize}
\item \concepto{EGC}: $\vect{p}^*$ vacía todos los mercados \vertema{3.A.21}; todos ven los mismos precios $\Rightarrow$ se cumplen las 3:
\begin{formula}
\centering $\mathrm{RMS}^A_{XY}=\mathrm{RMS}^B_{XY}=\dfrac{p_X}{p_Y}=\dfrac{\mathrm{CMg}_X}{\mathrm{CMg}_Y}=\mathrm{RMT}_{XY}$
\end{formula}
\item \concepto{1TFEB} (\autor{Arrow}, \autor{Debreu}, 1951): preferencias localmente no saciables $\Rightarrow$ todo EC es óptimo.
  \begin{itemize}
  \item Supuestos: mercados completos, precio-aceptantes, información perfecta, precios comunes.
  \item \emph{No} necesita convexidad (esa es para la existencia).
  \end{itemize}
\item \concepto{2TFEB}: con convexidad, todo óptimo es un EC tras transferencias de suma fija: separa eficiencia y equidad.
\end{itemize}

\sub{I.3. Fallos de mercado\label{rep:I.3}}
\begin{itemize}
\item \concepto{Fallo de mercado}: el EC no es óptimo; otra asignación factible mejora a alguien sin empeorar a nadie: los precios dejan de ser señales válidas.
\end{itemize}

\item Fallos = condición \emph{necesaria}, no \emph{suficiente} (\hyperref[rep:III]{III}); por \emph{equidad} se interviene aun sin fallos.
\end{itemize}

\bloque[17]{II. Externalidades y bienes públicos\label{rep:II}}
\begin{ideaclave}
En los 2~casos el mercado falla porque falta un precio; la intervención lo crea (impuesto, derechos, \autor{Lindahl}) o lo sustituye (estándar, provisión pública), y su gran obstáculo es la información.
\end{ideaclave}

\sub{II.1. Externalidades\label{rep:II.1}}
\subsub{a) Definición y clasificación}
\begin{itemize}
\item \concepto{Externalidad}: el bienestar o la producción de uno dependen \emph{directamente} (sin pasar por los precios) de las acciones de otro (\autor{Mas-Colell} et al.).
  \begin{itemize}
  \item «\emph{El que venga detrás, que arree}»: quien la genera no la paga.
  \end{itemize}
\item \autor{Viner} (1931): \concepto{tecnológicas} (fallo) frente a \concepto{pecuniarias} (vía precios: eficiente).
\item Positivas/negativas; en la producción/el consumo; agotables/no (= «mal público»).
\end{itemize}
\subsub{b) Asignación competitiva frente a óptima}
\subsubsub{i) Equilibrio general}
\begin{itemize}
\item Caja de \autor{Edgeworth}, $U_B(X_B,Y_B,X_A)$. Óptimo:
\begin{formula}
\centering $\mathrm{RMS}^A_{XY}+\dfrac{\partial U_B/\partial X_A}{\partial U_B/\partial Y_B}=\mathrm{RMS}^B_{XY}$
\end{formula}
  \begin{itemize}
  \item El mercado iguala $\mathrm{RMS}^A=\mathrm{RMS}^B=p_X/p_Y$: falta el 2.º término.
  \item Externalidad negativa $\Rightarrow$ $A$ consume demasiado $X$.
  \item Corrige $t=-p_Y\frac{\partial U_B/\partial X_A}{\partial U_B/\partial Y_B}$ en el óptimo: pigouviano en EG.
  \end{itemize}
\end{itemize}
\subsubsub{ii) Equilibrio parcial: la acería}
\begin{itemize}
\item Precio-aceptante a $\overline{p}$; $\mathrm{CMgE}$ = daño a la piscifactoría:
\funcionclave{\mathrm{CMgS}(q)=\mathrm{CMgP}(q)+\mathrm{CMgE}(q)}
  \begin{itemize}
  \item mercado: $\overline{p}=\mathrm{CMgP}(q^c)$; óptimo: $\overline{p}=\mathrm{CMgS}(q^*)$;
  \item $q^c>q^*$: \emph{sobreproducción}, pérdida $E$; positiva: infraproducción.
  \end{itemize}
\end{itemize}
\subsub{c) Intervención}
\begin{itemize}
\item \emph{Justificación}: falta un mercado (\autor{Arrow}); costes de transacción (polizón); derechos sin definir (\autor{Coase}).
\item Objetivo: nivel \emph{óptimo}, no cero. 5~mecanismos, de más a menos intervención.
\end{itemize}
\graficorepaso[0.92\linewidth]{externalidad-intervencion}{Impuesto pigouviano ($\mathrm{CMgP}+t^*$) y estándar ($\overline{q}=q^*$, oferta quebrada)}
\subsubsub{i) \autor{Pigou} (1920): impuestos y subvenciones}
\begin{itemize}
\item Impuesto por unidad = daño marginal \emph{en el óptimo}:
\begin{formula}
\centering $t^*=\mathrm{CMgE}(q^*)$;\quad $\max_q\ \overline{p}q-C(q)-t^*q\ \Rightarrow\ \overline{p}=\mathrm{CMgP}(q)+t^*$
\end{formula}
\item \emph{Áreas}: acería $A$; Estado $B+C$ ($=t^*q^*$); piscifactoría sufre $C$; se recupera $E$.
  \begin{itemize}
  \item Bienestar: $A+B-E\to A+B$.
  \item Recaudación sustituye impuestos distorsionantes: \concepto{doble dividendo} \vertema{4.B.13}.
  \item Externalidad positiva: subvención.
  \end{itemize}
\item \emph{Ventaja}: la reducción va donde es más barata.
\item \emph{Límites}:
  \begin{itemize}
  \item hay que conocer $\mathrm{CMgE}(q^*)$; el daño varía según lugar y momento;
  \item con poder de mercado, $t<$ daño marginal (\autor{Buchanan}, 1969: \emph{second best}).
  \end{itemize}
\end{itemize}
\subsubsub{ii) Estándares}
\begin{itemize}
\item $\overline{q}=q^*$ (cuotas, límites de vertido, matrículas): oferta quebrada en $\overline{q}$.
\item 1~empresa e información completa: mismo $q^*$ y bienestar $A+B$.
  \begin{itemize}
  \item Acería $A+B+C$: $B+C$ es \concepto{renta de escasez} (sin recaudación).
  \end{itemize}
\item \emph{Ventaja}: certeza (umbrales). \emph{Límites}: misma información; con varias empresas, reducción uniforme $\Rightarrow$ cara.
\end{itemize}
\subsubsub{iii) Negociación: \autor{Coase} (1960)}
\begin{itemize}
\item Daño \emph{recíproco}: si se negocia, basta definir quién tiene derecho al río. «\emph{Más vale un mal acuerdo que un buen pleito}».
\item \emph{Supuestos}: derechos definidos, costes de transacción nulos, información completa, cuasilineal.
\item \emph{Derechos de la acería} (parte de $q^c$): la piscifactoría paga entre $D$ y $D+E$ por bajar a $q^*$.
\item \emph{Derechos de la piscifactoría} (parte de 0): la acería paga entre $C$ y $A+B+C$ por llegar a $q^*$.
  \begin{itemize}
  \item Siempre $q^*$ y $A+B$; derechos y poder de negociación solo reparten \vertema{3.A.14}.
  \end{itemize}
\end{itemize}
\graficorepaso[0.7\linewidth]{coase-externalidad}{\autor{Coase}: quién paga a quién según los derechos (mismas áreas)}
\begin{itemize}
\item \concepto{Teorema de \autor{Coase}} (nombre de \autor{Stigler}, 1966): derechos bien definidos + costes de transacción nulos $\Rightarrow$ negociación eficiente \emph{sea quien sea el titular}.
  \begin{itemize}
  \item \autor{Pigou} y \autor{Coase}, mismo $q^*$: cambia quién necesita la información y quién se queda las rentas.
  \end{itemize}
\item \emph{Límites}:
  \begin{itemize}
  \item costes de transacción (muchos afectados; para \autor{Coase}, el mundo real) \vertema{3.A.15};
  \item información asimétrica (\autor{Myerson}-\autor{Satterthwaite}, 1983);
  \item efectos renta: sin cuasilineal, la eficiencia depende del reparto.
  \end{itemize}
\end{itemize}
\subsubsub{iv) Derechos de emisión (\emph{cap and trade})}
\begin{itemize}
\item \autor{Dales} (1968): tope $\overline{E}$, reparto (gratis o subasta) y comercio a $p_E$.
  \begin{enumerate}
  \item el tope fija la reducción $R_0$;
  \item $p_E$ = CMg de la instalación que hace la última reducción necesaria;
  \item las de coste $<p_E$ reducen y venden; las de coste $>p_E$ compran.
  \end{enumerate}
\end{itemize}
\graficorepaso[0.9\linewidth]{cap-and-trade-instalaciones}{5 instalaciones: el tope fija $R_0$ y la instalación marginal, el precio}
\begin{itemize}
\item Costes continuos: cada empresa reduce hasta
\begin{formula}
\centering $\mathrm{CMgR}_1(r_1)=\mathrm{CMgR}_2(r_2)=p_E$
\end{formula}
  \begin{itemize}
  \item = reducir el total al mínimo coste.
  \item \autor{Montgomery} (1972): \emph{cualquiera que sea el reparto inicial} (\autor{Coase} en un mercado creado por el Estado).
  \end{itemize}
\item \concepto{Precios frente a cantidades} (\autor{Weitzman}, 1974):
  \begin{itemize}
  \item con incertidumbre sobre costes de reducir: impuesto $\Rightarrow$ el error va a la cantidad; cuota $\Rightarrow$ al coste;
  \item \emph{cantidad} si el daño marginal crece más deprisa que el CMg de reducir (umbrales); \emph{precio} si es casi constante;
  \item híbridos: precio mínimo y máximo (\autor{Roberts}-\autor{Spence}, 1976).
  \end{itemize}
\item \concepto{RCDE UE} (Directiva 2003/87/CE): desde 2005, 4.ª fase 2021-2030; \autor{Dales}-\autor{Montgomery} más que \autor{Coase}.
  \begin{itemize}
  \item Tope $-4{,}3$\,\%/año (2024-2027); \autor{Weitzman} en el diseño: \emph{reserva de estabilidad} (2019).
  \item RCDE~2 (2028); MAFC desde 1/1/2026 contra la \emph{fuga de carbono} \vertema{3.B.31}.
  \item \emph{Límites}: precio volátil, rentas del reparto gratuito, fuga.
  \end{itemize}
\end{itemize}
\subsubsub{v) Integración}
\begin{itemize}
\item \autor{Meade} (1952): un dueño para acería y piscifactoría maximiza el beneficio conjunto e internaliza.
  \begin{itemize}
  \item Solución de mercado si negociar es caro y los afectados son pocos \vertema{3.A.15}.
  \end{itemize}
\end{itemize}

\sub{II.2. Bienes públicos\label{rep:II.2}}
\subsub{a) Definición y clasificación}
\begin{itemize}
\item Bien privado: \concepto{rival} y \concepto{excluible}.
\item \concepto{Bien público puro} (\autor{Samuelson}, 1954): \emph{no rival} (CMg de un usuario más nulo) y \emph{no excluible}: defensa, alumbrado, conocimiento básico, la mota.
\end{itemize}

\begin{tabular}{@{}>{\raggedright\arraybackslash}m{0.2\linewidth}>{\raggedright\arraybackslash}m{0.37\linewidth}>{\raggedright\arraybackslash}m{0.37\linewidth}@{}}
\toprule
\rowcolor{tceeazulpalido} & \textbf{Rival} & \textbf{No rival}\\
\midrule
\textbf{Excluible} & \textbf{privados} (alimentos, ropa) & \textbf{de club} (TV de pago, peaje)\\
\textbf{No excluible} & \textbf{recursos comunes} (caladeros, pastos, la pesca del río) & \textbf{públicos puros} (defensa, alumbrado, la mota)\\
\bottomrule
\end{tabular}\par}
\begin{itemize}
\item Clasificación de V. y \autor{E. Ostrom} (1977).
\item \concepto{Club} (\autor{Buchanan}, 1965): socios hasta que el ahorro de cuota = coste de congestión.
\item \concepto{Recursos comunes}: \emph{sobreexplotación}, no infraprovisión.
  \begin{itemize}
  \item \concepto{Tragedia de los comunes} (\autor{Hardin}, 1968); «\emph{lo que es del común es de ningún}».
  \item \autor{Gordon} (1954): libre acceso hasta captura \emph{media} $=c$; óptimo: captura \emph{marginal} $=c$ $\Rightarrow$ $N^L>N^*$.
  \item \autor{Ostrom} (1990): reglas comunitarias (Törbel, 1517; Tribunal de las Aguas de Valencia), 8~principios de diseño.
  \end{itemize}
\item \concepto{Tragedia de los anticomunes} (\autor{Heller}, 1998): muchos pueden excluir (patentes) $\Rightarrow$ infrautilización.
\item \concepto{Bienes preferentes} (\autor{Musgrave}, 1959): se infravaloran, no por su tecnología.
\end{itemize}
\subsub{b) Asignación competitiva frente a óptima}
\subsubsub{i) El óptimo: condición de \autor{Samuelson}}
\begin{itemize}
\item Planificador: $\max U_A(X_A,G)$ s.a. $U_B=\overline{U}_B$, $T(X_A+X_B,G)=0$:
\funcionclave{\mathrm{RMS}^A_{GX}+\mathrm{RMS}^B_{GX}=\mathrm{RMT}_{GX}}
  \begin{itemize}
  \item Valor social de una unidad = \emph{suma} de valoraciones (privado: se igualan).
  \item Gráfico de EG (\autor{Samuelson}, 1955): $\mathrm{FPC}_A$ = FPP $-$ $\overline{U}_B$; misma $G^*$, valoraciones distintas.
  \end{itemize}
\end{itemize}
\subsubsub{ii) Polizón e infraprovisión}
\begin{itemize}
\item Aportaciones voluntarias: cada uno para en $\mathrm{RMS}^i_{GX}=\mathrm{RMT}_{GX}$ $\Rightarrow$ $\sum_i\mathrm{RMS}^i=2\,\mathrm{RMT}>\mathrm{RMT}$: \concepto{infraprovisión}; \concepto{polizón}.
\item Dilema del prisionero \vertema{3.A.14}: mota de 120, cada vecino la valora en 100.
  \begin{itemize}
  \item Pagos $(40,40)$, $({-20},100)$, $(100,{-20})$, $(0,0)$: no contribuir es dominante.
  \end{itemize}
\end{itemize}
\subsubsub{iii) Equilibrio parcial: \autor{Bowen} (1943)}
\begin{itemize}
\item Valoración social = \emph{suma vertical} de disposiciones a pagar (privado: horizontal); vértice donde una curva llega a 0.
\item Óptimo: $\sum_i\mathrm{RMS}^i_{GX}(G^*)=\mathrm{CMg}_G$; si declaran menos, $\tilde{G}<G^*$, pérdida = triángulo.
\end{itemize}
\graficorepaso[0.8\linewidth]{bowen}{\autor{Bowen}: suma vertical, precios de \autor{Lindahl} e infraprovisión ($\tilde{G}<G^*$)}
\subsubsub{iv) Provisión privada como EN}
\begin{itemize}
\item $\max_{g_A}U_A(\overline{W}_A-g_A,g_A+g_B)$: reacción decreciente (\emph{sustitutos estratégicos}); $G^{\mathrm N}<G^*$.
  \begin{itemize}
  \item \emph{Concentración}, \emph{neutralidad} (\autor{Warr}, 1983), \emph{expulsión} 1~a 1; \emph{warm glow} (\autor{Andreoni}, 1990).
  \end{itemize}
\item \emph{Evidencia}: aportan 40-60\,\% al principio y cae al repetir (\autor{Ledyard}, 1995).
\end{itemize}
\subsub{c) Intervención: provisión pública}
\begin{itemize}
\item \concepto{Revelación de preferencias}: las $\mathrm{RMS}^i$ son privadas y a todos conviene infravalorar.
\item \concepto{Provisión pública} $\neq$ producción pública; ventajas: valoración social y coacción (impuestos).
\item 2~preguntas: \emph{cuánto} y \emph{cómo repartir el coste}.
\end{itemize}
\subsubsub{i) Según la capacidad de pago}
\begin{itemize}
\item Lo declarado no cambia lo pagado $\Rightarrow$ exagerar (muchísimo o nada): no informa.
\end{itemize}
\subsubsub{ii) Según el beneficio: \autor{Lindahl} (1919)}
\begin{itemize}
\item \autor{Wicksell} (1896); \concepto{precio personalizado} $p^i_G=\mathrm{RMS}^i_{GX}$ en $G^*$, con $p^A_G+p^B_G=\mathrm{RMT}_{GX}$.
  \begin{itemize}
  \item Cumple \autor{Samuelson}: óptimo y justo (beneficio).
  \item \emph{Problema}: vuelve el incentivo a infravalorar; único demandante de «su» mercado: no es precio-aceptante.
  \end{itemize}
\end{itemize}
\subsubsub{iii) \autor{Clarke}-\autor{Groves} (1971, 1973)}
\begin{itemize}
\item Mota indivisible de coste $C$, cuotas $c_i$ con $\sum c_i=C$; se construye si $\sum_i(\tilde v_i-c_i)\geq0$.
\item Solo paga el \concepto{pivotal} (cambia la decisión) lo que pierden los demás:
\begin{formula}
\centering $H_i=\Big|\sum_{j\neq i}(\tilde v_j-c_j)\Big|$ si $i$ es pivotal; $H_i=0$ si no
\end{formula}
\item \emph{Ejemplo}: coste 300, cuotas 100, valoraciones 40, 150, 170 $\Rightarrow$ netos $-60$, 50, 70: se construye.
  \begin{itemize}
  \item Sin el 3~suman $-10$: pivotal, paga 10~y gana 60.
  \end{itemize}
\item Verdad \emph{débilmente dominante} y decisión eficiente: idea \concepto{VCG} (\autor{Vickrey}, 1961).
\item \emph{Límites}:
  \begin{itemize}
  \item presupuesto no equilibrado (\autor{Green}-\autor{Laffont}, 1977: inevitable);
  \item exige cuasilineal (si no, \autor{Gibbard}-\autor{Satterthwaite} \vertema{3.A.24}); coaliciones; parece injusto.
  \end{itemize}
\end{itemize}
\subsubsub{iv) Otras formas de revelar}
\begin{itemize}
\item Votación (votante mediano $\neq$ \autor{Samuelson}) \vertema{3.A.24}; diseño de mecanismos (Nobel 2007); \autor{Tiebout} (1956) \vertema{4.B.14}.
\end{itemize}

\bloque[6,5]{III. Los fallos del sector público\label{rep:III}}
\begin{ideaclave}
El planificador ideal es informado, benevolente y capaz de comprometerse; ninguno de los 3~supuestos se cumple siempre: el fallo de mercado es condición necesaria, pero no suficiente, para intervenir.
\end{ideaclave}

\sub{III.1. Fallos del sector público\label{rep:III.1}}
\begin{itemize}
\item 3~tipos: \emph{información}, \emph{incentivos y proceso político}, \emph{compromiso}.
  \begin{itemize}
  \item En el río: el organismo de cuenca no conoce el daño, puede servir a la acería y no puede prometer que no hará la mota.
  \end{itemize}
\item \autor{Wolf} (1979), \concepto{fallos del no mercado}:
  \begin{itemize}
  \item \emph{internalidades} y objetivos privados;
  \item \emph{costes redundantes y crecientes};
  \item \emph{externalidades derivadas};
  \item \emph{desigualdad de poder} y privilegios.
  \end{itemize}
\end{itemize}
\subsub{a) Problemas de información}
\begin{itemize}
\item Cálculo económico socialista \vertema{3.A.21}:
  \begin{itemize}
  \item \autor{Mises} (1920): sin precios no hay cálculo;
  \item \autor{Lange} (1936): socialismo de mercado por prueba y error;
  \item \autor{Hayek} (1945): problema de \emph{conocimiento} disperso; \autor{Hurwicz} (1972): diseño de mecanismos.
  \end{itemize}
\item No conoce daños, valoraciones ni costes regulados (menú de contratos \vertema{3.A.20}: renta informativa).
\item Externalidades derivadas: la mota atrae casas a zona inundable; indemnizar cada pez muerto $\Rightarrow$ descuido.
\end{itemize}
\subsub{b) Incentivos y proceso político: la elección pública}
\subsubsub{i) Reglas de votación \vertema{3.A.24}}
\begin{itemize}
\item \concepto{Paradoja de \autor{Condorcet}} (1785): mayorías cíclicas $x\succ y\succ z\succ x$: manda la agenda.
\item \concepto{Teorema de imposibilidad} (\autor{Arrow}, 1951): con $\geq3$ opciones, solo la dictadura cumple las condiciones.
\item \concepto{Votante mediano} (\autor{Black}, 1948): preferencias unimodales, una dimensión.
  \begin{itemize}
  \item \autor{Downs} (1957): 2~partidos convergen (\autor{Hotelling} \vertema{3.A.18}).
  \item Eficiente solo si mediana = media de valoraciones.
  \end{itemize}
\item \concepto{Intercambio de votos} (\emph{logrolling}, \autor{Tullock}, 1959): se aprueban proyectos con beneficio $<$ coste.
\end{itemize}
\subsubsub{ii) Los agentes persiguen su interés}
\begin{itemize}
\item \autor{Buchanan}-\autor{Tullock} (1962; Nobel 1986).
\item \emph{Votantes}: \concepto{ignorancia racional} (\autor{Downs}).
\item \emph{Políticos}: \concepto{ciclo político} (\autor{Kalecki}, 1943; \autor{Nordhaus}, 1975; \autor{Rogoff}, 1990).
\item \emph{Burócratas}:
  \begin{itemize}
  \item \autor{Weber} (1922), optimista; ineficiencia X (\autor{Leibenstein}, 1966) \vertema{3.A.12};
  \item \autor{Niskanen} (1971): maximizan el presupuesto, $B(G)=C(G)$ $\Rightarrow$ $G'>G^*$.
  \end{itemize}
\item \concepto{Leviatán} (\autor{Brennan}-\autor{Buchanan}, 1980); \concepto{ilusión fiscal} (\autor{Puviani}, 1903).
\item \emph{Grupos de presión}:
  \begin{itemize}
  \item \concepto{búsqueda de rentas} (\autor{Tullock}, 1967; \autor{Krueger}, 1974) \vertema{3.A.17};
  \item \autor{Olson} (1965): ganan los grupos pequeños y concentrados;
  \item \concepto{captura del regulador} (\autor{Stigler}, 1971); \autor{Peltzman} (1976) \vertema{3.A.20}; \concepto{triángulo de hierro} \vertema{4.B.2}.
  \end{itemize}
\end{itemize}
\subsub{c) Compromiso: la inconsistencia temporal}
\begin{itemize}
\item \concepto{Inconsistencia temporal} (\autor{Kydland}-\autor{Prescott}, 1977): la política óptima anunciada deja de serlo cuando los agentes ya han decidido.
  \begin{itemize}
  \item «\emph{Antes de meter, prometo; y una vez metido, olvido lo prometido}».
  \item Zonas inundables: prometer no hacer la mota; con las casas hechas, se hace.
  \end{itemize}
\item \emph{Modelo}: 2~periodos, $x_1=X_1(a_1,a_2)$.
  \begin{itemize}
  \item Con compromiso, la CPO de $a_2$ incluye $\frac{\partial S}{\partial x_1}\frac{\partial X_1}{\partial a_2}$ (vía expectativas).
  \item Con discreción, $x_1$ ya dado: se olvida $\Rightarrow$ coinciden solo si $\partial X_1/\partial a_2=0$.
  \item Resultado consistente pero peor (EN sin promesa creíble \vertema{3.A.14}).
  \end{itemize}
\item \emph{Ejemplos}:
  \begin{itemize}
  \item impuesto sobre el capital (\autor{Fischer}, 1980); patentes; precio del carbono;
  \item rescates y amnistías: \emph{restricción presupuestaria blanda} (\autor{Kornai}, 1986) \vertema{3.A.13};
  \item inflación (\autor{Barro}-\autor{Gordon}, 1983) \vertema{3.A.36};
  \item \concepto{sesgo deficitario} (\autor{Alesina}-\autor{Tabellini}, 1990; fondo común) \vertema{3.A.39}.
  \end{itemize}
\item \emph{Salidas}: reglas, delegación (AIReF \vertema{4.B.21}), reputación; coste: menos flexibilidad (cláusulas de escape).
\end{itemize}

\sub{III.2. Soluciones\label{rep:III.2}}
\begin{itemize}
\item \emph{Incentivos y control}:
  \begin{itemize}
  \item retribución por productividad (¿cómo medir a un juez?);
  \item supervisión (¿quién controla al controlador?); presupuesto base cero.
  \end{itemize}
\item \emph{Reglas e instituciones independientes}:
  \begin{itemize}
  \item separar diseño y ejecución (análisis coste-beneficio independiente \vertema{4.B.6});
  \item reglas fiscales, constitución fiscal \vertema{4.B.2}; independencia.
  \end{itemize}
\item \emph{Competencia y mercado}: provisión pública con producción privada \vertema{4.B.15}; descentralización (\autor{Tiebout}) \vertema{4.B.14}.
\item Ninguna elimina el fallo: ¿mejora la intervención el resultado? \concepto{Segundo óptimo} (\autor{Lipsey}-\autor{Lancaster}, 1956) \vertema{3.A.22}.
\end{itemize}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}:
  \begin{itemize}
  \item EC eficiente si los precios lo recogen todo (1TFEB);
  \item externalidades: valoraciones privadas $\neq$ sociales; se internalizan con impuesto, estándar, derechos o mercados de derechos;
  \item bienes públicos: sumar valoraciones, pero polizón; financiación pública, \autor{Ostrom}, \autor{Clarke}-\autor{Groves};
  \item el Estado también falla: información, incentivos, compromiso.
  \end{itemize}
\item \textbf{Relevancia}: precio del carbono, gasto en I+D o defensa, regular sin captura, reglas fiscales y monetarias creíbles.
\item \textbf{Extensiones}: \emph{bienes públicos globales} \vertema{3.B.31}: polizón entre Estados; suma, eslabón más débil, mejor intento (\autor{Hirshleifer}, 1983); clubes climáticos (\autor{Nordhaus}, 2015).
\item \textbf{Opinión}: comparar un fallo con otro, no con un ideal.
  \begin{itemize}
  \item «Falacia del nirvana» (\autor{Demsetz}, 1969);
  \item ni ante «el mayor fallo de mercado que el mundo ha visto» (\autor{Stern}, 2006) basta señalarlo: robustos los que menos información piden; precios y cantidades conviven (RCDE con reserva).
  \end{itemize}
\item \textbf{Idea final}: «\emph{unos por otros, la casa sin barrer}»: poner precio a lo que ensucia, repartir la escoba con reglas y recordar que quien reparte las escobas también necesita reglas.
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Impuesto o estándar?} 1~empresa: misma eficiencia; el impuesto recauda $B+C$, el estándar deja renta de escasez.
\item \textbf{¿Por qué en \autor{Coase} los derechos no afectan a la eficiencia?} Sin costes de transacción y con cuasilineal se agotan las ganancias: solo cambia quién paga.
\item \textbf{¿Derechos o impuesto?} Cantidades si el daño marginal es más inclinado que el CMg de reducir (\autor{Weitzman}).
\item \textbf{¿Es \autor{Lindahl} óptimo?} Sí, pero no resuelve la revelación.
\item \textbf{¿Elige el votante mediano la $G$ eficiente?} Solo si su valoración coincide con la media.
\item \textbf{Inconsistencia temporal fuera de la política monetaria}: zonas inundables, impuesto sobre el capital, patentes, rescates.
\end{enumerate}
\begin{trampa}
\begin{itemize}
\item $t^*=\mathrm{CMgE}(q^*)$, no en $q^c$.
\item Bien público: suma \emph{vertical}.
\item 1TFEB no necesita convexidad; el 2TFEB sí.
\item \autor{Clarke}-\autor{Groves}: verdad \emph{débilmente} dominante; solo paga el pivotal; presupuesto no equilibrado.
\item Recursos comunes: sobreexplotación, no infraprovisión. Club: excluible, no rival.
\item Teorema de \autor{Coase}: el nombre es de \autor{Stigler} (1966).
\end{itemize}
\end{trampa}
