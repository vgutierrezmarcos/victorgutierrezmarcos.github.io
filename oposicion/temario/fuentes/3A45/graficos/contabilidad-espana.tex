% Contabilidad del crecimiento por periodos (medias anuales, puntos
% porcentuales) en 3 paneles con el mismo eje vertical: España (filas 0-5 del
% CSV), zona del euro de 20 (filas 6-9, desde 1996) y EE. UU. (filas 10-15).
% ../datos/contabilidad-crecimiento-esp-ea-eeuu.csv: AMECO (capital neto,
% horas, participación salarial ajustada; pesos de Törnqvist) y PWT 11.0
% (capital humano); la PTF es el residuo (consulta 1/11/2026).
% crec_pib = capital + horas + capital humano + PTF.
% Paleta: los factores en 2 familias (capital, dorado; capital humano, dorado
% claro, emparentado; horas, verde azulado) y la PTF en morado, para que se vea
% de un vistazo. El crecimiento del PIB, en negro (línea fina y puntos
% blancos), para no confundirlo con la PTF.
% Eje vertical de -3 a 4,5 (no de -2 a 4): en España, 2008-2013, las horas y la
% PTF restan 2,5 puntos, y en 1996-2007 los factores suman 4,1.
% Los valores del crecimiento del PIB van en una fila en lo alto de cada panel,
% rotulada «PIB:» a la izquierda del panel (a)
% (junto a los puntos chocarían con la línea o caerían sobre las barras).
% Capas (CRITERIOS §7): barras de los factores → barras de la PTF → ejes, cero
% y títulos → línea del PIB → puntos → rótulos y leyenda.
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/contabilidad-crecimiento-esp-ea-eeuu.csv}\conta
\colorlet{ccapital}{tceedorado}
\colorlet{ccaphum}{tceedorado!40}
\colorlet{choras}{dicempleo}
\colorlet{cptf}{tceemorado}
\pgfplotsset{
  % panel: #3 = número de periodos; las filas de cada panel, en \filas, y
  % el desplazamiento de x (primera fila), en \desp
  panelc/.style n args={3}{
    x=0.75cm, height=7cm, scale only axis=false,
    ybar stacked, stack negative=separate, bar width=0.42cm,
    xmin=-0.6, xmax=#3-0.4, ymin=-3, ymax=4.6, ytick={-3,-2,...,4},
    xtick={0,...,5}, xtick style={draw=none},
    xticklabel style={font=\scriptsize, align=center},
    yticklabel style={font=\footnotesize},
    axis x line*=bottom, axis y line*=left, tick align=outside,
    ymajorgrids, grid style={tceegris!25}, clip=false,
    dicapas},
  barra/.style={on layer=dicbarras, dicbarra=#1, forget plot, filas},
  pib/.style={stack plots=false, sharp plot, thin, black, forget plot, filas,
    preaction={draw=white, line width=1.6pt}},
  pibpuntos/.style={on layer=dicpuntos, stack plots=false, forget plot, filas,
    only marks, mark=*, mark size=1.8pt,
    mark options={fill=white, draw=black, line width=0.8pt, solid}},
  filas/.style={restrict expr to domain={\coordindex}{\filas}},
}
\tikzset{rotc/.style={font=\scriptsize, inner sep=1.5pt}}
\newcommand{\barras}{%
  \capa{1}{
    \addplot[barra=ccapital] table[x expr=\coordindex-\desp, y=contrib_capital_pp] {\conta};
    \addplot[barra=choras] table[x expr=\coordindex-\desp, y=contrib_horas_pp] {\conta};
    \addplot[barra=ccaphum] table[x expr=\coordindex-\desp, y=contrib_capital_humano_pp] {\conta};
  }
  \capa{2}{
    \addplot[barra=cptf] table[x expr=\coordindex-\desp, y=contrib_ptf_pp] {\conta};
  }}
\newcommand{\pibpanel}{%
  \capa{4}{\addplot[pib] table[x expr=\coordindex-\desp, y=crec_pib_pct] {\conta};}
  \capa{5}{\addplot[pibpuntos] table[x expr=\coordindex-\desp, y=crec_pib_pct] {\conta};}}
