% Asignaciones de la política de cohesión 2021-2027 por Estado (importe UE,
% FEDER + Fondo de Cohesión + FSE+ + FTJ, sin Interreg; última versión
% adoptada de los programas, 13/11/2026), en miles de millones de euros, con
% el importe por habitante en el rótulo. España en rojo.
% Datos: ../datos/cohesion-estados.csv (Comisión, cohesiondata, pbbz-hmfu).
% Capas: barras -> barra de España -> rótulos por habitante.

\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/cohesion-estados.csv}\coh
\colorlet{cohbarra}{tceeazul!60}
\begin{tikzpicture}[tcee]
  \begin{axis}[width=11cm, height=12cm, xbar, bar width=6.5pt, bar shift=0pt,
      xmin=0, xmax=92, ymin=-0.7, ymax=26.7,
      ytick={0,...,26}, yticklabels from table={\coh}{pais},
      yticklabel style={font=\scriptsize},
      xticklabel style={font=\footnotesize},
      xlabel={miles de millones de euros (en el rótulo, euros por habitante)}, xlabel style={font=\footnotesize},
      axis y line*=left, axis x line*=bottom, xmajorgrids, grid style={tceegris!20},
      dicapas]
    % 1: Estados
    \capa{1}{\addplot[on layer=dicbarras, dicbarra=cohbarra, point meta=explicit, nodes near coords={\pgfmathprintnumber[fixed,precision=0,1000 sep={.}]{\pgfplotspointmeta}}, nodes near coords style={font=\tiny, anchor=west}, x filter/.expression={\thisrow{es}==1 ? nan : x}]
      table[x=mm, y=n, meta=pc] {\coh};}
    % 2: España
    \capa{2}{\addplot[on layer=dicbarras, dicbarra=dicespana, x filter/.expression={\thisrow{es}==1 ? x : nan}]
      table[x=mm, y=n, meta=pc] {\coh};
      \dicrotulo[font=\scriptsize, text=dicespana, anchor=west]{(36.5,24)}{35,6 (724 €/hab.)}}
  \end{axis}
\end{tikzpicture}
\end{document}
