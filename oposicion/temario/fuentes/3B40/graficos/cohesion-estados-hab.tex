% FEDER + Fondo de Cohesión + FSE+ + FTJ, sin Interreg; última versión

% 1-1-2025), con la media de la UE-27 (815 euros). España en rojo.
% Datos: ../datos/cohesion-estados-hab.csv (Comisión, cohesiondata, pbbz-hmfu).

\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/cohesion-estados-hab.csv}\coh
\colorlet{cohbarra}{tceeazul!60}
\begin{tikzpicture}[tcee]
  \begin{axis}[width=10cm, height=10.5cm, xbar, bar width=5.5pt, bar shift=0pt,
      xmin=0, xmax=2800, ymin=-0.7, ymax=26.7,
      ytick={0,...,26}, yticklabels from table={\coh}{pais},
      yticklabel style={font=\scriptsize},
      xticklabel style={font=\footnotesize},
      xlabel={euros por habitante}, scaled x ticks=false, xticklabel={\pgfmathprintnumber[fixed,precision=0,1000 sep={.}]{\tick}}, xlabel style={font=\footnotesize},
      axis y line*=left, axis x line*=bottom, xmajorgrids, grid style={tceegris!20},
      dicapas]
    % 1: Estados
    \capa{1}{\addplot[on layer=dicbarras, dicbarra=cohbarra, point meta=x, nodes near coords={\pgfmathprintnumber[fixed,precision=0,1000 sep={.}]{\pgfplotspointmeta}}, nodes near coords style={font=\tiny, anchor=west}, x filter/.expression={\thisrow{es}==1 ? nan : x}]
      table[x=pc, y=n] {\coh};}
    % 2: España
    \capa{2}{\addplot[on layer=dicbarras, dicbarra=dicespana, x filter/.expression={\thisrow{es}==1 ? x : nan}]
      table[x=pc, y=n] {\coh};
      \dicrotulo[font=\scriptsize, text=dicespana, anchor=west]{(880,11)}{724}}
    % 3: media de la UE
    \capa{3}{\draw[dicreferencia, dashed, thick] (815,-0.7) -- (815,26.7);
      \dicrotulo[font=\scriptsize, text=dicreferencia, anchor=south west, rotate=90]{(815,0.5)}{media de la UE: 815}}
  \end{axis}
\end{tikzpicture}
\end{document}
