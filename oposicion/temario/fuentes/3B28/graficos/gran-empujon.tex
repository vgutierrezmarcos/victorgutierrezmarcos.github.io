% El gran empujón de Murphy, Shleifer y Vishny (1989), en el plano
% empleo-producción de un sector, como Krugman (1995, cap. 2).
% Autor: Víctor Gutiérrez Marcos
% Letras locales del anexo: N sectores, L trabajo total, F coste fijo (en
% trabajo), c trabajo por unidad con la técnica moderna, q producción (= ingreso,
% con precio 1), w salario del sector moderno (prima sobre el tradicional, 1).
% Coordenadas calculadas con F = 2, c = 0,4 y L/N = 5:
%   técnica tradicional:  q = L_j                         (pendiente 1)
%   técnica moderna:      q = (L_j - F)/c = 2,5 (L_j - 2)  (corta el eje en F)
%   A (se moderniza una sola empresa): vende q^A = L/N = 5, emplea
%     F + c q^A = 4; ingreso por trabajador q/L_j = 1,25
%   B (se modernizan todas): emplea L/N = 5 y vende q^B = (L/N - F)/c = 7,5;
%     q/L_j = 1,5
%   rectas de coste q = w L_j (beneficio nulo; la empresa gana si su punto
%   queda por encima): w_1 = 1,1 < 1,25 (gana sola), 1,25 < w_2 = 1,38 < 1,5
%   (pierde sola en A y gana si se modernizan todas, en B: gran empujón) y
%   w_3 = 1,7 > 1,5 (no gana nunca). Zona del gran empujón: el cono entre las
%   rectas de pendiente 1,25 y 1,5.

    \node[etiqueta,anchor=west,text=tceeazul,align=left] at (2.75,0.75) {técnica\\moderna};}
  % 4: puntos A (una empresa sola) y B (todas), con q^A y q^B

  % 5: rectas de coste w_1 < w_2 < w_3 y zona del gran empujón
  \capa{5}{\begin{pgfonlayer}{background}\fill[area2] (0,0) -- (7.2,9.0) -- (6.0,9.0) -- cycle;\end{pgfonlayer}

    \draw[thick,tceegris] (0,0) -- (4.95,8.42) node[etiqueta,left,text=tceegris] {$w_3$};
    \draw[thin,tceedorado] (6.5,8.3) -- (7.6,8.9)
      node[etiqueta,right,text=tceedorado,font=\footnotesize,align=left] {zona del\\gran empujón};}
