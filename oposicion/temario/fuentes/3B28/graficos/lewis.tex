% Modelo dual de Lewis (1954): el mercado de trabajo del sector moderno.
% Gráfico de referencia de la ficha lewis del canon (canon/modelos/lewis.md).
% Autor: Víctor Gutiérrez Marcos
% Coordenadas calculadas:
%   salario institucional  \overline w_A = 1,5
%   salario del sector moderno w_M = \overline w_A (1 + prima) = 2,5
%   oferta: horizontal en w_M hasta el punto de Lewis (L* = 6,3) y después
%           w = 2,5 + 0,35 (L - 6,3) + 0,45 (L - 6,3)^2
%   demandas D(K_i): w = 2,5 + 1,1 (L_i - L), paralelas, con cortes en
%           L_1 = 2,3, L_2 = 4,3 y L_3 = 6,3 (D(K_3) corta justo en el codo,
%           al mismo salario w_M)
%   las demandas se dibujan hasta w = 1,8 (por encima de \overline w_A)
%   beneficio con K_1: triángulo (0; 2,5), (0; 5,03), (2,3; 2,5)
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,y=0.95cm,
    declare function={oferta(\l)=2.5+0.35*(\l-6.3)+0.45*(\l-6.3)^2;
      dem(\l,\li)=2.5+1.1*(\li-\l);}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-1.0,-0.8) rectangle (10.6,6.8);
  % 1: ejes
  \capa{1}{\ejes{$L_M$}{$w$}{9.6}{6.6}}
  % 2: salario institucional y oferta horizontal en w_M
  \capa{2}{\draw[guia] (0,1.5) node[left,text=black] {$\overline w_A$} -- (9.3,1.5)
      node[etiqueta,below left,text=tceegris,font=\footnotesize] {salario institucional};
    \draw[curva1] (0,2.5) node[left,text=black] {$w_M$} -- (6.3,2.5);}
  % 3: tramo creciente de la oferta y punto de Lewis
  \capa{3}{\draw[curva1] plot[domain=6.3:8.55,samples=40,smooth] (\x,{oferta(\x)})
      node[etiqueta,above,text=tceemorado] {oferta};
    \draw[thin,tceegris] (6.4,2.5) -- (7.3,2.45);
    \node[etiqueta,anchor=west,font=\footnotesize] at (7.3,2.45) {punto de \textsc{Lewis}};}
  % 4: demanda D(K_1), empleo L_1 y beneficio
  \capa{4}{\fill[area] (0,{dem(0,2.3)}) -- (2.3,2.5) -- (0,2.5) -- cycle;
    \draw[curva2] (0,{dem(0,2.3)}) -- (2.94,{dem(2.94,2.3)})
      node[etiqueta,right,text=tceeazul] {$D(K_1)$};
    \node[punto] at (2.3,2.5) {};
    \draw[guia] (2.3,2.5) -- (2.3,0) node[below,text=black] {$L_1$};
    \node[etiqueta,text=tceemorado,font=\footnotesize] at (0.7,2.95) {beneficio};}
  % 5: D(K_2) y L_2
  \capa{5}{\draw[curvanueva] (1.8,{dem(1.8,4.3)}) -- (4.94,{dem(4.94,4.3)})
      node[etiqueta,right,text=tceerojo] {$D(K_2)$};
    \node[punto] at (4.3,2.5) {};
    \draw[guia] (4.3,2.5) -- (4.3,0) node[below,text=black] {$L_2$};}
  % 6: D(K_3), L_3 en el punto de Lewis y reinversión del beneficio
  \capa{6}{\draw[curvanueva] (3.8,{dem(3.8,6.3)}) -- (6.94,{dem(6.94,6.3)})
      node[etiqueta,right,text=tceerojo] {$D(K_3)$};
    \node[punto] at (6.3,2.5) {};
    \draw[guia] (6.3,2.5) -- (6.3,0) node[below,text=black] {$L_3$};
    \draw[flecha,tceedorado] (0.4,5.75) -- (4.1,5.75)
      node[etiqueta,midway,above,text=tceedorado,font=\footnotesize] {reinversión del beneficio};}
\end{tikzpicture}
\end{document}
