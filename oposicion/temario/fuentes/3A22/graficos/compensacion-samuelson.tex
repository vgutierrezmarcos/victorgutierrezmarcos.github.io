%   (4,371; 8,499), a 45° hasta (6,050; 7,348) y a la derecha hasta (7,586; 5,669); el tramo de la
%   FPU_2 entre esos extremos (ángulo paramétrico de 62,29° a 36,20°) mejora a los 2.

  % 1: ejes U_A, U_B

  % 2: FPU_1
  \capa{2}{\draw[curva2] plot[domain=90:0,samples=120,variable=\t] ({6.8*cos(\t)},{7.4*sin(\t)});}
