% Criterio de compensación de Hicks (3.A.22, anexo de criterios de compensación).
% Serie compensacion-*: misma escala, mismas FPU y mismos rótulos que compensacion-scitovsky.
% FPU_1: U_A²/6,8² + U_B²/9,3² = 1 (azul). FPU_2: U_A²/9,4² + U_B²/6,8² = 1 (verde).
% Paso de 1 = (1,63; 9,029) (en FPU_1) a 4 = (9,2; 1,395) (en FPU_2). Desde 1, el soborno
%   recorre la FPU_1 (ángulo paramétrico de 76,1° a 8,6°) hasta ζ = (6,723; 1,395), con la misma
%   U_B que 4 y menos U_A: no es Pareto superior a 4, se cumple Hicks.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.55cm,y=0.55cm,
    aux/.style={circle,fill=tceegris,inner sep=0.9pt},
    etqaux/.style={etiqueta,text=tceegris}]
  % 1: ejes U_A, U_B
  \capa{1}{\ejes{$U_A$}{$U_B$}{11}{10.3}}
  % 2: FPU_1 y FPU_2
  \capa{2}{\draw[curva2] plot[domain=90:0,samples=120,variable=\t] ({6.8*cos(\t)},{9.3*sin(\t)});}
  \capa{2}{\node[etiqueta,text=tceeazul,anchor=south east] at (6.75,0.1) {$\mathrm{FPU}_1$};}
  \capa{2}{\draw[curva3] plot[domain=90:0,samples=120,variable=\t] ({9.4*cos(\t)},{6.8*sin(\t)});}
  \capa{2}{\node[etiqueta,text=tceeverde,anchor=south west] at (9.45,0.1) {$\mathrm{FPU}_2$};}
  % 3: puntos 1 (inicial) y 4 (final)
  \capa{3}{\node[punto,label={[etiqueta]above right:$1$}] (p1) at (1.63,9.0289) {};}
  \capa{3}{\node[punto,label={[etiqueta]above right:$4$}] (p4) at (9.2,1.3953) {};}
  % 4: soborno por la FPU del estado inicial hasta ζ, que no es Pareto superior a 4
  \capa{4}{\draw[flecha,shorten <=3pt,shorten >=3pt] plot[domain=76.13:8.63,samples=60,variable=\t] ({6.8*cos(\t)+0.25*cos(\t)},{9.3*sin(\t)+0.25*sin(\t)});}
  \capa{4}{\draw[guia] (p4) -- (6.7230,1.3953);}
  \capa{4}{\node[aux,label={[etqaux]left:$\zeta$}] at (6.7230,1.3953) {};}
\end{tikzpicture}
