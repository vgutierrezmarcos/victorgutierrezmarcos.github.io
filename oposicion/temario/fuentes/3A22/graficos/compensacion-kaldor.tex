% Criterio de compensación de Kaldor (3.A.22, anexo de criterios de compensación).
% Serie compensacion-*: misma escala, mismas FPU y mismos rótulos que compensacion-scitovsky.
% FPU_1: U_A²/6,8² + U_B²/9,3² = 1 (azul). FPU_2: U_A²/9,4² + U_B²/6,8² = 1 (verde).
% Paso de 2 = (2,71; 6,511) (en FPU_2) a 3 = (6,38; 3,218) (en FPU_1). Desde 3, redistribuyendo
%   por la FPU_1 (ángulo paramétrico de 20,2° a 44,4°) se llega a γ = (4,855; 6,511), con la
%   misma U_B que 2 y más U_A: Pareto superior al inicial, se cumple Kaldor.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.55cm,y=0.55cm,
    aux/.style={circle,fill=tceegris,inner sep=0.9pt},
    etqaux/.style={etiqueta,text=tceegris}]
  % 1: ejes U_A, U_B
  \capa{1}{\ejes{$U_A$}{$U_B$}{11}{10.3}}
  % 2: FPU_1 y FPU_2
  \capa{2}{\draw[curva2] plot[domain=90:0,samples=120,variable=\t] ({6.8*cos(\t)},{9.3*sin(\t)});}
  \capa{2}{\node[etiqueta,text=tceeazul,anchor=south east] at (6.75,0.1) {$\mathrm{FPU}_1$};}
  \capa{2}{\draw[curva3] plot[domain=90:0,samples=120,variable=\t] ({9.4*cos(\t)},{6.8*sin(\t)});}
  \capa{2}{\node[etiqueta,text=tceeverde,anchor=south west] at (9.45,0.1) {$\mathrm{FPU}_2$};}
  % 3: punto inicial 2
  \capa{3}{\node[punto,label={[etiqueta]above:$2$}] (p2) at (2.71,6.5113) {};}
  % 4: punto final 3
  \capa{4}{\node[punto,label={[etiqueta]right:$3$}] (p3) at (6.38,3.2178) {};}
  % 5: redistribución por la FPU del estado final hasta γ, Pareto superior a 2
  \capa{5}{\draw[flecha,shorten <=3pt,shorten >=3pt] plot[domain=20.2:44.44,samples=40,variable=\t] ({6.8*cos(\t)-0.3*cos(\t)},{9.3*sin(\t)-0.3*sin(\t)});}
  \capa{5}{\draw[guia] (p2) -- (4.8553,6.5113);}
  \capa{5}{\node[aux,label={[etqaux]above right:$\gamma$}] at (4.8553,6.5113) {};}
\end{tikzpicture}
