\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: «no hay mal que por bien no venga»: si una distorsión no se quita, añadir otra puede mejorar.
  \begin{itemize}
  \item CBAM, régimen definitivo desde el 1/1/2026: lo mejor, un precio mundial único del CO\textsubscript{2}; como no existe, la UE grava el carbono en frontera: \emph{segundo óptimo}.
  \end{itemize}
\item \textbf{Relevancia}:
  \begin{itemize}
  \item \emph{teórica}: \concepto{economía del bienestar} = parte normativa de la micro: valora estados (asignación + reparto) y busca normas de política;
  \item \emph{de política}: toda reforma, impuesto o regulación lleva juicios de bienestar.
  \end{itemize}
\item \textbf{Contextualización}:
  \begin{itemize}
  \item \autor{Bentham} (1789) $\to$ marginalistas (UMg cardinal) $\to$ \autor{Pareto} (1906): ordinal;
  \item nueva economía del bienestar (años 30-40): compensación y FBS $\to$ teoremas (\autor{Arrow}, \autor{Debreu}) $\to$ \emph{second best} $\to$ información (\autor{Mirrlees}, \autor{Hurwicz});
  \item completa el EG \vertema{3.A.21}: aquel, si existe, es único y estable; este, si es \emph{bueno}.
  \end{itemize}
\item \textbf{Problemática}: ¿qué condiciones para que nadie mejore sin que otro empeore? ¿Las alcanza el mercado? ¿Lleva a cualquier óptimo? Si una falla, ¿cumplir las demás?
\item \textbf{Estructura}: \hyperref[rep:I]{I} define el óptimo; \hyperref[rep:II]{II}, cuándo lo alcanza el mercado; \hyperref[rep:III]{III}, qué hacer si no está al alcance.
\end{itemize}

\bloque[9,5]{I. Óptimo económico. Criterio de \autor{Pareto}\label{rep:I}}
\begin{ideaclave}
Un óptimo exige igualar 3 tipos de relaciones marginales (producción, consumo y entre ambas), sin precios ni mercados; pero hay infinitos, toda la GFPU, y la eficiencia no basta para elegir.
\end{ideaclave}

\sub{I.1. Definición y juicios de valor\label{rep:I.1}}
\begin{itemize}
\item \emph{Origen}: \autor{Pareto} (Lausana; \emph{Manuale}, 1906): valorar el EG sin medir ni comparar utilidades; versión moderna, \autor{Arrow} (1951), \autor{Debreu} (1959).
\item \concepto{Asignación} factible: $\sum_i\vect{x}_i\leq\sum_i\vect{\omega}_i+\sum_j\vect{y}_j$, $\vect{y}_j\in\mathcal{Y}_j$.
\item \concepto{\autor{Pareto} superior}: nadie peor y al menos uno mejor.
\item \concepto{Óptimo de \autor{Pareto}}: no hay factible \autor{Pareto} superior: no se mejora a nadie sin empeorar a otro.
  \begin{itemize}
  \item Superior $\neq$ óptimo; optimalidad \emph{fuerte} $\Rightarrow$ \emph{débil}.
  \item \emph{Ordinal}: solo si cada uno prefiere \vertema{3.A.8}.
  \end{itemize}
\item \emph{4 juicios de valor}:
  \begin{enumerate}
  \item \concepto{individualismo}: solo cuentan las utilidades;
  \item \concepto{no paternalismo}: cada uno, mejor juez de lo suyo;
  \item \concepto{benevolencia}: uno mejora y nadie empeora $\Rightarrow$ la sociedad mejora;
  \item \concepto{independencia del proceso}.
  \end{enumerate}
\end{itemize}

\sub{I.2. Condiciones de optimalidad paretiana\label{rep:I.2}}
\begin{itemize}
\item 3 condiciones: \emph{producción}, \emph{consumo} y \emph{global}; las 2 primeras, necesarias, no suficientes.
\end{itemize}
\subsub{a) Supuestos del modelo 2×2×2×2 (\autor{Bator}, 1957)}
\begin{itemize}
\item 2 bienes $X,Y$; 2 factores $L,K$ ($\overline{L},\overline{K}$); 2 consumidores (preferencias completas, transitivas, continuas, estr.\ monótonas y convexas); 2 empresas, isocuantas y conjuntos convexos; sin externalidades.
\item \concepto{Planificador benevolente} que lo sabe todo: solución \emph{tecnocrática}.
\item \emph{No se supone} UMg decreciente (ordinal) ni \autor{Inada}: basta la convexidad.
\end{itemize}
\subsub{b) Eficiencia en la producción}
\begin{itemize}
\item $\max F_X$ s.a. $F_Y(\overline{L}-L_X,\overline{K}-K_X)\geq\overline{Y}$:
\begin{formula}
\centering $\mathrm{RMST}^X_{LK}\equiv\dfrac{\mathrm{PMg}^X_L}{\mathrm{PMg}^X_K}=\dfrac{\mathrm{PMg}^Y_L}{\mathrm{PMg}^Y_K}\equiv\mathrm{RMST}^Y_{LK}$
\end{formula}
  \begin{itemize}
  \item tangencia de isocuantas: \concepto{curva de contrato} ($O_X$ a $O_Y$) $\to$ \concepto{FPP}; vértices, también eficientes.
  \end{itemize}
\item \emph{FPP}: pendiente negativa; en valor absoluto, la \concepto{RMT} (coste de oportunidad de $X$):
\begin{formula}
\centering $\mathrm{RMT}_{XY}\equiv-\dfrac{dY}{dX}\Big|_{\text{FPP}}=\dfrac{\mathrm{PMg}^Y_L}{\mathrm{PMg}^X_L}=\dfrac{\mathrm{CMg}_X}{\mathrm{CMg}_Y}$
\end{formula}
  \begin{itemize}
  \item forma: rend.\ decrecientes $\Rightarrow$ estr.\ cóncava; RCE: cóncava si difieren las intensidades, recta si coinciden; crecientes: puede ser convexa.
  \end{itemize}
\end{itemize}
\subsub{c) Eficiencia en el consumo}
\begin{itemize}
\item $\max U_A$ s.a. $U_B(\overline{X}-X_A,\overline{Y}-Y_A)\geq\overline{U}_B$:
\begin{formula}
\centering $\mathrm{RMS}^A_{XY}\equiv\dfrac{\partial U_A/\partial X_A}{\partial U_A/\partial Y_A}=\dfrac{\partial U_B/\partial X_B}{\partial U_B/\partial Y_B}\equiv\mathrm{RMS}^B_{XY}$
\end{formula}
  \begin{itemize}
  \item si no, un intercambio mejora a los 2;
  \item curva de contrato ($O_A$ a $O_B$) $\to$ \concepto{FPU} en $(U_A,U_B)$ (\autor{Allais}, 1943; \autor{Samuelson}, 1950).
  \end{itemize}
\item \emph{FPU}: decreciente; \emph{forma indeterminada} (ordinal: se dibuja cóncava por costumbre); una FPU por cada punto de la FPP.
\end{itemize}
\subsub{d) Eficiencia global}
\begin{itemize}
\item El planificador maximiza $U_A$ con $U_B\geq\overline{U}_B$, tecnologías y dotaciones ($\lambda_U,\lambda_X,\lambda_Y,\lambda_L,\lambda_K$); dividiendo las CPO, las \emph{3 condiciones de \autor{Pareto}}:
\funcionclave{\begin{aligned}&\mathrm{RMST}^X_{LK}=\mathrm{RMST}^Y_{LK}=\tfrac{\lambda_L}{\lambda_K}\\ &\mathrm{RMS}^A_{XY}=\mathrm{RMS}^B_{XY}=\tfrac{\lambda_X}{\lambda_Y}=\mathrm{RMT}_{XY}\end{aligned}}
  \begin{itemize}
  \item si $\mathrm{RMS}>\mathrm{RMT}$, $X$ se valora más de lo que cuesta: producir más $X$;
  \item multiplicadores = \concepto{precios sombra}: con precios proporcionales, cada agente elegiría lo asignado (\autor{Lange}-\autor{Lerner}): puerta de \hyperref[rep:II]{II};
  \item eficiente en producción y consumo $\not\Rightarrow$ global.
  \end{itemize}
\item \emph{Gráfico}: cada punto de la FPP fija una caja del consumo; en ella solo $E^*$ cumple $\mathrm{RMS}=\mathrm{RMT}$; envolvente de las FPU: \concepto{gran frontera de posibilidades de utilidad} (GFPU).
\end{itemize}
\graficorepaso[0.98\linewidth]{gfpu}{Construcción de la GFPU: caja del consumo dentro de la FPP con $\mathrm{RMS}=\mathrm{RMT}$ (izq.) y envolvente de las FPU (dcha.)}
\begin{itemize}
\item \emph{Forma}: ni FPU ni GFPU \emph{tienen por qué ser cóncavas}; puede haber \emph{hondonadas}.
\end{itemize}
\subsub{e) Elegir entre óptimos: la función de bienestar social}
\begin{itemize}
\item Infinitos óptimos (toda la GFPU) y \autor{Pareto} no los ordena: optimalidad \emph{necesaria, no suficiente} para el máximo bienestar.
\item \concepto{FBS} $W(U_A,U_B)$ (\autor{Bergson}, 1938; \autor{Samuelson}, 1947), creciente:
  \begin{itemize}
  \item \concepto{punto de felicidad} $c$: tangencia con la $W$ más alta;
  \item un punto ineficiente ($a$) puede ser mejor que uno eficiente y desigual ($d$); pero siempre hay uno en la GFPU que lo mejora;
  \item formas de $W$ (utilitarista, rawlsiana) \vertema{3.A.24}.
  \end{itemize}
\end{itemize}
\graficorepaso[0.55\linewidth]{fbs-gfpu}{FBS y GFPU: punto de felicidad $c$}

\sub{I.3. Valoración del criterio de \autor{Pareto}\label{rep:I.3}}
\begin{itemize}
\item \emph{Ventajas}: ordinal; pocos juicios de valor, aceptados; permite valorar el mercado.
\item \emph{Críticas}:
  \begin{enumerate}
  \item \emph{estático} (austriacos: eficiencia dinámica);
  \item \emph{favorece el statu quo}: veto de cada uno;
  \item \emph{independiente del proceso};
  \item \emph{ignora la distribución}: que uno lo tenga todo es óptimo (\autor{Sen}, 1970); \emph{justas} = eficientes + libres de envidia (\autor{Foley}, 1967);
  \item \emph{incompleto} (la principal): no ordena los óptimos.
  \end{enumerate}
\item \emph{Respuestas} (años 30-40):
  \begin{itemize}
  \item \emph{neoparetiana}: \concepto{criterios de compensación}: los ganadores \emph{podrían} compensar (\autor{Kaldor}, \autor{Hicks}, 1939); pueden contradecirse (\autor{Scitovsky}, 1941); ACB;
  \item \emph{neoutilitarista}: FBS de \autor{Bergson}-\autor{Samuelson} \vertema{3.A.24}.
  \end{itemize}
\end{itemize}

\bloque[10]{II. Teoremas fundamentales\label{rep:II}}
\begin{ideaclave}
EGC y óptimo se corresponden en los 2 sentidos: todo equilibrio es un óptimo (1TFEB) y todo óptimo puede ser un equilibrio tras redistribuir (2TFEB); cada sentido exige supuestos distintos, y en ellos está la política económica.
\end{ideaclave}
\begin{itemize}
\item \concepto{Equilibrio walrasiano} \vertema{3.A.21}: $\vect{p}^*$ y $(\vect{x}^*_i,\vect{y}^*_j)$; empresas maximizan $\vect{p}^*\cdot\vect{y}_j$, consumidores $U$ con su renta, todos los mercados se vacían.
  \begin{itemize}
  \item Con \concepto{transferencias de suma fija} $T_i$ ($\sum T_i=0$): \emph{equilibrio con transferencias}.
  \end{itemize}
\end{itemize}

\sub{II.1. Primer teorema (1TFEB)\label{rep:II.1}}
\subsub{a) Idea}
\begin{itemize}
\item (\autor{Arrow}, 1951; \autor{Debreu}, 1951): con preferencias \emph{localmente no saciables}, todo equilibrio walrasiano es un óptimo de \autor{Pareto}.
\item Formaliza la \emph{mano invisible} de \autor{Smith} (1776).
\end{itemize}
\subsub{b) Modelo: el EGC 2×2×2×2}
\begin{itemize}
\item \emph{Supuestos}: la economía de \autor{Bator} + mercados con $p_X,p_Y,w_L,w_K$ dados:
\end{itemize}
{\centering\small\renewcommand{\arraystretch}{1.05}\setlength{\tabcolsep}{2pt}%
\begin{tabular}{@{}>{\raggedright\arraybackslash}m{0.42\linewidth}>{\raggedright\arraybackslash}m{0.54\linewidth}@{}}
\toprule
\rowcolor{tceeazulpalido}\textbf{Supuesto} & \textbf{Si se levanta (fallo)}\\
\midrule
1. No saciedad local & renta sin gastar\\
2. Mercados completos & externalidades, bienes públicos \vertema{3.A.23}\\
3. Precio-aceptantes & poder de mercado \vertema{3.A.17}-\vertema{3.A.19}\\
4. Precios comunes & impuestos: exceso de gravamen \vertema{4.B.9}\\
5. Información perfecta & selección adversa, riesgo moral \vertema{3.A.13}\\
\bottomrule
\end{tabular}\par}
\begin{itemize}
\item \emph{No hace falta} convexidad ni rendimientos (aseguran que \emph{exista}): \emph{si existe}, es eficiente.
\item \emph{Desarrollo, en 4 pasos} (cada agente cumple su condición):
  \begin{enumerate}
  \item \emph{consumidores} (consumo): mismos precios $\Rightarrow$ valoran igual sin conocerse:
\begin{formula}
\centering $\mathrm{RMS}^A_{XY}=\dfrac{p_X}{p_Y}=\mathrm{RMS}^B_{XY}$
\end{formula}
  \item \emph{empresas en los factores} (producción): $p_j\mathrm{PMg}^j_L=w_L$, $p_j\mathrm{PMg}^j_K=w_K$:
\begin{formula}
\centering $\mathrm{RMST}^X_{LK}=\dfrac{w_L}{w_K}=\mathrm{RMST}^Y_{LK}$
\end{formula}
  \item \emph{empresas en los bienes} (global): $p_j=\mathrm{CMg}_j$ \vertema{3.A.16}; nadie sabe de tecnología ni preferencias ajenas:
\begin{formula}
\centering $\dfrac{p_X}{p_Y}=\dfrac{\mathrm{CMg}_X}{\mathrm{CMg}_Y}=\mathrm{RMT}_{XY}$
\end{formula}
  \item \emph{vaciado} de los 4 mercados (3 bastan, \autor{Walras}): hace compatibles los planes.
  \end{enumerate}
\item \emph{Resultado}: las condiciones de \hyperref[rep:I.2]{I.2} con $p_X/p_Y=\lambda_X/\lambda_Y$, $w_L/w_K=\lambda_L/\lambda_K$: \emph{los precios hacen de precios sombra}.
\item \emph{Gráfico}: desde $\vect{\omega}$, $E$ en la curva de contrato: la recta de precios, tangente a las 2 curvas.
\end{itemize}
\graficorepaso[0.6\linewidth]{caja-intercambio-teoremas}{1TFEB: $E$ en la curva de contrato; 2TFEB: de $\vect{\omega}$ a $\vect{\omega}'$ para alcanzar $E'$}
\begin{itemize}
\item \emph{Implicaciones}:
  \begin{enumerate}
  \item \emph{economía de información}: a cada agente le bastan los precios y lo suyo (\autor{Hayek}, 1945);
  \item \emph{papel del Estado}: corregir solo los fallos que rompen un supuesto;
  \item llega a \emph{un} óptimo (el de las dotaciones), no \emph{al} punto de felicidad.
  \end{enumerate}
\end{itemize}
\subsub{c) Extensiones: prueba general}
\begin{itemize}
\item \autor{Arrow}-\autor{Debreu} ($I$, $J$, $n$; sin derivadas, vale en esquinas), por reducción al absurdo:
  \begin{itemize}
  \item si una factible es \autor{Pareto} superior, quien mejora no podía pagarla ($>\overline{W}_i$) y quien queda igual tampoco por menos ($\geq\overline{W}_i$, no saciedad local);
  \item sumando, y con empresas que maximizan beneficio, su valor supera el de lo disponible: contradice la factibilidad.
  \end{itemize}
\item Vale con transferencias de suma fija (lo que usa el 2TFEB).
\end{itemize}
\subsub{d) Evidencia empírica}
\begin{itemize}
\item Subastas dobles de \autor{V. Smith} (1962): con pocos y solo su valoración o coste, convergen al equilibrio competitivo.
\item \autor{Gode}-\autor{Sunder} (1993): con robots al azar, eficiencia casi completa: lo hacen las reglas del mercado.
\end{itemize}
\subsub{e) Valoración}
\begin{itemize}
\item Nada de equidad.
\item Supuestos exigentes: cada \concepto{fallo de mercado} rompe uno de la tabla; con información imperfecta, ni óptimo restringido (\autor{Greenwald}-\autor{Stiglitz}, 1986).
\item Presupone la existencia; sin unicidad ni estabilidad; estático.
\end{itemize}

\sub{II.2. Segundo teorema (2TFEB)\label{rep:II.2}}
\subsub{a) Idea}
\begin{itemize}
\item (\autor{Arrow}, 1951; \autor{Debreu}, 1954): con preferencias y tecnologías \emph{convexas}, todo óptimo de \autor{Pareto} puede alcanzarse como EGC tras una \emph{redistribución de suma fija}.
\item Recíproco del 1TFEB: no hace falta sustituir el mercado.
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item \emph{Supuestos}:
  \begin{enumerate}
  \item los del 1TFEB;
  \item \emph{convexidad} de preferencias y producción (sin rend.\ crecientes): el nuevo y decisivo;
  \item transferencias de suma fija $T_i$;
  \item \emph{condición de renta}: cada uno puede comprar algo más barato que su cesta óptima.
  \end{enumerate}
\item \emph{Intuición}: en $E'$ la recta común separa $U_A'$ y $U_B'$: es la recta de precios; el Estado lleva $\vect{\omega}$ a $\vect{\omega}'$ en ella y el mercado hace el resto.
\end{itemize}
\pareja{segundo-teorema-gfpu}{2TFEB en utilidades: de $E$ al punto de felicidad $E'$}{segundo-teorema-no-convexo}{Sin convexidad: la tangente no sostiene $E'$}
\begin{itemize}
\item \emph{Demostración} (MWG §16.D):
  \begin{enumerate}
  \item lo preferido $V=\sum_iV_i$ y lo producible $\mathcal{Y}$, convexos;
  \item disjuntos (si no, no sería óptimo);
  \item \concepto{teorema del hiperplano separador} (\autor{Minkowski}, 1911): $\vect{p}\neq\vect{0}$ que los separa: los precios;
  \item \concepto{cuasiequilibrio} + condición de renta $\Rightarrow$ equilibrio, con $T_i$ que suman 0.
  \end{enumerate}
\item \emph{Sin convexidad}: con rend.\ crecientes, el óptimo no maximiza el beneficio a ningún precio \vertema{3.A.17}.
\item \emph{Implicaciones}:
  \begin{itemize}
  \item \emph{separación eficiencia-equidad}: distribución con suma fija, asignación al mercado;
  \item intervención \emph{ex ante} por equidad aunque no haya fallos \vertema{3.A.24};
  \item \emph{descentralización}: socialismo de mercado (\autor{Lange}, 1936; \autor{Lerner}, 1944).
  \end{itemize}
\end{itemize}
\subsub{c) Valoración: límites y soluciones}
\begin{itemize}
\item \emph{Límites}:
  \begin{enumerate}
  \item \emph{información}: la suma fija que redistribuye depende de lo inobservable (capacidad); si depende de lo observable, distorsiona (\autor{Mirrlees}, 1971);
  \item \emph{convexidad}: rend.\ crecientes $\Rightarrow$ $p=\mathrm{CMg}$ con pérdidas;
  \item \emph{cálculo}: fijar $\vect{\omega}'$ exige la información que el 1TFEB ahorraba;
  \item \emph{multiplicidad}: otros equilibrios desde $\vect{\omega}'$.
  \end{enumerate}
\item \emph{Soluciones}:
  \begin{itemize}
  \item imposición óptima: bienes (\autor{Ramsey}, 1927), renta (\autor{Mirrlees}, 1971) \vertema{4.B.11};
  \item eficiencia productiva (\autor{Diamond}-\autor{Mirrlees}, 1971); renta y no bienes (\autor{Atkinson}-\autor{Stiglitz}, 1976);
  \item \emph{tagging} (\autor{Akerlof}, 1978) \vertema{4.B.3}; sin suma fija: lógica del \emph{second best}.
  \end{itemize}
\end{itemize}

\bloque[7]{III. La teoría del \emph{second best}\label{rep:III}}
\begin{ideaclave}
Si una condición de \autor{Pareto} no puede cumplirse, cumplir las demás no garantiza acercarse al óptimo: el segundo óptimo no es lo que más se parece al óptimo. Con información privada, lo alcanzable es un óptimo restringido por los incentivos.
\end{ideaclave}
\begin{itemize}
\item \concepto{Primer óptimo}: todas las condiciones; \concepto{segundo óptimo}: el mejor con una restricción adicional.
\end{itemize}

\sub{III.1. Idea y resultado general\label{rep:III.1}}
\subsub{a) Idea}
\begin{itemize}
\item \emph{Visión tradicional}: cuantas más condiciones, más cerca; quitar distorsiones una a una (\emph{piecemeal}).
\item \autor{Meade} (1955), uniones aduaneras, lo intuye; \autor{Lipsey}-\autor{Lancaster} (1956): no necesariamente.
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item \emph{Supuestos}: $n$ bienes, consumidor representativo $U(\vect{x})$, tecnología $T(\vect{x})=0$; primer óptimo: $U_k/U_n=T_k/T_n$.
  \begin{itemize}
  \item \concepto{Restricción no trivial} sobre el bien 1 (impuesto: $\psi=1+t$):
\begin{formula}
\centering $\dfrac{U_1}{U_n}=\psi\,\dfrac{T_1}{T_n},\quad\psi\neq1$
\end{formula}
  \item origen: institucional, tecnológico, de mercado, informativo.
  \end{itemize}
\item \emph{Desarrollo}: $\max U$ s.a. $T=0$ ($\lambda_1$) y la restricción ($\lambda_2$):
\begin{formula}
\centering $\dfrac{U_k}{U_n}=\dfrac{T_k+\frac{\lambda_2}{\lambda_1}\Psi_k}{T_n+\frac{\lambda_2}{\lambda_1}\Psi_n}$
\end{formula}
  \begin{itemize}
  \item $\Psi_k$: derivadas cruzadas de 2.º orden, signo desconocido a priori.
  \end{itemize}
\item \concepto{Teorema general del segundo óptimo}: con una condición imposible, las demás dejan en general de ser deseables; hay que apartarse también de ellas, según sustituciones y complementariedades.
\item \emph{Gráfico}: $P$ (en la GFPU, cumple las posibles) en $W_1$; $Q$, \emph{dentro} de la GFPU, en $W_2>W_1$: segundo óptimo.
\end{itemize}
\pareja{second-best-gfpu}{\emph{Second best}: $Q$ (ineficiente) mejor que $P$ (eficiente accesible)}{montana-second-best}{Montaña: con el obstáculo $D$, mejor la cumbre $A$}
\begin{itemize}
\item \emph{Implicaciones}:
  \begin{enumerate}
  \item contar condiciones no informa;
  \item quitar una distorsión puede ↑, ↓ o no cambiar el bienestar;
  \item \emph{añadir} una puede mejorar si compensa otra (cine gravado y plataformas);
  \item la política por partes no tiene garantía general.
  \end{enumerate}
\end{itemize}
\subsub{c) Extensiones}
\begin{itemize}
\item \emph{Separabilidad} (\autor{Davis}-\autor{Whinston}, 1965): sector distorsionado separable $\Rightarrow$ \autor{Pareto} en el resto: rescata el \emph{piecemeal}.
\item \emph{Eficiencia productiva} (\autor{Diamond}-\autor{Mirrlees}, 1971): no gravar intermedios.
\item \emph{Corregir en el origen} (\autor{Bhagwati}-\autor{Ramaswami}, 1963): un arancel ante distorsión interna es solo segundo óptimo.
\end{itemize}

\sub{III.2. Aplicaciones\label{rep:III.2}}
\begin{itemize}
\item \emph{Uniones aduaneras} \vertema{3.B.10}: quitan aranceles entre socios y los mantienen fuera: creación frente a desviación de comercio (\autor{Viner}, 1950; \autor{Lipsey}, 1957).
\item \emph{Monopolio contaminante} \vertema{3.A.23}: produce poco (poder) y mucho (externalidad):
\begin{formula}
\centering $t^*=\mathrm{CMgE}(q^*)-\dfrac{p(q^*)}{|\varepsilon_{q,p}|}$
\end{formula}
  \begin{itemize}
  \item menor que el pigouviano; subvención si pesa más el poder (\autor{Buchanan}, 1969);
  \item romper el monopolio sin gravar la contaminación puede ↓ bienestar.
  \end{itemize}
\item \emph{Industria naciente} \vertema{3.B.8}: crédito o subvención; el arancel, segundo óptimo.
\item \emph{Instituciones} \vertema{3.B.28}: \autor{Rodrik} (2008), de segundo óptimo, no copiar «mejores prácticas»; FMI, Hong Kong, 1997: liberalizar capitales en plena crisis \vertema{3.B.17}.
\item \emph{Imposición y regulación}: complementos del ocio (\autor{Corlett}-\autor{Hague}, 1953) \vertema{4.B.11}; \autor{Ramsey}-\autor{Boiteux} \vertema{3.A.17}.
\end{itemize}

\sub{III.3. Información privada: diseño de mecanismos\label{rep:III.3}}
\begin{itemize}
\item \concepto{Eficiencia restringida}: ningún planificador con la misma información mejora a todos \vertema{3.A.13}.
  \begin{itemize}
  \item Abrir un mercado puede empeorar a todos (\autor{Hart}, 1975); \autor{Greenwald}-\autor{Stiglitz} (1986): la mano invisible es invisible porque no está (\autor{Stiglitz}, 2002).
  \end{itemize}
\item \concepto{Diseño de mecanismos} (\autor{Hurwicz}, 1972): reglas para que, cada uno en su interés, se alcance el resultado; \emph{principio de revelación} \vertema{3.A.14}.
\item \emph{VCG} (\autor{Vickrey}, 1961; \autor{Clarke}, 1971; \autor{Groves}, 1973): verdad dominante y eficiente (cada uno paga el daño que causa); sin presupuesto equilibrado (\autor{Green}-\autor{Laffont}, 1977).
  \begin{itemize}
  \item Sin pagos: \autor{Gibbard}-\autor{Satterthwaite} \vertema{3.A.24}; \emph{bayesianos} (\autor{d'Aspremont}-\autor{Gérard-Varet}, 1979): equilibrados, no siempre voluntarios.
  \end{itemize}
\item \autor{Myerson}-\autor{Satterthwaite} (1983): ninguno es eficiente, veraz, equilibrado y voluntario.
\end{itemize}

\sub{III.4. Valoración\label{rep:III.4}}
\begin{itemize}
\item Sin reglas generales (signo de $\Psi_k$): dice qué no afirmar, no qué hacer.
\item \emph{Respuestas}: análisis parcial, separabilidad, corregir en el origen, conocer el caso (\autor{Lipsey}, 2007).
\item No invalida los teoremas: enseña a usarlos.
\end{itemize}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}:
  \begin{itemize}
  \item óptimo = 3 igualdades marginales, infinitos (GFPU);
  \item 1TFEB: el mercado alcanza uno mirando precios; 2TFEB: con convexidad, cualquiera tras suma fija;
  \item si una condición falla, cumplir las demás no garantiza nada.
  \end{itemize}
\item \textbf{Relevancia}: vara de medir de la micro normativa: intervenir por eficiencia (1.º), por equidad (2.º).
\item \textbf{Extensiones}: \vertema{3.A.23} \vertema{3.A.24} \vertema{4.B.11} \vertema{3.A.14}.
\item \textbf{Opinión}: los teoremas ordenan las preguntas: qué supuesto falla, qué restricción no se quita.
\item \textbf{Idea final}: no hay mal que por bien no venga, pero solo quien conoce el óptimo sabe qué mal trae un bien.
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Qué agente cumple cada condición?} Consumidores, consumo; empresas en factores, producción; $p=\mathrm{CMg}$, global; el vaciado las hace compatibles.
\item \textbf{¿Necesitan convexidad los teoremas?} El 1.º no; el 2.º sí (hiperplano) y suma fija.
\item \textbf{¿Por qué no hay suma fija?} Depende de lo inobservable (\autor{Mirrlees}): imposición óptima, \emph{tagging}.
\item \textbf{¿Puede un impuesto ↑ el bienestar?} Sí: \emph{second best} ($t^*<$ pigouviano al monopolio).
\item \textbf{¿Segundo óptimo u óptimo restringido?} Distorsión dada frente a restricción de información.
\item \textbf{¿Qué criterio usa el ACB?} Compensación potencial (\autor{Kaldor}-\autor{Hicks}): no se paga y puede contradecirse (\autor{Scitovsky}).
\end{enumerate}

\begin{trampa}
\begin{itemize}
\item 1TFEB: \emph{no} necesita convexidad; el 2TFEB, sí.
\item $\mathrm{RMT}_{XY}=\mathrm{CMg}_X/\mathrm{CMg}_Y=\mathrm{PMg}^Y_L/\mathrm{PMg}^X_L$.
\item FPU y GFPU \emph{no tienen por qué ser cóncavas}.
\item El 1TFEB da \emph{un} óptimo, no el punto de felicidad.
\item \emph{Second best} $\neq$ «condiciones de segundo orden»; segundo óptimo $\neq$ óptimo restringido.
\end{itemize}
\end{trampa}
