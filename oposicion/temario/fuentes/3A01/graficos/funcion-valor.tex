% Función de valor de la teoría de las perspectivas (Tversky y Kahneman, 1992).
% z = x - r: v(z) = z^0,88 si z >= 0 y v(z) = -2,25 (-z)^0,88 si z < 0.
% Escala vertical 0,55. Con z = 2: v(2) = 2^0,88 = 1,84 y v(-2) = -2,25 · 1,84 = -4,14
%   (en el dibujo, 1,01 y -2,28): perder 2 duele 2,25 veces lo que alegra ganar 2.
% Pendiente infinita a ambos lados de r (beta < 1) y 2,25 veces mayor en las pérdidas.
% Elaboración propia a partir de Kahneman y Tversky (1979) y Tversky y Kahneman (1992).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,y=0.55cm,
    declare function={vg(\z)=\z^0.88; vp(\z)=-2.25*(-\z)^0.88;}]
  % 1: ejes con el punto de referencia en el cruce
  \capa{1}{\draw[eje] (-3.6,0) -- (4.4,0) node[below left=2pt and 0pt] {$x$};}
  \capa{1}{\draw[eje] (0,-6.4) -- (0,4) node[left] {$v$};}
  \capa{1}{\node[below right] at (0,0) {$r$};}
  \capa{1}{\node[etiqueta,font=\footnotesize,above] at (3.3,0) {ganancias};}
  \capa{1}{\node[etiqueta,font=\footnotesize,above] at (-3.0,0) {pérdidas};}
  % 2: ganancias: cóncava
  \capa{2}{\draw[curva1] plot[domain=0:4,samples=80] (\x,{vg(\x)});}
  % 3: pérdidas: convexa y más inclinada
  \capa{3}{\draw[curva1] plot[domain=-2.75:0,samples=80] (\x,{vp(\x)});}
  \capa{3}{\node[etiqueta,right,font=\footnotesize] at (4,{vg(4)}) {$v(x-r)$};}
  % 4: aversión a las pérdidas: ganar y perder lo mismo
  \capa{4}{\draw[guia] (2,0) node[below,text=black] {$+a$} -- (2,{vg(2)}) -- (0,{vg(2)}) node[right,text=black,font=\footnotesize,yshift=4pt] {$v(a)$};}
  \capa{4}{\draw[guia] (-2,0) node[above,text=black] {$-a$} -- (-2,{vp(-2)}) -- (0,{vp(-2)}) node[right,text=black,font=\footnotesize] {$v(-a)=-\lambda\,v(a)$};}
\end{tikzpicture}
\end{document}
