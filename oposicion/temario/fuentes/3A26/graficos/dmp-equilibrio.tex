% Equilibrio del modelo DMP (gráfico de referencia de la ficha dmp).
% Código base común con dmp-estatica.tex (mismos números):
%  Panel (theta, w), en unidades del dibujo:
%   curva salarial  w = (1-gamma) z + gamma (y + c theta) = 1,4 + 0,35 theta
%                    (y = 3,6, gamma = 0,5; ordenada (1-gamma) z + gamma y = 1,4)
%   creación de empleo  w = y - (r+s) c theta^alpha / A = 3,6 - 0,838 sqrt(theta)
%                    (alpha = 0,5); se cortan en theta* = 2,5, w* = 2,275.
%  Panel (U, V): tasas multiplicadas por 15; theta real = 0,32 theta del dibujo
%   (theta* = 0,8); curva de Beveridge V = [s(1-U)/(A U^alpha)]^(1/(1-alpha))
%   con s/A = 0,15: V = 5,0625 (1 - U/15)^2 / U en el dibujo; rayo V = 0,8 U;
%   U* = s/(s+f(theta*)) -> (2,1543; 1,7234).
% Capas 9-12: estática comparativa de cada parámetro sobre cada curva, como la
% imagen 6 del tema antiguo (signos del anexo B: curva salarial no depende de
% r, s ni A; creación de empleo w = y - (r+s) c theta^alpha / A).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-0.7,-0.8) rectangle (17.2,5.2);
  % ---------------- Panel izquierdo: (theta, w) ----------------------------
  % 1: ejes
  \capa{1}{\ejes{$\theta$}{$w$}{5.8}{4.6}}
  % 2: curva salarial, recta creciente
  \capa{2}{\draw[curva2] (0,1.4) -- (5.2,3.22) node[etiqueta, right, text=tceeazul] {curva salarial};}
  % 3: creación de empleo, decreciente desde w = y
  \capa{3}{\draw[curva3] plot[domain=0:2.2804, samples=60, smooth] ({(\x)^2},{3.6-0.838*\x})
      node[etiqueta, right, align=left, text=tceeverde] {creación\\de empleo};
    \node[left] at (0,3.6) {$y$};}
  % 4: equilibrio (theta*, w*)
  % theta* en el color del rayo V = theta* U del panel derecho (es el mismo número)
  \capa{4}{\node[punto] (E) at (2.5,2.275) {};
    \draw[guia, draw=tceedorado] (E) -- (2.5,0) node[below, text=tceedorado] {$\theta^*$};
    \draw[guia] (E) -- (0,2.275) node[left] {$w^*$};}
  % ---------------- Panel derecho: (U, V) ----------------------------------
  \begin{scope}[xshift=9.6cm]
    % 5: ejes
    \capa{5}{\ejes{$U$}{$V$}{5.8}{4.6}}
    % 6: curva de Beveridge, decreciente y convexa
    \capa{6}{\draw[curva1] plot[domain=1.12:5.3, samples=80, smooth] (\x,{5.0625*(1-\x/15)^2/\x})
        node[etiqueta, above right, align=left, text=tceemorado] {curva de\\Beveridge, $B_0$};}
    % 7: rayo de la tensión de equilibrio, V = theta* U
    \capa{7}{\draw[curva4] (0,0) -- (5.2,4.16) node[etiqueta, right, text=tceedorado] {$V=\theta^*U$};}
    % 8: equilibrio (U*, V*)
    \capa{8}{\node[punto] (B) at (2.1543,1.7234) {};
      \proyeccion{(B)}{$U^*$}{$V^*$}}
  \end{scope}
  % ---------------- Estática comparativa: qué mueve cada curva ------------
  % (flechas: hacia dónde se desplaza cada curva cuando sube el parámetro)
  % 9: la curva salarial sube con y, z, gamma y c
  \capa{9}{\draw[flecha] (4.4,2.94) -- (3.8,3.75)
      node[etiqueta, above, text=tceerojo] {$\uparrow y,\ \uparrow z,\ \uparrow\gamma,\ \uparrow c$};}
  % 10: la creación de empleo sube con y y A, y baja con c, r y s
  \capa{10}{\draw[flecha] (1.2,2.682) -- (1.75,3.4)
      node[etiqueta, above, text=tceerojo] {$\uparrow y,\ \uparrow A$};
    \draw[flecha] (4.0,1.924) -- (3.5,1.15)
      node[etiqueta, below, text=tceerojo] {$\uparrow c,\ \uparrow r,\ \uparrow s$};}
  \begin{scope}[xshift=9.6cm]
    % 11: el rayo gira hacia arriba si sube theta* (y, A) y hacia abajo si baja
    \capa{11}{\draw[flecha] (4.0,3.2) to[bend right=20] (3.3,3.75)
        node[etiqueta, above left, text=tceerojo] {$\uparrow y,\ \uparrow A$};
      \draw[flecha] (4.0,3.2) to[bend left=20] (4.6,2.65)
        node[etiqueta, below right, align=left, text=tceerojo] {$\uparrow z,\ \uparrow\gamma,$\\$\uparrow c,\ \uparrow r,\ \uparrow s$};}
    % 12: la curva de Beveridge se aleja del origen con s y con menos A
    \capa{12}{\draw[flecha] (3.3,0.95) -- (3.85,1.5)
        node[etiqueta, above, text=tceerojo] {$\uparrow s,\ \downarrow A$};}
  \end{scope}
\end{tikzpicture}
\end{document}
