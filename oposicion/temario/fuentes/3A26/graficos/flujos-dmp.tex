% Flujos del modelo DMP: población activa normalizada a 1, sin inactivos.
% Ocupados (1 - U) y parados (U); las vacantes (V) se abren con libre entrada
% y se cubren a la tasa q(theta). Notación del canon: s, f(theta), q(theta).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,
    estado/.style={rounded corners=6pt, draw=tceemorado, fill=tceemorado!12, very thick,
      minimum width=2.9cm, minimum height=1.15cm, align=center, font=\normalsize},
    vacante/.style={estado, draw=tceeverde, fill=tceeverde!12},
    transicion/.style={->, thick, black!75},
    tasa/.style={font=\normalsize, text=black, fill=none, inner sep=3pt, align=center}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-4.2,-4.6) rectangle (8.2,1.9);
  % 1: ocupados y parados
  \capa{1}{
    \node[estado] (O) at (0,0) {Ocupados\\$1-U$};
    \node[estado] (P) at (5.6,0) {Parados\\$U$};
  }
  % 2: separaciones, de ocupados a parados (por arriba)
  \capa{2}{
    \draw[transicion] (O.north east) to[out=35, in=145]
      node[tasa, midway, above] {$s$ (separaciones)} (P.north west);
  }
  % 3: salida del paro, de parados a ocupados (por abajo)
  \capa{3}{
    \draw[transicion] (P.south west) to[out=215, in=-35]
      node[tasa, midway, below] {$f(\theta)$ (encuentran empleo)} (O.south east);
  }
  % 4: vacantes: entran con libre entrada y se cubren a la tasa q(theta)
  \capa{4}{
    \node[vacante] (V) at (0,-3) {Vacantes\\$V$};
    \draw[transicion] (V.north) -- node[tasa, midway, left] {$q(\theta)$\\(se cubren)} (O.south);
    \draw[transicion] (-4.1,-3) -- node[tasa, midway, below, font=\small] {libre entrada\\(coste $c$)} (V.west);
  }
  % 5: estado estacionario
  \capa{5}{
    \node[font=\small, anchor=west] at (2.0,-3.0) {estado estacionario:};
    \node[font=\small, anchor=west] at (2.0,-3.6) {entradas al paro $s\,(1-U)$ $=$ salidas $f(\theta)\,U$};
  }
\end{tikzpicture}
\end{document}
