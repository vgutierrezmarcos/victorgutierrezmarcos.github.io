% Diagrama de fases del empleo con costes de ajuste cuadráticos, en (L, lambda).
% Es el diagrama de la q de Tobin (3.A.34) con L en lugar de K.
%   dL/dt = lambda/phi;  dlambda/dt = r lambda - [F_L(L) - w]
%   Con F_L lineal, F_L - w = -(L - L*)/2, y en unidades del dibujo r = 0,5,
%   phi = 1, L* = 4:
%   locus dlambda/dt = 0: lambda = [F_L(L) - w]/r = L* - L (pendiente -1)
%   raíces: mu = (r +- sqrt(r^2 + 2/phi))/2 = -0,5 y 1
%   senda estable: lambda = -0,5 (L - L*), MENOS inclinada que el locus
%     (siempre: |pendiente| = 1/(2(r + |mu_s|)) < 1/(2r));
%   senda inestable: lambda = L - L*.
%   Subida permanente de la productividad: L*_1 = 6; lambda salta en L = 4
%   de 0 a 1 (la nueva senda estable) y converge a (6, 0).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=1cm,y=1cm,
    dir/.style={->,thick,tceegris}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-0.8,-4.0) rectangle (10.2,4.2);
  % 1: ejes (lambda puede ser negativo: el eje horizontal va por el centro)
  \capa{1}{\draw[eje] (0,0) -- (8.8,0) node[below left=2pt and 0pt] {$L$};
    \draw[eje] (0,-3.7) -- (0,3.8) node[left] {$\lambda$};
    \node[left] at (0,0) {$0$};}
  % 2: los 2 loci y el estado estacionario
  \capa{2}{\draw[curva2] (0,0) -- (8.3,0);
    \node[etiqueta,text=tceeazul,anchor=south east] at (8.4,0.05) {$\dot L=0$};}
  \capa{2}{\draw[curva1] (0.7,3.3) -- (7.5,-3.5);
    \node[etiqueta,text=tceemorado,anchor=west] at (0.8,3.4) {$\dot\lambda=0$};}
  \capa{2}{\node[punto] (E) at (4,0) {};
    \node[below left=1pt and -1pt] at (E) {$L^*$};}
  % 3: flechas de dirección (encima de lambda = 0, L sube; a la derecha del
  % locus dlambda/dt = 0, lambda sube)
  \capa{3}{\draw[dir] (6.6,1.6) -- ++(0.6,0); \draw[dir] (6.6,1.6) -- ++(0,0.6);}
  \capa{3}{\draw[dir] (1.1,0.85) -- ++(0.6,0); \draw[dir] (1.1,0.85) -- ++(0,-0.6);}
  \capa{3}{\draw[dir] (2.6,-1.8) -- ++(-0.6,0); \draw[dir] (2.6,-1.8) -- ++(0,-0.6);}
  \capa{3}{\draw[dir] (6.9,-2.1) -- ++(-0.6,0); \draw[dir] (6.9,-2.1) -- ++(0,0.5);}
  % 4: la senda estable, menos inclinada que el locus, y la inestable
  \capa{4}{\draw[curva3] (0.5,1.75) -- node[pos=0.13,sloped,above,etiqueta,text=tceeverde] {senda estable} (7.5,-1.75);
    \draw[->,thick,tceeverde] (1.6,1.2) -- (2.6,0.7);
    \draw[->,thick,tceeverde] (6.6,-1.3) -- (5.6,-0.8);}
  \capa{4}{\draw[thick,dotted,tceegris] (1.3,-2.7) -- (7.3,3.3)
      node[etiqueta,above,text=tceegris] {senda inestable};}
  % 5: sube la productividad: el locus se desplaza, lambda salta y L converge
  \capa{5}{\draw[curvanueva,desplazada] (2.7,3.3) -- (8.5,-2.5);
    \node[etiqueta,text=tceerojo,anchor=west] at (8.5,-2.5) {$\dot\lambda=0'$};
    \node[punto,fill=tceerojo] (E1) at (6,0) {};
    \node[below left=1pt and -1pt,text=tceerojo] at (E1) {$L_1^*$};}
  \capa{5}{\draw[curvanueva] (2.6,1.7) -- (8.6,-1.3);
    \node[etiqueta,text=tceerojo,anchor=west] at (8.6,-1.3) {senda estable$'$};}
  \capa{5}{\draw[flecha,line width=1.2pt] (4,0.08) -- (4,0.92);
    \node[etiqueta,text=tceerojo,anchor=south west] at (4.04,0.06) {$\lambda$ salta};
    \draw[flecha] (4.3,1.08) -- (5.5,0.48);}
\end{tikzpicture}
