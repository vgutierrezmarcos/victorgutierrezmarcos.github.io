\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
% Curva de Beveridge (medias anuales; 2026 = media 1T-2T): tasa de paro 15-74
% frente a tasa de vacantes. Eurostat, jvs_q_r21 y une_rt_q/une_rt_m (consulta
% 31/10/2026). Mismo código, paleta y estilo que el gráfico de la 1D01
% (fuentes/1D01/graficos/beveridge-espana-zona-euro.tex), ampliado a 2008.
% España 2008-2009 en trazo discontinuo: la serie de vacantes salta entre
% 4T2009 y 1T2010 sin nota de ruptura (columna ruptura del CSV).
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/beveridge-espana-zona-euro.csv}\beveridge
% Filas del CSV: 0-18 zona del euro (2008-2026), 19-37 España (2008-2026);
% España: 19-20 (2008-2009) discontinuo, 20-21 (2009-2010) de puntos, 21-37 continuo.
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=12cm, height=8cm,
      xmin=5, xmax=28, xtick={6,8,...,26},
      ymin=0, ymax=3.6, ytick={0,0.5,...,3.5},
      xlabel={Tasa de paro, 15-74 años (\%)},
      ylabel={Tasa de vacantes (\%)},
      axis lines*=left, tick align=outside, clip=false,
      ticklabel style={font=\footnotesize},
      /pgf/number format/.cd, use comma, fixed, fixed zerofill=false,
      /pgfplots/.cd,
      grid=major, grid style={tceegris!20},
      legend style={at={(0.5,-0.16)}, anchor=north, legend columns=2,
        draw=none, font=\footnotesize, /tikz/every even column/.append style={column sep=10pt}},
      legend cell align=left, dicapas,
    ]
    % Paleta «dictamen»: zona del euro = dicze (círculos), España = dicespana
    % (cuadrados). Capas «dic» de tikz-tcee.tex: cuadrícula → ejes → líneas →
    % puntos → rótulos.
    % 2: zona del euro, 2008-2026 (línea y puntos)
    \capa{2}{
      \addplot[diclinea=dicze, thick, skip coords between index={19}{38},
        legend image post style={mark=*, mark size=1.6pt, mark options={fill=white}}]
        table[x=tasa_paro, y=tasa_vacantes] {\beveridge};
      \addlegendentry{Zona del euro, 2008-2026}
      \addplot[on layer=dicpuntos, forget plot, dicpuntos=dicze, mark size=1.6pt, skip coords between index={19}{38}]
        table[x=tasa_paro, y=tasa_vacantes] {\beveridge};
    }
    % 3: España, 2010-2026 continuo y 2008-2009 discontinuo, unido a 2010 con
    % una línea de puntos (línea y puntos)
    \capa{3}{
      \addplot[diclinea=dicespana, thick, skip coords between index={0}{21},
        legend image post style={mark=square*, mark size=1.5pt, mark options={fill=white}}]
        table[x=tasa_paro, y=tasa_vacantes] {\beveridge};
      \addlegendentry{España, 2010-2026}
      \addplot[diclinea=dicespana, thick, dashed, skip coords between index={0}{19},
        skip coords between index={21}{38},
        legend image post style={mark=square*, mark size=1.5pt, mark options={fill=white, solid}}]
        table[x=tasa_paro, y=tasa_vacantes] {\beveridge};
      \addlegendentry{España, 2008-2009 (antes del salto de la serie)}
      \addplot[diclinea=dicespana, thick, dotted, forget plot, skip coords between index={0}{20},
        skip coords between index={22}{38}]
        table[x=tasa_paro, y=tasa_vacantes] {\beveridge};
      \addplot[on layer=dicpuntos, forget plot, dicpuntos=dicespana, mark=square*, mark size=1.5pt, skip coords between index={0}{19}]
        table[x=tasa_paro, y=tasa_vacantes] {\beveridge};
    }
    % 4: rótulos de año
    \capa{4}{
      \dicrotulo[font=\scriptsize, text=dicze, anchor=east]{(7.45,1.23)}{2008}
      \dicrotulo[font=\scriptsize, text=dicze, anchor=east]{(6.55,3.2)}{2022}
      \dicrotulo[font=\scriptsize, text=dicze, anchor=east]{(6.15,2.2)}{2026}
      \dicrotulo[font=\scriptsize, text=dicespana, anchor=north]{(11.3,0.58)}{2008}
      \dicrotulo[font=\scriptsize, text=dicespana, anchor=north]{(17.9,0.58)}{2009}
      \dicrotulo[font=\scriptsize, text=dicespana, anchor=south]{(19.9,1.32)}{2010}
      \dicrotulo[font=\scriptsize, text=dicespana, anchor=west]{(26.3,0.57)}{2013}
      \dicrotulo[font=\scriptsize, text=dicespana, anchor=north]{(24.5,0.36)}{2014}
      \dicrotulo[font=\scriptsize, text=dicespana, anchor=south west]{(14.1,0.73)}{2019}
      \dicrotulo[font=\scriptsize, text=dicespana, anchor=south]{(13.1,0.96)}{2022}
      \dicrotulo[font=\scriptsize, text=dicespana, anchor=north]{(10.2,0.83)}{2026}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
