\begin{tikzpicture}[tcee,
    declare function={onda(\t)=2.2+1.2*sin(deg(pi*\t/2));
      cuad(\t)=3.6-2.8*exp(-(\t-2)/1.2);
      antes(\t)=0.8+1.7887*(exp((\t-2)/1.2)-exp(-1/1.2));
      despues(\t)=3.6-1.7887*exp(-(\t-2)/1.2);}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-0.9,-0.8) rectangle (17.3,5.0);
  % ---------------- Panel a: perturbación permanente -----------------------
  % 1: ejes, t_0 y el nuevo óptimo L_1*
  \capa{1}{\ejes{$t$}{$L_t$}{6.8}{4.3}
    \node[etiqueta,font=\small\itshape,anchor=south] at (3.4,4.5) {a) Perturbación permanente};
    \draw[guia] (2,0) node[below,text=black] {$t_0$} -- (2,4.0);
    \draw[guia] (0,3.6) node[left,text=black] {$L_1^*$} -- (6.4,3.6);
    \node[left] at (0,0.8) {$L_0$};}
  % 2: fijos, todo o nada: salto completo si la ganancia supera Phi_0; si no, nada
  \capa{2}{\draw[curva4,thick] (0,0.8) -- (2,0.8) -- (2,3.6) -- (6.2,3.6);
    \node[etiqueta,text=tceedorado,anchor=north west,align=left] at (2.15,3.5) {todo: salto completo\\si la ganancia supera $\Phi_0$};}
  \capa{2}{\draw[curva4,thick,dotted] (2,0.8) -- (6.2,0.8);
    \node[etiqueta,text=tceedorado,anchor=south] at (4.6,0.85) {o nada};}
  % ---------------- Panel b: perturbación transitoria ----------------------
  \begin{scope}[xshift=9.6cm]
    % 3: ejes y L*
    \capa{3}{\ejes{$t$}{$L_t$}{6.8}{4.3}
      \node[etiqueta,font=\small\itshape,anchor=south] at (3.4,4.5) {b) Perturbación transitoria (ciclo)};
      \draw[guia] (0,2.2) node[left,text=black] {$L^*$} -- (6.4,2.2);}
    \capa{3}{\draw[curva2,thick,tceeazul!35] plot[domain=0:6.2,samples=100,smooth] (\x,{onda(\x)});
      \node[etiqueta,text=tceeazul!60,anchor=south] at (1,3.45) {sin costes};}
    % 4: fijos, ante perturbaciones pequeñas y transitorias, casi siempre nada
    \capa{4}{\draw[curva4,thick] (0,2.2) -- (6.2,2.2);
      \node[etiqueta,text=tceedorado,align=center,anchor=north] at (3.45,0.92) {fijos: casi siempre nada\\(la plantilla no se toca)};}
  \end{scope}
\end{tikzpicture}
