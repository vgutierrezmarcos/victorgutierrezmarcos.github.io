% Banda de inacción con costes lineales de ajuste, sobre la demanda de trabajo
% a corto plazo del 3.A.25 (demanda-corto.tex: misma curva y escala).
%   F_L(L) = 8 - L/2;  w = 5 -> L* = 6
%   r phi_c = 0,8 -> w + r phi_c = 5,8 -> L_inf = 4,4
%   r phi_d = 1,2 -> w - r phi_d = 3,8 -> L_sup = 8,4  (phi_d > phi_c: la banda
%   es más ancha por la derecha, la de los despidos)
%   perturbación (misma que en demanda-corto): F_L'(L) = 9,5 - L/2
%   -> nuevo borde inferior L_inf,1 = 7,4; la empresa en L_0 = L* = 6 contrata
%   solo hasta 7,4 (no hasta el nuevo L*_1 = 9).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=1cm,y=0.6cm,
    declare function={fl(\l)=8-0.5*\l; fln(\l)=9.5-0.5*\l;}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-1.6,-1.5) rectangle (13.4,10.2);
  % 1: ejes y demanda de trabajo a corto plazo
  \capa{1}{\ejes{$L$}{$w,\ F_L$}{10}{9.8}}
  \capa{1}{\draw[curva1] plot[domain=0:9.2] (\x,{fl(\x)})
    node[etiqueta,below right=0pt and -4pt,align=left,text=tceemorado] {$F_L(L)$: demanda de\\trabajo a corto plazo};}
  % 2: el salario y el empleo sin costes de ajuste
  \capa{2}{\draw[curva2] (0,5) node[left,text=black] {$w$} -- (9.8,5);}
  \capa{2}{\node[punto] (E) at (6,5) {};
    \draw[guia] (E) -- (6,0) node[below,text=black] {$L^*$};}
  % 3: los umbrales de contratar (arriba) y de despedir (abajo)
  \capa{3}{\draw[curva2,dashed] (0,5.8) node[left,text=black] {$w+r\phi_c$} -- (9.8,5.8);
    \node[punto] (Linf) at (4.4,5.8) {};
    \draw[guia] (Linf) -- (4.4,0) node[below,text=black] {$\underline{L}$};}
  \capa{3}{\draw[curva2,dashed] (0,3.8) node[left,text=black] {$w-r\phi_d$} -- (9.8,3.8);
    \node[punto] (Lsup) at (8.4,3.8) {};
    \draw[guia] (Lsup) -- (8.4,0) node[below,text=black] {$\overline{L}$};}
  % 4: la banda de inacción
  \capa{4}{\begin{pgfonlayer}{background}
      \fill[tceesalvia] (4.4,0) rectangle (8.4,9.3);
      \draw[tceeverde,line width=2.5pt] (4.4,0) -- (8.4,0);
    \end{pgfonlayer}
    \node[etiqueta,align=center,text=tceeverde,anchor=south] at (6.4,7.6) {banda de inacción:\\no se toca la plantilla};}
  % 5: sube la productividad: la empresa contrata solo hasta el nuevo borde
  \capa{5}{\draw[curvanueva,desplazada] plot[domain=0:8.6] (\x,{fln(\x)});
    \node[etiqueta,above right,text=tceerojo] at (2.6,{fln(2.6)}) {$F_L'(L)$};
    \draw[flecha] (1.4,{fl(1.4)+0.3}) -- (1.4,{fln(1.4)-0.3});}
  \capa{5}{\node[punto,fill=tceerojo] (Linf1) at (7.4,5.8) {};
    \draw[guia,tceerojo] (Linf1) -- (7.4,0) node[below,text=tceerojo] {$\underline{L}_1$};
    \draw[flecha] (6,2.3) node[left,text=black] {$L_0$} -- (7.4,2.3);
    \node[etiqueta,text=tceerojo,align=right,anchor=east] at (4.25,2.3) {la empresa ajusta\\solo hasta el borde};}
\end{tikzpicture}
\end{document}
