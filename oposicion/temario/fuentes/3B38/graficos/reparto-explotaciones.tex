% Reparto de la ganancia del sostenimiento de precios entre una explotación
% pequeña y una grande (Baldwin y Wyplosz, 2020, cap. 9, fig. 9.5), con los
% precios del modelo del tema: p^W = 3 y p^I = 4.
% Pequeña: oferta p = 3,2 + 0,8 q (a p^W no produce; a p^I, q = 1);
%   ganancia = 0,5 x 1 x 0,8 = 0,4.
% Grande: oferta p = 1 + 0,5 q (q = 4 a p^W, 6 a p^I); ganancia = 5.
% Total (suma horizontal): q = 4 a p^W, 7 a p^I; ganancia 5,4, el 93 % para
% la grande. Misma escala en los 3 paneles.
% Capas: ejes y precios -> oferta pequeña y su ganancia -> oferta grande y
% su ganancia -> oferta total y ganancia total.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.6cm,y=0.9cm,
    declare function={pp(\x)=3.2+0.8*\x; pg(\x)=1+0.5*\x;},
    titulo/.style={font=\small\bfseries,anchor=south},
    precios/.style={}]
  \newcommand{\precios}[1]{%
    \draw[curva2] (0,3) -- (#1,3) node[etiqueta,right,text=tceeazul] {$p^W$};
    \draw[curvanueva] (0,4) -- (#1,4) node[etiqueta,right,text=tceerojo] {$p^I$};}
  % ---------- pequeña
  \begin{scope}
    \capa{1}{\ejes{$q$}{$p$}{3.6}{5.4}}
    \capa{1}{\node[titulo] at (1.6,5.45) {Explotación pequeña};}
    \capa{1}{\precios{3.2}}
    \capa{2}{\draw[curva3] (0,{pp(0)}) -- (2.2,{pp(2.2)}) node[etiqueta,above] {$q_P(p)$};}
    \capa{2}{\fill[area2] (0,{pp(0)}) -- (1,4) -- (0,4) -- cycle;}
    \capa{2}{\node[punto] at (1,4) {};}
    \capa{2}{\draw[guia] (1,4) -- (1,0) node[below,text=black] {$1$};}
    \capa{2}{\node[etiqueta,font=\footnotesize,text=tceedorado!80!black,align=center] at (1.6,1.6) {gana\\$0{,}4$};}
  \end{scope}
  % ---------- grande
  \begin{scope}[xshift=4.0cm]
    \capa{3}{\ejes{$q$}{$p$}{8.6}{5.4}}
    \capa{3}{\node[titulo] at (4.2,5.45) {Explotación grande};}
    \capa{3}{\precios{8.2}}
    \capa{3}{\draw[curva3] (0,{pg(0)}) -- (8,{pg(8)}) node[etiqueta,above] {$q_G(p)$};}
    \capa{4}{\fill[area2] (0,3) -- (4,3) -- (6,4) -- (0,4) -- cycle;}
    \capa{4}{\node[punto] at (4,3) {};}
    \capa{4}{\node[punto] at (6,4) {};}
    \capa{4}{\draw[guia] (4,3) -- (4,0) node[below,text=black] {$4$};}
    \capa{4}{\draw[guia] (6,4) -- (6,0) node[below,text=black] {$6$};}
    \capa{4}{\node[etiqueta,font=\footnotesize,text=tceedorado!80!black,align=center] at (6.4,1.6) {gana\\$5$};}
  \end{scope}
  % ---------- total
  \begin{scope}[xshift=10.6cm]
    \capa{5}{\ejes{$q$}{$p$}{9.0}{5.4}}
    \capa{5}{\node[titulo] at (4.4,5.45) {Total};}
    \capa{5}{\precios{8.6}}
    \capa{5}{\draw[curva3] (0,1) -- (4.4,3.2) -- (8.4,4.43) node[etiqueta,above] {$q(p)$};}
    \capa{6}{\fill[area2] (0,3) -- (4,3) -- (4.4,3.2) -- (7,4) -- (0,4) -- cycle;}
    \capa{6}{\node[punto] at (4,3) {};}
    \capa{6}{\node[punto] at (7,4) {};}
    \capa{6}{\draw[guia] (4,3) -- (4,0) node[below,text=black] {$4$};}
    \capa{6}{\draw[guia] (7,4) -- (7,0) node[below,text=black] {$7$};}
    \capa{6}{\node[etiqueta,font=\footnotesize,text=tceedorado!80!black,align=center] at (6.6,1.6) {ganan $5{,}4$:\\el 93\,\% la grande};}
  \end{scope}
\end{tikzpicture}
\end{document}
