% Curva de Laffer del señoreaje (Cagan, 1956). Gráfico de referencia de la
% ficha «cagan».
%   S(pi) = pi zeta e^{-alpha pi}, con zeta = 1 y alpha = 2:
%   máximo en pi* = 1/alpha = 0,5, S^max = zeta/(alpha e) = 0,1839.
%   Déficit S = 0,12: pi_B = 0,1679 y pi_A = 1,1142 (calculados con brentq).
%   Déficit S' = 0,22 > S^max: no hay estado estacionario.
% Escala: 1 unidad de pi = 3 cm; 1 unidad de S = 20 cm.
% Autor: Víctor Gutiérrez Marcos

  % 1: ejes (pi, S)
  \capa{1}{\ejes{$\pi$}{$S$}{2.75}{0.26}}

  \capa{2}{\draw[curva1] plot[domain=0:2.55, samples=160, smooth] (\x,{\x*exp(-2*\x)}) node[etiqueta, above right=0pt and -2pt, text=tceemorado] {$S(\pi)=\pi\,\zeta\,e^{-\alpha\pi}$};}
  \capa{2}{\proyeccion{(0.5,0.18394)}{$\pi^*=\frac{1}{\alpha}$}{$S^{\max}=\frac{\zeta}{\alpha e}$}}
