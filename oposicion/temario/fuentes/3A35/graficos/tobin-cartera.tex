% La cartera de Tobin (1958): dinero (seguro, no rinde) y bonos (tipo i,
% desviación típica sigma_B). Arriba, carteras posibles E[r_p] = (i/sigma_B)
% sigma_p, hasta sigma_p = sigma_B (todo en bonos), y curvas de indiferencia
% media-desviación E = u + 0,125 sigma_p^2 (crecientes y convexas: aversión al
% riesgo). Abajo, sigma_p = w_B sigma_B con w_B hacia abajo, de 0 a 1.
% Calibración: sigma_B = 4. Con i = 2 (pendiente 0,5) la tangencia es
% 0,25 sigma = 0,5: sigma_p* = 2, E* = 1 (u = 0,5), w_B* = 0,5.
% Con i' = 3 (pendiente 0,75): sigma_p*' = 3, E*' = 2,25 (u = 1,125),
% w_B*' = 0,75 (domina el efecto sustitución).
% Escalas: x = 1,4 cm por unidad de sigma; arriba 1,3 cm por unidad de E;
% abajo w_B = 1 a 2,6 cm (en el código, E va multiplicada por 1,3).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=1.4cm,y=1cm,
    declare function={ind(\s,\u)=1.3*(\u+0.125*\s^2); wy(\w)=-2.6*\w;}]
  % 1: cuadrante superior, ejes
  \capa{1}{\ejes{$\sigma_p$}{$\mathbb{E}[r_p]$}{5.1}{4.5}}
  % 2: recta de oportunidades, pendiente i/sigma_B (hasta todo en bonos)
  \capa{2}{\draw[curva2] (0,0) -- (4,2.6) node[etiqueta,right,text=tceeazul] {$\dfrac{i}{\sigma_{\mathrm B}}$};}
  % 3: curva de indiferencia y tangencia E
  \capa{3}{\draw[curva3] plot[domain=0:3.7,samples=60,smooth] (\x,{ind(\x,0.5)}) node[etiqueta,right,text=tceeverde] {$U_0$};}
  \capa{3}{\node[punto] (E) at (2,1.3) {};}
  \capa{3}{\node[etiqueta,below right] at (E) {$E$};}
  \capa{3}{\draw[guia] (E) -- (0,1.3) node[left,text=black] {$\mathbb{E}[r_p]^*$};}
  % 4: cuadrante inferior, ejes y recta sigma_p = w_B sigma_B
  \capa{4}{\draw[eje] (0,0) -- (0,{wy(1.17)}) node[below] {$w_{\mathrm B}$};}
  \capa{4}{\draw (0,{wy(1)}) -- (-0.07,{wy(1)}) node[left] {$1$};}
  \capa{4}{\draw[guia] (0,{wy(1)}) -- (4,{wy(1)});}
  \capa{4}{\draw[guia] (4,0) -- (4,{wy(1)});}
  \capa{4}{\node[below right,inner sep=2pt] at (4,0) {$\sigma_{\mathrm B}$};}
  \capa{4}{\draw[curva1] (0,0) -- (4,{wy(1)}) node[etiqueta,right,text=tceemorado] {$\sigma_p=w_{\mathrm B}\,\sigma_{\mathrm B}$};}
  % 5: proyección de sigma_p* a w_B* y el resto en dinero, 1 - w_B*
  \capa{5}{\draw[guia] (E) -- (2,{wy(0.5)}) -- (0,{wy(0.5)});}
  \capa{5}{\node[below left,inner sep=1.5pt] at (2,0) {$\sigma_p^*$};}
  \capa{5}{\node[left] at (0,{wy(0.5)}) {$w_{\mathrm B}^*$};}
  \capa{5}{\draw[decorate,decoration={brace,amplitude=4pt},thick,tceedorado] (-0.75,{wy(0.5)-0.05}) -- (-0.75,{wy(0)+0.02}) node[midway,left=5pt,align=right,text=black] {$w_{\mathrm B}^*$\\bonos};}
  \capa{5}{\draw[decorate,decoration={brace,amplitude=4pt},thick,tceedorado] (-0.75,{wy(1)}) -- (-0.75,{wy(0.5)-0.15}) node[midway,left=5pt,align=right,text=black] {$1-w_{\mathrm B}^*$\\dinero};}
  % 6: subida del tipo: la recta gira hacia arriba, nueva tangencia y nueva w_B
  \capa{6}{\draw[curvanueva,desplazada] (0,0) -- (4,3.9) node[etiqueta,right,text=tceerojo] {$\dfrac{i'}{\sigma_{\mathrm B}}$};}
  \capa{6}{\draw[curvanueva] plot[domain=0:3.45,samples=60,smooth] (\x,{ind(\x,1.125)}) node[etiqueta,above left,text=tceerojo] {$U_1$};}
  \capa{6}{\node[punto,fill=tceerojo] (E2) at (3,2.925) {};}
  \capa{6}{\node[etiqueta,below right,text=tceerojo] at (E2) {$E'$};}
  \capa{6}{\draw[guia] (E2) -- (3,{wy(0.75)}) -- (0,{wy(0.75)});}
  \capa{6}{\node[below left,inner sep=1.5pt,text=tceerojo] at (3,0) {$\sigma_p^{*\prime}$};}
  \capa{6}{\node[left,text=tceerojo] at (0,{wy(0.75)}) {$w_{\mathrm B}^{*\prime}$};}
\end{tikzpicture}
\end{document}
