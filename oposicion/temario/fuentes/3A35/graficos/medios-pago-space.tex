% Peso del efectivo en los pagos en el punto de venta (%), España y zona del
% euro, 2016, 2019, 2022 y 2024: a la izquierda en número de operaciones y a
% la derecha en valor. Datos: ../datos/efectivo-pdv-space.csv (BCE, SPACE
% 2024, gráfico 6; consulta 05/11/2026).
% Cifras de control (2024): España 57,4 (número) y 44,7 (valor); zona del
% euro 51,6 y 39,2. En 2016: España 87,1 y 67,8; zona del euro 78,8 y 53,8.
% Paleta «dictamen»: España dicespana, zona del euro dicze (como en el 3.A.26
% y el 3.A.39); el mismo color en los 2 paneles.
% Capas (CRITERIOS §7): barras → ejes → etiquetas de 2024 → leyenda.
% Los 2 paneles van en un solo axis (número en x = 0-3, valor en x = 4,5-7,5):
% con las capas de dicapas, un segundo axis desplazado (scope o at=) pierde el
% desplazamiento de las capas y el marco del dibujo.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/efectivo-pdv-space.csv}\efectivo
\begin{tikzpicture}[tcee]
\begin{axis}[width=14.6cm, height=6.6cm, ymin=0, ymax=100, ytick={0,20,...,100},
    xmin=-0.6, xmax=8.1, xtick={0,1,2,3,4.5,5.5,6.5,7.5},
    xticklabels={2016,2019,2022,2024,2016,2019,2022,2024},
    ybar, bar width=0.3,
    axis line style={draw=none}, tick style={draw=none},
    ticklabel style={font=\footnotesize},
    ylabel={\% de los pagos en tienda}, ylabel style={font=\footnotesize},
    clip=false,
    legend style={at={(0.5,-0.13)}, anchor=north, draw=none, font=\footnotesize, legend columns=2,
      /tikz/every even column/.append style={column sep=8pt}},
    legend cell align=left, dicapas]
  % cuadrícula de cada panel (bajo las barras)
  \foreach \y in {20,40,...,100} {
    \addplot[on layer=dicfondo, dicsuelto, tceegris!25] coordinates {(-0.6,\y) (3.6,\y)};
    \addplot[on layer=dicfondo, dicsuelto, tceegris!25] coordinates {(3.9,\y) (8.1,\y)};
  }
  % 1: barras (número a la izquierda, valor a la derecha)
  \capa{1}{
    \addplot[on layer=dicbarras, dicbarra=dicespana, forget plot, bar shift=-0.165] table[x expr=\coordindex, y=espana_numero] {\efectivo};
    \addplot[on layer=dicbarras, dicbarra=dicze, forget plot, bar shift=0.165] table[x expr=\coordindex, y=zona_euro_numero] {\efectivo};
    \addplot[on layer=dicbarras, dicbarra=dicespana, forget plot, bar shift=-0.165] table[x expr=\coordindex+4.5, y=espana_valor] {\efectivo};
    \addplot[on layer=dicbarras, dicbarra=dicze, forget plot, bar shift=0.165] table[x expr=\coordindex+4.5, y=zona_euro_valor] {\efectivo};
  }
  % 2: ejes de los 2 paneles y sus títulos
  \capa{2}{
    \dicraya{(-0.6,0) (3.6,0)}
    \dicraya{(-0.6,0) (-0.6,100)}
    \dicraya{(3.9,0) (8.1,0)}
    \dicraya{(3.9,0) (3.9,100)}
    \dicrotulo[font=\small,anchor=south,text height=1.8ex,text depth=0.4ex]{(1.5,102)}{En número de operaciones}
    \dicrotulo[font=\small,anchor=south,text height=1.8ex,text depth=0.4ex]{(6,102)}{En valor}
  }
  % 3: etiquetas de 2024
  \capa{3}{
    \dicrotulo[font=\footnotesize,text=dicespana,anchor=south,inner sep=1.5pt]{(2.835,57.4)}{57,4}
    \dicrotulo[font=\footnotesize,text=dicze,anchor=south,inner sep=1.5pt]{(3.165,51.6)}{51,6}
    \dicrotulo[font=\footnotesize,text=dicespana,anchor=south,inner sep=1.5pt]{(7.335,44.7)}{44,7}
    \dicrotulo[font=\footnotesize,text=dicze,anchor=south,inner sep=1.5pt]{(7.665,39.2)}{39,2}
  }
  % 4: leyenda
  \capa{4}{
    \addlegendimage{area legend, dicbarra=dicespana}
    \addlegendentry{España}
    \addlegendimage{area legend, dicbarra=dicze}
    \addlegendentry{Zona del euro}
  }
\end{axis}
\end{tikzpicture}
\end{document}
