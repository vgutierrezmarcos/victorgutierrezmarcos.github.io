% Coste de ajuste convexo de la inversión: Phi(I) = (phi/2) I^2, con
% Phi(0) = Phi'(0) = 0 y Phi'' > 0 (Romer, 2019, cap. 9).
%   Dibujo: Phi(I) = 0,6 I^2; I_1 = 1,35 -> Phi(I_1) = 1,094;
%   2 I_1 = 2,7 -> Phi(2 I_1) = 4,374 = 4 Phi(I_1): duplicar el ritmo de
%   inversión cuadruplica el coste de ajuste.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee]
  % 1: ejes: inversión (negativa a la izquierda: desinversión) y coste de ajuste
  \capa{1}{\draw[eje] (-3.4,0) -- (3.6,0) node[below left=2pt and 0pt] {$I$};
    \draw[eje] (0,0) -- (0,5.6) node[left] {$\Phi(I)$};
    \node[below=2pt] at (0,0) {$0$};
    \node[etiqueta,font=\footnotesize,text=tceegris,anchor=north] at (-2.0,-0.42) {desinversión};
    \node[etiqueta,font=\footnotesize,text=tceegris,anchor=north] at (2.0,-0.42) {inversión};}
  % 2: el coste de ajuste, convexo y con mínimo (y pendiente nula) en I = 0
  \capa{2}{\draw[curva1] plot[domain=-2.95:2.95,samples=80,smooth] (\x,{0.6*\x*\x})
      node[etiqueta,right] {$\Phi(I)=\frac{\phi}{2}I^2$};}
  % 3: un ritmo de inversión I_1 y su coste
  \capa{3}{\node[punto] at (1.35,1.094) {};
    \draw[guia] (1.35,0) node[below,text=black] {$I_1$} -- (1.35,1.094) -- (0,1.094) node[left,text=black] {$\Phi(I_1)$};}
  % 4: el doble de ritmo cuesta el cuádruple
  \capa{4}{\node[punto] at (2.7,4.374) {};
    \draw[guia] (2.7,0) node[below,text=black] {$2I_1$} -- (2.7,4.374) -- (0,4.374) node[left,text=black] {$4\,\Phi(I_1)$};}
\end{tikzpicture}
\end{document}
