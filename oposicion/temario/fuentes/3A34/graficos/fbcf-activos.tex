\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
% Formación bruta de capital fijo de España por tipo de activo, 1995-2025
% (% del PIB, precios corrientes, suma de los cuatro trimestres de cada año).
% INE, CNTR base 2020, tabla 67823 (consulta 31/10/2026); columnas *_pib de
% ../datos/fbcf-activos.csv. La última fila (2026_4T, suma 2025-T4 a 2026-T3)
% no se dibuja: el eje x sale del número de fila (1995 + \coordindex) y se
% salta la fila 31.
% Paleta del tema (CRITERIOS §7): construcción en una familia cálida
% (viviendas, tono oscuro; otros edificios y construcciones, claro),
% maquinaria y bienes de equipo en morado y propiedad intelectual en verde
% claro; los recursos biológicos cultivados (0,1-0,2 % del PIB) en gris. De
% abajo arriba, oscuro-claro-oscuro-claro: se distinguen también en grises.
% Capas (CRITERIOS §7): barras → ejes y rejilla → etiquetas → leyenda.
\definecolor{fbviviendas}{HTML}{8C3B12}
\definecolor{fbconstruccion}{HTML}{E8A964}
\colorlet{fbmaquinaria}{tceemorado}
\definecolor{fbintelectual}{HTML}{8CC79A}
\definecolor{fbbiologicos}{HTML}{8A8A8A}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/fbcf-activos.csv}\fbcf
% Ejes y rejilla desde la capa 2: antes, transparentes (no cambia la caja)
\ifnum\tceepaso<2
  \pgfplotsset{fbejes/.style={axis line style={draw=none}, major tick style={draw=none},
    ticklabel style={opacity=0}, ylabel style={opacity=0}, ymajorgrids=false}}
\else
  \pgfplotsset{fbejes/.style={}}
\fi
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=13cm, height=7.5cm,
      ybar stacked, bar width=0.24cm,
      xmin=1994.3, xmax=2025.7, ymin=0, ymax=32,
      xtick={1995,2000,...,2025}, ytick={0,5,...,30},
      xticklabel style={/pgf/number format/1000 sep=, font=\footnotesize},
      yticklabel style={font=\footnotesize},
      ylabel={\% del PIB},
      axis x line*=bottom, axis y line*=left, tick align=outside, clip=false,
      ymajorgrids, grid style={tceegris!25},
      legend style={at={(0.5,-0.13)}, anchor=north, legend columns=3,
        draw=none, font=\footnotesize,
        /tikz/every even column/.append style={column sep=8pt}},
      legend cell align=left, area legend,
      skip coords between index={31}{32},
      dicapas, fbejes,
    ]
    % 1: barras apiladas, de abajo arriba
    \capa{1}{
      \addplot[on layer=dicbarras, dicbarra=fbviviendas, forget plot]
        table[x expr=1995+\coordindex, y=viviendas_pib] {\fbcf};
      \addplot[on layer=dicbarras, dicbarra=fbconstruccion, forget plot]
        table[x expr=1995+\coordindex, y=otras_construcciones_pib] {\fbcf};
      \addplot[on layer=dicbarras, dicbarra=fbmaquinaria, forget plot]
        table[x expr=1995+\coordindex, y=maquinaria_equipo_pib] {\fbcf};
      \addplot[on layer=dicbarras, dicbarra=fbintelectual, forget plot]
        table[x expr=1995+\coordindex, y=propiedad_intelectual_pib] {\fbcf};
      \addplot[on layer=dicbarras, dicbarra=fbbiologicos, forget plot]
        table[x expr=1995+\coordindex, y=recursos_biologicos_pib] {\fbcf};
    }
    % 2: ejes y rejilla (fbejes) y línea del cero, encima de las barras
    \capa{2}{
      \dicraya{(1994.3,0) (2025.7,0)}
    }
    % 3: total en el máximo (2006), el mínimo (2013) y el último año (2025)
    \capa{3}{
      \dicrotulo[font=\footnotesize, anchor=south, inner sep=2pt]{(2006,30.05)}{30,0~\%}
      \dicrotulo[font=\footnotesize, anchor=south, inner sep=2pt]{(2013,18.75)}{17,6~\%}
      \dicrotulo[font=\footnotesize, anchor=south, inner sep=2pt]{(2025,20.97)}{21,0~\%}
    }
    % 4: leyenda (sin caja: toma el fondo del gráfico)
    \capa{4}{
      \addlegendimage{dicbarra=fbviviendas}
      \addlegendentry{Viviendas}
      \addlegendimage{dicbarra=fbconstruccion}
      \addlegendentry{Otros edificios y construcciones}
      \addlegendimage{dicbarra=fbmaquinaria}
      \addlegendentry{Maquinaria y bienes de equipo}
      \addlegendimage{dicbarra=fbintelectual}
      \addlegendentry{Propiedad intelectual}
      \addlegendimage{dicbarra=fbbiologicos}
      \addlegendentry{Recursos biológicos cultivados}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
