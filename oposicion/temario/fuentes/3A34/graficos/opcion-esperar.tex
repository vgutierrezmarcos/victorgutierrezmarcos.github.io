% El valor de la opción de invertir y el umbral de inversión (Dixit y Pindyck,
% 1994, cap. 5). Gráfico de referencia de la ficha opciones-reales (con O en
% vez de F, que en el tema es la producción).
%   O(V) = A V^beta_1 si V < V*;  O(V) = V - I si V >= V*
%   V* = beta_1/(beta_1 - 1) I,  A = (V* - I)/V*^beta_1 (tangencia: O(V*) = V* - I
%   y O'(V*) = 1)
%   I = 2, beta_1 = 2   -> V*  = 4,     A  = 0,125
%   más volatilidad, beta_1' = 1,6 -> V*' = 5,333, A' = 3,333/5,333^1,6 = 0,2289
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=1.2cm,y=1.05cm,
    declare function={o(\v)=0.125*\v^2; on(\v)=0.22888*\v^1.6;}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-1.2,-1.4) rectangle (8.2,5.6);
  % 1: ejes: valor del proyecto y valor de la opción
  \capa{1}{\ejes{$V$}{$O(V),\ V-I$}{7.4}{5.4}}
  % 2: el VAN de invertir ya, V - I, que se anula en V = I
  \capa{2}{\draw[curva2] (2,0) -- (7,5) node[etiqueta,right,text=tceeazul] {$V-I$};}
  \capa{2}{\node[below,align=center] at (2,0) {$I$\\[-1pt]{\footnotesize regla del VAN:}\\[-3pt]{\footnotesize $V=I$}};}
  % 3: el valor de la opción de invertir, convexo, por encima de V - I
  \capa{3}{\draw[curva1] plot[domain=0:4,samples=60,smooth] (\x,{o(\x)});}
  \capa{3}{\draw[thin,tceemorado] (2.75,{o(2.75)}) -- (2.2,2.0) node[etiqueta,above,text=tceemorado] {$O(V)$};}
  % 4: tangencia en V*: a partir de ahí, invertir (O = V - I)
  \capa{4}{\node[punto] (T) at (4,2) {};}
  \capa{4}{\draw[guia] (T) -- (4,0) node[below,text=black] {$V^*=\dfrac{\beta_1}{\beta_1-1}\,I$};}
  % 5: más volatilidad: la opción vale más y el umbral sube
  \capa{5}{\draw[curvanueva,desplazada] plot[domain=0:5.3333,samples=60,smooth] (\x,{on(\x)});}
  \capa{5}{\draw[thin,tceerojo] (4.6,{on(4.6)}) -- (3.6,3.6) node[etiqueta,above,text=tceerojo] {$O(V)$ con $\sigma'>\sigma$};}
  \capa{5}{\node[punto,fill=tceerojo] (T1) at (5.3333,3.3333) {};}
  \capa{5}{\draw[guia,tceerojo] (T1) -- (5.3333,0) node[below,text=tceerojo] {$V^{*\prime}$};}
\end{tikzpicture}
\end{document}
