% Riqueza neta, coste de los fondos e inversión (jerarquía financiera, Myers y
% Majluf, 1984; prima de financiación externa, Bernanke y Gertler, 1989).
%   oferta de fondos: r = 1,5 hasta RN = 3; desde RN, 1,5 + 0,12 (I - RN)^2
%   D_1: 3,7 - I      -> corta el tramo plano en I_1 = 2,2
%   D_2: 5,6 - 0,5 I  -> corta el tramo creciente en I_2 = 6,016 (2,592)
%   más riqueza neta, RN' = 4,5 -> I_2' = 6,862 (2,169); I_1 no cambia
% RN en redonda: N es el número de empresas del modelo de la q.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.85cm,y=1cm,
    declare function={of(\i)=1.5+0.12*(\i-3)^2; ofn(\i)=1.5+0.12*(\i-4.5)^2;
      dA(\i)=3.7-\i; dB(\i)=5.6-0.5*\i;}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-1.3,-0.9) rectangle (11.6,5.4);
  % 1: ejes: inversión de la empresa y coste de los fondos
  \capa{1}{\ejes{$I$}{coste de los fondos}{9.6}{5.2}}
  % 2: fondos propios: se invierten al tipo r hasta la riqueza neta RN
  \capa{2}{\draw[curva1] (0,1.5) node[left,text=black] {$r$} -- (3,1.5);}
  \capa{2}{\draw[guia] (3,1.5) -- (3,0) node[below,text=black] {$\mathrm{RN}$};}
  % 3: fondos externos: la prima crece con el endeudamiento
  \capa{3}{\draw[curva1] plot[domain=3:8.5,samples=50,smooth] (\x,{of(\x)})
      node[etiqueta,above,text=tceemorado,align=center] {oferta de\\fondos};}
  % 4: una empresa con pocos proyectos: su demanda corta el tramo plano
  \capa{4}{\draw[curva2] (0.4,{dA(0.4)}) -- (3.15,{dA(3.15)}) node[etiqueta,below right=-2pt and 0pt,text=tceeazul] {$D_1$};}
  \capa{4}{\node[punto] (A) at (2.2,1.5) {};}
  \capa{4}{\draw[guia] (A) -- (2.2,0) node[below,text=black] {$I_1$};}
  % 5: una empresa con más proyectos: su demanda corta el tramo creciente
  \capa{5}{\draw[curva3] (1.6,{dB(1.6)}) -- (9.6,{dB(9.6)}) node[etiqueta,right,text=tceeverde] {$D_2$};}
  \capa{5}{\node[punto] (B) at (6.0164,2.5918) {};}
  \capa{5}{\draw[guia] (B) -- (6.0164,0) node[below,text=black] {$I_2$};}
  % 6: más riqueza neta: el escalón se desplaza y sube la inversión de la segunda
  \capa{6}{\draw[curvanueva,desplazada] (3,1.5) -- (4.5,1.5) plot[domain=4.5:9.3,samples=50,smooth] (\x,{ofn(\x)});}
  \capa{6}{\draw[guia,tceerojo] (4.5,1.5) -- (4.5,0) node[below,text=tceerojo] {$\mathrm{RN}'$};}
  \capa{6}{\draw[flecha] (3.15,1.1) -- (4.35,1.1);}
  \capa{6}{\node[punto,fill=tceerojo] (B1) at (6.8615,2.1692) {};}
  \capa{6}{\draw[guia,tceerojo] (B1) -- (6.8615,0) node[below,text=tceerojo] {$I_2'$};}
\end{tikzpicture}
\end{document}
