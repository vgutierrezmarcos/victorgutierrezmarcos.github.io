\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: «si quieres algo bien hecho, hazlo tú mismo» frente a «zapatero, a tus zapatos».
  \begin{itemize}
  \item 1931: \autor{Coase}, 20 años, LSE, viaja a EE.\,UU.: si la planificación central es imposible, ¿por qué hay fábricas enormes, pequeñas economías planificadas dentro del mercado?
  \item ¿Por qué existen las empresas? ¿Y por qué no hay una sola?
  \end{itemize}
\item \textbf{Relevancia}:
  \begin{itemize}
  \item \emph{teórica}: 6 Nobel (\autor{Simon} 1978, \autor{Stigler} 1982, \autor{Coase} 1991, \autor{Williamson} 2009, \autor{Tirole} 2014, \autor{Hart}-\autor{Holmström} 2016);
  \item \emph{empírica}: economía de organizaciones (\autor{Simon}, 1991); márgenes crecientes;
  \item \emph{de política}: competencia y regulación \vertema{3.A.20}\vertema{3.B.39}\vertema{4.A.15}.
  \end{itemize}
\item \textbf{Contextualización}: \concepto{empresa}: unidad que reúne factores bajo una dirección común y vende en el mercado; neoclásicos: \concepto{caja negra}.
  \begin{itemize}
  \item 3 supuestos que se levantan, uno por apartado: maximiza $B$ (I); es una función de producción (II); estructura de mercado dada (III).
  \item Años 30, 3 frentes: objetivos (\autor{Berle}-\autor{Means}, 1932), organización (\autor{Coase}, 1937 $\to$ \autor{Williamson}, 1975), mercado (\autor{Chamberlin}, \autor{Robinson}, 1933); después, información y juegos \vertema{3.A.13}\vertema{3.A.14}.
  \item \concepto{TOI}: mercados con competencia imperfecta, conducta y bienestar; en sentido amplio, también la teoría de la empresa (\autor{Tirole}, 1988).
  \end{itemize}
\item \textbf{Problemática}: ¿por qué hay empresas? ¿Qué fija su tamaño eficiente y qué le impide crecer sin límite? ¿Buscan el beneficio si las dirigen no propietarios? ¿Poder de mercado: número de empresas o facilidad de entrar?
\item \textbf{Estructura}: de dentro hacia fuera: \hyperref[rep:I]{I}, tamaño y límites internos; \hyperref[rep:II]{II}, costes de transacción (mención especial); \hyperref[rep:III]{III}, TOI.
\end{itemize}

\bloque[7]{I. La empresa: el tamaño eficiente y sus límites\label{rep:I}}
\begin{ideaclave}
La tecnología fija un tamaño eficiente \emph{mínimo}, pero no un máximo; los límites al tamaño vienen de dentro: la dirección y los incentivos de quienes gobiernan la empresa.
\end{ideaclave}

\sub{I.1. Enfoque tecnológico: el tamaño eficiente\label{rep:I.1}}
\subsub{a) Idea y supuestos}
\begin{itemize}
\item Empresa = tecnología \vertema{3.A.11} $\to$ $C(\vect{w},q)$ \vertema{3.A.12}: explica \emph{tamaño}, \emph{configuración} de la industria e \emph{integración}.
\item Supuestos: misma tecnología, minimizan costes, $\vect{w}$ dado; demanda $x(p)$; LP con libre entrada y salida.
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item \concepto{Escala mínima eficiente} $\overline{q}$: menor $q$ con CMe LP mínimo, $\overline{c}=C(\overline{q})/\overline{q}$.
  \begin{itemize}
  \item Por debajo, economías de escala ($\mathrm{CMg}<\mathrm{CMe}$).
  \item CMe en \emph{L} (empírico) $\Rightarrow$ \emph{horquilla} de tamaños eficientes (\autor{Bain}, 1956; \autor{Stigler}, 1958).
  \end{itemize}
\item \concepto{Configuración eficiente}: con libre entrada y CMe en U, cada una produce $\overline{q}$ a $p^*=\overline{c}$ y
\begin{formula}
\centering $J^*=x(\overline{c})/\overline{q}$\quad (decide $\overline{q}$ \emph{relativa} a la demanda)
\end{formula}
  \begin{itemize}
  \item $\overline{q}$ pequeña: muchas, competencia posible;
  \item grande: oligopolio natural \vertema{3.A.19};
  \item subaditividad: monopolio natural.
  \end{itemize}
\item \concepto{Subaditividad} (\autor{Baumol}, 1977), \emph{misma} $C(\cdot)$ para todas $=$ definición de \concepto{monopolio natural}:
\begin{formula}
\centering $C\Big(\sum_{j}q_j\Big)<\sum_{j}C(q_j)$ para todo reparto
\end{formula}
  \begin{itemize}
  \item Economías de escala en todo el tramo: \emph{suficientes}, no necesarias \vertema{3.A.17}.
  \end{itemize}
\end{itemize}
\pareja{costes-l}{CMe LP en L: $\overline{q}$ y $\overline{c}$}{subaditividad}{Subaditividad: 1 empresa más barata aunque su CMe crezca}
\begin{itemize}
\item \concepto{Economías de alcance} (\autor{Panzar}-\autor{Willig}, 1981), por factores compartidos o \emph{públicos} (red, marca, conocimiento):
\begin{formula}
\centering $C(q_1,q_2)<C(q_1,0)+C(0,q_2)$
\end{formula}
\item \emph{Implicaciones}: escala $\to$ integración horizontal; alcance $\to$ diversificación; subaditividad $\to$ monopolio natural.
\end{itemize}
\subsub{c) Evidencia empírica}
\begin{itemize}
\item \concepto{Técnica del superviviente} (\autor{Stigler}, 1958): horquilla amplia, coherente con la L.
\item Distribución muy asimétrica: España, \dato{3,31 M de empresas; 54,4\,\% sin asalariados}{DIRCE 2025}.
  \begin{itemize}
  \item \concepto{Ley de \autor{Gibrat}} (1931): crecimiento independiente del tamaño (aprox.\ en las grandes) \vertema{3.B.4}.
  \end{itemize}
\item Concentración creciente: superestrellas (\autor{Autor} et al., 2020) o más barreras (\autor{Philippon}, 2019) $\to$ \hyperref[rep:III]{III}.
\end{itemize}
\subsub{d) Valoración}
\begin{itemize}
\item Explica la \emph{planta}, no la \emph{empresa}; no dice por qué las economías se aprovechan dentro y no con \emph{contratos entre empresas} (\autor{Teece}, 1980) $\to$ \hyperref[rep:II]{II}.
\end{itemize}

\sub{I.2. Objetivo y funcionamiento: los límites internos\label{rep:I.2}}
\subsub{a) Neoclásicos: la dirección como factor fijo}
\begin{itemize}
\item \autor{Kaldor}, \autor{Robinson} (1934): rendimientos decrecientes de la \concepto{capacidad de coordinación} $\Rightarrow$ el CMe acaba subiendo; \autor{Coase} lo usa como límite.
\item \concepto{Efecto \autor{Penrose}} (1959): limita la \emph{velocidad} de crecimiento, no el tamaño \vertema{3.B.4}.
\item \emph{Pérdida de control} (\autor{Williamson}, 1967): cada nivel transmite $\alpha\in(0,1)$; con $N$ niveles llega $\alpha^{N-1}$ $\Rightarrow$ tamaño óptimo finito.
\end{itemize}
\subsub{b) Gerencialistas (años 50-60)}
\begin{itemize}
\item \concepto{Separación propiedad-control} (\autor{Berle}-\autor{Means}, 1932):
  \begin{itemize}
  \item principal (accionista, $\max B$) frente a agente (directivo, $\max U$: retribución, prestigio, tamaño);
  \item \autor{Simon} (1955): \concepto{racionalidad limitada}, resultados \emph{satisfactorios}; \autor{Cyert}-\autor{March} (1963).
  \end{itemize}
\item Objetivo del directivo s.a.\ \concepto{beneficio mínimo} $\underline{B}$ (evita la venta de acciones o el cambio de directivos).
\end{itemize}
\subsubsub{Maximización de las ventas (\autor{Baumol}, 1958-1959)}
\begin{itemize}
\item Problema y condición:
\begin{formula}
\centering $\optim{\max}{q}{I(q)=p(q)\,q}{B(q)=I(q)-C(q)\geq\underline{B}}$\\[3pt]
$\mathrm{IMg}(q^B)=\dfrac{\lambda}{1+\lambda}\,\mathrm{CMg}(q^B)$,\quad $0<\mathrm{IMg}<\mathrm{CMg}$ si $\lambda>0$
\end{formula}
  \begin{itemize}
  \item Si en $\mathrm{IMg}=0$ ya $B>\underline{B}$: restricción no activa.
  \item Si no, $B(q^B)=\underline{B}$ con $q^B>q^M$: más $q$, menor $p$, más publicidad.
  \end{itemize}
\item \emph{Predicción contrastable}: $\uparrow$ coste fijo (o impuesto de suma fija) $\Rightarrow$ $q^B\downarrow$; $q^M$ no cambia.
\end{itemize}
\pareja{baumol}{\autor{Baumol}: $q^B>q^M$, con $B(q^B)=\underline{B}$}{williamson-gerencial}{\autor{Williamson} (1964): más gasto en personal}
\begin{itemize}
\item \emph{Utilidad de los directivos} (\autor{Williamson}, 1964): $U(S,M,I_D)$ s.a.\ $B(S)-M\geq\underline{B}$.
  \begin{itemize}
  \item ($S$ personal, $M$ emolumentos): \emph{gasta de más en personal}; en recesión, recortes mayores.
  \end{itemize}
\item \emph{Crecimiento} (\autor{Marris}, 1963-1964): máx.\ tasa de crecimiento equilibrado s.a.\ \emph{restricción de seguridad} (evitar la absorción) $\Rightarrow$ crece más de lo que maximiza el valor.
\item \emph{Valoración}: mantienen información perfecta; el conflicto solo importa con \emph{información asimétrica}.
  \begin{itemize}
  \item Bisagra: \concepto{ineficiencia X} (\autor{Leibenstein}, 1966): $C^{\mathrm{obs}}=(1+\chi)\,C(\vect{w},q)$, $\chi\geq0$, $\downarrow$ con la competencia.
  \item «El mejor de los beneficios del monopolio es una vida tranquila» (\autor{Hicks}, 1935).
  \end{itemize}
\end{itemize}
\subsub{c) La teoría de la agencia}
\begin{itemize}
\item \concepto{Relación de agencia}: el principal delega en un agente cuyas acciones no observa del todo \vertema{3.A.13}.
\item \concepto{Riesgo moral}: \emph{acción oculta}, \emph{conflicto de intereses} (esfuerzo costoso), \emph{riesgo y aversión} del agente.
\item \emph{Incentivos} frente a \emph{reparto del riesgo} (\autor{Holmström}, 1979): máx.\ beneficio esperado s.a.\ \concepto{participación} (PC) e \concepto{incentivos} (IC).
\item \autor{Jensen}-\autor{Meckling} (1976): \concepto{nexo de contratos}.
  \begin{itemize}
  \item \concepto{Costes de agencia} = supervisión + compromiso (\emph{bonding}) + \emph{pérdida residual}.
  \item Crecen con la separación y con los niveles: otro \emph{límite al tamaño}.
  \end{itemize}
\item \emph{Disciplina del mercado}:
  \begin{itemize}
  \item comparación (filtra el azar común); \emph{mercado de control} (\autor{Manne}, 1965); \emph{de directivos} (\autor{Fama}, 1980); competencia en el producto;
  \item sin ella: \emph{empresa pública} (\autor{Niskanen}, 1971: máx.\ presupuesto) \vertema{3.A.23};
  \item Enron (2001) $\to$ \concepto{gobierno corporativo}: «cumplir o explicar» (Código CNMV 2015/2020).
  \end{itemize}
\end{itemize}

\bloque[9,5]{II. La economía de los costes de transacción\label{rep:II}}
\begin{ideaclave}
La empresa existe porque usar el mercado cuesta; integra las transacciones con activos específicos, donde el oportunismo es más peligroso, y su frontera está en el margen en que organizar dentro y comprar cuestan lo mismo. Integrar también tiene costes: de ahí sus límites.
\end{ideaclave}

\sub{II.1. \autor{Coase} (1937): la frontera en el margen\label{rep:II.1}}
\begin{itemize}
\item 2 formas de coordinar: \emph{mercado} (precios) y \emph{empresa} (\emph{autoridad}): «islas de poder consciente» (\autor{Robertson}); dilema \emph{hacer o comprar}.
\item \concepto{Costes de transacción}: los de usar el mecanismo de precios.
  \begin{itemize}
  \item \emph{ex ante}: descubrir precios, buscar proveedores, negociar y redactar;
  \item \emph{ex post}: vigilar, coordinar, adaptar, resolver conflictos;
  \item $\uparrow$ con la duración y la incertidumbre.
  \end{itemize}
\item Organizar dentro también cuesta: rendimientos decrecientes de la dirección $\Rightarrow$ $\mathrm{CMg}^{\mathrm{org}}$ creciente.
\item Frontera en el \emph{margen}, no en totales; $m$ = transacciones organizadas dentro:
\funcionclave{\mathrm{CMg}^{\mathrm{org}}(m^{*})=\mathrm{CMg}^{\mathrm{tr}}(m^{*})}
\end{itemize}
\graficorepaso[0.5\linewidth]{coase-margen}{\autor{Coase}: tamaño $m^*$ donde organizar y comprar cuestan lo mismo en el margen}
\begin{itemize}
\item \emph{Implicaciones}: empresa mayor si organizar es barato y comprar caro.
  \begin{itemize}
  \item Teléfono, telégrafo $\Rightarrow$ empresas mayores; estandarización, plataformas $\Rightarrow$ subcontratación.
  \end{itemize}
\item \emph{Extensión}: \autor{Coase} (1960): sin costes de transacción, la negociación resuelve las externalidades sea quien sea el titular; con ellos, importa a quién se asignan \vertema{3.A.23}.
\item \emph{Valoración}: no dice \emph{qué} encarece una transacción: casi 40 años «más citado que usado».
\end{itemize}

\sub{II.2. \autor{Williamson}: especificidad y estructuras de gobierno\label{rep:II.2}}
\subsub{a) Idea}
\begin{itemize}
\item (1975, 1985): unidad de análisis, la \emph{transacción}; ¿qué \concepto{estructura de gobierno} la organiza con menos costes? (Carnegie; \emph{no} fue alumno de \autor{Coase}).
\end{itemize}
\subsub{b) Modelo}
\subsubsub{Supuestos de comportamiento}
\begin{itemize}
\item \concepto{Racionalidad limitada} (\autor{Simon}) $\Rightarrow$ contrato complejo \emph{incompleto}; \concepto{oportunismo}: interés propio con engaño.
\end{itemize}
\subsubsub{3 atributos de las transacciones (1979)}
\begin{itemize}
\item \concepto{Especificidad de los activos} $a$: pierden valor fuera de la relación (de localización, física, humana, dedicada, de marca).
\item \emph{Incertidumbre}; \emph{frecuencia} (no encarece: amortiza una estructura especializada).
\end{itemize}
\subsubsub{Cuasirrenta y \emph{hold-up}}
\begin{itemize}
\item \concepto{Cuasirrenta}: valor en la relación $-$ valor en el mejor uso alternativo (\autor{Klein}-\autor{Crawford}-\autor{Alchian}, 1978).
  \begin{itemize}
  \item Amenaza de romper $\Rightarrow$ \concepto{expropiación de la cuasirrenta} (\emph{hold-up}).
  \end{itemize}
\item \concepto{Transformación fundamental}: muchos oferentes \emph{ex ante} $\to$ \concepto{monopolio bilateral} \emph{ex post} \vertema{3.A.17}.
  \begin{itemize}
  \item $\Rightarrow$ infrainversión o salvaguardas.
  \end{itemize}
\end{itemize}
\subsubsub{Estructuras de gobierno (1991)}
\begin{itemize}
\item \emph{Mercado} (incentivos fuertes, protege mal); \emph{jerarquía} (autoridad, protege; incentivos débiles); \emph{híbridas} (contratos LP, franquicias, \emph{joint ventures}).
\item Con $C_{\text{merc}}(0)<C_{\text{híb}}(0)<C_{\text{jer}}(0)$ y $C'_{\text{merc}}>C'_{\text{híb}}>C'_{\text{jer}}>0$:
\funcionclave{C^{*}(a)=\min\big\{C_{\text{merc}}(a),C_{\text{híb}}(a),C_{\text{jer}}(a)\big\}}
  \begin{itemize}
  \item mercado si $a<a_1$; híbrida si $a_1<a<a_2$; jerarquía si $a>a_2$.
  \end{itemize}
\end{itemize}
\graficorepaso[0.52\linewidth]{gobierno-williamson}{Costes de gobierno según la especificidad: umbrales $a_1$ y $a_2$}
\begin{itemize}
\item \emph{Producción y gobierno} (\autor{Riordan}-\autor{Williamson}, 1985): el proveedor agrega demandas (escala) $\Rightarrow$ integrar exige más especificidad, $\hat a>\overline{a}$.
\end{itemize}
\subsub{c) Implicaciones}
\begin{itemize}
\item Poco específico $\to$ mercado (sea cual sea la frecuencia); intermedio $\to$ híbridas; muy específico, frecuente e incierto $\to$ empresa.
\item La integración vertical \emph{no} es indicio de monopolización.
\item Límites: \concepto{imposibilidad de la intervención selectiva} (1985, cap.~6):
  \begin{itemize}
  \item incentivos de \emph{baja potencia}; no puede comprometerse a intervenir solo cuando conviene;
  \item precios internos manipulables, \emph{costes burocráticos}, actividades de influencia.
  \end{itemize}
\end{itemize}
\subsub{d) Evidencia y valoración}
\begin{itemize}
\item Fisher Body-GM (1919-1926, discutido); automóvil (\autor{Monteverde}-\autor{Teece}, 1982); carbón (\autor{Joskow}, 1987).
\item Explica fusiones verticales, franquicias, externalización. Débil: conceptos difíciles de medir, comparación casi verbal; \emph{no dice por qué integrar elimina el oportunismo}.
\end{itemize}

\sub{II.3. Contratos incompletos: \autor{Grossman}-\autor{Hart} (1986), \autor{Hart}-\autor{Moore} (1990)\label{rep:II.3}}
\subsub{a) Idea}
\begin{itemize}
\item Al integrar cambia \emph{quién es dueño de los activos} (empresa = activos físicos de \emph{propiedad común}); la propiedad da poder porque el contrato es incompleto.
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item \emph{Supuestos}: variables observables pero \emph{no verificables} por un juez.
  \begin{itemize}
  \item \concepto{Derechos específicos} (los del contrato) frente a \concepto{derechos residuales de control} (decidir en lo no previsto; \autor{Hart}, 1995).
  \item \emph{Propiedad} = titularidad de los residuales, sobre activos físicos (no personas).
  \end{itemize}
\item \emph{Desarrollo}: el comprador invierte $e$ (coste $e$, no contratable); valor $y(e)$, $y'>0$, $y''<0$; opción exterior $y_0(e)$, $0\leq y_0'<y'$; \autor{Nash} simétrico, $\gamma=\tfrac12$ \vertema{3.A.14}:
\begin{formula}
\centering $\tfrac12\,y'(e^{*})+\tfrac12\,y_0'(e^{*})=1$\quad frente a\quad $y'(e^{o})=1$\\[2pt]
$y_0'<y'\ \Rightarrow\ e^{*}<e^{o}$: \textbf{infrainversión} (\emph{hold-up} formalizado)
\end{formula}
  \begin{itemize}
  \item Integrar sube $y_0'$ del comprador ($e^*\to e^o$), pero \emph{quita} incentivos al vendedor si también invierte.
  \end{itemize}
\end{itemize}
\subsub{c) Implicaciones y valoración}
\begin{itemize}
\item Propiedad a quien tiene la inversión más importante.
  \begin{itemize}
  \item Activos complementarios, mismo dueño; independientes, distintos; la parte imprescindible, propietaria.
  \item Óptimo: integración hacia atrás, hacia delante o no integración; coste de integrar = límite al tamaño.
  \end{itemize}
\item \emph{Extensiones}: prisiones privadas (\autor{Hart}-\autor{Shleifer}-\autor{Vishny}, 1997) \vertema{4.B.15}. \emph{Evidencia}: camiones (\autor{Baker}-\autor{Hubbard}, 2003).
\item Nobel 2016. Críticas: negociación \emph{ex post} eficiente; solo activos físicos (el capital humano no se posee).
\end{itemize}

\sub{II.4. Otros enfoques y política económica\label{rep:II.4}}
\begin{itemize}
\item \emph{Producción en equipo} (\autor{Alchian}-\autor{Demsetz}, 1972): escaqueo $\Rightarrow$ supervisor \concepto{perceptor del residuo}; sin poder de mando distinto de un contrato.
\item \emph{Sistema de incentivos} (\autor{Holmström}-\autor{Milgrom}, 1994); \emph{autoridad formal y real} (\autor{Aghion}-\autor{Tirole}, 1997); \emph{recursos y capacidades} (\autor{Penrose}, 1959; \autor{Wernerfelt}, 1984).
\item \emph{Política}: no condenar de antemano integraciones, franquicias, exclusivas ni concentraciones.
  \begin{itemize}
  \item Caso a caso: eficiencia frente a riesgo de cierre del mercado \vertema{3.B.39}\vertema{4.A.15}.
  \end{itemize}
\end{itemize}

\bloque[10]{III. La TOI: barreras a la entrada y mercados impugnables\label{rep:III}}
\begin{ideaclave}
El poder de mercado depende menos del número de empresas que de lo fácil que sea entrar: las barreras estratégicas solo funcionan si son creíbles, y sin costes hundidos basta la amenaza de entrada para disciplinar incluso a un monopolio.
\end{ideaclave}

\sub{III.1. Del enfoque neoclásico a \autor{Harvard} y \autor{Chicago}\label{rep:III.1}}
\begin{itemize}
\item \emph{Neoclásico}: caja negra precio-aceptante \vertema{3.A.16}; \autor{Chamberlin} y \autor{Robinson} (1933): la TOI se separa de la teoría de los precios \vertema{3.A.18}.
\end{itemize}
\subsub{a) Paradigma ECR: \autor{Harvard}}
\begin{itemize}
\item \concepto{Estructura-conducta-resultado} (\autor{Mason}, 1939; \autor{Bain}, 1951-1959):
  \begin{itemize}
  \item condiciones básicas $\to$ \emph{estructura} (concentración, entrada, diferenciación, integración);
  \item $\to$ \emph{conducta} (precios, colusión, publicidad, I+D, barreras);
  \item $\to$ \emph{resultado} (margen, beneficios, eficiencia, progreso técnico).
  \end{itemize}
\end{itemize}
\graficorepaso[0.56\linewidth]{ecr}{Paradigma ECR y relaciones de vuelta}
\begin{itemize}
\item \emph{Medida}: \concepto{ratio de concentración} e \concepto{índice de \autor{Herfindahl}-\autor{Hirschman}}:
\begin{formula}
\centering $\mathrm{CR}_m=\sum_{i=1}^{m}s_i$,\quad $\mathrm{IHH}=\sum_{i=1}^{n}(100\,s_i)^{2}\in(0;\,10.000]$
\end{formula}
\item \autor{Bain} (1951): más beneficio con más concentración $\Rightarrow$ actuar sobre la estructura (Standard Oil, 1911; AT\&T, 1984).
\item \emph{Críticas}: causalidad no unidireccional, estático, correlación sin teoría, concentración por eficiencia.
\end{itemize}
\subsub{b) La escuela de \autor{Chicago}}
\begin{itemize}
\item \autor{Stigler}, \autor{McGee}, \autor{Demsetz}: concentración por \emph{eficiencia} (\autor{Demsetz}, 1973: las pequeñas no ganan más).
  \begin{itemize}
  \item Cárteles inestables (\autor{Stigler}, 1964); depredación irracional (\autor{McGee}, 1958).
  \item Poder temporal; barreras duraderas casi siempre \emph{legales}.
  \item \emph{Competencia por el mercado} (\autor{Demsetz}, 1968) \vertema{3.A.20}.
  \end{itemize}
\item Política: bienestar del consumidor, tolerancia, desregulación. El debate se resuelve en las \emph{barreras}.
\end{itemize}

\sub{III.2. Las barreras a la entrada\label{rep:III.2}}
\subsub{a) Definición y tipos}
\begin{itemize}
\item \autor{Bain} (1956): lo que permite fijar \emph{a LP} $p>\mathrm{CMe}$ mín.\ sin atraer la entrada.
  \begin{itemize}
  \item 3 fuentes: economías de escala, ventajas absolutas en costes, diferenciación.
  \end{itemize}
\item \autor{Stigler} (1968): coste que soporta el entrante y no soportaron las instaladas (la escala no es barrera con la misma tecnología).
\item \concepto{Estructurales}: escala frente a demanda, \concepto{costes hundidos}, ventajas absolutas, marca y costes de cambio, efectos de red, legales.
\item \concepto{Estratégicas}: precio límite, capacidad, proliferación de marcas, depredación, elevar costes de rivales; deben ser \emph{creíbles} \vertema{3.A.14}.
\item Entrada \concepto{bloqueada} (precio de monopolio y nadie entra), \concepto{disuadida} (cambia su conducta), \concepto{acomodada} (mejor dejar entrar).
\end{itemize}
\subsub{b) El precio límite}
\begin{itemize}
\item \autor{Bain} (1949), \autor{Sylos-Labini} (1957), \autor{Modigliani} (1958).
\item \emph{Postulado de \autor{Sylos-Labini}}: el entrante cree que la instalada mantiene $q_{\mathrm I}$ y entra con al menos $\overline{q}$; no entra si $q_{\mathrm I}+\overline{q}\geq x(\overline{c})$:
\funcionclave{q^{L}=x(\overline{c})-\overline{q},\qquad p^{L}=p(q^{L})>\overline{c}}
  \begin{itemize}
  \item $p^M\leq p^L$: bloqueada.
  \item Margen $\uparrow$ con $\overline{q}/x(\overline{c})$ y con demanda menos elástica; lineal: $p^L=c+2\sqrt{b\,\mathrm{CF}}$.
  \end{itemize}
\end{itemize}
\graficorepaso[0.5\linewidth]{precio-limite}{Precio límite: con $q^L$, al entrante de escala $\overline{q}$ el precio no le cubre $\overline{c}$}
\begin{itemize}
\item \emph{Crítica}: amenaza \emph{no creíble}: tras la entrada conviene acomodar; no es ENPS.
\end{itemize}
\subsub{c) El compromiso de capacidad}
\begin{itemize}
\item \autor{Schelling} (1960, «quemar las naves»), \autor{Spence} (1977), \autor{Dixit} (1980).
\item Pagos (entrante, instalada): no entra $(0,B^M)$; acomoda $(1,B^D)$; lucha $(-1,B^G)$; $B^G<B^D<B^M$ (ej.: 4, 1, $-1$).
  \begin{itemize}
  \item Coste hundido CH \emph{antes}; 3 etapas; inducción hacia atrás.
  \item Disuasión ENPS si
\begin{formula}
\centering $B^{D}-B^{G}<\mathrm{CH}<B^{M}-B^{D}$\quad (ej.: $2<\mathrm{CH}<3$)
\end{formula}
  \end{itemize}
\end{itemize}
\graficorepaso[0.66\linewidth]{disuasion-arbol}{Árbol de disuasión con $\mathrm{CH}=2{,}5$: invierte, lucharía y el entrante no entra}
\begin{itemize}
\item Da credibilidad que la inversión sea \emph{hundida}, observable y previa; \autor{Dixit}: sin capacidad ociosa.
\end{itemize}
\subsub{d) Otras conductas}
\begin{itemize}
\item Precio límite como \emph{señal} (\autor{Milgrom}-\autor{Roberts}, 1982; EBP): en el separador entra el mismo entrante; solo el agrupador disuade más.
\item \emph{Depredación}: cadena de tiendas (\autor{Selten}, 1978); con incertidumbre, \emph{reputación} (\autor{Kreps}-\autor{Wilson}, 1982); \autor{Areeda}-\autor{Turner}: $p<\mathrm{CVMe}$.
\item \emph{Proliferación de marcas} (\autor{Schmalensee}, 1978); \emph{elevar costes de rivales} (\autor{Salop}-\autor{Scheffman}, 1983).
\item Abuso de posición dominante: art.\ 102 TFUE \vertema{3.B.39}.
\end{itemize}

\sub{III.3. Los mercados impugnables\label{rep:III.3}}
\subsub{a) Idea}
\begin{itemize}
\item \autor{Baumol} (1982); \autor{Baumol}-\autor{Panzar}-\autor{Willig} (1982): disciplina la \emph{amenaza de entrada}, no el número; \emph{entrar y salir} (\emph{hit and run}).
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item \concepto{Mercado perfectamente impugnable}:
  \begin{enumerate}
  \item \emph{misma tecnología} y demanda;
  \item entrada y salida libres \emph{sin costes hundidos} (fijos, sí);
  \item las instaladas \emph{tardan en reaccionar} más que los consumidores.
  \end{enumerate}
  \begin{itemize}
  \item \emph{No} exige muchas empresas, rendimientos constantes ni producto único.
  \end{itemize}
\item Configuración \concepto{factible} (producciones suman la demanda, nadie pierde) y \concepto{sostenible} (ningún entrante gana con $p_{\mathrm E}\leq p$ y $q_{\mathrm E}\leq x(p_{\mathrm E})$). Solo son sostenibles:
\funcionclave{p=\mathrm{CMe}(q),\ q=x(p)\ \text{(1 empresa)};\quad p=\mathrm{CMg}=\mathrm{CMe}\ \text{($\geq2$)}}
\item Propiedades:
  \begin{itemize}
  \item beneficio nulo; eficiencia productiva (sin ineficiencia X); nunca $p<\mathrm{CMg}$;
  \item con $\geq2$, cada una en $\overline{q}$;
  \item con 1 (monopolio natural), $p=\mathrm{CMe}>\mathrm{CMg}$: \emph{second best} con presupuesto equilibrado sin regulador \vertema{3.A.17}\vertema{3.A.22};
  \item varios productos: sin subvenciones cruzadas.
  \end{itemize}
\end{itemize}
\graficorepaso[0.5\linewidth]{impugnables}{Monopolio natural impugnable: la amenaza de entrada lleva $p$ al CMe}
\subsub{c) Implicaciones, evidencia y valoración}
\begin{itemize}
\item Mirar barreras y costes hundidos, no el número: \concepto{competencia efectiva} (Ley 15/2007) \vertema{4.A.15}.
  \begin{itemize}
  \item Industrias de red: separar red y servicios, acceso de terceros \vertema{3.A.20}.
  \end{itemize}
\item \emph{Evidencia}: aviación, desregulación de EE.\,UU.\ 1978 (\autor{Bailey}-\autor{Panzar}, 1981); tarifas según competidores reales: impugnable de forma imperfecta (\autor{Morrison}-\autor{Winston}, 1987).
\item \emph{Frágil}: un pequeño coste hundido y reacción rápida $\Rightarrow$ precio de monopolio (\autor{Schwartz}-\autor{Reynolds}, 1983; \autor{Stiglitz}, 1987): discontinuidad.
  \begin{itemize}
  \item Valor: \emph{referencia normativa}.
  \end{itemize}
\end{itemize}

\sub{III.4. La Nueva TOI\label{rep:III.4}}
\begin{itemize}
\item Teoría de juegos (\autor{Tirole}, 1988) \vertema{3.A.14}: pregunta de \autor{Harvard} + rigor de \autor{Chicago}; causalidad en los 2 sentidos.
\item \emph{NOIE} (\autor{Bresnahan}, 1989): mercado a mercado, estima la conducta; parámetro $\theta$ (\autor{Bresnahan}, 1982; giros de la demanda):
\begin{formula}
\centering $p+\theta\,Q\,p'(Q)=\mathrm{CMg}\ \Leftrightarrow\ \mathcal{L}=\theta/|\varepsilon_{Q,p}|$
\end{formula}
  \begin{itemize}
  \item $\theta=0$ competencia (o \autor{Bertrand}), $1/n$ \autor{Cournot}, 1 monopolio o cártel.
  \end{itemize}
\item \autor{Tirole} (Nobel 2014): no hay recetas universales.
  \begin{itemize}
  \item \emph{Cierre de mercados} (\autor{Hart}-\autor{Tirole}, 1990); regulación con información asimétrica (\autor{Laffont}-\autor{Tirole}, 1986, 1993) \vertema{3.A.20}.
  \item \emph{Mercados de 2 lados} (\autor{Rochet}-\autor{Tirole}, 2003): precio nulo en un lado no es depredación.
  \end{itemize}
\end{itemize}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}: la caja negra se abre:
  \begin{itemize}
  \item tecnología $\to$ tamaño mínimo; límites internos (dirección, objetivos, agencia);
  \item empresa porque usar el mercado cuesta; frontera según especificidad y derechos residuales;
  \item poder de mercado según las condiciones de entrada.
  \end{itemize}
\item \textbf{Relevancia}: competencia efectiva y barreras; no condenar integraciones ni franquicias \vertema{3.B.39}\vertema{4.A.15}; separar red y servicios \vertema{3.A.20}.
\item \textbf{Extensiones}: \vertema{3.A.13} \vertema{3.A.14} \vertema{3.A.16}--\vertema{3.A.19}; fusiones \vertema{3.B.4}; cadenas globales de valor \vertema{3.B.18}; plataformas y DMA (2022).
\item \textbf{Opinión}: humildad del regulador ante prácticas que resuelven problemas de contratación; pero márgenes crecientes: vigilar las barreras, sobre todo digitales.
\item \textbf{Idea final}: \autor{Lenin}, una sola fábrica; los profesores de \autor{Coase}, un solo mercado.
  \begin{itemize}
  \item Los 2 cuestan: hay empresas porque los contratos no lo prevén todo; no hay una sola porque la jerarquía también cuesta.
  \item Islas de organización en un mar de mercado, con su tamaño en el margen.
  \end{itemize}
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Diferencia entre \autor{Coase} y los derechos de propiedad?} \autor{Coase}: por qué existe la empresa; GHM: qué cambia al integrar (propiedad e incentivos a invertir).
\item \textbf{¿La saturación de un mercado implica más competencia?} No: con costes hundidos, la proliferación de marcas o de capacidad puede ser barrera.
\item \textbf{¿Por qué el precio límite no es creíble?} Tras la entrada conviene acomodar: no es ENPS; hace falta compromiso hundido o información asimétrica.
\item \textbf{¿Qué precio en un monopolio natural impugnable?} $p=\mathrm{CMe}>\mathrm{CMg}$: segundo óptimo sin regulación.
\item \textbf{¿Qué pasa en \autor{Baumol} si sube el coste fijo?} Con la restricción activa, $q^B$ baja; el maximizador de $B$ no cambia.
\item \textbf{¿Por qué el Nobel de \autor{Hart}?} Contratos incompletos: derechos residuales de control y frontera de la empresa.
\item \textbf{¿Riesgo moral y selección adversa?} Acción oculta tras el contrato; información oculta antes \vertema{3.A.13}.
\end{enumerate}
\begin{trampa}
\begin{itemize}
\item Impugnables: \emph{sin costes hundidos}, no rendimientos constantes (fijos sí puede haber).
\item Monopolio impugnable: $p=\mathrm{CMe}$, no $p=\mathrm{CMg}$.
\item \autor{Grossman}-\autor{Hart}, 1986; \autor{Hart}-\autor{Moore}, 1990. \autor{Williamson} \emph{no} fue alumno de \autor{Coase}.
\item \autor{Baumol} (1959): ventas \emph{sujeto a beneficio mínimo}.
\item GM compró el 60\,\% de Fisher Body en 1919 y el resto en 1926.
\item Frecuencia: no encarece; justifica una estructura especializada.
\item Subaditividad: escala \emph{suficiente}, no necesaria.
\end{itemize}
\end{trampa}
