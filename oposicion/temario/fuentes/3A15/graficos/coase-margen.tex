% El tamaño de la empresa según Coase (1937). Elaboración propia.
% CMg^org(m) = 0,5 + 0,08 m^2 (creciente: rendimientos decrecientes de la dirección).
% CMg^tr(m)  = 3,2 - 0,1 m (casi plano).
% Corte: 0,08 m^2 + 0,1 m - 2,7 = 0 -> m* = 5,218 (CMg = 2,678).
% Teléfono y telégrafo: CMg^org' = 0,5 + 0,05 m^2 -> m*' = 6,416 (CMg = 2,558).
% Estandarización, plataformas: CMg^tr' = 2,4 - 0,1 m (con CMg^org) -> m*'' = 4,288 (CMg = 1,971).
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=0.95cm,y=1cm,
    declare function={org(\m)=0.5+0.08*\m^2; tr(\m)=3.2-0.1*\m;}]
  \useasboundingbox (-1.2,-1.2) rectangle (11.0,5.3);
  % 1: ejes
  \capa{1}{\ejes{$m$}{euros}{8.6}{5}}
  % 2: coste marginal de organizar dentro
  \capa{2}{\draw[curva1] plot[domain=0:7.4,samples=80,smooth] (\x,{org(\x)}) node[etiqueta,right] {$\mathrm{CMg}^{\mathrm{org}}$};}
  % 3: coste marginal de la transacción en el mercado
  \capa{3}{\draw[curva2] (0,{tr(0)}) -- (8,{tr(8)}) node[etiqueta,right] {$\mathrm{CMg}^{\mathrm{tr}}$};}
  % 4: corte y tamaño de la empresa
  \capa{4}{\node[punto] (E) at (5.218,2.678) {};}
  \capa{4}{\draw[guia] (E) -- (5.218,0) node[below,text=black] {$m^{*}$};}
  % 5: hacer y comprar
  \capa{5}{\draw[<-,thick,tceeverde] (2.6,0.3) -- (5.1,0.3) node[midway,above,etiqueta,font=\footnotesize,text=tceeverde] {hacer};}
  \capa{5}{\draw[->,thick,tceerojo] (5.34,0.3) -- (7.8,0.3) node[midway,above,etiqueta,font=\footnotesize,text=tceerojo] {comprar};}
  % 6: abaratar la coordinación: CMg^org más bajo, m* mayor
  \capa{6}{\draw[curvanueva] plot[domain=0:8,samples=80,smooth] (\x,{0.5+0.05*\x^2}) node[etiqueta,right] {$\mathrm{CMg}^{\mathrm{org}\prime}$};}
  \capa{6}{\node[etiqueta,font=\footnotesize,text=tceerojo,anchor=north west] at (8.05,3.45) {teléfono, telégrafo};}
  \capa{6}{\node[punto,fill=tceerojo] (E1) at (6.416,2.558) {};}
  \capa{6}{\draw[guia] (E1) -- (6.416,0) node[below,text=black] {$m^{*\prime}$};}
  % 7: abaratar el mercado: CMg^tr más bajo, m* menor
  \capa{7}{\draw[curva2,desplazada] (0,2.4) -- (8,1.6) node[etiqueta,right] {$\mathrm{CMg}^{\mathrm{tr}\prime}$};}
  \capa{7}{\node[etiqueta,font=\footnotesize,text=tceeazul,anchor=north west,align=left] at (0.15,2.1) {estandarización,\\ plataformas};}
  \capa{7}{\node[punto,fill=tceeazul] (E2) at (4.288,1.971) {};}
  \capa{7}{\draw[guia] (E2) -- (4.288,0) node[below,text=black] {$m^{*\prime\prime}$};}
\end{tikzpicture}
\end{document}
