% Maximización de la utilidad de los directivos. Elaboración propia a partir de
% Williamson (1964).
% Beneficio discrecional: P(S) = B(S) - \underline{B} = 4 - 0,4 (S - 3)^2, máximo en S^M = 3.
% Curvas de indiferencia del directivo: y = U / S^a (convexas y decrecientes).
% Tangencia en S* = 4,2: P = 3,424, P' = -0,96, a = 0,96·4,2/3,424 = 1,17757,
%   U_1 = 3,424·4,2^a = 18,4396 (U_0 = 0,7 U_1 corta; U_2 = 1,3 U_1 no llega).
% Retribución ligada al beneficio: curva más plana, a' = 0,48·3,6/3,856 = 0,44813,
%   tangente en S** = 3,6 (P = 3,856, P' = -0,48), U' = 3,856·3,6^a' = 6,8634.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=1.05cm,y=1cm,
    declare function={ben(\s)=4-0.4*(\s-3)^2; ci(\s,\u)=\u/\s^1.17757; cp(\s)=6.8634/\s^0.44813;}]
  \useasboundingbox (-1.5,-1.5) rectangle (8.6,6.2);
  % 1: ejes
  \capa{1}{\ejes{$S$}{$B-\underline{B}$}{7.6}{6}}
  % 2: curva de beneficio y su máximo
  \capa{2}{\draw[curva1] plot[domain=0:6.16,samples=100,smooth] (\x,{ben(\x)}) node[etiqueta,above right] {$B(S)-\underline{B}$};}
  \capa{2}{\node[punto,fill=tceemorado] (M) at (3,4) {};}
  \capa{2}{\draw[guia] (M) -- (3,0) node[below,text=black] {$S^{M}$};}
  % 3: curvas de indiferencia del directivo
  \capa{3}{\draw[curva2,thick] plot[domain=2.52:7.2,samples=80,smooth] (\x,{ci(\x,12.9077)}) node[etiqueta,right] {$U_0$};}
  \capa{3}{\draw[curva2] plot[domain=3.37:7.2,samples=80,smooth] (\x,{ci(\x,18.4396)}) node[etiqueta,right] {$U_1$};}
  \capa{3}{\draw[curva2,thick] plot[domain=4.2:7.2,samples=80,smooth] (\x,{ci(\x,23.9715)}) node[etiqueta,right] {$U_2$};}
  % 4: tangencia
  \capa{4}{\node[punto] (T) at (4.2,3.424) {};}
  \capa{4}{\draw[guia] (T) -- (4.2,0) node[below,text=black] {$S^{*}$};}
  % 5: gasto de más en personal
  \capa{5}{\draw[decorate,decoration={brace,amplitude=4pt,mirror},thick,tceerojo] (3,-0.62) -- (4.2,-0.62)
    node[midway,below=5pt,etiqueta,font=\footnotesize,text=tceerojo] {gasto de más en personal};}
  % 6: retribución ligada al beneficio: curva más plana y nueva tangencia
  \capa{6}{\draw[curvanueva] plot[domain=1.6:7.2,samples=80,smooth] (\x,{cp(\x)}) node[etiqueta,right] {$U'$};}
  \capa{6}{\node[punto,fill=tceerojo] (T2) at (3.6,3.856) {};}
  \capa{6}{\draw[guia] (T2) -- (3.6,0) node[below,text=black] {$S^{**}$};}
\end{tikzpicture}
\end{document}
