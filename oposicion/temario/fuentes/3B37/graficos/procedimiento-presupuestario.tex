% El procedimiento presupuestario anual (art. 314 TFUE): proyecto de la
% Comisión, posición del Consejo, lectura del Parlamento, Comité de
% Conciliación y votación del texto conjunto, con sus plazos y mayorías; a la
% derecha, las salidas en que el presupuesto queda adoptado y, a la
% izquierda, aquellas en que la Comisión presenta un nuevo proyecto. Al
% margen, las doceavas partes provisionales (art. 315). Fuente: arts. 314 y
% 315 TFUE.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\newcommand{\nocortar}{\hyphenpenalty=10000 \exhyphenpenalty=10000 }
\newcommand{\etapa}[2]{{\bfseries\color{tceemorado}#1}\hfill{\bfseries\color{tceedorado!70!black}#2}\par\vspace{1pt}}
\begin{tikzpicture}[tcee,every node/.append style={execute at begin node=\nocortar},
    paso/.style={draw=tceemorado,thick,fill=tceemorado!8,rounded corners=2pt,
      align=left,text width=6.7cm,inner sep=4pt,font=\footnotesize,anchor=north},
    si/.style={draw=tceeverde,thick,fill=tceeverde!10,rounded corners=2pt,
      align=left,text width=3.45cm,inner sep=4pt,font=\footnotesize,anchor=west},
    no/.style={draw=tceerojo,thick,fill=tceerojo!7,rounded corners=2pt,
      align=left,text width=3.15cm,inner sep=4pt,font=\footnotesize,anchor=east},
    nota/.style={draw=tceegris,thick,dashed,fill=tceegris!6,rounded corners=2pt,
      align=left,text width=3.15cm,inner sep=4pt,font=\footnotesize,anchor=north east},
    sigue/.style={->,thick,tceegris},
    salesi/.style={->,thick,tceeverde},
    saleno/.style={->,thick,tceerojo},
    cond/.style={font=\footnotesize,align=left,anchor=west,text width=3.3cm,text=tceegris!70!black}]
  % 1: proyecto de la Comisión
  \capa{1}{\node[paso] (p1) at (8,0) {\etapa{Proyecto de la Comisión}{antes del 1-IX}
      dentro de los techos del MFP; en la práctica, en junio};}
  % 2: posición del Consejo
  \capa{2}{\node[paso] (p2) at ($(p1.south)+(0,-0.55)$) {\etapa{Posición del Consejo}{antes del 1-X}
      por mayoría cualificada};}
  \capa{2}{\draw[sigue] (p1) -- (p2);}
  % 3: lectura del Parlamento
  \capa{3}{\node[paso] (p3) at ($(p2.south)+(0,-0.55)$) {\etapa{Lectura del Parlamento Europeo}{42 días}
      sobre la posición del Consejo};}
  \capa{3}{\draw[sigue] (p2) -- (p3);}
  \capa{3}{\node[si] (s3) at ($(p3.east)+(0.6,0)$) {\textbf{Adoptado}\\la aprueba o no se pronuncia};}
  \capa{3}{\draw[salesi] (p3) -- (s3);}
  % 4: el Consejo ante las enmiendas
  \capa{4}{\node[paso] (p4) at ($(p3.south)+(0,-0.85)$) {\etapa{El Consejo ante las enmiendas}{10 días}
      puede aprobarlas todas};}
  \capa{4}{\draw[sigue] (p3) -- node[cond] {enmiendas por mayoría de sus miembros} (p4);}
  \capa{4}{\node[si] (s4) at ($(p4.east)+(0.6,0)$) {\textbf{Adoptado}\\aprueba todas las enmiendas};}
  \capa{4}{\draw[salesi] (p4) -- (s4);}
  % 5: Comité de Conciliación
  \capa{5}{\node[paso] (p5) at ($(p4.south)+(0,-0.85)$) {\etapa{Comité de Conciliación}{21 días}
      27 miembros del Consejo y 27 diputados buscan un texto conjunto; la Comisión media};}
  \capa{5}{\draw[sigue] (p4) -- node[cond] {si no, se convoca el Comité} (p5);}
  \capa{5}{\node[no] (n5) at ($(p5.west)+(-0.6,0)$) {\textbf{Nuevo proyecto}\\sin texto conjunto};}
  \capa{5}{\draw[saleno] (p5) -- (n5);}
  % 6: votación del texto conjunto
  \capa{6}{\node[paso] (p6) at ($(p5.south)+(0,-0.85)$) {\etapa{Votación del texto conjunto}{14 días}
      el Parlamento (mayoría de los votos emitidos) y el Consejo (mayoría cualificada)};}
  \capa{6}{\draw[sigue] (p5) -- node[cond] {texto conjunto} (p6);}
  \capa{6}{\node[si,anchor=north west] (s6) at ($(p6.north east)+(0.6,0)$) {\textbf{Adoptado}\\lo aprueban los 2, o uno y el otro no se pronuncia; o lo rechaza solo el Consejo y el PE confirma sus enmiendas (mayoría de miembros y 3/5 de los votos)};}
  \capa{6}{\draw[salesi] (p6.east) -- (p6.east -| s6.west);}
  \capa{6}{\node[no,anchor=north east] (n6) at ($(p6.north west)+(-0.6,0)$) {\textbf{Nuevo proyecto}\\lo rechazan los 2, o solo el Parlamento};}
  \capa{6}{\draw[saleno] (p6.west) -- (p6.west -| n6.east);}
  % 7: sin presupuesto el 1 de enero
  \capa{7}{\node[nota] (dz) at ($(p1.north west)+(-0.6,0)$) {\textbf{Si el 1 de enero no hay presupuesto}: doceavas partes provisionales (art. 315): cada mes, hasta 1/12 de los créditos del año anterior};}
\end{tikzpicture}
