\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/gastos-composicion.csv}\gas

\colorlet{gagr}{tceeverde!75}\colorlet{gcoh}{tceeazul!75}\colorlet{gcomp}{tceedorado!75}

\begin{tikzpicture}[tcee]

      xtick={2000,2005,...,2025}, minor xtick={2000,...,2025}, xticklabel style={/pgf/number format/1000 sep=,font=\footnotesize},
      yticklabel style={font=\footnotesize}, ylabel={miles de millones de euros}, ylabel style={font=\footnotesize},
      axis x line*=bottom, axis y line*=left, ymajorgrids, grid style={tceegris!20},
      legend style={at={(0.02,0.98)},anchor=north west,font=\scriptsize,draw=none,legend columns=2},
      legend cell align=left, area legend, dicapas]
    \capa{1}{\addplot[on layer=dicbarras, dicbarra=gagr] table[x=anio,y=agricultura] {\gas}; \addlegendentry{agricultura y pesca}}
    \capa{2}{\addplot[on layer=dicbarras, dicbarra=gcoh] table[x=anio,y=cohesion] {\gas}; \addlegendentry{cohesión}}

    \capa{5}{\addplot[on layer=dicbarras, dicbarra=gadm] table[x=anio,y=admin] {\gas}; \addlegendentry{administración}}
    \capa{6}{\addplot[on layer=dicbarras, dicbarra=gotr] table[x=anio,y=otros] {\gas}; \addlegendentry{otros}}
    \capa{7}{\addplot[on layer=dicbarras, dicbarra=gngeu] table[x=anio,y=ngeu] {\gas}; \addlegendentry{NextGenerationEU}}
