% Crecimiento sesgado hacia el exportable en un país grande (Bhagwati, 1958):
% con mejora del bienestar (izquierda) y empobrecedor (derecha). Calculado:
%   FPP0: X = 4 cos t, Y = 3,5 sin t; FPP1: X = 5 cos t, Y = 3,5 sin t (sesgo
%     hacia X). Tangencia con pendiente -p: X = a^2 p/d, Y = b^2/d,
%     d = sqrt(a^2 p^2 + b^2) = renta en unidades de Y.
%   CIS Cobb-Douglas U = X^0,2 Y^0,8 (Y = k X^-0,25): C = (0,2 d/p; 0,8 d).
%   Inicial, RRI0 = 1: P0 = (3,010; 2,305), C0 = (1,063; 4,252), X + Y = 5,315,
%     CIS0 k = 4,3175.
%   Crecimiento a precios constantes (paralela discontinua): X + Y = 6,103,
%     tangente a FPP1 en (4,096; 2,007).
%   Izquierda, RRI1 = 0,85: P1 = (3,860; 2,225), C1 = (1,295; 4,405),
%     Y = 5,506 - 0,85 X; CIS1 k = 4,699 (bienestar +7 %).
%   Derecha, RRI1 = 0,55: P1 = (3,089; 2,752), C1 = (1,618; 3,561),
%     Y = 4,451 - 0,55 X; CIS1 k = 4,016 (bienestar -5,6 %).
% Elaboración propia a partir de Bhagwati (1958) y Gandolfo (2014), 307-308.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=1cm,y=1cm,
    declare function={cis(\x,\k)=\k*\x^(-0.25);}]
  % ---------- Panel izquierdo: mejora del bienestar ----------
  % 1: ejes
  \capa{1}{\ejes{$X$}{$Y$}{6.4}{6.4}}
  \capa{1}{\node[etiqueta,anchor=south,font=\normalsize] at (3.2,6.7) {mejora del bienestar};}
  % 2: FPP0, RRI0 tangente en P0, consumo C0 en la CIS0
  \capa{2}{\draw[curva1] plot[domain=0:90,samples=60,smooth] ({4*cos(\x)},{3.5*sin(\x)});}
  \capa{2}{\node[etiqueta,text=tceemorado,below] at (4,0) {$\mathrm{FPP}_0$};}
  \capa{2}{\draw[very thick,black] (0.25,5.065) -- (4.4,0.915) node[etiqueta,below,font=\footnotesize] {$\mathrm{RRI}_0$};}
  \capa{2}{\node[punto,label={[etiqueta]below left:$P_0$}] at (3.010,2.305) {};}
  \capa{2}{\draw[thin,tceeverde] plot[domain=0.42:3.2,samples=60] (\x,{cis(\x,4.3175)}) node[etiqueta,right,text=tceeverde,font=\footnotesize] {$\mathrm{CIS}_0$};}
  \capa{2}{\node[punto,label={[etiqueta]below left:$C_0$}] at (1.063,4.252) {};}
  % 3: FPP1 y la paralela a RRI0 tangente a ella (crecimiento a precios constantes)
  \capa{3}{\draw[curvanueva] plot[domain=0:90,samples=60,smooth] ({5*cos(\x)},{3.5*sin(\x)});}
  \capa{3}{\node[etiqueta,text=tceerojo,below] at (5,0) {$\mathrm{FPP}_1$};}
  \capa{3}{\draw[very thick,black,desplazada] (2.8,3.303) -- (5.6,0.503) node[etiqueta,right,font=\footnotesize] {$\mathrm{d}\overline W_0$};}
  % 4: RRI1 poco menos inclinada, P1, C1 y CIS1 más alta
  \capa{4}{\draw[very thick,tceerojo] (0.6,4.996) -- (5.2,1.086) node[etiqueta,above right,text=tceerojo,font=\footnotesize] {$\mathrm{RRI}_1$};}
  \capa{4}{\node[punto,label={[etiqueta]above right:$P_1$}] at (3.860,2.225) {};}
  \capa{4}{\draw[thin,tceeverde] plot[domain=0.55:3.0,samples=60] (\x,{cis(\x,4.699)}) node[etiqueta,right,text=tceeverde,font=\footnotesize] {$\mathrm{CIS}_1$};}
  \capa{4}{\node[punto,label={[etiqueta]above right:$C_1$}] at (1.295,4.405) {};}
  % ---------- Panel derecho: crecimiento empobrecedor ----------
  \begin{scope}[xshift=8.2cm]
  % 5: ejes
  \capa{5}{\ejes{$X$}{$Y$}{6.4}{6.4}}
  \capa{5}{\node[etiqueta,anchor=south,font=\normalsize] at (3.2,6.7) {crecimiento empobrecedor};}
  % 6: FPP0, RRI0, P0, C0 y CIS0
  \capa{6}{\draw[curva1] plot[domain=0:90,samples=60,smooth] ({4*cos(\x)},{3.5*sin(\x)});}
  \capa{6}{\node[etiqueta,text=tceemorado,below] at (4,0) {$\mathrm{FPP}_0$};}
  \capa{6}{\draw[very thick,black] (0.25,5.065) -- (4.4,0.915) node[etiqueta,below,font=\footnotesize] {$\mathrm{RRI}_0$};}
  \capa{6}{\node[punto,label={[etiqueta]below left:$P_0$}] at (3.010,2.305) {};}
  \capa{6}{\draw[thin,tceeverde] plot[domain=2.6:0.42,samples=60] (\x,{cis(\x,4.3175)}) node[etiqueta,above,text=tceeverde,font=\footnotesize] {$\mathrm{CIS}_0$};}
  \capa{6}{\node[punto,label={[etiqueta]above right:$C_0$}] at (1.063,4.252) {};}
  % 7: FPP1 y la paralela discontinua
  \capa{7}{\draw[curvanueva] plot[domain=0:90,samples=60,smooth] ({5*cos(\x)},{3.5*sin(\x)});}
  \capa{7}{\node[etiqueta,text=tceerojo,below] at (5,0) {$\mathrm{FPP}_1$};}
  \capa{7}{\draw[very thick,black,desplazada] (3.4,2.703) -- (5.6,0.503) node[etiqueta,right,font=\footnotesize] {$\mathrm{d}\overline W_0$};}
  % 8: RRI1 mucho menos inclinada, P1, C1 y CIS1 más baja
  \capa{8}{\draw[very thick,tceerojo] (0.5,4.176) -- (5.6,1.371) node[etiqueta,right,text=tceerojo,font=\footnotesize] {$\mathrm{RRI}_1$};}
  \capa{8}{\node[punto] at (3.089,2.752) {};}
  \capa{8}{\node[etiqueta,anchor=south west] at (3.17,2.78) {$P_1$};}
  \capa{8}{\draw[thin,tceeverde] plot[domain=0.5:2.4,samples=60] (\x,{cis(\x,4.016)}) node[etiqueta,right,text=tceeverde,font=\footnotesize] {$\mathrm{CIS}_1$};}
  \capa{8}{\node[punto] (C1) at (1.618,3.561) {};}
  \capa{8}{\node[etiqueta] (c1) at (0.7,2.6) {$C_1$};}
  \capa{8}{\draw[thin,tceegris] (c1.north east) -- (C1);}
  \end{scope}
\end{tikzpicture}
