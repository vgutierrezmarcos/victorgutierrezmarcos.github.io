% Progreso técnico neutral en X en la caja de los factores, con los precios de
% los bienes dados (misma caja que 3B05/graficos/rybczynski-caja.tex).
% L̄ = 6, K̄ = 5 (1 unidad de L = 1,25 cm); O_X abajo a la izquierda, O_Y arriba
% a la derecha.
% Cobb-Douglas: K_X/L_X = lambda K_Y/L_Y con lambda = 4 en todo punto eficiente,
% así que la curva de contrato es K_X = 20 L_X/(6 + 3 L_X), por encima de la
% diagonal (X intensivo en capital). El progreso neutral en X no la cambia.
% E_0: K_X/L_X = 2 y K_Y/L_Y = 0,5 -> 1,5 L_X = 2 -> (L_X, K_X) = (1,3333; 2,6667).
% Sube r/w y los 2 sectores usan menos capital por trabajador (mismo lambda):
% E_1: K_X/L_X = 1,6 y K_Y/L_Y = 0,4 -> 1,2 L_X = 2,6 -> (2,1667; 3,4667)
%   (sobre la curva de contrato: 20·2,1667/12,5 = 3,4667).
% Y: de (L_Y, K_Y) = (4,6667; 2,3333) a (3,8333; 1,5333): menos de los 2;
% X: más trabajo y más capital. El rayo de X sigue más inclinado que el de Y.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=1.25cm,y=1cm,
    llave/.style={decorate,decoration={brace,amplitude=5pt},thin,tceegris}]
  % Marco fijo para que todas las capas del vídeo tengan el mismo tamaño
  \path (-1.45,-1.4) rectangle (6.95,5.75);
  % 1: la caja L̄ × K̄ con O_X y O_Y y sus llaves
  \capa{1}{
    \draw[thick] (0,0) rectangle (6,5);
    \node[below left,text=tceemorado] at (0,0) {$O_X$};
    \node[above right,text=tceeverde] at (6,5) {$O_Y$};
    \node[etiqueta] at (-0.25,4.7) {$K_X$};
    \node[etiqueta] at (5.75,-0.25) {$L_X$};
    \node[etiqueta] at (5.72,5.22) {$L_Y$};
    \node[etiqueta] at (6.25,4.6) {$K_Y$};
    \draw[llave] (6,-0.65) -- (0,-0.65) node[midway,below=5pt,text=black] {$\bar L$};
    \draw[llave] (6.6,5) -- (6.6,0) node[midway,right=5pt,text=black] {$\bar K$};
  }
  % 2: curva de contrato
  \capa{2}{\draw[curva4] plot[domain=0:6,samples=120] (\x,{20*\x/(6+3*\x)});}
  \capa{2}{\node[etiqueta,text=tceedorado,anchor=north west,align=left] at (4.1,1.6) {curva de\\contrato};}
  \capa{2}{\draw[->,thin,tceedorado] (4.6,1.65) to[bend right=20] (4.95,4.72);}
  % 3: rayos iniciales (X más inclinado: intensivo en capital) y E_0
  \capa{3}{\draw[curva1] (0,0) -- (2.5,5);}
  \capa{3}{\node[etiqueta,text=tceemorado,anchor=south east] at (2.48,5.03) {$K/L$ de $X$};}
  \capa{3}{\draw[curva3] (6,5) -- (0,2);}
  \capa{3}{\node[etiqueta,text=tceeverde,anchor=east] at (-0.03,2.0) {$K/L$ de $Y$};}
  \capa{3}{\node[punto] (E0) at (1.3333,2.6667) {};}
  \capa{3}{\node[etiqueta] at (0.82,2.6) {$E_0$};}
  % 4: sube r/w: los 2 rayos pierden pendiente y su corte E_1 se aleja de O_X
  \capa{4}{\draw[curvanueva] (0,0) -- (3.125,5);}
  \capa{4}{\node[etiqueta,text=tceerojo,anchor=south west] at (3.15,5.03) {$K/L$ de $X$};}
  \capa{4}{\draw[curvanueva] (6,5) -- (0,2.6);}
  \capa{4}{\node[etiqueta,text=tceerojo,anchor=east] at (-0.03,2.6) {$K/L$ de $Y$};}
  \capa{4}{\node[punto] (E1) at (2.1667,3.4667) {};}
  \capa{4}{\node[etiqueta,anchor=south east] at (2.1,3.55) {$E_1$};}
  % 5: de E_0 a E_1: X emplea más trabajo y produce más; Y, menos
  \capa{5}{\draw[flecha,shorten <=3pt,shorten >=3pt] (E0) -- (E1);}
\end{tikzpicture}
\end{document}
