% Diagrama tomahawk del modelo centro-periferia (Krugman, 1991; Fujita,
% Krugman y Venables, 1999). Gráfico de referencia del canon (centro-periferia),
% compartido por el 3.B.6 y el 3.B.10.
% La rama interior (el «hacha») se ha calculado resolviendo el modelo con
% mu = 0,4 y sigma = 5 (omega_A = omega_B para lambda_A en (1/2, 1)) y
% reescalando phi_tau para que los puntos de sostenibilidad y de ruptura
% queden en 0,35 y 0,62: la forma es la del modelo (tangente vertical en el
% punto de ruptura), las posiciones son ilustrativas.
\begin{document}
\begin{tikzpicture}[tcee,x=7cm,y=5cm]
  \def\fs{0.35}% punto de sostenibilidad
  \def\fr{0.62}% punto de ruptura
  % 1: ejes y marcas
  \capa{1}{%
    \draw[eje] (0,0) -- (1.16,0) node[below left=2pt and 0pt] {$\phi_\tau$};
    \draw[eje] (0,0) -- (0,1.14) node[left] {$\lambda_A$};
    \node[below left] at (0,0) {$0$};
    \draw (1,0) -- ++(0,-0.02) node[below] {$1$};
    \draw (0,0.5) -- ++(-0.014,0) node[left] {$\tfrac12$};
    \draw (0,1) -- ++(-0.014,0) node[left] {$1$};
    \draw[guia] (1,0) -- (1,1);
  }
  % 2: equilibrio simétrico estable hasta el punto de ruptura
  \capa{2}{\draw[curva1] (0,0.5) -- (\fr,0.5);}
  % 3: equilibrio simétrico inestable desde el punto de ruptura
  \capa{3}{\draw[curva2,desplazada] (\fr,0.5) -- (1,0.5);}
  % 4: aglomeración en A (lambda_A = 1) y en B (lambda_A = 0): inestable antes
  % del punto de sostenibilidad, estable desde él
  \capa{4}{%
    \draw[curva2,desplazada] (0,1) -- (\fs,1);
    \draw[curva2,desplazada] (0,0) -- (\fs,0);
    \draw[curva1] (\fs,1) -- (1,1);
    \draw[curva1] (\fs,0) -- (1,0);
  }
  % 5: equilibrios interiores inestables (el hacha)
  \capa{5}{%
    \draw[curva2,desplazada] plot[smooth] coordinates {(0.6200,0.5000)
      (0.6200,0.5050) (0.6195,0.5200) (0.6170,0.5500) (0.6081,0.6000)
      (0.5933,0.6500) (0.5730,0.7000) (0.5475,0.7500) (0.5170,0.8000)
      (0.4818,0.8500) (0.4422,0.9000) (0.3983,0.9500) (0.3699,0.9800)
      (0.3500,1.0000)};
    \draw[curva2,desplazada] plot[smooth] coordinates {(0.6200,0.5000)
      (0.6200,0.4950) (0.6195,0.4800) (0.6170,0.4500) (0.6081,0.4000)
      (0.5933,0.3500) (0.5730,0.3000) (0.5475,0.2500) (0.5170,0.2000)
      (0.4818,0.1500) (0.4422,0.1000) (0.3983,0.0500) (0.3699,0.0200)
      (0.3500,0.0000)};
  }
  % 6: puntos de sostenibilidad y de ruptura e histéresis
  \capa{6}{%
    \draw[guia] (\fs,0) node[below=1pt,text=black] {$\phi_\tau^{\mathrm{s}}$} -- (\fs,1);
    \draw[guia] (\fr,0) node[below=1pt,text=black] {$\phi_\tau^{\mathrm{r}}$} -- (\fr,1);
    \draw[thick,decorate,decoration={brace,amplitude=5pt}]
      (\fs,1.04) -- (\fr,1.04) node[etiqueta,midway,above=5pt] {histéresis};
    \node[etiqueta,align=center,font=\footnotesize] at (\fs/2,0.79)
      {dispersión\\(simétrico estable)};
    \node[etiqueta,align=center,font=\footnotesize] at ({(1+\fr)/2},0.79)
      {aglomeración\\(centro-periferia)};
  }
  % leyenda
  \capa{6}{%
    \draw[curva1] (0.70,0.30) -- ++(0.07,0) node[etiqueta,right,black] {estable};
    \draw[curva2,desplazada] (0.70,0.20) -- ++(0.07,0) node[etiqueta,right,black] {inestable};
  }
\end{tikzpicture}
\end{document}
