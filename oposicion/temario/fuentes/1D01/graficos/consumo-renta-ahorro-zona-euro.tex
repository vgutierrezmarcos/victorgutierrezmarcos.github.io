\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
% Zona del euro (EA21), 2012-2025: consumo real y renta bruta disponible real
% de los hogares (% de variación, eje izquierdo) y tasa de ahorro bruto
% (% de la renta disponible, eje derecho). Eurostat (consulta 23/10/2026).
% Paleta «dictamen»: consumo = dicconspriv (el del gráfico de contribuciones),
% renta = dicrenta, ahorro = dicahorro. Capas: sombreado de 2020 → barras →
% cero → línea del ahorro → puntos → rótulos (capas «dic» de tikz-tcee.tex:
% las barras del eje izquierdo quedan debajo de todo lo del derecho).
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/consumo-renta-ahorro-zona-euro.csv}\hogares
\pgfplotsset{
  hogares/.style={
    width=14cm, height=7.5cm, scale only axis=false,
    xmin=2011.4, xmax=2025.6, xtick={2012,...,2025},
    xticklabel style={/pgf/number format/1000 sep=, font=\footnotesize},
    yticklabel style={font=\footnotesize},
    tick align=outside, clip=false,
  },
  leyenda linea corta/.style={legend image code/.code={%
    \draw[##1,mark repeat=2,mark phase=2] plot coordinates {(0cm,0cm) (0.3cm,0cm) (0.6cm,0cm)};}},
}
% Capas «dic» (tikz-tcee.tex): fuera de los axis, porque hay varios en el dibujo.
\pgfplotsset{dicapas}
\begin{tikzpicture}[tcee]
  % Eje izquierdo: tasas de variación real
  \begin{axis}[hogares,
      ybar, bar width=0.28cm,
      ymin=-9, ymax=9, ytick={-8,-6,...,8},
      ylabel={Tasa de variación real (\%)},
      axis x line*=bottom, axis y line*=left,
      ymajorgrids, grid style={tceegris!25},
    ]
    % 1: 2020 resaltado (área sombreada, lo primero)
    \capa{1}{
      \dicsombra[fill=tceedorado, opacity=0.12]{2019.5}{-9}{2020.5}{9}
    }
    % 2: crecimiento del consumo real
    \capa{2}{
      \addplot[on layer=dicbarras, dicbarra=dicconspriv] table[x=anio, y=consumo_real] {\hogares};
    }
    % 3: crecimiento de la renta disponible real
    \capa{3}{
      \addplot[on layer=dicbarras, dicbarra=dicrenta] table[x=anio, y=renta_disponible_real] {\hogares};
    }
    % 4: eje del cero, encima de las barras
    \capa{4}{
      \dicraya{(2011.4,0) (2025.6,0)}
    }
  \end{axis}
  % Eje derecho: tasa de ahorro (se dibuja después: queda encima de las barras)
  \begin{axis}[hogares,
      ymin=10, ymax=21, ytick={10,12,...,20},
      ylabel={Tasa de ahorro (\% de la renta disponible)},
      axis x line=none, axis y line*=right,
      legend style={at={(0.5,-0.12)}, anchor=north, legend columns=3,
        draw=none, font=\footnotesize, /tikz/every even column/.append style={column sep=8pt}},
      legend cell align=left,
    ]
    % 5: tasa de ahorro (línea) y leyenda
    \capa{5}{
      \addlegendimage{area legend, dicbarra=dicconspriv}
      \addlegendentry{Consumo real de los hogares (\%)}
      \addlegendimage{area legend, dicbarra=dicrenta}
      \addlegendentry{Renta disponible real (\%)}
      \addplot[diclineahalo=dicahorro, leyenda linea corta,
        legend image post style={mark=*, mark size=1.8pt, mark options={fill=white}}]
        table[x=anio, y=tasa_ahorro] {\hogares};
      \addlegendentry{Tasa de ahorro (eje derecho)}
    }
    % 6: puntos de la tasa de ahorro
    \capa{6}{
      \addplot[on layer=dicpuntos, forget plot, dicpuntos=dicahorro] table[x=anio, y=tasa_ahorro] {\hogares};
    }
    % 7: rótulos de 2020 (en coordenadas del eje derecho: y = 10 + (y_izq + 9)·11/18)
    \capa{7}{
      \dicrotulo[font=\footnotesize,text=dicconspriv!80!black,anchor=north east,inner sep=1pt]{(2020,10.764)}{$-7{,}7$}
      \dicrotulo[font=\footnotesize,text=dicrenta,anchor=north west,inner sep=1pt]{(2020.05,15.256)}{$-0{,}3$}
      \dicrotulo[font=\footnotesize,text=dicahorro,anchor=south west,inner sep=2pt]{(2020.1,19.31)}{$19{,}3$}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
