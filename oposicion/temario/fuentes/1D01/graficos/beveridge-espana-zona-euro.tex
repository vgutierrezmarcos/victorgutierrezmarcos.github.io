\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
% Curva de Beveridge (medias anuales; 2026 = media 1T-2T): tasa de paro 15-74
% frente a tasa de vacantes. Eurostat, jvs_q_r21 y une_rt_q (consulta 23/10/2026).
% España desde 2010: la serie de vacantes salta entre 4T2009 y 1T2010 (ver CSV).
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/beveridge-espana-zona-euro.csv}\beveridge
% Filas del CSV: 0-17 zona del euro (2009-2026), 18-35 España (2009-2026);
% España se dibuja desde la fila 19 (2010).
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=12cm, height=8cm,
      xmin=5, xmax=28, xtick={6,8,...,26},
      ymin=0, ymax=3.6, ytick={0,0.5,...,3.5},
      xlabel={Tasa de paro, 15-74 años (\%)},
      ylabel={Tasa de vacantes (\%)},
      axis lines*=left, tick align=outside, clip=false,
      ticklabel style={font=\footnotesize},
      /pgf/number format/.cd, use comma, fixed, fixed zerofill=false,
      /pgfplots/.cd,
      grid=major, grid style={tceegris!20},
      legend style={at={(0.5,-0.16)}, anchor=north, legend columns=2,
        draw=none, font=\footnotesize, /tikz/every even column/.append style={column sep=10pt}},
      legend cell align=left, dicapas,
    ]
    % Paleta «dictamen»: zona del euro = dicze (círculos), España = dicespana
    % (cuadrados). Capas «dic» de tikz-tcee.tex: cuadrícula → ejes → líneas →
% puntos → rótulos.
    % 1: zona del euro, 2009-2026
    \capa{1}{
      \addplot[diclinea=dicze, thick, skip coords between index={18}{36},
        legend image post style={mark=*, mark size=1.6pt, mark options={fill=white}}]
        table[x=tasa_paro, y=tasa_vacantes] {\beveridge};
      \addlegendentry{Zona del euro, 2009-2026}
    }
    % 2: España, 2010-2026
    \capa{2}{
      \addplot[diclinea=dicespana, thick, skip coords between index={0}{19},
        legend image post style={mark=square*, mark size=1.5pt, mark options={fill=white}}]
        table[x=tasa_paro, y=tasa_vacantes] {\beveridge};
      \addlegendentry{España, 2010-2026}
    }
    % 3: puntos (un año cada uno)
    \capa{3}{
      \addplot[on layer=dicpuntos, forget plot, dicpuntos=dicze, mark size=1.6pt, skip coords between index={18}{36}]
        table[x=tasa_paro, y=tasa_vacantes] {\beveridge};
      \addplot[on layer=dicpuntos, forget plot, dicpuntos=dicespana, mark=square*, mark size=1.5pt, skip coords between index={0}{19}]
        table[x=tasa_paro, y=tasa_vacantes] {\beveridge};
    }
    % 4: años extremos y 2020 de cada serie
    \capa{4}{
      \dicrotulo[font=\scriptsize, text=dicze, anchor=east]{(9.6,1.07)}{2009}
      \dicrotulo[font=\scriptsize, text=dicze, anchor=east]{(6.15,2.2)}{2026}
      \dicrotulo[font=\scriptsize, text=dicze, anchor=north west]{(7.9,1.88)}{2020}
      \dicrotulo[font=\scriptsize, text=dicespana, anchor=south]{(19.9,1.25)}{2010}
      \dicrotulo[font=\scriptsize, text=dicespana, anchor=east]{(10.05,0.9)}{2026}
      \dicrotulo[font=\scriptsize, text=dicespana, anchor=north]{(15.5,0.57)}{2020}
    }
    % 5: rótulos de 2013 (paro máximo) y 2022 (vacantes máximas en la zona del euro)
    \capa{5}{
      \dicrotulo[font=\footnotesize, text=dicespana, anchor=north, align=center]{(26.1,0.57)}{2013\\paro máximo}
      \dicrotulo[font=\footnotesize, text=dicze, anchor=north, align=center]{(12.2,1.35)}{2013}
      \dicrotulo[font=\footnotesize, text=dicze, anchor=south west, align=left]{(6.7,3.2)}{2022: vacantes máximas (3,2\,\%)}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
