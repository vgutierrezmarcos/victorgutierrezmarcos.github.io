\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
% El ciclo como oscilación del PIB alrededor de su tendencia.
% Tendencia: T(t) = 1 + 0,32 t. PIB: T(t) + 0,55 sen(100°·(t − 0,2)).
% Los giros reales (derivada nula) están en la fase 109,5° (pico) y 250,5°
% (valle), porque la tendencia sube: t = 1,295, 2,705 y 4,895.
\begin{tikzpicture}[tcee, xscale=1.1,
    declare function={tend(\t)=1+0.32*\t; pib(\t)=tend(\t)+0.55*sin(100*(\t-0.2));}]
  % 1: ejes
  \capa{1}{
    \ejes{$t$}{PIB real (log)}{6.6}{4.4}
  }
  % 2: tendencia
  \capa{2}{
    \draw[curva4, desplazada, domain=0.2:6.0] plot (\x,{tend(\x)}) node[etiqueta, right] {tendencia};
  }
  % 3: PIB
  \capa{3}{
    \draw[curva1, domain=0.2:6.0, samples=120, smooth] plot (\x,{pib(\x)}) node[etiqueta, right] {PIB};
  }
  % 4: pico y valle
  \capa{4}{
    \node[punto, label={above:pico}] at (1.295,{pib(1.295)}) {};
    \node[punto, label={below:valle}] at (2.705,{pib(2.705)}) {};
    \node[punto, label={above:pico}] at (4.895,{pib(4.895)}) {};
  }
  % 5: contracción (pico → valle) y expansión (valle → pico)
  \capa{5}{
    \draw[decorate, decoration={brace, amplitude=5pt, mirror}] (1.295,0.85) -- (2.705,0.85)
      node[midway, below=5pt, etiqueta] {contracción};
    \draw[decorate, decoration={brace, amplitude=5pt, mirror}] (2.705,0.85) -- (4.895,0.85)
      node[midway, below=5pt, etiqueta] {expansión};
    \draw[guia] (1.295,0.85) -- (1.295,{pib(1.295)});
    \draw[guia] (2.705,0.85) -- (2.705,{pib(2.705)-0.45}); % hasta el rótulo «valle», sin tacharlo
    \draw[guia] (4.895,0.85) -- (4.895,{pib(4.895)});
  }
  % 6: brecha negativa (PIB por debajo de la tendencia)
  \capa{6}{
    \draw[flecha, <->] (3.1,{pib(3.1)}) -- (3.1,{tend(3.1)});
    \node[etiqueta, anchor=north west, text=tceerojo, font=\footnotesize, align=left] at (3.2,1.52) {brecha\\negativa}; % entre las guías, sin cruzarlas
  }
\end{tikzpicture}
\end{document}
