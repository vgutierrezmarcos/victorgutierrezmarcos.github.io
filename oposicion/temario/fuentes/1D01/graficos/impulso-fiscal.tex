\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
% Impulso fiscal = −Δ saldo primario ajustado por el ciclo (pp del PIB
% potencial; positivo = expansivo), 2019-2027 (2026-2027, proyecciones).
% FMI, Fiscal Monitor, base de datos; cálculo propio (consulta 23/10/2026).
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/impulso-fiscal.csv}\impulso
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=15cm, height=7.5cm,
      ybar=0pt, bar width=0.14,
      xmin=2018.45, xmax=2027.55, xtick={2019,...,2027},
      xticklabel style={/pgf/number format/1000 sep=},
      ymin=-7.5, ymax=9, ytick={-6,-4,...,8},
      ticklabel style={font=\footnotesize},
      ylabel={Puntos del PIB potencial},
      axis lines*=left, tick align=outside, clip=false,
      ymajorgrids, grid style={tceegris!25},
      legend style={at={(0.5,-0.12)}, anchor=north, legend columns=5,
        draw=none, font=\footnotesize, /tikz/every even column/.append style={column sep=8pt}},
      legend cell align=left, area legend, dicapas,
    ]
    % Paleta «dictamen»: zona del euro (azul marino) y España (rojo) en color;
    % el resto, en grises de distinta claridad. Capas «dic»: cuadrícula → sombreado de la previsión
    % → barras → cero → rótulos.
    % 1: marca de las previsiones (área sombreada, lo primero)
    \capa{1}{
      \dicsombra[fill=tceegris, opacity=0.10]{2025.5}{-7.5}{2027.55}{9}
    }
    % 2: barras por país (zona del euro y España, juntas y en color)
    \capa{2}{
      \addplot[on layer=dicbarras, dicbarra=dicze] table[x=anio, y=zona_euro] {\impulso};
      \addlegendentry{Zona del euro}
      \addplot[on layer=dicbarras, dicbarra=dicespana] table[x=anio, y=espana] {\impulso};
      \addlegendentry{España}
      \addplot[on layer=dicbarras, dicbarra=dicotro] table[x=anio, y=eeuu] {\impulso};
      \addlegendentry{EE.\,UU.}
      \addplot[on layer=dicbarras, dicbarra=dicotroclaro] table[x=anio, y=reino_unido] {\impulso};
      \addlegendentry{Reino Unido}
      \addplot[on layer=dicbarras, dicbarra=dicotromasclaro] table[x=anio, y=japon] {\impulso};
      \addlegendentry{Japón}
    }
    % 3: eje del cero, encima de las barras
    \capa{3}{
      \dicraya{(2018.45,0) (2027.55,0)}
    }
    % 4: rótulos (sentido del impulso y previsión)
    \capa{4}{
      \dicrotulo[font=\scriptsize, text=tceegris, anchor=north west]{(2018.5,8.9)}{$+$ expansivo}
      \dicrotulo[font=\scriptsize, text=tceegris, anchor=south west]{(2018.5,-7.4)}{$-$ contractivo}
      \dicrotulo[font=\footnotesize, text=tceegris, anchor=north, inner sep=1pt]{(2026.5,8.9)}{Previsión}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
