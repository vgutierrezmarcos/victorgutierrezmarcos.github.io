\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
% Zona del euro, enero de 2017 a septiembre de 2026: IAPC general y sin energía
% ni alimentos (%) y contribuciones de sus componentes (pp). Eurostat,
% prc_hicp_minr y prc_hicp_ctr (consulta 23/10/2026). Eje x: n.º de mes desde
% enero de 2017 (fila del CSV); los años se rotulan en el centro de cada uno.
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/iapc-zona-euro.csv}\iapc
\pgfplotsset{
  leyenda linea corta/.style={legend image code/.code={%
    \draw[##1,mark repeat=2,mark phase=2] plot coordinates {(0cm,0cm) (0.3cm,0cm) (0.6cm,0cm)};}},
}
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=15cm, height=8cm,
      ybar stacked, stack negative=separate, bar width=0.8,
      xmin=-1, xmax=117,
      % marcas en el cambio de año y el año rotulado entre ellas
      xtick={-0.5,11.5,...,107.5,116.5}, x tick label as interval,
      xticklabels={2017,2018,2019,2020,2021,2022,2023,2024,2025,2026},
      ymin=-1.5, ymax=11.5, ytick={-1,0,...,11},
      ticklabel style={font=\footnotesize},
      ylabel={\% y puntos porcentuales},
      axis lines*=left, tick align=outside, clip=false,
      ymajorgrids, grid style={tceegris!25},
      legend style={at={(0.5,-0.1)}, anchor=north, legend columns=3,
        draw=none, font=\footnotesize, /tikz/every even column/.append style={column sep=8pt}},
      legend cell align=left, area legend,
    ]
    % 1: ejes, cero y años
    \capa{1}{
      \draw[thin,black] (axis cs:-1,0) -- (axis cs:117,0);
    }
    % 2: energía
    \capa{2}{
      \addplot[fill=tceerojo!75, draw=none]
        table[x expr=\coordindex, y=c_energia] {\iapc};
      \addlegendentry{Energía}
    }
    % 3: alimentos, bienes industriales no energéticos y servicios
    \capa{3}{
      \addplot[fill=tceedorado!80, draw=none]
        table[x expr=\coordindex, y=c_alimentos] {\iapc};
      \addlegendentry{Alimentos, alcohol y tabaco}
      \addplot[fill=tceeverde!70, draw=none]
        table[x expr=\coordindex, y=c_bienes_ind_no_energ] {\iapc};
      \addlegendentry{Bienes industriales no energéticos}
      \addplot[fill=tceeazul!60, draw=none]
        table[x expr=\coordindex, y=c_servicios] {\iapc};
      \addlegendentry{Servicios}
    }
    % 4: IAPC general
    \capa{4}{
      \addplot[stack plots=false, sharp plot, leyenda linea corta, very thick, tceemorado]
        table[x expr=\coordindex, y=iapc] {\iapc};
      \addlegendentry{IAPC general (\%)}
      \node[font=\footnotesize, text=tceemorado, anchor=south, inner sep=2pt]
        at (axis cs:69,10.6) {oct.\ 2022: 10,6\,\%};
    }
    % 5: subyacente (sin energía ni alimentos)
    \capa{5}{
      \addplot[stack plots=false, sharp plot, leyenda linea corta, very thick, dashed, black]
        table[x expr=\coordindex, y=iapc_sin_energia_alimentos] {\iapc};
      \addlegendentry{Sin energía ni alimentos (\%)}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
