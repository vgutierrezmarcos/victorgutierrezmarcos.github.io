\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
% Zona del euro (EA21), 2012-2025: crecimiento del PIB real (%) y
% contribuciones de la demanda (pp). Eurostat, nama_10_gdp (consulta 23/10/2026).
% Paleta «dictamen» (tikz-tcee.tex): demanda interna en gama cálida (consumo
% en naranjas, FBCF en verde, existencias en gris con trama) y externa en azul.
% Orden de capas (CRITERIOS §7): barras → cero → línea del PIB → puntos → rótulos.
% Leyenda de la línea del mismo ancho que la de las barras (columnas alineadas)
\pgfplotsset{leyenda linea corta/.style={legend image code/.code={%
  \draw[##1,mark repeat=2,mark phase=2] plot coordinates {(0cm,0cm) (0.3cm,0cm) (0.6cm,0cm)};}}}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/contribuciones-pib-zona-euro.csv}\contrib
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=14cm, height=7.5cm,
      ybar stacked, stack negative=separate, bar width=0.5cm,
      xmin=2011.4, xmax=2025.6,
      xtick={2012,...,2025},
      xticklabel style={/pgf/number format/1000 sep=, font=\footnotesize},
      ymin=-7, ymax=7, ytick={-6,-4,...,6},
      yticklabel style={font=\footnotesize},
      ylabel={\% y puntos porcentuales},
      axis lines*=left, tick align=outside, clip=false,
      ymajorgrids, grid style={tceegris!25},
      legend style={at={(0.5,-0.12)}, anchor=north, legend columns=3,
        draw=none, font=\footnotesize, /tikz/every even column/.append style={column sep=8pt}},
      legend cell align=left, area legend,
    ]
    % 1-2: consumo privado y público (familia del consumo, naranjas)
    % (stack negative=separate: las positivas se apilan hacia arriba y las
    % negativas hacia abajo, cada una en su lado del cero)
    \capa{1}{
      \addplot[dicbarra=dicconspriv] table[x=anio, y=consumo_privado] {\contrib};
      \addlegendentry{Consumo privado}
    }
    \capa{2}{
      \addplot[dicbarra=dicconspub] table[x=anio, y=consumo_publico] {\contrib};
      \addlegendentry{Consumo público}
    }
    % 3: inversión (verde)
    \capa{3}{
      \addplot[dicbarra=dicfbcf] table[x=anio, y=fbcf] {\contrib};
      \addlegendentry{Formación bruta de capital fijo}
    }
    % 4: existencias (gris con trama, aparte)
    \capa{4}{
      \addplot[dicbarratrama=dicexistencias] table[x=anio, y=existencias] {\contrib};
      \addlegendentry{Variación de existencias}
    }
    % 5: demanda externa neta (azul: la gama fría, frente a la cálida de la interna)
    \capa{5}{
      \addplot[dicbarra=dicexterna] table[x=anio, y=demanda_externa_neta] {\contrib};
      \addlegendentry{Demanda externa neta}
    }
    % 6: eje del cero, encima de las barras
    \capa{6}{
      \draw[diccero] (axis cs:2011.4,0) -- (axis cs:2025.6,0);
    }
    % 7: crecimiento del PIB real (línea) sobre las barras
    \capa{7}{
      \addplot[stack plots=false, sharp plot, diclineahalo=dicpib, leyenda linea corta,
        legend image post style={mark=*, mark size=1.8pt, mark options={fill=white, line width=0.9pt}}]
        table[x=anio, y=pib_real] {\contrib};
      \addlegendentry{Crecimiento del PIB real (\%)}
    }
    % 8: puntos del PIB
    \capa{8}{
      \addplot[forget plot, stack plots=false, dicpuntos=dicpib]
        table[x=anio, y=pib_real] {\contrib};
    }
    % 9: rótulos de los extremos (2020 y 2021)
    \capa{9}{
      \node[font=\footnotesize, text=dicpib, anchor=east, inner sep=2pt]
        at (axis cs:2019.72,-6.0) {$-6{,}0$};
      \node[font=\footnotesize, text=dicpib, anchor=west, inner sep=2pt]
        at (axis cs:2021.28,6.5) {$+6{,}5$};
    }
  \end{axis}
\end{tikzpicture}
\end{document}
