\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
% Precio del Brent (media mensual, al contado), enero de 2005 a septiembre de
% 2026, en dólares y en euros por barril. EIA (RBRTE); en euros con el tipo de
% referencia del BCE (consulta 23/10/2026). Eje x: n.º de mes desde enero de
% 2005 (fila del CSV); marcas en enero de cada año. Paleta «dictamen»: dólares
% en el bermellón de la energía, euros en el azul de la zona del euro.
% Capas «dic» (tikz-tcee.tex): cuadrícula → ejes → líneas → rótulos.
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/brent.csv}\brent
\pgfplotsset{
  leyenda linea corta/.style={legend image code/.code={%
    \draw[##1] (0cm,0cm) -- (0.6cm,0cm);}},
}
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=15cm, height=7.5cm,
      xmin=-3, xmax=264,
      xtick={0,24,...,264}, xticklabels={2005,2007,2009,2011,2013,2015,2017,2019,2021,2023,2025,2027},
      minor xtick={12,36,...,252},
      ymin=0, ymax=145, ytick={0,20,...,140},
      ticklabel style={font=\footnotesize},
      ylabel={Por barril},
      axis lines*=left, tick align=outside, clip=false,
      ymajorgrids, grid style={tceegris!25},
      legend style={at={(0.5,-0.12)}, anchor=north, legend columns=2,
        draw=none, font=\footnotesize, /tikz/every even column/.append style={column sep=10pt}},
      legend cell align=left, dicapas,
    ]
    % 1: ejes (los dibuja el entorno axis)
    % 2: en dólares
    \capa{2}{
      \addplot[diclinea=dicenergia, leyenda linea corta] table[x expr=\coordindex, y=usd_barril] {\brent};
      \addlegendentry{Dólares por barril}
    }
    % 3: en euros
    \capa{3}{
      \addplot[diclinea=dicze, leyenda linea corta] table[x expr=\coordindex, y=eur_barril] {\brent};
      \addlegendentry{Euros por barril}
    }
    % 4: rótulos de los picos (jul. 2008, jun. 2022, abr. y sep. 2026)
    \capa{4}{
      \dicrotulo[font=\footnotesize, anchor=south, align=center]{(42,133)}{jul.\ 2008\\\textcolor{dicenergia}{133\,\$}}
      \dicrotulo[font=\footnotesize, anchor=south, align=center]{(209,123)}{jun.\ 2022\\\textcolor{dicenergia}{123\,\$}}
      \addplot[on layer=axis descriptions, dicsuelto, guia, solid] coordinates {(210,116) (230,110)};
      \dicrotulo[font=\footnotesize, anchor=west, inner sep=1pt, text=dicze]{(230,110)}{116\,€}
      \dicrotulo[font=\footnotesize, anchor=south, align=center]{(253,118)}{abr.\ y sep.\ 2026\\\textcolor{dicenergia}{117\,\$ y 114\,\$}}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
