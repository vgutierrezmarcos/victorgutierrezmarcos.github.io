\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
% España, 1T2019-2T2026: PIB por persona en edad de trabajar (15-64) y sus dos
% componentes, PIB por hora y horas por persona en edad de trabajar; variación
% acumulada desde 4T2019 (%, aproximación logarítmica: los componentes suman el
% total). Eurostat (namq_10_gdp, namq_10_a10_e, lfsq_pganws); cálculo propio
% (consulta 23/10/2026). Eje x: n.º de trimestre desde 1T2019 (fila del CSV).
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/pib-por-pet-espana.csv}\pet
\pgfplotsset{
  leyenda linea corta/.style={legend image code/.code={%
    \draw[##1,mark repeat=2,mark phase=2] plot coordinates {(0cm,0cm) (0.3cm,0cm) (0.6cm,0cm)};}},
}
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=14cm, height=8cm,
      ybar stacked, stack negative=separate, bar width=0.7,
      xmin=-0.7, xmax=29.7,
      % marcas en el cambio de año y el año rotulado entre ellas
      xtick={-0.5,3.5,...,27.5,29.5}, x tick label as interval,
      xticklabels={2019,2020,2021,2022,2023,2024,2025,2026},
      ymin=-34, ymax=12, ytick={-30,-25,...,10},
      ticklabel style={font=\footnotesize},
      ylabel={Variación acumulada desde 4T2019 (\%)},
      axis lines*=left, tick align=outside, clip=false,
      ymajorgrids, grid style={tceegris!25},
      legend style={at={(0.5,-0.1)}, anchor=north, legend columns=3,
        draw=none, font=\footnotesize, /tikz/every even column/.append style={column sep=8pt}},
      legend cell align=left, area legend, dicapas,
    ]
    % Paleta «dictamen»: productividad = dicproductividad, horas = dichoras,
    % PIB por persona = dicpib. Capas «dic»: cuadrícula → barras → cero → línea → puntos → rótulo.
    % 1: PIB por hora trabajada
    \capa{1}{
      \addplot[on layer=dicbarras, dicbarra=dicproductividad]
        table[x expr=\coordindex, y=pib_por_hora] {\pet};
      \addlegendentry{PIB por hora trabajada}
    }
    % 2: horas por persona en edad de trabajar
    \capa{2}{
      \addplot[on layer=dicbarras, dicbarra=dichoras]
        table[x expr=\coordindex, y=horas_por_pet] {\pet};
      \addlegendentry{Horas por persona en edad de trabajar}
    }
    % 3: eje del cero, encima de las barras
    \capa{3}{
      \dicraya{(-0.7,0) (29.7,0)}
    }
    % 4: PIB por persona en edad de trabajar (la suma)
    \capa{4}{
      \addplot[stack plots=false, sharp plot, leyenda linea corta, diclineahalo=dicpib,
        legend image post style={mark=*, mark size=1.5pt, mark options={fill=white}}]
        table[x expr=\coordindex, y=pib_por_pet] {\pet};
      \addlegendentry{PIB por persona en edad de trabajar}
    }
    % 5: puntos
    \capa{5}{
      \addplot[on layer=dicpuntos, forget plot, stack plots=false, dicpuntos=dicpib, mark size=1.5pt]
        table[x expr=\coordindex, y=pib_por_pet] {\pet};
    }
    % 6: rótulo del último dato
    \capa{6}{
      \dicrotulo[font=\footnotesize, text=dicpib, anchor=south east, inner sep=2pt]{(29.3,6.2)}{2T2026: $+5{,}5\,\%$ (por hora, $+1{,}7$; horas, $+3{,}9$)}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
