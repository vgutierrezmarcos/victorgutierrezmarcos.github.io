\bloque{I. Perfil cíclico de la economía\label{rep:I}}
\begin{ideaclave}
\textbf{Qué hay que contestar}: ¿en qué fase está la economía y desde cuándo? ¿Qué tira del crecimiento, la demanda interna o la externa; el empleo o la productividad? ¿Crece por encima o por debajo de su potencial?
\end{ideaclave}
\graficorepaso[0.97\linewidth]{esquema-perfil-ciclico}{Los 3 pasos de la respuesta, siempre los mismos: fases, motor (demanda y oferta) y diagnóstico anticipado.}
\begin{itemize}[itemsep=1.5pt]
\item Empieza con una \textbf{frase ejecutiva}, como el FMI o el Banco de España: dónde está la economía, qué la ha sostenido, qué la frena y qué riesgos tiene. Se escribe \textbf{al final}, en un hueco que se deja antes de la primera respuesta.
\item Los \textbf{gráficos} y las filas poco habituales del cuadernillo \textbf{no suelen ser gratuitos} (días de huelga, uso de la capacidad, precios de activos o materias primas): si están, es para usarlos.
\end{itemize}

\sub{I.1-I.3. El PIB, las fases y sus medidores\label{rep:I.1}}
\begin{itemize}[itemsep=1.5pt]
\item El \textbf{PIB} es el valor de mercado de los bienes y servicios nuevos y finales producidos en un territorio; se descompone por el \textbf{gasto}, la \textbf{producción} (suma de los VAB) y la \textbf{renta}.
\funcionclave{\mathrm{PIB}=C+I+G+(X-M),\qquad \mathrm{RNB}=\mathrm{PIB}+\mathrm{RP}}
  \begin{itemize}
  \item $G$ es solo el consumo público: la inversión pública va en $I$.
  \item \textbf{PIB frente a RNB}: $\mathrm{RNB}<\mathrm{PIB}$ en un deudor neto o con mucha IED (beneficios repatriados); $\mathrm{RNB}>\mathrm{PIB}$ en un acreedor neto o con muchos trabajadores fuera.
  \end{itemize}
\item \textbf{Crecimiento del PIB real}: el signo y el nivel dicen si crece y cuánto; su variación, si se acelera. \emph{Orientativo} (avanzadas): $<0$ contracción; 0-1~\% estancamiento; 1-2~\% moderado; 2-3~\% por encima del potencial; $\geq3$~\% sólido. En los emergentes, un 3~\% puede ser débil.
\item \textbf{Brecha de producción}: distancia del PIB al \textbf{potencial}, el que se produce con los factores a su intensidad normal y sin tensiones inflacionistas. Referencia: 0.
\funcionclave{\mathrm{OG}_t\equiv\frac{Y_t-Y^p_t}{Y^p_t}\times100,\qquad \gamma_{Y^p,t}\approx\gamma_{Y,t}-\Delta\mathrm{OG}_t}
  \begin{itemize}
  \item \textbf{Negativa}: recursos ociosos, inflación a la baja. \textbf{Positiva}: cuellos de botella, riesgo de sobrecalentamiento.
  \item Crecer con fuerza sin cerrar la brecha no es un cuadro débil: el potencial crece mucho. Cerrarla creciendo poco no es buena señal.
  \item El potencial se estima (función de producción, filtro de \autor{Hodrick}-\autor{Prescott}) y se revisa mucho.
  \end{itemize}
\end{itemize}
\graficorepaso[0.97\linewidth]{fases-ciclo-brecha}{\textbf{Recuperación} (crece más que el potencial con brecha aún negativa), \textbf{expansión} (brecha positiva que se abre), \textbf{desaceleración} (crece menos que el potencial) y \textbf{recesión} (el PIB cae).}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Pasar de caer un 7~\% a caer un 1~\% no es una recuperación}: mientras el PIB cae, la economía sigue en contracción. \textbf{Recesión} (regla práctica): 2 trimestres seguidos de caída; formas posibles: V, U, W o L.
\item \textbf{Cómo se escribe}: visión de conjunto y lo atípico como excepción, sin contar la serie año a año; si hay más de un ciclo, se dice al principio.
\item \textbf{Alta frecuencia}: dicen cómo está la economía \emph{ahora}.
  \begin{itemize}
  \item \textbf{PMI} $>50$: la actividad \emph{crece} frente al mes anterior (no que esté alta). \textbf{ESI}: media de largo plazo 100.
  \item \textbf{Curva de tipos} (largo menos corto): \textbf{creciente}, expansión esperada (o solo prima de liquidez); \textbf{plana}, se esperan bajadas que compensan la prima; \textbf{invertida}, suele anticipar una recesión.
  \item Coincidentes: producción industrial, afiliación, ventas. Retrasados: paro, subyacente, tipos de los préstamos.
  \end{itemize}
\end{itemize}
\graficorepaso[0.84\linewidth]{comovimientos}{Procíclicas, contracíclicas y adelantadas: los adelantados giran primero; los coincidentes, 6-12 meses después; los retrasados, 12-24 meses después.}

\sub{I.4. Demanda: el motor\label{rep:I.4}}
\begin{itemize}[itemsep=1.5pt]
\item Cada \textbf{contribución} es el peso del componente en $t-1$ por su crecimiento:
\funcionclave{\begin{gathered}\gamma_{Y,t}=\underbrace{\tfrac{C_{t-1}}{Y_{t-1}}\gamma_{C,t}+\tfrac{I_{t-1}}{Y_{t-1}}\gamma_{I,t}+\tfrac{G_{t-1}}{Y_{t-1}}\gamma_{G,t}}_{\text{demanda interna}}\\+\underbrace{\tfrac{X_{t-1}}{Y_{t-1}}\gamma_{X,t}-\tfrac{M_{t-1}}{Y_{t-1}}\gamma_{M,t}}_{\text{demanda externa}}\end{gathered}}
  \begin{itemize}
  \item Se miden en \textbf{puntos porcentuales}: si el PIB crece un 3~\% y la interna aporta 2~pp, no ha crecido un 2~\%.
  \item En las crisis la interna resta (\textbf{procíclica}) y la externa suma; si la externa «toma el relevo», ¿suben las exportaciones (fortaleza) o solo caen las importaciones (caída de la renta)?
  \item \textbf{Contribución real frente a saldo nominal}: la contribución externa va en \emph{volumen} y el saldo comercial en \emph{valor}; con el petróleo más caro, la primera puede mejorar mientras el segundo empeora (\textbf{efecto precio}). Si el cuadro trae las 2, se dice.
  \end{itemize}
\item \textbf{Perspicaz, no descriptivo}: no contar el signo año a año, sino «la demanda interna tira en la bonanza y se desploma en las recesiones, cuando la externa amortigua la caída».
\end{itemize}
\graficorepaso[0.88\linewidth]{contribuciones-pib-zona-euro}{Contribuciones al PIB de la zona del euro: la demanda interna explica casi todo el crecimiento salvo en las crisis; en 2020 se hunde el consumo privado y en 2023 restan las existencias.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Consumo privado}: el mayor componente (53~\% del PIB en la UE), el menos volátil de los privados (los hogares lo suavizan según su \textbf{renta permanente}) y procíclico.
\funcionclave{C=f\big(\overset{+}{Y^d_{\text{real}}},\overset{+}{\text{riqueza}},\overset{-}{r^{cp}},\overset{-}{\pi},\overset{+}{\text{crédito}},\overset{-}{\text{deuda}},\overset{-}{\text{incert.}}\big)}
  \begin{itemize}
  \item Manda la \textbf{renta disponible real}: empleo, salario real ($W/\mathrm{IPC}$), rentas del capital, impuestos y transferencias. La renta de \textbf{más empleo} se consume más que la de \textbf{salarios más altos}.
  \item \textbf{Inflación esperada}, ambigua (resta renta real, pero adelanta compras); el \textbf{petróleo} pesa mucho en la renta real (demanda inelástica).
  \item \textbf{Equivalencia ricardiana}: si se esperan más impuestos para pagar la deuda de hoy, se ahorra más.
  \end{itemize}
\end{itemize}
\graficorepaso[0.84\linewidth]{consumo-renta-ahorro-zona-euro}{Hogares de la zona del euro: en 2020 la renta apenas cae y el consumo se desploma; el ahorro sube al 19~\%. El \textbf{forzoso} se gasta al reabrir; el \textbf{precautorio}, despacio: por eso la demanda embalsada sale más lenta de lo esperado.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Inversión} (FBCF, existencias, objetos valiosos): es \textbf{demanda hoy y oferta mañana} y el componente más volátil (\emph{animal spirits}).
\funcionclave{I=f\big(\overset{-}{r^{lp}},\overset{+}{\gamma^e_Y},\overset{+}{\mathrm{OG}},\overset{+}{\mathrm{EBE}},\overset{+}{\text{crédito}},\overset{-}{\text{incert.}}\big)}
  \begin{itemize}
  \item Con brecha positiva se amplía capacidad (\textbf{acelerador}); las \textbf{existencias} son muy procíclicas (no se reponen en las recesiones).
  \item \textbf{Tipo real a largo}: el cuadro casi nunca da las expectativas; por orden, deflactar el nominal a largo con el deflactor, usar el real a corto o suponer una inflación. Decir siempre qué se ha hecho.
  \item \textbf{Expulsión} (\emph{crowding-out}): \textbf{real}, si la inversión pública se superpone a la privada; \textbf{financiera}, si el Estado acapara la financiación.
  \end{itemize}
\item \textbf{Cómo se escribe}: los 2 o 3 determinantes que el cuadro deja ver, ligados a sus años; los que faltan, con condicionales.
\item \textbf{Demanda externa}: 1 o 2 frases (su contribución y si la explican las exportaciones o las importaciones); los determinantes, en \hyperref[rep:IV]{IV}. \textbf{Penetración de importaciones}, $M/\mathrm{PIB}$: un salto brusco apunta a sobrecalentamiento.
\end{itemize}

\sub{I.5-I.6. Oferta y diagnóstico anticipado\label{rep:I.5}}
\begin{itemize}[itemsep=1.5pt]
\item A corto plazo manda la demanda; a largo, la oferta. Con la \textbf{productividad aparente del trabajo}, $\mathrm{PAT}=Y/L$:
\funcionclave{\gamma_Y=\gamma_A+\alpha\,\gamma_K+(1-\alpha)\,\gamma_L\quad\Rightarrow\quad\gamma_Y=\gamma_{\mathrm{PAT}}+\gamma_L}
  \begin{itemize}
  \item Empleo con PAT estancada: patrón \textbf{extensivo}; PAT al alza: \textbf{intensivo}, más sostenible (a largo plazo solo la productividad eleva la renta por habitante).
  \item Si el empleo cae más que el PIB, la PAT sube de forma \textbf{espuria}.
  \end{itemize}
\item \textbf{Sectores}: ¿cambia la estructura productiva?, ¿tira un sector del resto?
  \begin{itemize}
  \item \textbf{primario}: acíclico, baja elasticidad-renta; \textbf{energía}: precios fijados fuera, riesgo de \textbf{enfermedad holandesa}; \textbf{construcción}: grandes eslabonamientos, efecto riqueza y simbiosis con el crédito;
  \item \textbf{sector dominante} (petróleo, turismo): si pesa mucho en el VAB, las exportaciones o los ingresos públicos, su precio mueve todo el cuadro;
  \item un tejido muy \textbf{atomizado} limita la innovación, la financiación y la productividad.
  \end{itemize}
\item \textbf{Diagnóstico anticipado}: «parece una economía avanzada que crece con la demanda interna, pero con un déficit exterior persistente y una deuda pública elevada». Sin adivinar el país; lo estructural, con condicionales.
\end{itemize}

\bloque{II. Mercados de factores\label{rep:II}}
\begin{ideaclave}
\textbf{Qué hay que contestar}: ¿cuánto empleo crea el crecimiento? ¿Qué tipo de paro hay? ¿Los salarios siguen a la productividad? ¿Hay financiación para crecer?
\end{ideaclave}

\sub{II.1. Mercado de trabajo\label{rep:II.1}}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Cantidades}, con $L$ ocupados, $U$ parados, $A=L+U$ activos y $P_{16}$ la población de 16 y más años:
\funcionclave{\text{tasa de actividad}=\frac{A}{P_{16}},\qquad u=\frac{U}{A}}
  \begin{itemize}
  \item \textbf{Ojo con la edad}: el INE usa 16 y más años; Eurostat y la OCDE, 15-64 (España: 74,9~\% y 58~\%).
  \item \emph{Oficial} (PDM): paro $>10$~\%; caída de la tasa de actividad de más de 0,2~pp en 3 años. \emph{Orientativo}: pleno empleo con un paro del 3-6~\%.
  \item Empleo y paro pueden subir a la vez si llegan activos (inmigración). Si el paro baja porque cae la actividad, no es buena noticia: \textbf{efecto desánimo}.
  \item \textbf{Acaparamiento de trabajo}: en la recesión se retiene plantilla y se recortan horas; el empleo va retrasado y la PAT es \textbf{procíclica}. Un PIB que repunta sin empleo no es un «milagro» de productividad.
  \end{itemize}
\item \textbf{Productividad}: crece por \textbf{intensificación del capital}, capital intangible o \textbf{progreso técnico}; a largo plazo eleva la renta per cápita y los salarios reales.
\item \textbf{Costes laborales unitarios}: $\mathrm{CLU}=WL/Y=W/\mathrm{PAT}$, la medida de competitividad en costes más usada.
\end{itemize}
\graficorepaso[0.97\linewidth]{esquema-clu-margen}{De la productividad al margen ($P$, deflactor del PIB; $P=(1+\mu)\,\mathrm{CLU}$): los márgenes suben exactamente cuando caen los CLU reales.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Cómo se leen}: CLU nominales que crecen \textbf{más que la inflación}, presiones de costes y menos margen; \textbf{más que en los socios}, pérdida de competitividad-precio; poco con paro alto, \textbf{devaluación interna}.
\item \textbf{Umbrales}: \emph{oficial} (PDM), CLU nominales $>+9$~\% en 3 años en el euro (12~\% fuera); \emph{orientativo}, salarios que crecen como la productividad más el objetivo de inflación.
\item \textbf{Demanda de trabajo}: depende negativamente de los CLU \emph{reales}; \textbf{oferta}: del salario real, la demografía y la inmigración. La indexación a la inflación pasada y la negociación sectorial (\autor{Calmfors}-\autor{Driffill}) alimentan la segunda ronda.
\item \textbf{NAIRU}: el paro compatible con una inflación estable; se aproxima con el paro de los años de brecha nula y vale para toda la serie (NAWRU de la Comisión: 6-7~\% en el euro, 12~\% en España).
\item \textbf{Histéresis}: el paro pasado eleva el de equilibrio (\emph{insiders} y \emph{outsiders}, larga duración); la demanda tiene entonces efectos a largo plazo.
\end{itemize}
\graficorepaso[0.97\linewidth]{esquema-tipos-paro}{\textbf{Tipos de paro}: la receta depende del diagnóstico. Ojo con el \textbf{efecto composición}: en una recesión se pierden los empleos de salario bajo y el salario medio sube sin que suba ninguno; no prueba rigidez.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{\autor{Okun}}: medio punto menos de paro por punto de crecimiento sobre el potencial (EE.~UU.). \textbf{\autor{Phillips}} con expectativas:
\funcionclave{\pi_t=\pi^e_t-a\,(U_t-U^n)+\varepsilon_t}
\item \textbf{Curva de \autor{Beveridge}}: moverse \emph{a lo largo} es el ciclo; \emph{desplazarse hacia fuera}, peor emparejamiento. Vacantes al alza sin bajar el paro: problema estructural.
\end{itemize}
\graficorepaso[0.78\linewidth]{beveridge-espana-zona-euro}{\autor{Beveridge}: la zona del euro se mueve con el ciclo (vacantes máximas, 3,2~\%, en 2022); España, con pocas vacantes (0,9~\%) y mucho paro, está lejos del origen.}

\sub{II.2. Mercado de capitales\label{rep:II.2}}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Qué mirar}: el crédito nuevo, su coste y el diferencial entre préstamos y depósitos (más estrecho, mejor intermediación). Un \textbf{racionamiento} (crédito que cae con demanda y tipos estables) frena la inversión.
\item Un sistema que funciona mal financia \textbf{empresas zombis} y deja sin crédito proyectos rentables.
\item \textbf{Fuentes}: \textbf{crédito bancario} (pymes y hogares), \textbf{valores} (grandes empresas), \textbf{capital riesgo} (emergentes) y banca pública. Muy bancarizada, depende de la salud de sus bancos (\hyperref[rep:V.2]{V.2}).
\end{itemize}

\bloque{III. Competitividad\label{rep:III}}
\begin{ideaclave}
\textbf{Qué hay que contestar}: ¿está controlada la inflación? ¿Viene de la demanda o de los costes, de dentro o de fuera? ¿Es transitoria o persistente? ¿Qué política requiere?
\end{ideaclave}
\begin{itemize}[itemsep=1.5pt]
\item 2 caras: \textbf{precio} (inflación, costes, tipo de cambio) y \textbf{no precio} (calidad, diferenciación, mercados).
\end{itemize}

\sub{III.1. Precios e inflación\label{rep:III.1}}
\begin{itemize}[itemsep=1.5pt]
\item La \textbf{inflación} es el aumento sostenido y generalizado del nivel de precios, no el cambio de un precio relativo: $\pi_t=(\mathrm{IPC}_t/\mathrm{IPC}_{t-1}-1)\times100$.
\end{itemize}
\graficorepaso[0.97\linewidth]{esquema-ipc-deflactor}{IPC (\autor{Laspeyres}, cesta fija, con importaciones) y deflactor (\autor{Paasche}, solo lo producido dentro). \uline{El salario real del trabajador se deflacta con el IPC; el coste real para la empresa, con el deflactor.}}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Subyacente}: sin los componentes volátiles (INE: alimentos no elaborados y energía; BCE: energía y todos los alimentos); \textbf{hay que decir cuál se usa}. Dice si el choque se extiende al resto de la cesta.
\item \textbf{Efecto base}: la tasa interanual baja cuando sale de la comparación una subida excepcional, aunque los precios no bajen:
\funcionclave{\pi^{12}_t-\pi^{12}_{t-1}\approx\pi^{m}_t-\pi^{m}_{t-12}}
\end{itemize}
\graficorepaso[0.88\linewidth]{iapc-zona-euro}{IAPC de la zona del euro: el brote de 2021-2022 empieza en la energía y pasa a alimentos y bienes; la subyacente baja más despacio, por los servicios.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{¿Choque de oferta o de demanda?} \uline{Precios y cantidades en el mismo sentido, demanda; en sentido contrario, oferta.} Con el cuadro anual, solo a grandes rasgos (inflación frente a PIB) y con condicionales; un choque puede volverse otro (pandemia: de oferta a demanda).
\end{itemize}
\graficorepaso[0.8\linewidth]{esquema-origen-inflacion}{De dónde viene la inflación: el triángulo de \autor{Gordon} (demanda, costes e inercia).}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{De demanda}: la interna crece más que el PIB, brecha positiva, crédito y agregados dinámicos, estímulo fiscal.
\item \textbf{De costes}: CLU, energía, depreciación, impuestos indirectos; márgenes amplios señalan poca competencia. \textbf{Importada}: precios de importación que crecen más que el IPC.
\item \textbf{A largo plazo}, fenómeno monetario: $\pi\approx\gamma_M-\gamma_Y$ (un M3 muy por encima del PIB nominal no cuadra con inflación baja).
\item \textbf{Un choque de costes llega en 3 pasos}:
  \begin{enumerate}
  \item \textbf{directos}: sube la energía que compran los hogares;
  \item \textbf{indirectos}: las empresas trasladan el coste de sus insumos a sus precios;
  \item \textbf{segunda ronda}: los salarios piden recuperar poder adquisitivo y se arma la espiral precios-salarios.
  \end{enumerate}
\uline{Lo que preocupa al banco central es la segunda ronda}: se ve en la subyacente (servicios), los salarios negociados, los CLU y las expectativas. La inflación pesa más en los hogares de renta baja (energía y alimentos).
\item \textbf{¿Transitoria o persistente?} \textbf{Transitoria}: concentrada en energía y alimentos, subyacente contenida, expectativas ancladas. \textbf{Persistente}: la subyacente se generaliza, salarios y márgenes se aceleran, brecha positiva, paro bajo la NAIRU.
\item \textbf{Qué política} (objetivo del BCE: \textbf{2~\% simétrico} a medio plazo):
  \begin{itemize}
  \item de demanda: subir tipos, sin que la fiscal empuje en contra;
  \item de oferta y transitoria: el banco central puede «mirar a través» y vigilar la segunda ronda; si las expectativas se desanclan, subir;
  \item en una unión monetaria: fiscal y oferta. Con un choque de relación real de intercambio, un \textbf{pacto de rentas} reparte la pérdida y evita la espiral. Siempre, más competencia.
  \end{itemize}
\item \textbf{Deflación}: \textbf{de demanda}, mala (se aplazan compras, sube la deuda real: \autor{Fisher}; en el límite de los tipos sube el tipo real); \textbf{de oferta}, puede ser «buena». Receta: monetaria expansiva; en el límite, compras de activos y fiscal.
\end{itemize}

\sub{III.2. Tipo de cambio\label{rep:III.2}}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Nominal} $E$ (convención directa: moneda nacional por unidad de extranjera; si sube, se \textbf{deprecia}); \textbf{efectivo} (media ponderada por el comercio); \textbf{real}, la competitividad-precio:
\funcionclave{q\equiv\frac{EP^*}{P},\qquad q^{ef}=\prod_i\Big(\frac{E_iP^*_i}{P}\Big)^{\omega_i}}
  \begin{itemize}
  \item \textbf{Antes de leer el signo, hay que comprobar la convención}: el BCE y el Banco de España publican el índice al revés (sube = apreciación).
  \item \emph{Oficial} (PDM): apreciación real $>3$~\% en 3 años en el euro (10~\% fuera).
  \end{itemize}
\item \textbf{PPA relativa}, $\%\Delta E\approx\pi-\pi^*$: con más inflación que los socios y el nominal estable, el tipo real se aprecia y se pierde competitividad. Una apreciación real por \textbf{convergencia} (\autor{Balassa}-\autor{Samuelson}: productividad de los comercializables) no preocupa; por \textbf{exceso de demanda} (no comercializables caros por el crédito), sí.
\item \textbf{Traspaso}: la parte de la depreciación que pasa a los precios (0,3: una depreciación del 10~\% los sube un 3~\%); mayor con inflación alta y poco creíble y comercio en dólares.
\item \textbf{\autor{Marshall}-\autor{Lerner}}: $|\varepsilon_{X,q}|+|\varepsilon_{M,q}|>1$. \textbf{Curva J}: primero domina el \textbf{efecto valor} (importar cuesta más, de inmediato) y después el \textbf{efecto volumen}; se cumple si el saldo mejora en 1 o 2 años.
\end{itemize}
\graficorepaso[0.94\linewidth]{esquema-devaluacion}{Una devaluación: un solo efecto positivo (a medio plazo, con \autor{Marshall}-\autor{Lerner}) frente a 6 costes o riesgos; nunca se recomienda sola.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Régimen}: con libertad de capitales, un tipo fijo obliga a renunciar a la política monetaria (\textbf{trilema}). La intervención se ve en las \textbf{reservas} (caen si se frena una depreciación: \emph{miedo a flotar}); comprar divisas sin esterilizar sube la base monetaria.
\item Un \textbf{tipo fijo como ancla} baja la inflación, pero si converge despacio el tipo real se aprecia mes a mes: origen de muchas crisis cambiarias. Para \textbf{defenderlo}: subir tipos (daña la actividad), gastar reservas (se agotan) o controlar salidas; si cae, anclar las expectativas.
\item \textbf{Ante una oleada de entradas de capital}: dejar apreciar (si no está sobrevalorada); intervenir y esterilizar (temporal y con \textbf{coste cuasi fiscal}); fiscal más restrictiva; en último recurso, controles temporales a la entrada.
\end{itemize}

\sub{III.3. Competitividad no precio\label{rep:III.3}}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Cuota de mercado} que crece aunque se aprecie el tipo real: no precio fuerte (paradoja de \autor{Kaldor}). Perder cuota sin perder precio: problema de oferta exportadora. \emph{Oficial} (PDM): resultados de exportación $-3$~\% en 3 años.
\item \textbf{Tasa de cobertura}, $X/M$ ($>100$: las exportaciones cubren las importaciones). \textbf{Comercio intraindustrial}: valor añadido medio o alto (\autor{Grubel}-\autor{Lloyd}, $1-|X_i-M_i|/(X_i+M_i)$).
\item \textbf{Aranceles de un tercero}: efecto \textbf{directo} (exportaciones a ese destino), \textbf{indirecto} (cadenas de valor, demanda mundial), \textbf{desviación de comercio} (importaciones baratas que presionan a los productores locales), incertidumbre y tipo de cambio. La protección se mide con el arancel medio simple o el ponderado por el comercio (que infravalora los muy altos).
\item \textbf{Integración comercial asimétrica}: con más apertura ganan los sectores con ventaja comparativa y pierden los manufactureros menos productivos (plazos de desarme, ayudas a la reasignación y formación).
\item \textbf{Recomendación}: I+D, más tamaño empresarial, diversificar mercados, promoción comercial y acuerdos comerciales.
\end{itemize}

\bloque{IV. Sector exterior\label{rep:IV}}
\begin{ideaclave}
\textbf{Qué hay que contestar}: ¿hay capacidad o necesidad de financiación y de qué subcuenta viene? ¿Es de ahorro o de inversión, pública o privada? ¿Cómo se financia? ¿Qué posición acumula? ¿Es sostenible?
\end{ideaclave}
\begin{itemize}[itemsep=1.5pt]
\item \uline{Ser deudor no es malo en sí: lo que importa es la sostenibilidad}. Los desequilibrios externos suelen ser síntoma de los internos.
\end{itemize}

\sub{IV.1. Capacidad o necesidad de financiación\label{rep:IV.1}}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{La identidad de la balanza de pagos, desarrollada} (MBP6):
\funcionclave{\begin{aligned}&\underbrace{\overbrace{(X-M)+\mathrm{RP}+\mathrm{RS}}^{\mathrm{CC}\text{: cuenta corriente}}+\overbrace{\mathrm{CK}}^{\text{c. de capital}}}_{\text{capacidad }(+)\text{ o necesidad }(-)\text{ de financiación}}+\overbrace{\mathrm{EO}}^{\text{err. y omis.}}\\[2pt]&=\overbrace{\mathrm{CF}}^{\text{c. financiera}}=\underbrace{\mathrm{ID}+\mathrm{IC}+\mathrm{D}+\mathrm{OI}+\Delta R}_{\text{por instrumentos}}\end{aligned}}
y la cuenta corriente, a su vez, es
\funcionclave{\begin{aligned}\mathrm{CC}&=\underbrace{(X-M)}_{\text{bienes y serv.}}+\underbrace{\mathrm{RP}+\mathrm{RS}}_{\text{rentas}}=\underbrace{(Y-A)}_{\text{renta}-\text{absorción}}+\mathrm{RP}+\mathrm{RS}\\[2pt]&=\underbrace{S-I}_{\text{ahorro}-\text{inversión}}=\underbrace{(S_P-I_P)}_{\text{privado}}+\underbrace{(T-G)}_{\text{público}}\end{aligned}}
  \begin{itemize}
  \item $A=C+I+G$, la absorción; ID, IC, D y OI: inversión directa, de cartera, derivados y otra inversión; $\Delta R$, reservas.
  \item \textbf{Signos} (MBP6): una salida de capitales es CF positiva; una necesidad de financiación sale negativa.
  \item \textbf{3 lecturas} del déficit: produce poco o gasta mucho ($Y<A$); ahorra poco o invierte mucho ($S<I$); ahorra poco el sector privado o el público.
  \end{itemize}
\item \textbf{Por subcuentas}: \textbf{bienes y servicios} (competitividad, renta exterior, energía); \textbf{rentas primarias} (intereses y beneficios: PIB frente a RNB); \textbf{secundarias} (remesas, estables y contracíclicas; aportaciones a la UE); \textbf{cuenta de capital} (fondos europeos: se suma a la corriente).
\item \emph{Oficial} (PDM): saldo corriente medio de 3 años $<-4$~\% o $>+6$~\% del PIB. \emph{Orientativo}: apertura, $(X+M)/\mathrm{PIB}$, cerrada por debajo del 15-20~\% y muy abierta por encima del 75~\%.
\item \textbf{Relación real de intercambio}, $P_X/P_M$: si sube, enriquecimiento o pérdida de competitividad (sin inflación marcada, lo primero).
\item \textbf{Exportaciones e importaciones} ($q$ al alza = depreciación real):
\funcionclave{\begin{gathered}X=f\big(\overset{+}{q},\overset{+}{\text{no precio}},\overset{+}{Y^*},\overset{-}{\text{dem. interna}}\big)\\ M=f\big(\overset{-}{q},\overset{-}{\text{no precio}},\overset{+}{Y}\big)\end{gathered}}
  \begin{itemize}
  \item Cuentan también la \textbf{oferta exportadora}, ser \textbf{precio-aceptante} (materias primas), la \textbf{diversificación} y el \textbf{contenido importado} de las exportaciones.
  \end{itemize}
\item \textbf{Ahorro e inversión}: \uline{un déficit corriente es un exceso de inversión sobre el ahorro nacional}. Por \textbf{poco ahorro} es peor que por \textbf{mucha inversión} productiva; si es \textbf{privado}, riesgo de corrección brusca al cortarse el crédito; si es \textbf{público}, de sostenibilidad fiscal.
\end{itemize}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Déficits gemelos}: si el saldo privado no cambia, más déficit público es más déficit exterior, por 2 vías:
\begin{formula}
\centering\footnotesize $\downarrow(T-G)\Rightarrow\uparrow i\Rightarrow\text{entran capitales}\Rightarrow\text{apreciación}\Rightarrow\downarrow(X-M)$\\[1pt]
$\downarrow(T-G)\Rightarrow\uparrow Y\Rightarrow\uparrow M\Rightarrow\downarrow(X-M)$
\end{formula}
\item Un déficit \textbf{no es malo en sí}, ni un superávit bueno. Razonables: financiar el \textbf{desarrollo} (si la inversión rinde más que la deuda), amortiguar un \textbf{choque temporal}, o un superávit por \textbf{envejecimiento} o recursos agotables. No lo son el déficit de \textbf{sobrecalentamiento} ni el \textbf{superávit persistente} por inversión insuficiente (por eso el PDM vigila también el $+6$~\%).
\end{itemize}

\sub{IV.2. Cómo se financia\label{rep:IV.2}}
\graficorepaso[0.97\linewidth]{esquema-cuenta-financiera}{\uline{No es lo mismo financiarse con capital que con deuda}: de más estable (inversión directa) a más volátil; abajo, 3 referencias de reservas suficientes, que también cuestan si sobran.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{¿Hace falta financiación excepcional?} La pregunta clave es si el país atiende sus pagos exteriores sin usar reservas, sin crédito del FMI y sin dejar de pagar.
\item \textbf{Qué mueve la IED y la cartera}: el diferencial de tipos, las perspectivas y el clima para el capital extranjero.
  \begin{itemize}
  \item La cartera se revierte de golpe (\emph{flight to liquidity, to safety}); con tipos altos y moneda estable atrae \emph{carry trade}, que sale al cambiar las expectativas.
  \item Las entradas dependen de factores de \textbf{empuje} (fuera: tipos de la Reserva Federal, apetito por el riesgo) y de \textbf{atracción} (dentro): el \emph{ciclo financiero mundial}.
  \end{itemize}
\item \textbf{Son flujos}: una cartera del $-3$~\% del PIB es lo que entró ese año, no lo que tienen los extranjeros; los saldos, en la PIIN.
\item \textbf{Ajuste según el régimen}: con tipo \textbf{fijo}, el banco central cubre el déficit con reservas y el ajuste llega por los agregados (si se agotan, crisis cambiaria); con tipo \textbf{flexible}, por el tipo de cambio.
\end{itemize}

\sub{IV.3. PIIN y deuda externa\label{rep:IV.3}}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{PIIN} $\equiv$ activos $-$ pasivos exteriores; la \textbf{deuda externa}, solo los pasivos exigibles (sin acciones, IED en capital ni derivados). \emph{Oficial} (PDM): PIIN $<-35$~\% del PIB.
\item \textbf{La composición importa tanto como el nivel}: mejor instrumentos no exigibles, plazos largos, moneda nacional y \textbf{tenedores} residentes (Japón, Italia).
\item \textbf{Círculo flujo-stock}: más pasivos traen más intereses (rentas primarias), que empeoran la cuenta corriente y exigen más financiación: los déficits persistentes se alimentan solos.
\item \textbf{Efecto valoración} (otras variaciones: precios y tipo de cambio). En un deudor neto:
  \begin{itemize}
  \item \textbf{mejora} la PIIN si caen la bolsa o los bonos nacionales en manos de no residentes, si se aprecia la moneda frente a la de su deuda o si suben los activos exteriores;
  \item \textbf{empeora} en los casos contrarios, pero no siempre es más vulnerabilidad (bolsa que sube por mejores perspectivas).
  \end{itemize}
\item \textbf{Indicadores de deuda externa}: deuda y \textbf{servicio} (intereses y amortizaciones) sobre las \textbf{exportaciones}; reservas sobre la deuda a corto a \textbf{vencimiento residual}; tipo implícito, vida media y \textbf{tasa de renovación}; diferenciales (EMBI, CDS), que saltan antes de las crisis.
\item \textbf{Enfoque de balances}: 4 descalces (\textbf{plazos}, \textbf{monedas}, \textbf{estructura de capital} y \textbf{solvencia}); la deuda en divisas \emph{entre residentes} no se ve en la PIIN, pero estalla si se deprecia la moneda.
\end{itemize}

\sub{IV.4. ¿Es sostenible?\label{rep:IV.4}}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Dinámica}: con PIIN negativa, el crecimiento nominal la mejora aunque se siga pidiendo prestado (los intereses ya están en la CC):
\funcionclave{\mathrm{PIIN}_t-\mathrm{PIIN}_{t-1}\approx\mathrm{CF}_t+\mathrm{OV}_t-\frac{\gamma^N_t}{1+\gamma^N_t}\,\mathrm{PIIN}_{t-1}}
\item \textbf{Saldo corriente sin intereses que estabiliza la deuda externa} $D^*$ (con $i^*$ su tipo implícito); si $i^*>\gamma^N$, hace falta superávit de bienes y servicios:
\funcionclave{\mathrm{CC}^{\text{sin int.}}=\frac{i^*-\gamma^N}{1+\gamma^N}\,D^*}
\item \textbf{Determinantes}:
\funcionclave{\begin{gathered}\text{Sostenibilidad}=f\big(\overset{-}{\text{deuda}},\overset{+}{\gamma_Y},\overset{-}{i},\overset{?}{\pi},\\ \overset{-}{\text{absorción}},\overset{-}{E},\overset{+}{\text{acceso a los mercados}}\big)\end{gathered}}
  \begin{itemize}
  \item La inflación solo licúa la deuda en moneda nacional; con deuda en divisas, una depreciación la encarece (\textbf{pecado original}).
  \item \textbf{Frenazo súbito}: si se corta la financiación, el ajuste llega por la \textbf{compresión de importaciones} (hundir consumo e inversión).
  \item \emph{Orientativo}: deuda externa preocupante por encima del 100~\% del PIB (no sirve para el euro); mejor la PIIN y su composición.
  \end{itemize}
\item \textbf{Financiar o ajustar}, según el choque: \uline{lo temporal se financia; lo permanente se ajusta}. El ajuste combina políticas \textbf{reductoras del gasto} (fiscal y monetaria) y \textbf{desviadoras del gasto} (tipo de cambio real, competitividad); con brecha negativa, más de las segundas. Ante la duda, prudencia.
\end{itemize}

\bloque{V. Políticas\label{rep:V}}
\begin{itemize}[itemsep=1.5pt]
\item Las de \textbf{demanda} (monetaria, fiscal) estabilizan el ciclo; las de \textbf{oferta} elevan el potencial. En cada una: \textbf{tono}, relación con el ciclo y eficacia.
\end{itemize}

\sub{V.1. Política monetaria\label{rep:V.1}}
\begin{ideaclave}
\textbf{Qué hay que contestar}: ¿qué objetivo e instrumentos tiene el banco central? ¿Qué tono y por qué? ¿Ha funcionado? ¿Qué debería hacer ahora?
\end{ideaclave}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Diseño}: apenas altera la actividad a largo plazo; su papel básico es \textbf{controlar la inflación} (BCE: 2~\% simétrico). Instrumentos: tipos oficiales, financiación a plazo largo, compras de activos, orientación sobre los tipos futuros; con exceso de liquidez manda la facilidad de depósito.
\item \textbf{Objetivo aparente}: si sube con fuerza al subir la inflación, objetivo de inflación; si baja en plena estanflación, pesa la actividad; si frena la apreciación o la depreciación, objetivo cambiario. En una unión monetaria responde al ciclo del conjunto.
\item \textbf{Tono}: por el \emph{nivel} del tipo real a corto, $r_t\approx i_t-\pi_t$, frente al natural $r^n$, no por si el nominal sube o baja; se completa con la \textbf{regla de \autor{Taylor}}:
\funcionclave{\begin{gathered}i_t=r^*+\pi^*+1{,}5\,(\pi_t-\pi^*)+0{,}5\,\mathrm{OG}_t\\ r^*=\pi^*=2~\%\end{gathered}}
\end{itemize}
\graficorepaso[0.97\linewidth]{esquema-tono-politicas}{El tono: monetario por el nivel del tipo real; fiscal por la variación del saldo estructural primario.}
\begin{itemize}[itemsep=1.5pt]
\item En el límite inferior de los tipos, el tono se ve en el \textbf{balance} y en M3. Si el nominal sube pero la inflación sube más, el tipo real cae (aunque la subida indique un endurecimiento).
\end{itemize}
\graficorepaso[0.88\linewidth]{tipos-balance-bce}{BCE: con exceso de liquidez guía el tipo de la facilidad de depósito; el balance del Eurosistema pasó de 2 billones (2014) a 8,8 (2022) y a 5,9 (octubre de 2026).}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Retardos}: el efecto máximo sobre los precios llega, de media, a los 29 meses; una política que no ha bajado la inflación en un año no ha fracasado necesariamente.
\item \textbf{Contracíclica} si sube los tipos con la brecha positiva y los baja con la negativa; con el desfase, puede acabar siendo procíclica.
\item \textbf{Menos eficaz} con un sistema financiero poco desarrollado o dolarización; con \textbf{dominancia fiscal} (el banco central acaba financiando el déficit: déficit y deuda altos, tipos reales muy negativos); o con el \textbf{canal del crédito roto} (crédito que cae aunque bajen los tipos: sanear la banca, no bajar más los tipos). Con mucha deuda privada a tipo variable, la transmisión es más fuerte.
\end{itemize}
\graficorepaso[0.86\linewidth]{esquema-canales-monetaria}{5 canales de transmisión: funciona si baja la inflación (con 1 o 2 años de retraso), se frena el crédito y se aprecia la moneda.}

\sub{V.2. Estabilidad financiera y macroprudencial\label{rep:V.2}}
\begin{ideaclave}
\textbf{Qué hay que contestar}: ¿es resiliente el sistema? ¿Quién está endeudado? ¿En qué punto del ciclo financiero está? ¿Es la banca solvente, líquida y rentable? ¿Hay burbujas? ¿Qué deben hacer la regulación y la política macroprudencial?
\end{ideaclave}
\begin{itemize}[itemsep=1.5pt]
\item Orden fijo: \textbf{endeudamiento → banca → riesgos → políticas}, tras una frase de resumen («el sistema parece resiliente, aunque…»); la banca, en el orden CAMELS (capital, activos, gestión, beneficios, liquidez).
\item \textbf{Endeudamiento} (\emph{oficial}, PDM): hogares $>55$~\% del PIB; empresas $>85$~\%; flujos de crédito $>14$ y 13~\% de su deuda. \textbf{Brecha crédito-PIB} $>2$~pp: activar el colchón anticíclico. \textbf{Bucle bancario-soberano}: la razón de ser de la unión bancaria.
\item \textbf{Solvencia}: capital sobre \textbf{activos ponderados por riesgo} (APR), según su calidad: \textbf{CET1}, \textbf{Tier~1} y \textbf{total}. Apalancamiento (Tier~1 sobre la exposición sin ponderar): mínimo del 3~\%.
\end{itemize}
\graficorepaso[0.97\linewidth]{esquema-basilea}{Basilea~III: mínimos por calidad del capital más colchones (\% de los APR). Un capital que cumple muy justo puede no bastar ante un choque.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Calidad de los activos}: dudosos (más de 90 días) y cobertura; la morosidad va retrasada respecto al ciclo. \textbf{Liquidez}: LCR y NSFR $\geq100$~\%; LTD $>100$~\%, dependencia mayorista.
\item \textbf{Rentabilidad}: ROE frente al \textbf{coste del capital} (8-12~\%). Tipos bajos estrechan el margen (efecto precio) pero dan más crédito (efecto volumen).
\item \textbf{Empresas}: cobertura de intereses (EBITDA sobre gastos financieros) por debajo de 1: no pagan su deuda con lo que generan; son la morosidad de mañana.
\end{itemize}
\graficorepaso[0.97\linewidth]{esquema-semaforo-banca}{Semáforo de la banca: las debilidades se refuerzan (más dudosos, más provisiones, menos beneficio, menos capital). España, 2026: ROE del 16~\% (UE, 11~\%), CET1 del 14~\% (UE, 16~\%).}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Riesgos}: \textbf{vivienda} (fundamentos o burbuja: precio muy por encima de renta y alquileres, crédito con criterios relajados; \emph{oficial}, precio nominal $>+9$~\% anual); auge del crédito; préstamos en divisas; tipos muy bajos mucho tiempo; bolsa que sube más que los beneficios.
\item \textbf{Regulación}: \textbf{micro} (cada entidad; los 3 pilares de Basilea) y \textbf{macro} (el sistema). Unión bancaria: MUS, MUR y garantía de depósitos pendiente.
\item \textbf{Recomendación tipo}: en la fase alcista, colchón anticíclico o sectorial y límites \textbf{LTV}, \textbf{LTI} y \textbf{DSTI}; en la bajista, liberar colchones y limitar dividendos. En una crisis, \uline{distinguir liquidez (prestamista de última instancia) de solvencia (recapitalizar o resolver)}.
\end{itemize}

\sub{V.3. Política fiscal y deuda pública\label{rep:V.3}}
\begin{ideaclave}
\textbf{Qué hay que contestar}: ¿qué tono tiene la política fiscal y es adecuado al ciclo? ¿Es sostenible la deuda? ¿Qué riesgos tiene y cómo se reducen? ¿Qué exigen las reglas?
\end{ideaclave}
\begin{itemize}[itemsep=1.5pt]
\item Lo recomendable es una política \textbf{neutral}, que deje actuar a los \textbf{estabilizadores automáticos}; la discrecional, solo ante una recesión grave o un sobrecalentamiento.
\item \textbf{Los 5 saldos}: $b$ total; $b^p$ primario; $b^{ca}=b-\varepsilon\cdot\mathrm{OG}$ ajustado por el ciclo; $b^e$ estructural (sin medidas puntuales); $b^{pe}$ estructural primario. Su variación es el tono:
\funcionclave{\begin{gathered}b^{pe}=b-\varepsilon\cdot\mathrm{OG}-\text{puntuales}+\text{intereses}\\ \text{esfuerzo}_t=\Delta b^{pe}_t\end{gathered}}
  \begin{itemize}
  \item $\Delta b^{pe}>0$, \textbf{contractiva}; $<0$, \textbf{expansiva}; $\approx0$, neutral. $\varepsilon\approx0{,}5$-$0{,}6$ en la UE.
  \item \uline{Si un saldo negativo se hace más negativo, ha \emph{bajado}: es expansiva}. Con déficits muy altos, mirar también el nivel.
  \item \textbf{Engaña en los auges}: el ajuste cíclico no quita los ingresos extraordinarios de un auge de activos (inmobiliarios, plusvalías); mirar también el gasto, que es lo que vigila la regla de la UE.
  \end{itemize}
\end{itemize}
\graficorepaso[0.86\linewidth]{impulso-fiscal}{Impulso fiscal ($-\Delta b^{pe}$, FMI): positivo, política expansiva.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Procíclica} en muchos emergentes: pocos estabilizadores, ciclo mal medido, sin acceso a la financiación en las recesiones y gasto de los ingresos del auge.
\item \textbf{Efectos no keynesianos}: una consolidación creíble podría animar el consumo, pero casi siempre reduce la actividad a corto plazo. Los \textbf{multiplicadores} son mayores para la inversión pública y las transferencias focalizadas que para las rebajas generales de impuestos, y mayores con tipos en su límite.
\item \textbf{Reglas de la UE} (\emph{oficial}): déficit del 3~\% y deuda del 60~\% del PIB; senda de \textbf{gasto primario neto} en un plan fiscal-estructural de 4 años (hasta 7); la deuda baja 1~pp al año si supera el 90~\%. Reglas de \textbf{saldo} (procíclicas), de \textbf{deuda} (anclan) y de \textbf{gasto} (controlables y anticíclicas).
\item \textbf{Dinámica de la deuda} (ADD, ajuste déficit-deuda); \uline{lo que decide si se sostiene es la diferencia entre el tipo y el crecimiento nominal}:
\funcionclave{d_t-d_{t-1}=(g_t-\tau_t)+\underbrace{\frac{i_t-\gamma^N_t}{1+\gamma^N_t}\,d_{t-1}}_{\text{bola de nieve}}+\mathrm{ADD}_t}
\end{itemize}
\graficorepaso[0.97\linewidth]{esquema-deuda-estabilizadora}{Con $i>\gamma^N$ hace falta superávit primario. Ejemplo: deuda del 100~\%, $i=3$~\% y $\gamma^N=4$~\%: basta un déficit primario de casi 1 punto.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Con los datos del examen, hay que decirlo}: el cuadro da la deuda \textbf{bruta} (debería usarse la \textbf{neta}) y el tipo \textbf{a corto} (lo relevante es el tipo \textbf{medio} de la deuda viva).
\item \textbf{Riesgos}: macroeconómicos (menos crecimiento, subidas de tipos, envejecimiento, \textbf{pasivos contingentes}: avales, empresas públicas, colaboraciones público-privadas) y financieros (deuda en divisas o a corto, tenedores no residentes, calificación).
\item \textbf{Necesidades brutas de financiación}: déficit más vencimientos a 12 meses; miden la \emph{liquidez}, no la solvencia.
\item \textbf{Cómo se financia el déficit}: fuera (apreciación, riesgo cambiario); banco central (inflación: dominancia fiscal); banca nacional (expulsión y bucle bancario-soberano); no bancario (tipos más altos).
\end{itemize}
\graficorepaso[0.86\linewidth]{inflacion-cuentas-publicas}{La inflación y las cuentas públicas: a corto plazo \textbf{mejoran} (ingresos rápidos, progresividad en frío, PIB nominal mayor); a medio, \textbf{empeoran} (intereses y gastos indexados con retraso).}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Otros análisis}: estabilizadores, $b-b^{ca}=\varepsilon\cdot\mathrm{OG}$; ingresos y gastos frente a países parecidos (sostenibilidad y eficiencia, no tamaño); gasto corriente frente a inversión; deuda a corto y vida media; deuda regional (corresponsabilidad fiscal).
\end{itemize}
\graficorepaso[0.9\linewidth]{escala-calificaciones}{Por debajo de BBB$-$/Baa3 (S\&P y Fitch / Moody's) la deuda deja de ser grado de inversión.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Recomendación tipo}: consolidación \textbf{gradual, creíble y anunciada pronto}; la \textbf{calidad} importa tanto como el tamaño (proteger inversión y redes focalizadas, ampliar bases); reglas y techo de gasto; alargar la vida media.
\item \textbf{En una crisis, según su origen}: si es \emph{fiscal}, consolidar; si es financiera o empresarial con cuentas sanas, dejar actuar a los estabilizadores o estimular, con un plan de retirada anunciado pronto.
\end{itemize}

\sub{V.4-V.5. \emph{Policy mix} y políticas de oferta\label{rep:V.4}}
\graficorepaso[0.8\linewidth]{policy-mix}{\uline{Es coherente si cada política ayuda a la otra, no si van en el mismo sentido}. Según el FMI, en 2020 el 77~\% de las economías aplicó ambas expansivas y en 2022 el 74~\%, ambas contractivas.}
\begin{itemize}[itemsep=1.5pt]
\item Fiscal contractiva con monetaria expansiva: la combinación clásica para consolidar con deuda alta y brecha negativa. Fiscal expansiva con monetaria contractiva: suben los tipos y se aprecia la moneda; poco coherente salvo que la expansión sea temporal y focalizada. En una \textbf{unión monetaria} solo queda la fiscal, que compensa si el ciclo del país se separa del de la zona.
\item \textbf{Oferta}, según el cuello de botella: progreso técnico (I+D), capital (incentivos, infraestructuras), capital humano (FP dual), competencia (subvenciones transversales, no sectoriales) e instituciones (seguridad jurídica, justicia eficaz).
\item \textbf{Paro}: la receta, según su tipo (\hyperref[rep:II.1]{II.1}); en todos, políticas activas bien focalizadas y evaluadas, y prestaciones de cuantía decreciente ligadas a la búsqueda de empleo; para el de larga duración, planes de choque con incentivos a su contratación y formación.
\end{itemize}

\begin{trampa}
errores típicos:
\begin{itemize}
\item las contribuciones van en \textbf{pp}, no en \%;
\item hay que comprobar la \textbf{convención del tipo de cambio} antes de leer el signo;
\item un PMI $>50$ indica que la actividad crece, no que esté alta;
\item las tasas de actividad se comparan con la \textbf{misma edad};
\item los intereses de la deuda externa ya están en la cuenta corriente;
\item la PAT puede subir solo porque se destruye empleo poco productivo;
\item el tono monetario va por el \textbf{nivel} del tipo real; el fiscal, por la \textbf{variación} de $b^{pe}$.
\end{itemize}
