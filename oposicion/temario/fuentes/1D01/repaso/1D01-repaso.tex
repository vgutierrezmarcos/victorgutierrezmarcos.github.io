\bloque{I. Perfil cíclico de la economía\label{rep:I}}
\begin{ideaclave}
\textbf{Qué hay que contestar}: ¿en qué fase está la economía y desde cuándo? ¿Qué tira del crecimiento? ¿Crece por encima o por debajo de su potencial?
\end{ideaclave}
\graficorepaso[0.97\linewidth]{rep-perfil}{Los 3 pasos de la respuesta, siempre los mismos.}
\begin{itemize}[itemsep=1.5pt]
\item Es la \textbf{pregunta más larga} y casi nunca falta (13 de 14 dictámenes de 2011 a 2026).
\item Empieza con una \textbf{frase ejecutiva}, que se escribe al final en un hueco: dónde está la economía y qué la ha sostenido, qué la frena y qué riesgos tiene.
  \begin{itemize}
  \item Ejemplo (FMI, España, 2026): la economía «siguió creciendo más que la zona del euro pese al impacto adverso de la guerra en Oriente Medio».
  \end{itemize}
\end{itemize}

\sub{I.1. Las fases y sus medidores\label{rep:I.1}}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Crecimiento del PIB real}: el signo y el nivel dicen si crece; su variación, si se acelera.
  \begin{itemize}
  \item \emph{Orientativo} (avanzadas): $<0$ contracción; 0-1~\% estancamiento; 1-2~\% moderado; 2-3~\% sobre el potencial; $\geq3$~\% sólido.
  \item En un emergente, un 3~\% puede ser débil.
  \end{itemize}
\item \textbf{Brecha de producción}: distancia del PIB al \textbf{potencial}, el que se produce sin tensiones inflacionistas. Referencia: 0.
\funcionclave{\mathrm{OG}_t\equiv\frac{Y_t-Y^p_t}{Y^p_t}\times100,\qquad \Delta\mathrm{OG}_t\approx\gamma_{Y,t}-\gamma_{Y^p,t}}
  \begin{itemize}
  \item \textbf{Negativa}: recursos ociosos. \textbf{Positiva}: cuellos de botella y riesgo de inflación.
  \item Crecer mucho sin cerrar la brecha no es débil: el potencial crece mucho.
  \item El potencial se estima y se revisa mucho: no se divide el ciclo solo con él.
  \end{itemize}
\end{itemize}
\graficorepaso[0.9\linewidth]{fases-ciclo-brecha}{Las 4 fases: \textbf{recuperación} y \textbf{expansión} si el PIB crece más que el potencial; \textbf{desaceleración} y \textbf{recesión} si crece menos o cae.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Recesión} (regla práctica): 2 trimestres seguidos de caída del PIB real. Su forma: V, U, W o L.
\item \textbf{PIB frente a RNB}, con $\mathrm{RNB}=\mathrm{PIB}+\mathrm{RP}$:
  \begin{itemize}
  \item $\mathrm{RNB}<\mathrm{PIB}$: deudor neto o mucha inversión extranjera directa;
  \item $\mathrm{RNB}>\mathrm{PIB}$: acreedor neto o muchos trabajadores fuera.
  \end{itemize}
\item \textbf{Alta frecuencia}: un \textbf{PMI} $>50$ indica que la actividad \emph{crece}, no que esté alta; el \textbf{ESI} tiene media 100; una \textbf{curva de tipos invertida} anticipa una recesión.
  \begin{itemize}
  \item Coincidentes: producción industrial, afiliación, ventas. Retrasados: paro, subyacente, tipos de los préstamos.
  \item Si los mensuales caen mientras el PIB anual aún crece, la economía se desacelera.
  \end{itemize}
\end{itemize}

\sub{I.2. Demanda: el motor\label{rep:I.2}}
\begin{itemize}[itemsep=1.5pt]
\item Cada \textbf{contribución} es el peso del componente en $t-1$ por su crecimiento:
\funcionclave{\begin{gathered}\gamma_{Y,t}=\underbrace{\tfrac{C_{t-1}}{Y_{t-1}}\gamma_{C,t}+\tfrac{I_{t-1}}{Y_{t-1}}\gamma_{I,t}+\tfrac{G_{t-1}}{Y_{t-1}}\gamma_{G,t}}_{\text{demanda interna}}\\+\underbrace{\tfrac{X_{t-1}}{Y_{t-1}}\gamma_{X,t}-\tfrac{M_{t-1}}{Y_{t-1}}\gamma_{M,t}}_{\text{demanda externa}}\end{gathered}}
  \begin{itemize}
  \item Se miden en \textbf{puntos porcentuales}; las importaciones restan.
  \item En las crisis la demanda interna resta (\textbf{procíclica}) y la externa suma.
  \item Si la externa «toma el relevo»: ¿suben las exportaciones o solo caen las importaciones?
  \end{itemize}
\end{itemize}
\graficorepaso[0.97\linewidth]{contribuciones-pib-zona-euro}{Contribuciones al PIB de la zona del euro (Eurostat): la demanda interna explica casi todo el crecimiento, salvo en las crisis.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Consumo privado}: el mayor componente (53~\% del PIB en la UE) y el menos volátil de los privados.
\funcionclave{C=f\big(\overset{+}{Y^d_{\text{real}}},\overset{+}{\text{riqueza}},\overset{-}{r^{cp}},\overset{-}{\pi},\overset{+}{\text{crédito}},\overset{-}{\text{deuda}},\overset{-}{\text{incert.}}\big)}
\end{itemize}
\graficorepaso[0.97\linewidth]{consumo-renta-ahorro-zona-euro}{Hogares de la zona del euro: el consumo sigue a la renta disponible; en 2020, ahorro forzoso.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Inversión}: el componente más volátil; es demanda hoy y oferta mañana.
\funcionclave{I=f\big(\overset{-}{r^{lp}},\overset{+}{\gamma^e_Y},\overset{+}{\mathrm{OG}},\overset{+}{\mathrm{EBE}},\overset{+}{\text{crédito}},\overset{-}{\text{incert.}}\big)}
\item \textbf{Cómo se escribe}: los 2 o 3 determinantes que el cuadro deja ver, con sus años; los que faltan, con condicionales.
\item \textbf{Demanda externa}: 1 o 2 frases; sus determinantes van en el sector exterior.
\end{itemize}
\graficorepaso[0.97\linewidth]{pib-comercio-mundial}{El comercio mundial es más volátil que el PIB (cae un 10~\% en 2009 y un 8~\% en 2020).}

\sub{I.3. Oferta y diagnóstico\label{rep:I.3}}
\begin{itemize}[itemsep=1.5pt]
\item Con la \textbf{productividad aparente del trabajo}, $\mathrm{PAT}=Y/L$:
\begin{formula}
\centering $\gamma_Y=\gamma_A+\alpha\,\gamma_K+(1-\alpha)\,\gamma_L\quad\Rightarrow\quad\gamma_Y=\gamma_{\mathrm{PAT}}+\gamma_L$
\end{formula}
  \begin{itemize}
  \item Con empleo y PAT estancada, patrón \textbf{extensivo}; con PAT al alza, \textbf{intensivo} y más sostenible.
  \item Si el empleo cae más que el PIB, la PAT sube de forma \textbf{espuria}.
  \end{itemize}
\item \textbf{Sectores}: si uno (petróleo, turismo) pesa mucho en el VAB, las exportaciones o los ingresos, su precio mueve todo el cuadro.
\end{itemize}
\graficorepaso[0.97\linewidth]{brent}{El Brent: grandes picos en 2008, 2011-2012, 2022 y 2026; en euros pesa también el tipo de cambio.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Diagnóstico anticipado}: «parece una economía avanzada que crece con la demanda interna, pero con un déficit exterior persistente y una deuda pública elevada».
  \begin{itemize}
  \item Sin adivinar el país; lo estructural, con condicionales y 2 o 3 datos.
  \end{itemize}
\end{itemize}

\bloque{II. Mercados de factores\label{rep:II}}
\begin{ideaclave}
\textbf{Qué hay que contestar}: ¿cuánto empleo crea el crecimiento? ¿Qué tipo de paro hay? ¿Los salarios siguen a la productividad? ¿Llega el crédito?
\end{ideaclave}

\sub{II.1. Mercado de trabajo\label{rep:II.1}}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Cantidades}, con $L$ ocupados, $U$ parados y $A=L+U$ activos:
\funcionclave{\text{tasa de actividad}=\frac{A}{P_{16}},\qquad u=\frac{U}{A}}
  \begin{itemize}
  \item \textbf{Ojo con la edad}: el INE usa 16 y más años; Eurostat y la OCDE, 15-64.
  \item \emph{Oficial} (PDM): paro $>10$~\%. \emph{Orientativo}: pleno empleo con un paro del 3-6~\%.
  \item Empleo y paro pueden subir a la vez si llegan más activos (inmigración).
  \item En la pandemia el empleo apenas reflejó la caída del PIB, por los esquemas de mantenimiento del empleo.
  \end{itemize}
\end{itemize}
\graficorepaso[0.97\linewidth]{empleo-horas-actividad-zona-euro}{Zona del euro: el empleo supera el nivel de 2019 más que las horas (baja la jornada media); el paro, cerca de su mínimo.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Productividad}: la PAT recoge también el capital y la tecnología; crece por \textbf{intensificación del capital}, capital intangible o \textbf{progreso técnico}. A largo plazo es la que eleva la renta per cápita y los salarios reales.
\end{itemize}
\graficorepaso[0.97\linewidth]{pib-por-pet-espana}{España: desde 2019 el PIB por persona en edad de trabajar crece sobre todo por más horas por persona y poco por la productividad por hora.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Costes laborales unitarios} (CLU): lo que cuesta en salarios producir una unidad.
\end{itemize}
\graficorepaso[0.97\linewidth]{rep-clu}{De la productividad al margen: las 4 fórmulas encadenadas.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Umbrales de los CLU}:
  \begin{itemize}
  \item \emph{oficial} (PDM): CLU nominales $>+9$~\% en 3 años en el euro (12~\% fuera);
  \item \emph{orientativo}: salarios que crecen como la productividad más el objetivo de inflación mantienen estables los CLU reales.
  \end{itemize}
\end{itemize}
\graficorepaso[0.97\linewidth]{rep-paro}{\textbf{Tipos de paro}: la receta depende del diagnóstico; cómo se ve cada uno en el cuadro y qué receta tiene.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{NAIRU}: el paro compatible con una inflación estable; se aproxima con el paro de los años de brecha nula.
\item \textbf{\autor{Okun}}: medio punto menos de paro por punto de crecimiento sobre el potencial (EE.~UU.). \textbf{\autor{Phillips}} con expectativas:
\begin{formula}
\centering $\pi_t=\pi^e_t-a\,(U_t-U^n)+\varepsilon_t$
\end{formula}
\item \textbf{Curva de \autor{Beveridge}}: moverse \emph{a lo largo} de ella es el ciclo; \emph{desplazarse hacia fuera}, peor emparejamiento.
\end{itemize}
\graficorepaso[0.85\linewidth]{beveridge-espana-zona-euro}{\autor{Beveridge}: la zona del euro se mueve con el ciclo; España, con pocas vacantes y mucho paro, está lejos del origen.}

\sub{II.2. Mercado de capitales\label{rep:II.2}}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Qué mirar}: el crédito nuevo, su coste y el diferencial entre préstamos y depósitos (cuanto más estrecho, mejor intermediación).
\item Un sistema que funciona mal financia \textbf{empresas zombis} y deja sin crédito proyectos rentables.
\item El endeudamiento, la banca y los riesgos van en \hyperref[rep:V.2]{V.2}.
\end{itemize}

\bloque{III. Competitividad\label{rep:III}}
\begin{ideaclave}
\textbf{Qué hay que contestar}: ¿está controlada la inflación? ¿Viene de la demanda o de los costes, de dentro o de fuera? ¿Es transitoria o persistente? ¿Qué política requiere?
\end{ideaclave}
\begin{itemize}[itemsep=1.5pt]
\item Tiene 2 caras: \textbf{precio} (inflación, costes, tipo de cambio) y \textbf{no precio} (calidad, diferenciación, mercados).
\end{itemize}

\sub{III.1. Precios e inflación\label{rep:III.1}}
\graficorepaso[0.97\linewidth]{rep-indices}{IPC y deflactor: qué compara cada uno y por qué se separan.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Inflación subyacente}: sin los componentes volátiles (energía, alimentos). Dice si el choque se extiende: \textbf{efectos de segunda ronda}.
\item \textbf{Efecto base}: la tasa interanual baja cuando sale de la comparación una subida excepcional.
\begin{formula}
\centering $\pi^{12}_t-\pi^{12}_{t-1}\approx\pi^{m}_t-\pi^{m}_{t-12}$
\end{formula}
\end{itemize}
\graficorepaso[0.97\linewidth]{iapc-zona-euro}{IAPC de la zona del euro por componentes: la energía explica el brote; la subyacente muestra si se extiende.}
\graficorepaso[0.82\linewidth]{rep-inflacion}{De dónde viene la inflación: el triángulo de \autor{Gordon}.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{¿Transitoria o persistente?}
  \begin{itemize}
  \item \textbf{Transitoria}: energía y alimentos, subyacente contenida, expectativas ancladas.
  \item \textbf{Persistente}: subyacente que se generaliza (servicios), salarios y márgenes al alza, brecha positiva.
  \end{itemize}
\item \textbf{Umbral oficial}: el objetivo del BCE, \textbf{2~\% simétrico} a medio plazo.
\item \textbf{Recomendación tipo}:
  \begin{itemize}
  \item si es de demanda, subir tipos, sin estímulo fiscal en contra;
  \item si es de oferta y transitoria, «mirar a través» y vigilar la segunda ronda;
  \item en una unión monetaria, fiscal y oferta; siempre, más competencia.
  \end{itemize}
\end{itemize}

\sub{III.2. Tipo de cambio\label{rep:III.2}}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Tipo de cambio real}: si sube, hay depreciación real y más competitividad-precio.
\funcionclave{\begin{gathered}q\equiv\frac{EP^*}{P}\\ |\varepsilon_{X,q}|+|\varepsilon_{M,q}|>1\ \ (\text{\autor{Marshall}-\autor{Lerner}})\end{gathered}}
  \begin{itemize}
  \item \textbf{Ojo}: el BCE y el Banco de España publican el índice al revés (sube = apreciación).
  \item \emph{Oficial} (PDM): apreciación real $>3$~\% en 3 años en el euro (10~\% fuera).
  \item \textbf{Curva J}: la balanza empeora antes de mejorar.
  \end{itemize}
\item \textbf{Traspaso}: la parte de la depreciación que pasa a los precios; mayor con inflación alta y poco creíble.
\item \textbf{Intervención}: se ve en las reservas. Comprar divisas sin esterilizar sube la base monetaria (\textbf{trilema}).
\end{itemize}

\sub{III.3. Competitividad no precio\label{rep:III.3}}
\begin{itemize}[itemsep=1.5pt]
\item Una \textbf{cuota de mercado} que crece aunque se aprecie el tipo real indica no precio fuerte (\textbf{paradoja de \autor{Kaldor}}).
\item \textbf{Recomendación}: I+D, más tamaño empresarial, diversificación y promoción comercial.
\end{itemize}

\bloque{IV. Sector exterior\label{rep:IV}}
\begin{ideaclave}
\textbf{Qué hay que contestar}: ¿hay capacidad o necesidad de financiación y de dónde viene? ¿Es de ahorro o de inversión, pública o privada? ¿Cómo se financia? ¿Es sostenible?
\end{ideaclave}
\begin{itemize}[itemsep=1.5pt]
\item Ha caído en \textbf{todos los dictámenes}. Ser deudor no es malo en sí: importa la \textbf{sostenibilidad}.
\end{itemize}

\sub{IV.1. Capacidad o necesidad de financiación\label{rep:IV.1}}
\graficorepaso[0.97\linewidth]{rep-balanza}{La identidad de la balanza de pagos, desarrollada (MBP6).}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Signos} (MBP6): una salida de capitales es CF positiva; una necesidad de financiación sale negativa.
\item \textbf{La cuenta de capital} (fondos europeos) se suma a la corriente.
\item \emph{Oficial} (PDM): saldo corriente medio de 3 años $<-4$~\% o $>+6$~\% del PIB.
\item \textbf{Exportaciones e importaciones} (con $q$ al alza = depreciación real):
\funcionclave{\begin{gathered}X=f\big(\overset{+}{q},\overset{+}{\text{no precio}},\overset{+}{Y^*}\big)\\ M=f\big(\overset{-}{q},\overset{-}{\text{no precio}},\overset{+}{Y}\big)\end{gathered}}
\item \textbf{Ahorro e inversión}:
  \begin{itemize}
  \item un déficit por \textbf{poco ahorro} es peor que uno por \textbf{mucha inversión} productiva;
  \item si viene del \textbf{sector privado}, hay riesgo de corrección brusca; si del \textbf{público}, de sostenibilidad;
  \item \textbf{déficits gemelos}: más déficit público, más déficit exterior (no siempre).
  \end{itemize}
\end{itemize}

\sub{IV.2. Financiación y posición\label{rep:IV.2}}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Cómo se financia}: no es lo mismo con capital que con deuda.
\end{itemize}
\graficorepaso[0.97\linewidth]{rep-financiacion}{La cuenta financiera por instrumentos y la adecuación de las reservas (demasiadas también cuestan).}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{PIIN}: activos menos pasivos exteriores. \emph{Oficial} (PDM): $<-35$~\% del PIB.
  \begin{itemize}
  \item Importa la composición: no exigibles, plazos largos, moneda nacional.
  \item Mejora con el crecimiento nominal aunque el país siga pidiendo prestado:
\funcionclave{\mathrm{PIIN}_t-\mathrm{PIIN}_{t-1}\approx\mathrm{CF}_t+\mathrm{OV}_t-\frac{\gamma^N_t}{1+\gamma^N_t}\,\mathrm{PIIN}_{t-1}}
  \end{itemize}
\item \textbf{Sostenibilidad}: un país no puede mantener un déficit crónico.
\funcionclave{\begin{gathered}\text{Sostenibilidad}=f\big(\overset{-}{\text{deuda}},\overset{+}{\gamma_Y},\overset{-}{i},\overset{?}{\pi},\\ \overset{-}{\text{absorción}},\overset{-}{E},\overset{+}{\text{acceso a los mercados}}\big)\end{gathered}}
La inflación solo licúa la deuda en moneda nacional (\textbf{pecado original}).
\item \textbf{Recomendación tipo}: más ahorro nacional, competitividad, inversión directa antes que deuda y reservas suficientes.
\end{itemize}

\bloque{V. Políticas\label{rep:V}}
\begin{itemize}[itemsep=1.5pt]
\item Las de \textbf{demanda} (monetaria, fiscal) estabilizan el ciclo; las de \textbf{oferta} elevan el potencial. En cada una se juzga el \textbf{tono}, la relación con el ciclo y la eficacia.
\end{itemize}

\sub{V.1. Política monetaria\label{rep:V.1}}
\begin{ideaclave}
\textbf{Qué hay que contestar}: ¿qué objetivo e instrumentos tiene? ¿Qué tono y por qué? ¿Ha funcionado? ¿Qué debería hacer ahora?
\end{ideaclave}
\begin{itemize}[itemsep=1.5pt]
\item Cae en 11 de 14 dictámenes, a menudo con una coletilla sobre el sistema financiero.
\item \textbf{Tono}: el tipo real a corto frente al natural; se completa con la \textbf{regla de \autor{Taylor}}:
\funcionclave{\begin{gathered}i_t=r^*+\pi^*+1{,}5\,(\pi_t-\pi^*)+0{,}5\,\mathrm{OG}_t\\ r^*=\pi^*=2~\%\end{gathered}}
\end{itemize}
\graficorepaso[0.97\linewidth]{rep-tono}{El tono: monetario por el nivel del tipo real; fiscal por la variación del saldo estructural primario.}
\begin{itemize}[itemsep=1.5pt]
\item En el límite inferior de los tipos, el tono se ve en el \textbf{balance} y en M3.
\item Si el nominal sube pero la inflación sube más, el tipo real cae.
\end{itemize}
\graficorepaso[0.97\linewidth]{tipos-balance-bce}{BCE: el tipo de la facilidad de depósito guía la política; el balance crece con las compras de activos y baja desde 2022.}
\graficorepaso[0.92\linewidth]{rep-canales}{Canales de transmisión: funcionan si baja la inflación, se frena el crédito y se aprecia la moneda.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Dominancia fiscal}: el banco central financia el déficit y pierde el control de la inflación.
\item Es \textbf{contracíclica} si sube los tipos con la brecha positiva y los baja con la negativa.
\end{itemize}

\sub{V.2. Estabilidad financiera y macroprudencial\label{rep:V.2}}
\begin{ideaclave}
\textbf{Qué hay que contestar}: ¿es resiliente el sistema? ¿Quién está endeudado? ¿Es la banca solvente, líquida y rentable? ¿Hay burbujas? ¿Qué debe hacer la política macroprudencial?
\end{ideaclave}
\begin{itemize}[itemsep=1.5pt]
\item Orden fijo: \textbf{endeudamiento → banca → riesgos → políticas}, tras una frase de resumen («el sistema parece resiliente, aunque…»).
\item \textbf{Endeudamiento} (\emph{oficial}, PDM): hogares $>55$~\% del PIB; empresas $>85$~\%.
  \begin{itemize}
  \item \textbf{Brecha crédito-PIB} $>2$~pp: activar el colchón anticíclico.
  \item \textbf{Bucle bancario-soberano}: la razón de ser de la unión bancaria.
  \end{itemize}
\item \textbf{Solvencia}: capital sobre \textbf{activos ponderados por riesgo} (APR); apalancamiento (Tier~1 sobre la exposición total) de al menos el 3~\%.
\end{itemize}
\graficorepaso[0.97\linewidth]{rep-basilea}{Basilea~III: mínimos por calidad del capital más colchones (\% de los APR).}
\graficorepaso[0.97\linewidth]{rep-semaforo}{Semáforo de la banca: las debilidades se refuerzan (más dudosos, menos beneficio, menos capital).}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Liquidez}: LCR y NSFR de al menos el 100~\%; una LTD $>100$~\% indica dependencia de la financiación mayorista.
\item \textbf{Riesgos}: vivienda (\emph{oficial}: precio $>+9$~\% anual), auge del crédito, préstamos en divisas, tipos bajos mucho tiempo.
\item \textbf{Regulación}: \textbf{micro} (cada entidad; los 3 pilares de Basilea) y \textbf{macro} (el sistema). Unión bancaria: MUS, MUR y garantía de depósitos pendiente.
\item \textbf{Recomendación tipo}:
  \begin{itemize}
  \item en la fase alcista, colchón anticíclico y límites \textbf{LTV}, \textbf{LTI} y \textbf{DSTI};
  \item en la bajista, liberar colchones y limitar dividendos;
  \item con fragilidad, sanear, recapitalizar y resolver ordenadamente.
  \end{itemize}
\end{itemize}

\sub{V.3. Política fiscal y deuda pública\label{rep:V.3}}
\begin{ideaclave}
\textbf{Qué hay que contestar}: ¿qué tono tiene y es adecuado al ciclo? ¿Es sostenible la deuda? ¿Qué riesgos tiene? ¿Qué exigen las reglas?
\end{ideaclave}
\begin{itemize}[itemsep=1.5pt]
\item Lo recomendable es una política \textbf{neutral}, que deje actuar a los \textbf{estabilizadores automáticos}.
\item \textbf{Los 5 saldos}: $b$ total; $b^p$ primario; $b^{ca}$ ajustado por el ciclo; $b^e$ estructural; $b^{pe}$ estructural primario.
\funcionclave{b^{ca}=b-\varepsilon\cdot\mathrm{OG},\qquad \text{esfuerzo}_t=\Delta b^{pe}_t}
  \begin{itemize}
  \item $\Delta b^{pe}>0$, \textbf{contractiva}; $<0$, \textbf{expansiva}; $\approx0$, neutral.
  \item \textbf{Ojo}: si un saldo negativo se hace más negativo, ha \emph{bajado}: es expansiva.
  \end{itemize}
\end{itemize}
\graficorepaso[0.97\linewidth]{impulso-fiscal}{Impulso fiscal ($-\Delta b^{pe}$, FMI): positivo, política expansiva.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Reglas de la UE} (\emph{oficial}): déficit del 3~\% y deuda del 60~\% del PIB; senda de gasto primario neto en un plan de 4 años (hasta 7).
\item \textbf{Dinámica de la deuda} (efecto bola de nieve):
\funcionclave{d_t-d_{t-1}=(g_t-\tau_t)+\frac{i_t-\gamma^N_t}{1+\gamma^N_t}\,d_{t-1}+\mathrm{ADD}_t}
\end{itemize}
\graficorepaso[0.97\linewidth]{rep-deuda}{Saldo primario que estabiliza la deuda: $\tau-g=\frac{i-\gamma^N}{1+\gamma^N}\,d$.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Inflación y cuentas públicas}: a corto plazo mejoran (ingresos rápidos, PIB nominal mayor); a medio, empeoran (intereses y gastos indexados con retraso).
\end{itemize}
\graficorepaso[0.97\linewidth]{inflacion-cuentas-publicas}{El calendario de los efectos de la inflación en las cuentas públicas.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Estabilizadores automáticos}: $b-b^{ca}=\varepsilon\cdot\mathrm{OG}$ mide cuánto del saldo explica el ciclo; si apenas se separan, son débiles.
\item \textbf{Riesgos}: menos crecimiento, subidas de tipos, envejecimiento, deuda en divisas o a corto, calificación cerca del grado especulativo.
\end{itemize}
\graficorepaso[0.97\linewidth]{rep-calificaciones}{Por debajo de BBB$-$/Baa3 la deuda deja de ser grado de inversión.}
\begin{itemize}[itemsep=1.5pt]
\item \textbf{Recomendación tipo}: consolidación \textbf{gradual, creíble y anunciada pronto}; reglas y techo de gasto; alargar la vida media; reformas que eleven el potencial.
\end{itemize}

\sub{V.4. \emph{Policy mix} y políticas de oferta\label{rep:V.4}}
\graficorepaso[0.97\linewidth]{policy-mix}{\emph{Policy mix}: es coherente si cada política ayuda a la otra, no si van en el mismo sentido.}
\begin{itemize}[itemsep=1.5pt]
\item En una \textbf{unión monetaria} solo queda la fiscal, que compensa si el ciclo del país se separa del de la zona.
\item \textbf{Oferta}, según el cuello de botella: I+D, capital, capital humano, competencia e instituciones.
\item \textbf{Paro}: la receta, según su tipo (cuadro de \hyperref[rep:II.1]{II.1}); en todos, políticas activas bien focalizadas y prestaciones ligadas a la búsqueda de empleo.
\end{itemize}
