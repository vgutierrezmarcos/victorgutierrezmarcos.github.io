% Mundell-Fleming: expansión fiscal con tipo de cambio flexible y movilidad
% imperfecta del capital: alta (izquierda) y baja (derecha). Plano (Y, i);
% IS azul, LM morado, BP verde. La LM no se mueve (el banco central controla
% M); el tipo de cambio desplaza la IS y la BP hasta cortarse sobre la LM.
%   IS: i = 2,5 - (Y - 3); IS': i = 2,5 - (Y - 4,5); LM: i = 2,5 + 0,8 (Y - 3)
%   BP alta: i = 2,5 + 0,35 (Y - 3);  BP baja: i = 2,5 + 1,6 (Y - 3)
%   B = IS' ∩ LM = (3,833; 3,167)
%   Alta (apreciación): C = (3,4; 2,82) sobre la LM; IS'' y BP' pasan por C
%   Baja (depreciación): C = (4,4; 3,62) sobre la LM; IS'' y BP' pasan por C

      isa(\y)=2.82-(\y-3.4); bpan(\y)=2.82+0.35*(\y-3.4);

  \capa{2}{\draw[curva3] (0.3,{bpa(0.3)}) -- (6.6,{bpa(6.6)}) node[etiqueta,right,text=tceeverde] {BP};}
