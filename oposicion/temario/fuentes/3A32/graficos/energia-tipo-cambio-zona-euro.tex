% El petróleo en dólares y en euros y el tipo de cambio del euro, medias
% mensuales, enero de 2019-octubre de 2026.
% Datos: ../datos/energia-tipo-cambio-zona-euro-2019-2026.csv (EIA, Brent
% RBRTE; BCE, EXR.M.USD.EUR.SP00.A; brent_eur = brent_usd / usd_por_eur;
% consulta 11/11/2026). Eje x: n.º de mes desde enero de 2019 (fila del CSV):
% 2019-01 = 0, 2021-01 = 24, 2022-06 = 41, 2026-10 = 93.
% Cifras de control: ene. 2021, 54,77 $ y 45,00 €; jun. 2022, 122,71 $ y
% 116,14 € (+124 % y +158 %); oct. 2026, 123,93 $ y 110,64 €; mínimo del
% tipo de cambio, 0,9826 $/€ (oct. 2022).
% Paleta «dictamen»: el Brent en dólares, en el bermellón de la energía (como
% brent del 1.D.1); en euros, el mismo tono oscurecido (familia de la
% energía); el tipo de cambio, en el azul de la zona del euro, fino.
% Capas (CRITERIOS §7): banda → ejes y paridad → líneas → puntos → etiquetas
% y leyenda.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\colorlet{dicenergiaeur}{dicenergia!55!black}
\begin{document}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/energia-tipo-cambio-zona-euro-2019-2026.csv}\datos
\pgfplotsset{
  leyenda linea corta/.style={legend image code/.code={%
    \draw[##1] (0cm,0cm) -- (0.6cm,0cm);}},
  comun/.style={width=14cm, height=7cm, xmin=-1, xmax=95,
    xtick={0,12,...,84}, xticklabels={2019,2020,2021,2022,2023,2024,2025,2026},
    minor xtick={6,18,...,90},
    ticklabel style={font=\footnotesize}, label style={font=\footnotesize},
    tick align=outside, clip=false,
    /pgf/number format/use comma, /pgf/number format/1000 sep={}},
}
\begin{tikzpicture}[tcee]
  % ---------------------------------------------- eje izquierdo: el Brent
  \begin{axis}[comun,
      ymin=0, ymax=140, ytick={0,20,...,140},
      axis x line*=bottom, axis y line*=left,
      ylabel={Dólares o euros por barril},
      ymajorgrids, grid style={tceegris!25},
      legend style={at={(0.5,-0.13)}, anchor=north, legend columns=3,
        draw=none, font=\footnotesize, /tikz/every even column/.append style={column sep=10pt}},
      legend cell align=left, dicapas]
    % 1: banda: el enganche de enero de 2021 a junio de 2022
    \capa{1}{
      \dicsombra[fill=dicotromasclaro]{24}{0}{41}{140}
    }
    % 2: la paridad (1 dólar por euro: 35 en esta escala, 1,0 en la de la derecha)
    \capa{2}{
      \dicraya[dicumbral, thick]{(-1,35) (95,35)}
      \dicrotulo[font=\scriptsize, text=tceegris, anchor=south west]{(52,35.5)}{paridad (1\,\$ por 1\,€)}
    }
    % 3: Brent en dólares
    \capa{3}{
      \addplot[diclinea=dicenergia, forget plot] table[x expr=\coordindex, y=brent_usd] {\datos};
    }
    % 4: Brent en euros
    \capa{4}{
      \addplot[diclinea=dicenergiaeur, forget plot] table[x expr=\coordindex, y=brent_eur] {\datos};
    }
    % 6: junio de 2022 y octubre de 2026
    \capa{6}{
      \addplot[on layer=dicpuntos, dicpuntos=dicenergia, forget plot] coordinates {(41,122.71) (93,123.93)};
      \addplot[on layer=dicpuntos, dicpuntos=dicenergiaeur, forget plot] coordinates {(41,116.14) (93,110.64)};
    }
    % 7: etiquetas y leyenda
    \capa{7}{
      \dicrotulo[font=\scriptsize, text=tceegris, anchor=south, align=center]{(32.5,3)}{ene.\ 2021-jun.\ 2022:\\\textcolor{dicenergia}{+124\,\% en dólares}\\\textcolor{dicenergiaeur}{+158\,\% en euros}}
      \dicrotulo[font=\footnotesize, anchor=south, align=center]{(41,126)}{jun.\ 2022\\\textcolor{dicenergia}{122,7\,\$} y \textcolor{dicenergiaeur}{116,1\,€}}
      \dicrotulo[font=\footnotesize, anchor=south, align=center]{(85.5,127)}{oct.\ 2026\\\textcolor{dicenergia}{123,9\,\$} y \textcolor{dicenergiaeur}{110,6\,€}}
      \addlegendimage{diclinea=dicenergia, leyenda linea corta}
      \addlegendentry{Brent, dólares por barril}
      \addlegendimage{diclinea=dicenergiaeur, leyenda linea corta}
      \addlegendentry{Brent, euros por barril}
      \addlegendimage{diclinea=dicze, thick, leyenda linea corta}
      \addlegendentry{Dólares por euro (eje derecho)}
    }
  \end{axis}
  % ---------------------------------------------- eje derecho: dólares por euro
  \begin{axis}[comun,
      ymin=0.9, ymax=1.3, ytick={0.9,1.0,...,1.3},
      yticklabel style={/pgf/number format/fixed, /pgf/number format/precision=1,
        /pgf/number format/fixed zerofill},
      axis y line*=right, axis x line=none,
      ylabel={Dólares por euro}, ylabel style={text=dicze},
      dicapas]
    % 5: dólares por euro
    \capa{5}{
      \addplot[diclinea=dicze, thick, forget plot] table[x expr=\coordindex, y=usd_por_eur] {\datos};
    }
    % 7: mínimo del euro
    \capa{7}{
      \dicrotulo[font=\scriptsize, text=dicze, anchor=north west]{(45.6,0.982)}{0,98\,\$/€ (oct.\ 2022)}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
