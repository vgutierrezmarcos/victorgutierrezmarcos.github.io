% Mercado de factores del H-O-S (3.B.5, III.2): K/L en abscisas, w/r en ordenadas.
% Elaboración propia (cualitativa), a partir de Krugman, Obstfeld y Melitz
% (2018), cap. 5. Relaciones capital-trabajo óptimas de cada sector (CES, σ = 0,6):
%   (K/L)_X = 1,8 (w/r)^0,6 (X intensivo en capital, a la derecha),
%   (K/L)_Y = 0,8 (w/r)^0,6.
% Economía: K/L = λ (K/L)_X + (1 - λ)(K/L)_Y, con λ = parte del trabajo en X.
% Dotación K̄/L̄ = 2,4.
%   Antes (λ = 0,4): K/L = 1,2 (w/r)^0,6 = 2,4 -> (w/r)_0 = 2^(1/0,6) = 3,175.
%   A se abre y produce más X (λ = 0,7): K/L = 1,5 (w/r)^0,6 = 2,4 ->
%   (w/r)_1 = 1,6^(1/0,6) = 2,190: w/r baja.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,x=1.1cm,y=1.1cm]
  % 1: ejes
  \capa{1}{\ejes{$K/L$}{$w/r$}{5.6}{4.8}}
  % 2: relaciones capital-trabajo de cada sector, crecientes en w/r
  \capa{2}{\draw[curva3] plot[domain=0.25:4.4,samples=80] ({0.8*\x^0.6},\x)
    node[etiqueta,above] {$(K/L)_Y$};}
  \capa{2}{\draw[curva1] plot[domain=0.25:3.0,samples=80] ({1.8*\x^0.6},\x)
    node[etiqueta,right] {$(K/L)_X$};}
  % 3: dotación de la economía
  \capa{3}{\draw[thick,black] (2.4,0) node[below] {$\overline K/\overline L$} -- (2.4,4.5);}
  % 4: demanda conjunta (media ponderada) y w/r de equilibrio
  \capa{4}{\draw[curva2] plot[domain=0.25:4.4,samples=80] ({1.2*\x^0.6},\x)
    node[etiqueta,right] {$K/L$ (economía)};}
  \capa{4}{\node[punto] (E0) at (2.4,3.175) {};}
  \capa{4}{\draw[guia] (E0) -- (0,3.175) node[left,text=black] {$(w/r)_0$};}
  % 5: al abrirse A produce más X: la demanda conjunta se acerca a la de X y w/r baja
  \capa{5}{\draw[curvanueva,desplazada] plot[domain=0.25:3.6,samples=80] ({1.5*\x^0.6},\x)
    node[etiqueta,right] {$K/L'$};}
  \capa{5}{\node[punto] (E1) at (2.4,2.190) {};}
  \capa{5}{\draw[guia] (E1) -- (0,2.190) node[left,text=black] {$(w/r)_1$};}
  \capa{5}{\draw[flecha] (-1.3,3.1) -- (-1.3,2.26);}
  \capa{5}{\node[etiqueta,text=tceerojo,font=\footnotesize,align=left,anchor=west] at (3.4,1.0) {apertura de $A$:\\[-1pt]más producción de $X$};}
\end{tikzpicture}
\end{document}
