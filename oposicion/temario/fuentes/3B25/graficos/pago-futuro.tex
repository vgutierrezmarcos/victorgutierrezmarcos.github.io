% Pago del futuro a vencimiento: comprador S_T - F_0 (izquierda) y vendedor
% F_0 - S_T (derecha). Escala: 1 cm = 40 euros en los 2 ejes (pendiente 1).
% F_0 = 100 (x = 2,5). Ejemplo: S_T = 140 (x = 3,5) da +40 al comprador y
% -40 al vendedor; S_T = 60 (x = 1,5), al revés. Suma cero.
\begin{tikzpicture}[tcee,scale=0.85,
    gana/.style={fill=tceeverde,fill opacity=0.15,draw=none},
    pierde/.style={fill=tceerojo,fill opacity=0.13,draw=none},
    formula/.style={draw=tceemorado,rounded corners=2pt,inner sep=4pt,font=\small}]
  % 1: panel izquierdo, ejes y título
  \capa{1}{\draw[eje] (0,0) -- (5.6,0) node[below left] {$S_T$};}
  \capa{1}{\draw[eje] (0,-2.9) -- (0,3) node[left] {Pago};}
  \capa{1}{\node[etiqueta,font=\normalsize] at (2.8,3.9) {Comprador (largo)};}
  \capa{1}{\node[formula] at (2.8,3.25) {Pago $=S_T-F_0$, con $F_0=100$};}
  % 2: zonas de ganancia y de pérdida
  \capa{2}{\fill[gana] (2.5,0) -- (5,0) -- (5,2.5) -- cycle;}
  \capa{2}{\fill[pierde] (0,0) -- (2.5,0) -- (0,-2.5) -- cycle;}
  % 3: recta de pendiente +1 que corta en F_0
  \capa{3}{\draw[curva1] (0,-2.5) -- (5,2.5) node[etiqueta,pos=0.84,above,sloped,text=tceemorado,font=\footnotesize] {pendiente $+1$};}
  \capa{3}{\node[punto] at (2.5,0) {};}
  \capa{3}{\node[etiqueta,above left] at (2.5,0) {$F_0$};}
  \capa{3}{\node[etiqueta,left] at (0,-2.5) {$-F_0$};}
  % 4: ejemplo: S_T = 140 y S_T = 60
  \capa{4}{\draw[guia] (3.5,0) -- (3.5,1) -- (0,1);}
  \capa{4}{\node[punto] at (3.5,1) {};}
  \capa{4}{\node[etiqueta,below,font=\footnotesize] at (3.5,0) {$140$};}
  \capa{4}{\node[etiqueta,left,text=tceeverde] at (0,1) {$+40$};}
  \capa{4}{\draw[guia] (1.5,0) -- (1.5,-1) -- (0,-1);}
  \capa{4}{\node[punto] at (1.5,-1) {};}
  \capa{4}{\node[etiqueta,above,font=\footnotesize] at (1.5,0) {$60$};}
  \capa{4}{\node[etiqueta,left,text=tceerojo] at (0,-1) {$-40$};}
  % 5: lectura de las zonas
  \capa{5}{\node[etiqueta,text=tceeverde,align=center,font=\footnotesize] at (4.55,0.75) {gana si\\$S_T>F_0$;\\sin límite};}
  \capa{5}{\node[etiqueta,text=tceerojo,align=center,font=\footnotesize] at (2.75,-1.7) {pierde si $S_T<F_0$;\\como mucho $F_0$};}
  \begin{scope}[xshift=7.4cm]
    % 6: panel derecho, ejes y título
    \capa{6}{\draw[eje] (0,0) -- (5.6,0) node[above left] {$S_T$};}
    \capa{6}{\draw[eje] (0,-2.9) -- (0,3) node[left] {Pago};}
    \capa{6}{\node[etiqueta,font=\normalsize] at (2.8,3.9) {Vendedor (corto)};}
    \capa{6}{\node[formula] at (2.8,3.25) {Pago $=F_0-S_T$, con $F_0=100$};}
    % 7: zonas
    \capa{7}{\fill[gana] (0,0) -- (2.5,0) -- (0,2.5) -- cycle;}
    \capa{7}{\fill[pierde] (2.5,0) -- (5,0) -- (5,-2.5) -- cycle;}
    % 8: recta de pendiente -1
    \capa{8}{\draw[curva2] (0,2.5) -- (5,-2.5) node[etiqueta,pos=0.84,below,sloped,text=tceeazul,font=\footnotesize] {pendiente $-1$};}
    \capa{8}{\node[punto] at (2.5,0) {};}
    \capa{8}{\node[etiqueta,below left] at (2.5,0) {$F_0$};}
    \capa{8}{\node[etiqueta,left] at (0,2.5) {$F_0$};}
    % 9: el mismo ejemplo, con el signo contrario
    \capa{9}{\draw[guia] (3.5,0) -- (3.5,-1) -- (0,-1);}
    \capa{9}{\node[punto] at (3.5,-1) {};}
    \capa{9}{\node[etiqueta,above,font=\footnotesize] at (3.5,0) {$140$};}
    \capa{9}{\node[etiqueta,left,text=tceerojo] at (0,-1) {$-40$};}
    \capa{9}{\draw[guia] (1.5,0) -- (1.5,1) -- (0,1);}
    \capa{9}{\node[punto] at (1.5,1) {};}
    \capa{9}{\node[etiqueta,below,font=\footnotesize] at (1.5,0) {$60$};}
    \capa{9}{\node[etiqueta,left,text=tceeverde] at (0,1) {$+40$};}
    % 10: lectura de las zonas
    \capa{10}{\node[etiqueta,text=tceeverde,align=center,font=\footnotesize] at (2.75,1.7) {gana si $S_T<F_0$;\\como mucho $F_0$};}
    \capa{10}{\node[etiqueta,text=tceerojo,align=center,font=\footnotesize] at (4.55,-0.75) {pierde si\\$S_T>F_0$;\\sin límite};}
  \end{scope}
  % 11: suma cero entre los 2 paneles
  \capa{11}{\node[etiqueta,align=center,font=\footnotesize,text=tceemorado] at (6.55,-3.75)
    {Suma cero: $(S_T-F_0)+(F_0-S_T)=0$ para cualquier $S_T$. Si $S_T=140$, el comprador gana 40 y el vendedor pierde 40.};}
\end{tikzpicture}
