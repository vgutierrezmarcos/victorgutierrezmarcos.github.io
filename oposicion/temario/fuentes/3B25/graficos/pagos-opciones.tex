% Infografía de página entera: pago (gris discontinuo) y beneficio (color) a
% vencimiento de las 4 posiciones simples, con las fórmulas de cada una debajo
% y la suma cero entre comprador y vendedor. K = 2,4; c = p = 0,9.
% Umbrales: K+c = 3,3 y K-p = 1,5.
\begin{tikzpicture}[tcee,scale=0.75,
    pago/.style={thick,dashed,tceegris},
    ficha/.style={draw=#1,thick,rounded corners=3pt,fill=#1!6,align=left,
      text width=6.6cm,inner sep=5pt,font=\small,anchor=north},
    espejo/.style={<->,very thick,tceegris}]
  % #1 título; #2 posición del rótulo K
  \newcommand{\panel}[2]{%
    \draw[eje] (0,0) -- (5.9,0) node[below left] {$S_T$};
    \draw[eje] (0,-3) -- (0,3.1) node[left,font=\footnotesize,align=right] {Pago /\\beneficio};
    \node[etiqueta,font=\large] at (2.9,3.7) {#1};
    \draw (2.4,0.08) -- (2.4,-0.08);
    \node[#2] at (2.4,0) {$K$};}
  % 1: comprador de call
  \capa{1}{\panel{Comprador de \emph{call}}{below}}
  % 2: pago, beneficio, -c y umbral K+c
  \capa{2}{\draw[pago] (0,0) -- (2.4,0) -- (5.2,2.8);
    \draw[curva1] (0,-0.9) -- (2.4,-0.9) -- (5.2,1.9);
    \node[etiqueta,left] at (0,-0.9) {$-c$};
    \node[punto] at (3.3,0) {};
    \node[etiqueta,below right] at (3.3,0) {$K+c$};}
  % 3: fórmulas del comprador de call
  \capa{3}{\node[ficha=tceemorado] at (2.9,-3.5) {%
    \textbf{Pago}: $\max(S_T-K,\,0)$\\
    \textbf{Beneficio}: $\max(S_T-K,\,0)-c$\\
    Gana si $S_T>K+c$\\
    Pierde como mucho $c$\\Gana sin límite\\
    \emph{Expectativa}: \textcolor{tceeverde}{alcista}};}
  \begin{scope}[xshift=9.6cm]
    % 4: vendedor de call
    \capa{4}{\panel{Vendedor de \emph{call}}{above}}
    % 5: pago, beneficio, c y umbral K+c
    \capa{5}{\draw[pago] (0,0) -- (2.4,0) -- (5.2,-2.8);
      \draw[curva2] (0,0.9) -- (2.4,0.9) -- (5.2,-1.9);
      \node[etiqueta,left] at (0,0.9) {$c$};
      \node[punto] at (3.3,0) {};
      \node[etiqueta,above right] at (3.3,0) {$K+c$};}
    % 6: fórmulas del vendedor de call
    \capa{6}{\node[ficha=tceeazul] at (2.9,-3.5) {%
      \textbf{Pago}: $-\max(S_T-K,\,0)$\\
      \textbf{Beneficio}: $c-\max(S_T-K,\,0)$\\
      Gana si $S_T<K+c$\\
      Gana como mucho $c$\\Pierde sin límite\\
      \emph{Expectativa}: \textcolor{tceerojo}{bajista} o de calma};}
  \end{scope}
  % 7: suma cero entre comprador y vendedor de la call
  \capa{7}{\draw[espejo] (6.4,1.2) -- (9.0,1.2);
    \node[etiqueta,align=center,font=\footnotesize,text=tceegris] at (7.7,0.4) {suma cero:\\lo que gana uno\\lo pierde el otro};}
  \begin{scope}[yshift=-11.2cm]
    % 8: comprador de put
    \capa{8}{\panel{Comprador de \emph{put}}{below}}
    % 9: pago, beneficio, -p, máximo K-p y umbral K-p
    \capa{9}{\draw[pago] (0,2.4) -- (2.4,0) -- (5.2,0);
      \draw[curva1] (0,1.5) -- (2.4,-0.9) -- (5.2,-0.9);
      \node[etiqueta,left] at (0,1.5) {$K-p$};
      \node[etiqueta,left] at (0,-0.9) {$-p$};
      \node[punto] at (1.5,0) {};
      \node[etiqueta,below left] at (1.5,0) {$K-p$};}
    % 10: fórmulas del comprador de put
    \capa{10}{\node[ficha=tceemorado] at (2.9,-3.5) {%
      \textbf{Pago}: $\max(K-S_T,\,0)$\\
      \textbf{Beneficio}: $\max(K-S_T,\,0)-p$\\
      Gana si $S_T<K-p$\\
      Pierde como mucho $p$\\Gana como mucho $K-p$\\
      \emph{Expectativa}: \textcolor{tceerojo}{bajista}};}
  \end{scope}
  \begin{scope}[xshift=9.6cm,yshift=-11.2cm]
    % 11: vendedor de put
    \capa{11}{\panel{Vendedor de \emph{put}}{above}}
    % 12: pago, beneficio, p, pérdida máxima -(K-p) y umbral K-p
    \capa{12}{\draw[pago] (0,-2.4) -- (2.4,0) -- (5.2,0);
      \draw[curva2] (0,-1.5) -- (2.4,0.9) -- (5.2,0.9);
      \node[etiqueta,left] at (0,0.9) {$p$};
      \node[etiqueta,left] at (0,-1.5) {$-(K-p)$};
      \node[punto] at (1.5,0) {};
      \node[etiqueta,above left] at (1.5,0) {$K-p$};}
    % 13: fórmulas del vendedor de put
    \capa{13}{\node[ficha=tceeazul] at (2.9,-3.5) {%
      \textbf{Pago}: $-\max(K-S_T,\,0)$\\
      \textbf{Beneficio}: $p-\max(K-S_T,\,0)$\\
      Gana si $S_T>K-p$\\
      Gana como mucho $p$\\Pierde como mucho $K-p$\\
      \emph{Expectativa}: \textcolor{tceeverde}{alcista} o de calma};}
  \end{scope}
  % 14: suma cero entre comprador y vendedor de la put
  \capa{14}{\draw[espejo] (6.4,-10.0) -- (9.0,-10.0);
    \node[etiqueta,align=center,font=\footnotesize,text=tceegris] at (7.7,-10.8) {suma cero:\\lo que gana uno\\lo pierde el otro};}
  % 15: leyenda común
  \capa{15}{\node[etiqueta,align=center,font=\footnotesize] at (7.7,-19.6) {%
    \tikz\draw[thick,dashed,tceegris] (0,0) -- (0.6,0); pago (lo que da el contrato en $T$)\qquad
    \tikz\draw[very thick,tceemorado] (0,0) -- (0.6,0); \tikz\draw[very thick,tceeazul] (0,0) -- (0.6,0); beneficio (pago menos la prima pagada, o más la cobrada)};}
\end{tikzpicture}
