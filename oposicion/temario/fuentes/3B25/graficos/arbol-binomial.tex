% Árbol binomial de un periodo (mismo estilo de ramas y rótulos que el árbol
% de opciones reales del 3.B.2). Acción: S_0 = 100, u = 1,25, d = 0,8.
% Call con K = 110: c_u = 15, c_d = 0. r = 10 %: pi = (1,1-0,8)/0,45 = 2/3,
% c = (2/3 * 15)/1,1 = 9,09; Delta = 15/45 = 1/3.
\begin{tikzpicture}[tcee,
    nodo/.style={draw=tceemorado,thick,rounded corners=2pt,minimum width=1.7cm,minimum height=0.75cm,font=\small},
    nodoc/.style={nodo,draw=tceedorado},
    rama/.style={->,thick},
    sobre/.style={font=\small,above,sloped,inner sep=2pt},
    bajo/.style={font=\small,below,sloped,inner sep=2pt,yshift=-9pt}]
  % 1: árbol de la acción, con t = 0 y t = 1
  \capa{1}{\node[nodo] (S0) at (0,0) {$S_0=100$};}
  \capa{1}{\node[nodo] (Su) at (3.4,1.4) {$uS_0=125$};}
  \capa{1}{\node[nodo] (Sd) at (3.4,-1.4) {$dS_0=80$};}
  \capa{1}{\draw[rama] (S0) -- (Su);}
  \capa{1}{\draw[rama] (S0) -- (Sd);}
  \capa{1}{\node[etiqueta,text=tceegris] at (0,-2.4) {$t=0$};}
  \capa{1}{\node[etiqueta,text=tceegris] at (3.4,-2.4) {$t=1$};}
  \capa{1}{\node[etiqueta,font=\footnotesize,text=tceegris] at (1.7,2.5) {acción};}
  \begin{scope}[xshift=7cm]
    % 2: árbol de la call (K = 110)
    \capa{2}{\node[nodoc] (c0) at (0,0) {$c=\,?$};}
    \capa{2}{\node[nodoc] (cu) at (3.4,1.4) {$c_u=15$};}
    \capa{2}{\node[nodoc] (cd) at (3.4,-1.4) {$c_d=0$};}
    \capa{2}{\draw[rama] (c0) -- (cu);}
    \capa{2}{\draw[rama] (c0) -- (cd);}
    \capa{2}{\node[etiqueta,text=tceegris] at (0,-2.4) {$t=0$};}
    \capa{2}{\node[etiqueta,text=tceegris] at (3.4,-2.4) {$t=1$};}
    \capa{2}{\node[etiqueta,font=\footnotesize,text=tceegris] at (1.7,2.5) {\emph{call} con $K=110$};}
    % 3: probabilidades neutrales al riesgo
    \capa{3}{\path (c0) -- (cu) node[sobre,pos=0.5] {$\pi=2/3$};}
    \capa{3}{\path (c0) -- (cd) node[bajo,pos=0.58] {$1-\pi=1/3$};}
    % 4: resultado
    \capa{4}{\node[etiqueta,text=tceemorado,anchor=north] at (0,-0.6) {$c\approx9{,}09$};}
    \capa{4}{\node[etiqueta,text=tceemorado,anchor=north] at (0,-1.1) {$\Delta=1/3$};}
  \end{scope}
\end{tikzpicture}
