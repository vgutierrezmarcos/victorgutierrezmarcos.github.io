% Árbol binomial de un periodo (mismo estilo de ramas y rótulos que el árbol
% de opciones reales del 3.B.2). Acción: S_0 = 100, u = 1,25, d = 0,8.
% Call con K = 110: c_u = max(125-110,0) = 15, c_d = max(80-110,0) = 0.
% r = 10 %: pi = (1,1-0,8)/0,45 = 2/3; con pi la acción rinde r:
% 2/3*125 + 1/3*80 = 110 = 100*1,1. c = (2/3*15 + 1/3*0)/1,1 = 9,09;
% Delta = 15/45 = 1/3.
\begin{tikzpicture}[tcee,
    nodo/.style={draw=tceemorado,thick,rounded corners=2pt,minimum width=1.7cm,minimum height=0.75cm,font=\small},
    nodoc/.style={nodo,draw=tceedorado},
    rama/.style={->,thick},
    sobre/.style={font=\small,above,sloped,inner sep=2pt},
    bajo/.style={font=\small,below,sloped,inner sep=2pt},
    cuenta/.style={etiqueta,font=\footnotesize,align=center,anchor=north}]
  % 1: árbol de la acción, con t = 0 y t = 1
  \capa{1}{\node[nodo] (S0) at (0,0) {$S_0=100$};}
  \capa{1}{\node[nodo] (Su) at (3.4,1.4) {$uS_0=125$};}
  \capa{1}{\node[nodo] (Sd) at (3.4,-1.4) {$dS_0=80$};}
  \capa{1}{\draw[rama] (S0) -- (Su);}
  \capa{1}{\draw[rama] (S0) -- (Sd);}
  \capa{1}{\node[etiqueta,text=tceegris] at (0,-2.25) {$t=0$};}
  \capa{1}{\node[etiqueta,text=tceegris] at (3.4,-2.25) {$t=1$};}
  \capa{1}{\node[etiqueta,font=\normalsize] at (1.7,2.6) {Acción};}
  % 2: las mismas probabilidades neutrales al riesgo en el árbol de la acción
  \capa{2}{\path (S0) -- (Su) node[sobre,pos=0.5] {$\pi=2/3$};}
  \capa{2}{\path (S0) -- (Sd) node[bajo,pos=0.4] {$1-\pi=1/3$};}
  % 3: comprobación: con pi la acción rinde el tipo sin riesgo
  \capa{3}{\node[cuenta,text=tceemorado] at (1.7,-2.75)
    {$\pi=\dfrac{(1+r)-d}{u-d}=\dfrac{1{,}1-0{,}8}{1{,}25-0{,}8}=\dfrac{2}{3}$\\[6pt]
     $\tfrac23\cdot125+\tfrac13\cdot80=110=100\cdot(1+r)$\\[1pt]
     con $\pi$, la acción rinde el tipo sin riesgo};}
  \begin{scope}[xshift=7.6cm]
    % 4: árbol de la call (K = 110), con el cálculo de cada pago
    \capa{4}{\node[nodoc] (c0) at (0,0) {$c=\,?$};}
    \capa{4}{\node[nodoc,minimum width=3.6cm] (cu) at (4.2,1.4) {$c_u=\max(125-110,\,0)=15$};}
    \capa{4}{\node[nodoc,minimum width=3.6cm] (cd) at (4.2,-1.4) {$c_d=\max(80-110,\,0)=0$};}
    \capa{4}{\draw[rama] (c0) -- (cu);}
    \capa{4}{\draw[rama] (c0) -- (cd);}
    \capa{4}{\node[etiqueta,text=tceegris] at (0,-2.25) {$t=0$};}
    \capa{4}{\node[etiqueta,text=tceegris] at (4.2,-2.25) {$t=1$};}
    \capa{4}{\node[etiqueta,font=\normalsize] at (2.1,2.6) {\emph{Call} con $K=110$};}
    % 5: probabilidades neutrales al riesgo
    \capa{5}{\path (c0) -- (cu) node[sobre,pos=0.36] {$\pi=2/3$};}
    \capa{5}{\path (c0) -- (cd) node[bajo,pos=0.34] {$1-\pi=1/3$};}
    % 6: resultado, por las 2 vías
    \capa{6}{\node[cuenta,text=tceemorado] at (2.1,-2.75)
      {$c=\dfrac{\tfrac23\cdot15+\tfrac13\cdot0}{1{,}1}\approx9{,}09$\\[6pt]
       réplica: $\Delta=\dfrac{15-0}{125-80}=\dfrac13$ de acción y una deuda de $24{,}24$\\[1pt]
       $\tfrac13\cdot100-24{,}24\approx9{,}09$};}
  \end{scope}
\end{tikzpicture}
\end{document}
