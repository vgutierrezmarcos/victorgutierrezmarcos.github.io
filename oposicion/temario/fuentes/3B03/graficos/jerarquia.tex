% Jerarquía financiera (Myers y Majluf, 1984; Myers, 1984): la escalera de la
% oferta de fondos. Adaptado de fuentes/3A34/graficos/coste-fondos.tex (misma
% escala, colores y demandas) para que se reconozca igual.
%   fondos internos: coste 1,5 hasta I = 3;
%   deuda: 2,2 + 0,1 (I - 3) hasta la capacidad de endeudamiento, I = 6,5 (2,55);
%   acciones nuevas: 3,6 + 0,06 (I - 6,5) (descuento por selección adversa).
%   D_1: 3,7 - I      -> corta los fondos internos en I_1 = 2,2
%   D_2: 5,6 - 0,5 I  -> corta la deuda en I_2 = 3,7/0,6 = 6,167 (2,517)
% Tramos rotulados por fuente (sin r ni RN del 3.A.34, para no mezclar notación).

  \path (-1.3,-0.9) rectangle (11.6,5.4);
  % 1: ejes: fondos y coste de los fondos
  \capa{1}{\ejes{fondos, $I$}{coste de los fondos}{9.6}{5.2}}
  % 2: fondos internos (beneficios retenidos): los más baratos
  \capa{2}{\draw[curva1] (0,1.5) -- (3,1.5);}
  \capa{2}{\node[etiqueta,below,text=tceemorado,font=\footnotesize,align=center] at (0.85,1.5) {fondos\\internos};}

  \capa{3}{\draw[curva1] (3,1.5) -- (3,2.2) -- (6.5,2.55);}
  \capa{3}{\node[etiqueta,above,text=tceemorado,font=\footnotesize,rotate=5.7] at (4.75,2.375) {deuda};}
  % 4: acciones nuevas: el salto del descuento por selección adversa
  \capa{4}{\draw[curva1] (6.5,2.55) -- (6.5,3.6) -- (9.3,3.768);}
  \capa{4}{\node[etiqueta,above=2pt,text=tceemorado,font=\footnotesize,rotate=3.4] at (7.9,3.72) {acciones nuevas};}
  \capa{4}{\node[etiqueta,right,text=tceemorado,align=left] at (9.3,3.768) {oferta de\\fondos};}

  \capa{5}{\draw[curva2] (0.4,{dA(0.4)}) -- (3.15,{dA(3.15)}) node[etiqueta,below right=-2pt and 0pt,text=tceeazul] {$D_1$};}

  \capa{5}{\draw[guia] (A) -- (2.2,0) node[below,text=black] {$I_1$};}
  % 6: una empresa con más proyectos: llega a la deuda
  \capa{6}{\draw[curva3] (1.6,{dB(1.6)}) -- (9.6,{dB(9.6)}) node[etiqueta,right,text=tceeverde] {$D_2$};}
  \capa{6}{\node[punto] (B) at (6.1667,2.5167) {};}
