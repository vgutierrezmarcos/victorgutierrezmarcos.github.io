\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,
    caja/.style={draw=#1,thick,rounded corners=2pt,fill=#1!12,align=center,
      text width=3.9cm,minimum height=8mm,inner sep=3pt},
    paso/.style={->,thick,#1},
    impuesto/.style={etiqueta,font=\footnotesize,align=#1}]
  % 1: raíz
  \capa{1}{\node[caja=black,fill=tceesalvia] (R) at (0,0) {1~€ de resultado de explotación};}
  % 2: rama de la deuda
  \capa{2}{\node[caja=tceeverde] (D1) at (-3,-1.6) {pagado como interés:\\ $1$};}
  \capa{2}{\draw[paso=tceeverde] (R.south) -- (D1.north);}
  \capa{2}{\node[caja=tceeverde] (D2) at (-3,-6.4) {queda\\ $1-t_D$};}
  \capa{2}{\draw[paso=tceeverde] (D1.south) -- node[impuesto=right,left=3pt,text=tceeverde] {IRPF de los\\intereses, $t_D$} (D2.north);}
  % 3: rama de las acciones
  \capa{3}{\node[caja=tceemorado] (E1) at (3,-1.6) {pagado a los accionistas:\\ $1$};}
  \capa{3}{\draw[paso=tceemorado] (R.south) -- (E1.north);}
  \capa{3}{\node[caja=tceemorado] (E2) at (3,-4) {$1-t_{IS}$};}
  \capa{3}{\draw[paso=tceemorado] (E1.south) -- node[impuesto=left,right=3pt,text=tceemorado] {impuesto sobre\\sociedades, $t_{IS}$} (E2.north);}
  \capa{3}{\node[caja=tceemorado] (E3) at (3,-6.4) {queda\\ $(1-t_{IS})(1-t_E)$};}
  \capa{3}{\draw[paso=tceemorado] (E2.south) -- node[impuesto=left,right=3pt,text=tceemorado] {IRPF de las\\acciones, $t_E$} (E3.north);}
  % 4: comparación
  \capa{4}{\node[etiqueta,align=center] at (0,-7.9) {la deuda conviene si\quad $1-t_D>(1-t_{IS})(1-t_E)$};}
\end{tikzpicture}
