\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: «quien parte y reparte se queda con la mejor parte».
  \begin{itemize}
  \item 5/8/2024: un juez de EE.\,UU.: «Google es un monopolista» (89,2\,\% de las búsquedas).
  \item ¿Se queda siempre con la mejor parte quien vende solo? ¿Qué le limita? ¿Qué le queda al resto?
  \end{itemize}
\item \textbf{Relevancia}:
  \begin{itemize}
  \item teórica: caso más sencillo de poder de mercado (IMg, \autor{Lerner});
  \item empírica: margen sobre el CMg en EE.\,UU., 21\,\% (1980) $\to$ 61\,\% (2016);
  \item política: competencia \vertema{3.B.39}, regulación \vertema{3.A.20}, salario mínimo \vertema{3.A.25}.
  \end{itemize}
\item \textbf{Contextualización}: \autor{Cournot} (1838) $\to$ \autor{Marshall}, \autor{Pigou}, \autor{Robinson}, \autor{Lerner} $\to$ \autor{Harberger}, \autor{Coase}, \autor{Baumol} $\to$ información asimétrica.
  \begin{itemize}
  \item \concepto{Monopolio}: único vendedor de un bien sin sustitutivos cercanos, protegido por barreras.
  \item \concepto{Poder de mercado}: fijar $p>\mathrm{CMg}$ sin perder todas las ventas $\Rightarrow$ falla el 1TFEB \vertema{3.A.22}.
  \end{itemize}
\item \textbf{Problemática}: ¿cómo fija $p$ y $q$ y cuánto cuesta? ¿Qué cambia si discrimina, si la tecnología pide una empresa o si vende varios bienes? ¿Y si el poder está en quien compra, o en los 2~lados?
\item \textbf{Estructura}: \hyperref[rep:I]{I}, monopolio básico (vara de medir); \hyperref[rep:II]{II}, se levantan supuestos; \hyperref[rep:III]{III}, el poder cambia de lado. Hilo: la \concepto{regla de la elasticidad inversa}.
\end{itemize}

\bloque[8]{I. Teoría del monopolio\label{rep:I}}
\begin{ideaclave}
Produce menos y cobra más que la competencia; el margen lo limita la elasticidad de la demanda, y la pérdida de bienestar va más allá de las unidades que deja de producir.
\end{ideaclave}

\sub{I.1. Caracterización: el escenario base\label{rep:I.1}}
\begin{itemize}
\item Lista común de las estructuras de mercado \vertema{3.A.16}: se mantienen racionalidad, información perfecta y producto homogéneo.
\item \emph{Se levantan 2}:
  \begin{itemize}
  \item mercado atomizado: \textbf{un solo vendedor}, \emph{precio-decisor}; compradores precio-aceptantes, sin reventa;
  \item libre entrada: \textbf{barreras} \vertema{3.A.15}, que protegen el beneficio a LP.
    \begin{itemize}
    \item estratégicas (disuasión); legales (patentes, licencias); tecnológicas (subaditividad, \hyperref[rep:II.2]{II.2}).
    \end{itemize}
  \end{itemize}
\item \emph{Supuestos propios}, que se levantan luego: precio uniforme (\hyperref[rep:II.1]{II.1}), sin ventajas de tamaño (\hyperref[rep:II.2]{II.2}), un bien y una planta (\hyperref[rep:II.3]{II.3}), precio-aceptante en factores (\hyperref[rep:III.1]{III.1}).
\item \emph{Consecuencia}: afronta toda la demanda; elige \emph{un punto} de ella (precio \emph{o} cantidad).
\end{itemize}

\sub{I.2. Equilibrio del monopolio\label{rep:I.2}}
\subsub{a) Idea}
\begin{itemize}
\item Para vender una unidad más tiene que bajar el precio de \emph{todas}.
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item \emph{Supuestos}: $p(q)$ derivable, $p'<0$ (cuasilineal \vertema{3.A.8}); $C(q)$ con $\mathrm{CMg}>0$, ya minimizado \vertema{3.A.12}.
\item \emph{Desarrollo}: con la \concepto{condición de participación}:
\begin{formula}
\centering $\optim{\max}{q}{B(q)=p(q)\,q-C(q)}{q\geq0,\ B(q)\geq0}$\\[3pt]
$\underbrace{p(q^M)+p'(q^M)\,q^M}_{\mathrm{IMg}}=\mathrm{CMg}(q^M)$;\quad $\mathrm{IMg}=p\Big(1-\dfrac{1}{|\varepsilon_{q,p}|}\Big)$
\end{formula}
  \begin{itemize}
  \item \concepto{IMg}: $p$ de la unidad adicional menos lo que pierde en las demás, $p'q<0$ (\autor{Cournot}, 1838).
  \item Competencia: $|\varepsilon|\to\infty$ $\Rightarrow$ $\mathrm{IMg}=p$.
  \end{itemize}
\item \emph{Índice de \autor{Lerner}} (1934), \concepto{regla de la elasticidad inversa}:
\funcionclave{\mathcal{L}\equiv\frac{p^M-\mathrm{CMg}(q^M)}{p^M}=\frac{1}{|\varepsilon_{q,p}|}}
\end{itemize}
\graficorepaso[0.62\linewidth]{equilibrio-monopolio}{$\mathrm{IMg}=\mathrm{CMg}$ en $q^M$; precio sobre la demanda; competencia en $q^c$, $p^c$}
\begin{itemize}
\item \emph{Tramo elástico}: $\mathrm{CMg}>0\Rightarrow\mathrm{IMg}>0\Rightarrow|\varepsilon|>1$.
\item \emph{CSO}: $\mathrm{IMg}'<\mathrm{CMg}'$: vale con CMg decreciente si corta desde abajo.
\item \emph{No hay curva de oferta}: $q$ depende de la elasticidad, no solo de la altura de la demanda.
\item \emph{Viabilidad} en $q^M$ (no en el mín.\ del CMe):
  \begin{itemize}
  \item CP: $p(q^M)\geq\mathrm{CVMe}(q^M)$; LP: $p(q^M)\geq\mathrm{CMe}(q^M)$.
  \end{itemize}
\item \emph{Implicaciones}: $p^M>\mathrm{CMg}$, $q^M<q^c$; beneficios a LP; no necesariamente en el mín.\ del CMe LP.
\end{itemize}
\subsub{c) Extensión: impuestos}
\begin{itemize}
\item Unitario $t$: $\mathrm{IMg}=\mathrm{CMg}+t$; traslación según la \emph{curvatura} (\autor{Bulow}-\autor{Pfleiderer}, 1983).
  \begin{itemize}
  \item lineal: $p$ sube $t/2$; isoelástica: \emph{más} que $t$;
  \item sobre beneficios o de suma fija: $q$ no cambia.
  \end{itemize}
\end{itemize}

\sub{I.3. Análisis de eficiencia y bienestar\label{rep:I.3}}
\subsub{a) Idea}
\begin{itemize}
\item Comparar con la competencia, que maximiza $\mathrm{ET}=\mathrm{EC}+\mathrm{EP}$ \vertema{3.A.22}.
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item \emph{Supuestos}: demanda = valoración marginal; CMg = coste de oportunidad; mismos costes.
\item \emph{Desarrollo}:
  \begin{itemize}
  \item rectángulo $(p^M-p^c)\,q^M$: \emph{transferencia}, no pérdida;
  \item \concepto{triángulo de \autor{Harberger}}: unidades que valen más de lo que cuestan y no se producen:
\begin{formula}
\centering $\mathrm{PIE}\approx\tfrac12\,|\varepsilon|\,\mathcal{L}^2\,p^Mq^M\ \overset{|\varepsilon|=1/\mathcal{L}}{=}\ \tfrac12\,\mathcal{L}\,p^Mq^M=\tfrac12\,B$
\end{formula}
  \item crece con el \emph{cuadrado} del margen; $B/2$ exacto con demanda lineal y CMg constante.
  \end{itemize}
\item \emph{Otras ineficiencias}:
  \begin{itemize}
  \item \emph{en factores}: $\mathrm{IPMg}_i=w_i<\mathrm{VPMg}_i$: \emph{explotación monopolística} (\autor{Robinson});
  \item \emph{productiva}: no produce en el mín.\ del CMe LP;
  \item \concepto{ineficiencia X} (\autor{Leibenstein}, 1966): $C^{\mathrm{obs}}=(1+\chi)\,C(q)$; «vida tranquila» (\autor{Hicks}, 1935);
  \item \concepto{búsqueda de rentas} (\autor{Tullock}, 1967): \autor{Posner} (1975), se disipa todo el rectángulo \vertema{4.B.2}.
  \end{itemize}
\end{itemize}
\graficorepaso[0.62\linewidth]{bienestar-monopolio}{Transferencia, triángulo de \autor{Harberger}, rentas e ineficiencia X}
\begin{itemize}
\item \emph{Implicaciones}: pérdida entre el triángulo (cota inferior) y triángulo + X + rentas (superior); mayor cuanto más inelástica la demanda.
\end{itemize}
\subsub{c) Extensiones: eficiencia dinámica}
\begin{itemize}
\item \autor{Schumpeter} (1942): el beneficio financia la innovación; \autor{Arrow} (1962): \emph{efecto reemplazo}.
\item Evidencia: U invertida (\autor{Aghion} et al., 2005) \vertema{3.A.44}.
\end{itemize}
\subsub{d) Evidencia y valoración}
\begin{itemize}
\item \autor{Harberger} (1954): $<0{,}1$\,\% de la renta; \autor{Posner}, \autor{Cowling}-\autor{Mueller} (1978): mucho más.
\item Suele ser ineficiente, pero no siempre hay que desmantelarlo; intervenir cuesta \vertema{3.A.23}.
\end{itemize}

\bloque[11,5]{II. Desarrollos de la teoría del monopolio\label{rep:II}}
\begin{ideaclave}
Al levantar cada supuesto, el monopolista adapta la regla de la elasticidad inversa, y el juicio de bienestar puede cambiar de signo.
\end{ideaclave}

\sub{II.1. Discriminación de precios\label{rep:II.1}}
\begin{itemize}
\item \concepto{Discriminación}: el mismo bien a precios distintos sin diferencias de coste (\autor{Varian}, 1989).
\item \emph{3 condiciones}: poder de mercado; información sobre la disposición a pagar; sin reventa.
\item \emph{Clasificación} (\autor{Pigou}, 1920; hoy por la información, \autor{Tirole}):
  \begin{itemize}
  \item \concepto{1.\textsuperscript{er} grado}: observa cada valoración;
  \item \concepto{2.º}: conoce la distribución, menú común (\emph{autoselección});
  \item \concepto{3.\textsuperscript{er}}: observa una señal, precio por grupo.
  \item «Cuarto grado» (intertemporal): autoselección en el tiempo.
  \end{itemize}
\end{itemize}
\subsub{a) Primer grado}
\begin{itemize}
\item Cada unidad a su precio de reserva $\Rightarrow$ $\mathrm{IMg}=p$ $\Rightarrow$ $q^*=q^c$.
\item \emph{Eficiente}, pero EC $=0$: problema de \emph{equidad}. Hoy, precios personalizados.
\end{itemize}
\subsub{b) Segundo grado: autoselección}
\begin{itemize}
\item \concepto{Tarifa no lineal} (\autor{Mussa}-\autor{Rosen} 1978, \autor{Maskin}-\autor{Riley} 1984) \vertema{3.A.13}.
\item \emph{Supuestos}: tipos $t_1<t_2$ ($\nu$, $1-\nu$); utilidad $t_iu(q)-T$; CMg $c$; menú $\{(q_i,T_i)\}$.
\item \emph{Desarrollo}: 2~restricciones de participación (PC) y 2~de incentivos (IC); se saturan $\mathrm{PC}_1$ e $\mathrm{IC}_2$:
\begin{formula}
\centering $T_2=t_2u(q_2)-\underbrace{(t_2-t_1)\,u(q_1)}_{\text{renta informativa}}$\\[2pt]
$t_2u'(q_2)=c$;\quad $t_1u'(q_1)>c$
\end{formula}
\item \emph{Implicaciones}:
  \begin{itemize}
  \item \emph{no distorsión arriba}: el alto, cantidad eficiente + \concepto{renta informativa};
  \item \emph{distorsión abajo}: menos al bajo, para abaratar la renta del alto;
  \item excluir al bajo si $\nu t_1\leq(1-\nu)(t_2-t_1)$.
  \end{itemize}
\item \emph{Calidad}: versión baja \emph{degradada} (\emph{damaged goods}, \autor{Deneckere}-\autor{McAfee}).
\end{itemize}
\graficorepaso[0.6\linewidth]{autoseleccion}{2 tipos: eficiente arriba, distorsión abajo, renta informativa}
\begin{itemize}
\item \emph{Tarifa en dos partes} $T=A+pq$ (\autor{Oi}, 1971, Disneyland):
  \begin{itemize}
  \item cuotas distintas: $p=c$, $A_i=\mathrm{EC}_i$ = 1.\textsuperscript{er} grado;
  \item cuota común: $A=\mathrm{EC}_1(p)$, $p>c$; o excluir a los bajos.
  \end{itemize}
\end{itemize}
\subsub{c) Tercer grado}
\begin{itemize}
\item \emph{Supuestos}: 2~grupos $p_k(q_k)$, precio lineal en cada uno, sin reventa.
\item \emph{Desarrollo}: IMg agregado (suma horizontal) $=$ CMg:
\begin{formula}
\centering $\mathrm{IMg}_1=\mathrm{IMg}_2=\mathrm{CMg}(q_1+q_2)\ \Rightarrow\ \mathcal{L}_k=\dfrac{1}{|\varepsilon_k|}$
\end{formula}
\item \emph{Implicaciones}: más caro el grupo \emph{menos elástico}.
  \begin{itemize}
  \item ET solo sube si \emph{aumenta la cantidad total} (\autor{Schmalensee} 1981, \autor{Varian} 1985);
  \item lineales y todos servidos: $Q$ igual, ET baja (\autor{Robinson});
  \item favorable si abre un mercado (medicamentos en países pobres).
  \end{itemize}
\end{itemize}
\graficorepaso[0.92\linewidth]{tercer-grado}{IMg agregado = CMg; precio mayor en el mercado menos elástico}
\subsub{d) Intertemporal: conjetura de \autor{Coase} (1972)}
\begin{itemize}
\item Bien duradero, revisiones frecuentes, consumidores pacientes $\Rightarrow$ $p\to\mathrm{CMg}$: compite con su yo futuro.
  \begin{itemize}
  \item Inconsistencia temporal \vertema{3.A.14}; salidas: \emph{leasing}, obsolescencia, ediciones (\autor{Bulow}, 1982).
  \end{itemize}
\end{itemize}

\sub{II.2. Monopolio natural\label{rep:II.2}}
\subsub{a) Uniproducto: existencia y sostenibilidad}
\begin{itemize}
\item \concepto{Monopolio natural} $\Leftrightarrow$ costes \concepto{subaditivos} en la producción relevante (\autor{Baumol}, 1977):
\begin{formula}
\centering $C\Big(\textstyle\sum_{j}q_j\Big)<\textstyle\sum_{j}C(q_j)$
\end{formula}
  \begin{itemize}
  \item economías de escala \emph{suficientes, no necesarias}; relativo a la \emph{demanda};
  \item redes (electricidad, gas, agua, vías), no los servicios que circulan.
  \end{itemize}
\item \concepto{Sostenible}: un precio que cubre costes y al que ningún entrante gana (\autor{Panzar}-\autor{Willig}, 1977).
  \begin{itemize}
  \item demanda antes del mín.\ del CMe ($q^0$): sostenible; entre $q^0$ y $q^1$: puede no serlo;
  \item mercados impugnables: $p=\mathrm{CMe}$ \vertema{3.A.15}.
  \end{itemize}
\end{itemize}
\graficorepaso[0.66\linewidth]{subaditividad-sostenibilidad}{Subaditividad ($q<q^1$) y sostenibilidad}
\subsub{b) Multiproducto}
\begin{itemize}
\item \concepto{Economías de alcance} $C(q_1,q_2)<C(q_1,0)+C(0,q_2)$: \emph{necesarias, no suficientes}.
\item Sostenible con precios \concepto{libres de subsidios cruzados} (\autor{Faulhaber}, 1975): pruebas del coste aislado y del incremental.
\end{itemize}
\subsub{c) Regulación}
\begin{itemize}
\item \emph{Idea}: ni desmantelar ni dejar libre: eficiencia asignativa frente a viabilidad.
\item \concepto{Primer óptimo} $p=\mathrm{CMg}$ (\autor{Hotelling}, 1938): pérdidas $\Rightarrow$ subvención; cada euro cuesta $1+\lambda$.
\item \concepto{Segundos óptimos}, sin subvención:
  \begin{itemize}
  \item $p=\mathrm{CMe}$: beneficio nulo, menos unidades;
  \item tarifa en dos partes (\autor{Coase}, 1946): $p=\mathrm{CMg}$, $A=\mathrm{CF}/n$;
  \item punta y valle (\autor{Boiteux}, \autor{Steiner}): $p_v=b$, $p_p=b+\beta$;
  \item \autor{Ramsey}-\autor{Boiteux} (multiproducto):
\begin{formula}
\centering $\dfrac{p_i-\mathrm{CMg}_i}{p_i}=\dfrac{\lambda}{1+\lambda}\cdot\dfrac{1}{|\varepsilon_i|}$
\end{formula}
  \end{itemize}
\end{itemize}
\graficorepaso[0.66\linewidth]{regulacion-monopolio-natural}{Pérdidas con $p=\mathrm{CMg}$; beneficio nulo con $p=\mathrm{CMe}$}
\begin{itemize}
\item \emph{Implicaciones}: \emph{\autor{Lerner} amortiguado}; $\lambda/(1+\lambda)$ crece con $\lambda$ ($0$: $p=\mathrm{CMg}$; $\infty$: monopolio).
  \begin{itemize}
  \item recarga lo inelástico; misma regla que la imposición óptima \vertema{4.B.11}.
  \end{itemize}
\item \emph{Valoración}: exige conocer costes y elasticidades; puede ser regresiva.
  \begin{itemize}
  \item Con información asimétrica \vertema{3.A.20}: tasa de rentabilidad (\autor{Averch}-\autor{Johnson}, 1962), \emph{price cap} $\mathrm{IPC}-X$ (\autor{Littlechild}), \autor{Laffont}-\autor{Tirole}, captura (\autor{Stigler}).
  \end{itemize}
\end{itemize}

\sub{II.3. Producción conjunta\label{rep:II.3}}
\begin{itemize}
\item \emph{Idea}: el multiproducto \emph{internaliza} relaciones de demanda y de costes.
\end{itemize}
\subsub{a) Monopolio multiproducto}
\begin{itemize}
\item Demandas independientes: $\mathcal{L}_i=1/|\varepsilon_i|$. Dependientes:
\begin{formula}
\centering $\mathcal{L}_i=\dfrac{1}{|\varepsilon_{ii}|}+\displaystyle\sum_{j\neq i}\dfrac{(p_j-\mathrm{CMg}_j)\,q_j}{p_i\,q_i}\cdot\dfrac{\varepsilon_{ji}}{|\varepsilon_{ii}|}$
\end{formula}
  \begin{itemize}
  \item sustitutivos: precios \emph{más altos};
  \item complementarios: \emph{más bajos}, incluso $\mathcal{L}_i<0$ (consolas, impresoras).
  \end{itemize}
\item \concepto{Complementariedad de costes} ($\partial^2C/\partial q_1\partial q_2<0$): produce más de los 2 (aprendizaje).
\end{itemize}
\subsub{b) Proporciones fijas y multiplanta}
\begin{itemize}
\item \concepto{Oferta conjunta} (\autor{Marshall}; carne y piel): $\mathrm{IMg}_1+\mathrm{IMg}_2=\mathrm{CMg}$ (suma vertical); si un IMg $<0$, desecha.
\item Multiplanta: $\mathrm{IMg}(Q)=\mathrm{CMg}_1=\mathrm{CMg}_2$: como un cártel \vertema{3.A.19}.
\item \emph{Bienestar}: mejor que monopolistas separados con complementarios; peor con sustitutivos.
\end{itemize}

\bloque[7]{III. Otras estructuras relacionadas\label{rep:III}}
\begin{ideaclave}
El monopsonio es el espejo del monopolio: compra menos y paga menos de lo que vale el factor. Con poder en los 2~lados, decide la negociación.
\end{ideaclave}

\sub{III.1. Monopsonio\label{rep:III.1}}
\subsub{a) Idea}
\begin{itemize}
\item \concepto{Monopsonio}: único comprador (\autor{Robinson}, 1933): empresa de la comarca, hospital público, gran distribución.
\item Para comprar más debe \emph{subir} el precio de todo lo que compra.
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item \emph{Supuestos}: precio-aceptante en el bien; $q=F(L)$; oferta de trabajo $w_L(L)$, $w_L'>0$; salario único.
\item \emph{Desarrollo}: $\max_L\ pF(L)-w_L(L)\,L$ $\Rightarrow$ $\mathrm{VPMg}_L=$ \concepto{gasto marginal} $\mathrm{GMg}_L=w_L+w_L'L$:
\funcionclave{\frac{\mathrm{VPMg}_L-w_L}{w_L}=\frac{1}{\varepsilon_{L,w_L}}}
  \begin{itemize}
  \item espejo de \autor{Lerner}: brecha mayor cuanto menos elástica la oferta;
  \item salario en la \emph{oferta}: $w^M=w_L(L^M)$; GMg lineal con doble pendiente.
  \end{itemize}
\item No hay curva de demanda de trabajo; isocostes curvas.
\item \emph{4 casos}: si además es monopolista, $\mathrm{IPMg}_L=\mathrm{GMg}_L$: se suman las 2~explotaciones (monopolística y monopsonística).
\item \emph{Implicaciones}: $L^M<L^c$, $w^M<w^c$, \emph{sin paro involuntario}.
\item \emph{Salario mínimo} $\underline{w}$: oferta y GMg horizontales hasta $L^S(\underline{w})$:
  \begin{itemize}
  \item $\underline{w}\leq w^M$: nada;
  \item $w^M<\underline{w}\leq w^c$: el empleo \emph{sube} (máximo en $w^c$);
  \item $\underline{w}>w^c$: paro, pero más empleo que $L^M$ si $\underline{w}\leq\mathrm{VPMg}_L(L^M)$ \vertema{3.A.25}.
  \end{itemize}
\end{itemize}
\pareja{monopsonio}{Menos empleo y salario que en competencia}{salario-minimo}{El empleo sube hasta $w^c$}
\subsub{c) Extensiones, evidencia y valoración}
\begin{itemize}
\item \emph{Monopsonio dinámico} (\autor{Manning}, 2003): fricciones con muchas empresas.
\item \autor{Card}-\autor{Krueger} (1994): NJ sube el mínimo (4,25 $\to$ 5,05 \$) frente a Pensilvania: sin caída del empleo.
\item \autor{Yeh} et al.\ (2022): IPMg 53\,\% sobre el salario.
\item \autor{Deb} et al.\ (2022): salarios estancados, 75\,\% por monopolio y 25\,\% por monopsonio.
\item Explica por qué el mínimo no siempre destruye empleo.
\end{itemize}

\sub{III.2. Monopolio bilateral\label{rep:III.2}}
\subsub{a) Idea}
\begin{itemize}
\item Único vendedor frente a único comprador (sindicato y monopsonista; componente y montador).
\item \autor{Bowley} (1928): \emph{indeterminado}; decide el poder de negociación. Como la caja de \autor{Edgeworth} \vertema{3.A.21}.
\end{itemize}
\subsub{b) Modelo}
\begin{itemize}
\item \emph{Supuestos}: sindicato con $U_S=wL-v(L)$, $v'(L)=w_L(L)$; empresa con $B=pF(L)-wL$.
\item \emph{Extremos}:
  \begin{itemize}
  \item manda la empresa: $L^M$, $w^M$;
  \item manda el sindicato (\autor{Dunlop}, 1944): $w^S>w^c$ y paro;
  \item entre $w^M$ y $w^c$, \emph{poder compensatorio} (\autor{Galbraith}, 1952).
  \end{itemize}
\item \emph{Negociación eficiente} (\autor{Nash}, 1950), $\gamma$ = poder del sindicato:
\begin{formula}
\centering $\max_{w,L}\ (U_S-U_S^0)^{\gamma}(B-B^0)^{1-\gamma}$\\[2pt]
$\Rightarrow\ L^*=L^c$;\quad $w^*L^*=v+U_S^0+\gamma\,S$
\end{formula}
\item \emph{Implicaciones}: cantidad \emph{eficiente} y \emph{sin $\gamma$}; el poder solo \emph{reparte}.
  \begin{itemize}
  \item \autor{Rubinstein} (1982): gana el más paciente \vertema{3.A.14}; incentivo a integrarse.
  \end{itemize}
\end{itemize}
\graficorepaso[0.72\linewidth]{monopolio-bilateral}{Extremos $w^M$ y $w^S$; cantidad eficiente $L^c$}
\subsub{c) Extensiones y valoración}
\begin{itemize}
\item \concepto{Doble marginalización} (\autor{Spengler}, 1950): 2~márgenes en cadena.
  \begin{itemize}
  \item lineal: separados, la mitad de $Q$ que integrados; integrar o tarifa en dos partes mejora a todos.
  \end{itemize}
\item Derecho a gestionar frente a contratos eficientes (\autor{McDonald}-\autor{Solow}, 1981) \vertema{3.A.27}.
\item \emph{Evidencia}: negociación colectiva, distribución y productores, medicamentos con patente.
\item Desplaza la pregunta al poder de negociación y a las instituciones.
\end{itemize}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}:
  \begin{itemize}
  \item $\mathcal{L}=1/|\varepsilon|$; coste social más allá del triángulo;
  \item discriminar extrae excedente; subaditividad $\Rightarrow$ regular (\autor{Lerner} amortiguado); varios bienes: internaliza;
  \item monopsonio, espejo sin paro; 2~lados: negociación.
  \end{itemize}
\item \textbf{Relevancia}: competencia \vertema{3.B.39}, redes \vertema{3.A.20}, salario mínimo \vertema{3.A.25}; márgenes crecientes.
\item \textbf{Extensiones}: \vertema{3.A.18} \vertema{3.A.19} \vertema{3.A.15} \vertema{3.A.23} \vertema{3.A.14} \vertema{3.A.27}.
\item \textbf{Opinión}: el problema no es el número de empresas, sino la \emph{falta de disciplina} («el mejor beneficio del monopolio es una vida tranquila», \autor{Hicks}, 1935):
  \begin{itemize}
  \item un monopolio impugnable o bien regulado puede ser la forma más barata de producir; una patente, el premio de la innovación;
  \item defensa de la competencia frente a las conductas;
  \item regulación con incentivos frente al monopolio natural;
  \item atención al poder de compra en el trabajo.
  \end{itemize}
\item \textbf{Idea final}: quien parte y reparte se queda con la mejor parte, pero no con toda; y lo que se queda no compensa lo que la sociedad pierde.
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Por qué no hay curva de oferta?} $q$ depende de la elasticidad, no solo de la altura de la demanda.
\item \textbf{¿Solo el triángulo de \autor{Harberger}?} No: más ineficiencia X y búsqueda de rentas.
\item \textbf{¿Por qué se distorsiona al tipo bajo?} Abarata la renta informativa del alto.
\item \textbf{¿Cuándo mejora el bienestar el tercer grado?} Solo si aumenta la cantidad total.
\item \textbf{¿Escala necesaria para el monopolio natural?} No: suficiente; la definición es la subaditividad.
\item \textbf{¿Paro con monopsonio?} No; un mínimo entre $w^M$ y $w^c$ sube el empleo.
\end{enumerate}
\begin{trampa}
\begin{itemize}
\item Primer grado: \emph{eficiente}, EC $=0$. Segundo: \emph{sin distorsión arriba}.
\item \autor{Ramsey}: $\lambda/(1+\lambda)$ crece con $\lambda$. Escala: suficiente; alcance: necesaria.
\item Monopsonio: salario en la \emph{oferta}, no en el GMg. Bilateral: cantidad sin $\gamma$.
\end{itemize}
\end{trampa}
