% Inflación de oferta y de demanda en Estados Unidos (Shapiro, 2022): deflactor
% del consumo privado (PCE), tasa interanual (%) y contribuciones (pp) de los
% componentes de demanda, ambiguos y de oferta, enero de 2019-septiembre de 2026.
% Datos: ../datos/pce-oferta-demanda-shapiro-2019-2026.csv (Banco de la Reserva
% Federal de San Francisco, «Supply- and Demand-Driven PCE Inflation», hoja de
% cálculo descargada el 13/11/2026; columnas «headline, y/y»). Eje x: n.º de mes
% desde enero de 2019 (fila del CSV): 2019-01 = 0, 2026-09 = 92.
% Cifras de control: jun. 2022: demanda 2,76, ambiguo 0,95, oferta 3,43, total
% 7,14; sep. 2026: 1,17, 1,20, 1,28, total 3,65.
% Paleta: oferta en bermellón (como la energía), demanda en azul oscuro,
% ambiguo en gris; total en negro.
% Capas (CRITERIOS §7): barras → cero → línea del total → rótulo.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\usepackage{pgfplotstable}
\begin{document}
\pgfplotstableread[col sep=comma,comment chars={\#}]{../datos/pce-oferta-demanda-shapiro-2019-2026.csv}\pce
\pgfplotsset{
  leyenda linea corta/.style={legend image code/.code={%
    \draw[##1] plot coordinates {(0cm,0cm) (0.6cm,0cm)};}},
}
\begin{tikzpicture}[tcee]
  \begin{axis}[
      width=15cm, height=7.5cm,
      ybar stacked, stack negative=separate, bar width=0.8,
      xmin=-1, xmax=93,
      xtick={-0.5,11.5,...,83.5,92.5}, x tick label as interval,
      xticklabels={2019,2020,2021,2022,2023,2024,2025,2026},
      ymin=-1.5, ymax=8, ytick={-1,0,...,8},
      ticklabel style={font=\footnotesize},
      ylabel={\% y puntos porcentuales},
      axis lines*=left, tick align=outside, clip=false,
      ymajorgrids, grid style={tceegris!25},
      legend style={at={(0.5,-0.1)}, anchor=north, legend columns=4,
        draw=none, font=\footnotesize, /tikz/every even column/.append style={column sep=8pt}},
      legend cell align=left, area legend, dicapas,
    ]
    % 1: componentes de demanda
    \capa{1}{
      \addplot[on layer=dicbarras, fill=dicze, draw=none]
        table[x expr=\coordindex, y=demanda] {\pce};
      \addlegendentry{Demanda}
    }
    % 2: componentes de oferta
    \capa{2}{
      \addplot[on layer=dicbarras, fill=dicenergia, draw=none]
        table[x expr=\coordindex, y=oferta] {\pce};
      \addlegendentry{Oferta}
    }
    % 3: ambiguos
    \capa{3}{
      \addplot[on layer=dicbarras, fill=dicotroclaro, draw=none]
        table[x expr=\coordindex, y=ambiguo] {\pce};
      \addlegendentry{Ambiguos}
    }
    % 4: eje del cero
    \capa{4}{
      \dicraya{(-1,0) (93,0)}
    }
    % 5: inflación total
    \capa{5}{
      \addplot[stack plots=false, sharp plot, leyenda linea corta, diclineahalo=dicinflacion]
        table[x expr=\coordindex, y=total] {\pce};
      \addlegendentry{Inflación del PCE (\%)}
    }
    % 6: rótulo del máximo
    \capa{6}{
      \dicrotulo[font=\footnotesize, text=dicinflacion, anchor=south, inner sep=2pt]{(41,7.2)}{jun.\ 2022: 7,1\,\%}
    }
  \end{axis}
\end{tikzpicture}
\end{document}
