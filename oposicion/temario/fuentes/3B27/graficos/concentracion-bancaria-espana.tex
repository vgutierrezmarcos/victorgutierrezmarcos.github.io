% Concentración bancaria en España frente a la zona del euro, 1999-2025, en 2
% paneles: izquierda, cuota de las 5 mayores entidades en los activos (CR5, %);
% derecha, índice de Herfindahl-Hirschman de los activos (IHH, 0-10.000).
% Datos: ../datos/concentracion-bancaria-espana.csv (BCE, SSI; consulta del
% 16/11/2026). España: 2007, CR5 41,0 % e IHH 459; 2025, 67,4 % y 1.306.
% Zona del euro (media simple de los 20 países): 2007, 61,6 % y 1.231; 2025,
% 71,3 % y 1.511.
% Sombreado: la reestructuración del sector en España, 2010-2015 (rótulo, no
% dato del CSV).
% Orden de capas: sombreado -> ejes -> líneas -> puntos -> rótulos, panel a panel.
% Colores: España morado (como en el resto del tema), zona del euro dorado.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}

\pgfplotsset{cbejes/.style={width=7.6cm, height=5.6cm, scale only axis=false,

    x tick label style={/pgf/number format/1000 sep={}},
    /pgf/number format/use comma, /pgf/number format/1000 sep={.},
    axis lines*=left, tick align=outside, clip=false,
    ticklabel style={font=\footnotesize}, label style={font=\small},
    title style={font=\small\bfseries}}}
\colorlet{cbes}{tceemorado}
\colorlet{cbea}{tceedorado}
\begin{document}
