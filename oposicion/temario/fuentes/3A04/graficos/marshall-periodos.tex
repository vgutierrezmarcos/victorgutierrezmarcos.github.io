% El mercado de pescado de Marshall: equilibrio temporal y equilibrio normal.
% Elaboración propia a partir de De Vroey (2016, fig. 1.1).
% Demanda inicial  D_0: p = 0,6 + 4/(q + 0,3); demanda nueva D_1: p = 1 + 6/(q + 0,3).
% Oferta de cada día (periodo de mercado) MS: vertical en la cantidad desembarcada.
% Oferta normal NS: p = a + b q^2, creciente y convexa, que pasa por A y por C:
%   b = (D_1(3,4) - D_0(2,6))/(3,4^2 - 2,6^2) = 0,133815; a = 1,074722.
% A = (2,6; 1,979): D_0 corta a MS_0 y a NS (equilibrio inicial).
% B = (2,6; 3,069): D_1 corta a MS_0 (equilibrio temporal; cuasirrentas).
% C = (3,4; 2,622): D_1 corta a NS y a MS_1 (equilibrio normal), p_A < p_C < p_B.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,
    declare function={dcero(\q)=0.6+4/(\q+0.3); duno(\q)=1+6/(\q+0.3);
      ns(\q)=1.074722+0.133815*\q^2;}]
  % 1: ejes
  \capa{1}{\ejes{$q$}{$p$}{6.2}{5.2}}
  % 2: demanda inicial
  \capa{2}{\draw[curva1] plot[domain=0.9:5.5,samples=80,smooth] (\x,{dcero(\x)})
    node[etiqueta,right] {$D_0$};}
  % 3: oferta normal (con tiempo para ajustar la flota)
  \capa{3}{\draw[curva3] plot[domain=0.9:4.55,samples=80,smooth] (\x,{ns(\x)})
    node[etiqueta,above right=0pt and -2pt] {$NS$};}
  % 4: oferta de cada día: el pescado desembarcado
  \capa{4}{\draw[curva2] (2.6,0) -- (2.6,4.85) node[etiqueta,above] {$MS_0$};}
  % 5: equilibrio inicial A
  \capa{5}{\node[punto] (A) at (2.6,1.9793) {};}
  \capa{5}{\node[etiqueta,below right] at (A) {$A$};}
  \capa{5}{\draw[guia] (A) -- (0,1.9793) node[left,text=black] {$p_A$};}
  \capa{5}{\node[below,text=black] at (2.6,0) {$q_A$};}
  % 6: aumenta la demanda (la epidemia encarece la carne)
  \capa{6}{\draw[curvanueva,desplazada] plot[domain=1.4:5.6,samples=80,smooth] (\x,{duno(\x)})
    node[etiqueta,right] {$D_1$};}
  \capa{6}{\draw[flecha] (1.6,{dcero(1.6)+0.12}) -- (2.1,{duno(2.1)-0.12});}
  % 7: equilibrio temporal B: la cantidad no cambia y sube el precio
  \capa{7}{\node[punto] (B) at (2.6,3.0690) {};}
  \capa{7}{\node[etiqueta,above right] at (B) {$B$};}
  \capa{7}{\draw[guia] (B) -- (0,3.0690) node[left,text=black] {$p_B$};}
  % 8: con tiempo salen más barcos: la oferta de cada día aumenta
  \capa{8}{\draw[curva2,desplazada] (3.4,0) -- (3.4,4.85) node[etiqueta,above] {$MS_1$};}
  \capa{8}{\draw[flecha] (2.68,4.3) -- (3.32,4.3);}
  % 9: equilibrio normal C
  \capa{9}{\node[punto] (C) at (3.4,2.6216) {};}
  \capa{9}{\node[etiqueta,below right] at (C) {$C$};}
  \capa{9}{\draw[guia] (C) -- (0,2.6216) node[left,text=black] {$p_C$};}
  \capa{9}{\node[below,text=black] at (3.4,0) {$q_C$};}
\end{tikzpicture}
\end{document}
