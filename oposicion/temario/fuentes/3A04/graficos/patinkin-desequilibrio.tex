% Patinkin (1956): el paro involuntario como ajuste lento. Redibujo de las
% figuras 3.1 y 3.2 de De Vroey (2016, págs. 60-61) en un solo gráfico por capas.
% Panel a, trabajo (salario real w):  L^S: w = 0,5 + 0,6 L;  L^D nocional: w = 4 - 0,7 L
%   -> A = (2,692; 2,115). Con la producción en Y_1, el empleo es N_1 = 1,15 (punto K).
% Panel b, bienes (gasto E y producción Y): E = Y (45 grados); oferta Y = S(w_0, K_0) en Y_0 = 3,6;
%   E_0(Y) = 3,6 + 0,35 (Y - 3,6) - 0,03 (Y - 3,6)^2 (corta la de 45 grados en B);
%   E_1 = E_0 - 1,1 -> G en Y_1 = 1,75 (C, sobre la oferta);  E_2 = E_0 - 0,5 -> Y_2 = 2,8.
% Autor: Víctor Gutiérrez Marcos
\documentclass[tikz,border=4pt]{standalone}
\input{../../../latex/tikz-tcee}
\begin{document}
\begin{tikzpicture}[tcee,
    declare function={e0(\y)=3.6+0.35*(\y-3.6)-0.03*(\y-3.6)^2;}]
  \path (-0.9,-1.4) rectangle (14.6,5.6);
  % 1: ejes
  \capa{1}{\ejes{$L$}{$w$}{5.2}{4.9}}
  \capa{1}{\node[etiqueta, font=\small\itshape, anchor=south] at (2.6,5.15) {a) Mercado de trabajo};}
  \begin{scope}[xshift=7.6cm]
    \capa{1}{\ejes{$Y$}{$E$}{5.8}{4.9}}
    \capa{1}{\node[etiqueta, font=\small\itshape, anchor=south] at (2.9,5.15) {b) Mercado de bienes};}
  \end{scope}
  % 2: el mercado de trabajo en equilibrio (A)
  \capa{2}{\draw[curva1] (0.3,{4-0.7*0.3}) -- (5.0,{4-0.7*5.0}) node[right, etiqueta, text=tceemorado] {$L^D$ nocional};}
  \capa{2}{\draw[curva2] (0,0.5) -- (5.0,3.5) node[right, etiqueta, text=tceeazul] {$L^S$};}
  \capa{2}{\node[punto] at (2.692,2.115) {};
    \node[etiqueta, above right] at (2.692,2.115) {$A$};
    \draw[guia] (0,2.115) node[left, text=black] {$w_0$} -| (2.692,0) node[below, text=black] {$N_0$};}
  % 3: el mercado de bienes en equilibrio (B)
  \begin{scope}[xshift=7.6cm]
    \capa{3}{\draw[thick, black] (0,0) -- (4.7,4.7) node[right, etiqueta] {$E=Y$};}
    \capa{3}{\draw[curva4] (3.6,0) node[below, text=black] {$Y_0$} -- (3.6,4.9) node[anchor=east, etiqueta, text=tceedorado] at (3.55,4.75) {$Y=S(w_0,K_0)$};}
    \capa{3}{\draw[curva1] plot[domain=0.6:5.6, samples=40] (\x,{e0(\x)}) node[right, etiqueta, text=tceemorado] {$E_0$};
      \node[punto] at (3.6,3.6) {}; \node[etiqueta, above left] at (3.6,3.6) {$B$};}
    % 4: cae la demanda de bienes: E_1, C (exceso de oferta) y G (se produce Y_1)
    \capa{4}{\draw[curvanueva, desplazada] plot[domain=0.6:5.6, samples=40] (\x,{e0(\x)-1.1}) node[right, etiqueta, text=tceerojo] {$E_1$};
      \node[punto] at (3.6,{e0(3.6)-1.1}) {}; \node[etiqueta, below right] at (3.6,{e0(3.6)-1.1}) {$C$};
      \node[punto] at (1.75,1.75) {}; \node[etiqueta, above left] at (1.75,1.75) {$G$};
      \draw[guia] (1.75,1.75) -- (1.75,0) node[below, text=black] {$Y_1$};}
    \capa{4}{\draw[decorate, decoration={brace, amplitude=3pt, mirror}, thick, tceerojo] (1.75,-0.55) -- (3.6,-0.55)
        node[midway, below=3pt, etiqueta, text=tceerojo, font=\scriptsize] {exceso de oferta de bienes};}
  \end{scope}
  % 5: las empresas contratan menos al mismo salario real: K, fuera de las 2 curvas
  \capa{5}{\node[punto, tceerojo] at (1.15,2.115) {};
    \node[etiqueta, above left, text=tceerojo] at (1.15,2.115) {$K$};
    \draw[guia] (1.15,2.115) -- (1.15,0) node[below, text=black] {$N_1$};
    \draw[decorate, decoration={brace, amplitude=3pt, mirror}, thick, tceerojo] (1.15,-0.55) -- (2.692,-0.55)
      node[midway, below=3pt, etiqueta, text=tceerojo, font=\scriptsize] {paro involuntario};}
  % 6: el efecto saldos reales devuelve la demanda hacia E_0 (vía E_2)
  \begin{scope}[xshift=7.6cm]
    \capa{6}{\draw[curva3, desplazada] plot[domain=0.6:5.6, samples=40] (\x,{e0(\x)-0.5}) node[right, etiqueta, text=tceeverde] {$E_2$};
      \draw[->, thick, tceeverde] (4.9,{e0(4.9)-1.0}) -- (4.9,{e0(4.9)-0.1});
      \node[etiqueta, text=tceeverde, font=\scriptsize, align=left, anchor=north west] at (3.75,1.3) {efecto saldos reales:\\$P\downarrow\ \Rightarrow\ M/P\uparrow\ \Rightarrow\ E\uparrow$};}
  \end{scope}
\end{tikzpicture}
\end{document}
