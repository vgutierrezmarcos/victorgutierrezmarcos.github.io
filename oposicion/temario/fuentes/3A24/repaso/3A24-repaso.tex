\bloque[2,5]{Introducción}
\begin{itemize}
\item \textbf{Enganche}: «\emph{nunca llueve a gusto de todos}»: ¿puede una sociedad decidir con coherencia si cada uno quiere una cosa?
  \begin{itemize}
  \item Dic. 2023: un solo Estado bloquea en el Consejo Europeo la revisión del MFP (50.000~M€ para Ucrania; unanimidad); acuerdo de los 27~el 1/2/2024.
  \item Con unanimidad, cualquiera bloquea; con mayoría, la minoría pierde y la mayoría puede no saber qué quiere.
  \end{itemize}
\item \textbf{Relevancia}:
  \begin{itemize}
  \item \emph{Teórica}: \concepto{economía del bienestar} = parte normativa de la micro: juzgar situaciones y dar normas; las 2~exigen \emph{agregar}.
  \item \emph{Política}: toda política mejora a unos y empeora a otros.
  \item Nobel: \autor{Arrow} (1972), \autor{Buchanan} (1986), \autor{Sen} (1998), \autor{Hurwicz}-\autor{Maskin}-\autor{Myerson} (2007).
  \end{itemize}
\item \textbf{Contextualización}:
  \begin{itemize}
  \item \autor{Bentham} (1789): suma de utilidades $\to$ \autor{Pareto} (1906), ordinal; \autor{Robbins} (1932): comparar = juicio de valor.
  \item $\to$ compensación (\autor{Kaldor}, \autor{Hicks}, 1939) y FBS (\autor{Bergson}, 1938; \autor{Samuelson}, 1947).
  \item $\to$ \autor{Arrow} (1951) recupera \autor{Borda} y \autor{Condorcet}: \concepto{elección colectiva}. Cierra el bloque \vertema{3.A.22}, \vertema{3.A.23}.
  \end{itemize}
\item \textbf{Problemática}:
  \begin{itemize}
  \item ¿Cómo elegir entre los infinitos óptimos de \autor{Pareto}? ¿Cómo medir la desigualdad?
  \item ¿Hay una regla que agregue preferencias razonablemente? Si no, ¿a qué renunciar? ¿Y si se puede mentir?
  \end{itemize}
\item \textbf{Estructura}: eje = teorema de \autor{Arrow}.
  \begin{itemize}
  \item \hyperref[rep:I]{I}: de la incompletitud de \autor{Pareto} al teorema.
  \item \hyperref[rep:II]{II}: 1.ª escapatoria, rechazar la metodología (FBS); va después aunque es anterior.
  \item \hyperref[rep:III]{III}: 2.ª, aceptarla y renunciar a un axioma.
  \end{itemize}
\end{itemize}

\bloque[8,5]{I. Elección colectiva y teorema de \autor{Arrow}\label{rep:I}}
\begin{ideaclave}
Con $\geq3$ alternativas, ninguna regla que use solo ordenaciones individuales cumple a la vez 5~condiciones mínimas de racionalidad y democracia: toda regla razonable sacrifica alguna.
\end{ideaclave}

\sub{I.1. Contexto: incompletitud del criterio de \autor{Pareto}\label{rep:I.1}}
\subsub{a) Criterio de \autor{Pareto}}
\begin{itemize}
\item \concepto{Óptimo de \autor{Pareto}}: no se puede mejorar a nadie sin empeorar a otro; 4~juicios de valor; ordinal \vertema{3.A.22}.
  \begin{itemize}
  \item 2×2×2×2: $\mathrm{RMST}^X_{LK}=\mathrm{RMST}^Y_{LK}$, $\mathrm{RMS}^A_{XY}=\mathrm{RMS}^B_{XY}=\mathrm{RMT}_{XY}$ $\Rightarrow$ \concepto{GFPU}.
  \item 1TFEB: EC = óptimo; 2TFEB: cualquier óptimo es un EC (convexidad, suma fija).
  \end{itemize}
\item Pero es \emph{incompleto}: no ordena los puntos de la GFPU (necesario, no suficiente).
\end{itemize}
\graficorepaso[0.78\linewidth]{gfpu}{GFPU: envolvente de las FPU de cada punto de la FPP; \autor{Pareto} no ordena sus puntos}
\subsub{b) Criterios adicionales}
\begin{itemize}
\item \emph{Neoparetiana} (LSE): \concepto{compensación} (\autor{Kaldor}, \autor{Hicks}, 1939): los ganadores \emph{podrían} compensar.
  \begin{itemize}
  \item Comparación implícita; contradicciones (\autor{Scitovsky}, 1941).
  \end{itemize}
\item \emph{Neoutilitarista} (Harvard): FBS con juicios explícitos.
\item \autor{Arrow} cambia la pregunta: ¿puede $W$ salir de las preferencias de los individuos?
\end{itemize}

\sub{I.2. Teorema de imposibilidad de \autor{Arrow} (1951)\label{rep:I.2}}
\subsub{a) Idea}
\begin{itemize}
\item \emph{Social Choice and Individual Values}: qué condiciones debe cumplir cualquier procedimiento y si existe alguno.
\item Solo \emph{ordenaciones} (como pedía \autor{Robbins}); método \emph{axiomático} (como \autor{Debreu}).
\end{itemize}
\subsub{b) Modelo}
\subsubsub{i) Supuestos}
\begin{itemize}
\item $I$ finito; $\geq3$ alternativas: poco ($x$), algo ($y$), mucho ($z$) en el portal de 3~vecinos.
\item \emph{Inputs}: ordenaciones $R_i$ (completas, reflexivas, transitivas \vertema{3.A.8}); \emph{perfil} = lista de las $I$.
\item \emph{Output}: \concepto{regla de agregación} (\emph{social welfare function}):
\funcionclave{R_G=f(R_1,R_2,\dots,R_I)}
\item \textbf{5 condiciones}:
  \begin{itemize}
  \item \textbf{(U) Dominio universal}: definida para cualquier perfil.
  \item \textbf{(T) Transitividad}: $R_G$ completa y transitiva.
  \item \textbf{(P) \autor{Pareto} débil}: $x\succ_iy\ \forall i\Rightarrow x\succ_Gy$; con el débil el teorema es más fuerte.
  \item \textbf{(\textup{I}) IAI}: el orden \emph{social} de $x$ e $y$ depende solo de cómo ordena cada uno $x$ e $y$: impide usar la intensidad.
  \item \textbf{(D) No dictadura}: nadie impone siempre su orden.
  \end{itemize}
\end{itemize}
\subsubsub{ii) Desarrollo: el teorema}
\begin{itemize}
\item \concepto{Teorema de imposibilidad}: con $\geq3$ alternativas e $I$ finito, ninguna regla cumple U, T, P, I y D.
  \begin{itemize}
  \item Al revés: U, T, P e I $\Rightarrow$ dictadura.
  \item 1.ª ed. con un error (\autor{Blau}, 1957); 2.ª ed., 1963.
  \end{itemize}
\item \emph{Idea de la prueba} (\autor{Geanakoplos}, 2005):
  \begin{enumerate}
  \item \emph{Extremos}: si todos ponen $y$ en un extremo, la sociedad también.
  \item \emph{Votante pivote}: se sube $y$ de última a primera individuo a individuo; salta de golpe: quien la sube entonces, $n$, es \emph{pivote}.
  \item \emph{El pivote es dictador} (con la IAI): contradice D.
  \end{enumerate}
\item «\emph{Donde hay patrón no manda marinero}»: con U, T, P e I siempre hay un patrón.
\end{itemize}
\subsubsub{iii) Implicaciones}
\begin{itemize}
\item Mayoría por parejas incumple T (\autor{Condorcet}); \autor{Borda}, la IAI; unanimidad no ordena muchos pares; dictadura, D.
\item Razón última: \emph{pérdida de información}: $I$ ordenaciones $\to$ 1, sin comparar ganancias y pérdidas.
\item Toda regla tiene un defecto: hay que decir cuál.
\end{itemize}

\bloque[9]{II. 1.ª escapatoria: las funciones de bienestar social\label{rep:II}}
\begin{ideaclave}
La FBS completa el criterio de \autor{Pareto} solo a cambio de juicios de valor explícitos y comparaciones interpersonales; cada forma es una idea de la justicia, y cada una lleva a una medida de la desigualdad.
\end{ideaclave}

\sub{II.1. Rechazo de la metodología de \autor{Arrow}\label{rep:II.1}}
\begin{itemize}
\item \autor{Little} (1952), \autor{Bergson} (1954): juicios éticos, no un procedimiento para votar.
  \begin{itemize}
  \item Fase \emph{deliberativa} (FBS) y \emph{decisiva} (\autor{Arrow}).
  \end{itemize}
\item \autor{Samuelson} (1977): un solo perfil; pero \autor{Kemp}-\autor{Ng}, \autor{Parks} (1976): imposibilidad también con uno $\Rightarrow$ salida: comparaciones.
\end{itemize}

\sub{II.2. La FBS de \autor{Bergson} y \autor{Samuelson}\label{rep:II.2}}
\subsub{a) Idea y modelo}
\begin{itemize}
\item \autor{Bergson} (1938), \autor{Samuelson} (1947): bienestar social = función de las utilidades:
\funcionclave{W=W(U_1,\dots,U_I),\quad \frac{\partial W}{\partial U_i}>0}
  \begin{itemize}
  \item $\partial W/\partial U_i$ = \concepto{utilidad marginal social} de $i$.
  \end{itemize}
\item \emph{Propiedades}:
  \begin{itemize}
  \item \emph{individualismo} y \emph{coherencia con \autor{Pareto}} (de \autor{Samuelson});
  \item \emph{anonimato}; \emph{cuasiconcavidad} (aversión a la desigualdad); \emph{comparabilidad}.
  \end{itemize}
\item \emph{Medir $\neq$ comparar}: exige \emph{comparabilidad}, no cardinalidad. \concepto{Bases informativas} (\autor{Sen}, 1970):
  \begin{itemize}
  \item ordinal o cardinal no comparable: imposibilidad;
  \item niveles: \emph{leximin} (\autor{Hammond}, 1976); unidades: utilitarismo (\autor{d'Aspremont}-\autor{Gevers}, 1977);
  \item escala de razón: isoelástica.
  \end{itemize}
\item \concepto{CIS}: curvas de nivel de $W$; convexas solo con aversión a la desigualdad:
\begin{formula}
\centering $-\left.\dfrac{dU_B}{dU_A}\right|_{\overline{W}}=\dfrac{\partial W/\partial U_A}{\partial W/\partial U_B}$
\end{formula}
\end{itemize}
\subsub{b) Implicaciones}
\begin{itemize}
\item \emph{Planificador}: $\max W$ s.a. GFPU $\Rightarrow$ pendiente CIS = pendiente GFPU.
  \begin{itemize}
  \item \concepto{Punto de felicidad} (\emph{optimum optimorum}): máximo \emph{global}; ej. $c=(5{,}39;\,4{,}89)$.
  \end{itemize}
\end{itemize}
\begin{itemize}
\item \emph{\autor{Pareto} necesario, no suficiente}: $a$ (ineficiente, en $W_2$) es mejor que $d$ (eficiente, en $W_1$).
\item \emph{Tecnocrática}: el EGC coincide con $c$ solo por casualidad; 2TFEB: suma fija + mercado $\Rightarrow$ $c$.
\item \emph{Disyuntiva equidad-eficiencia}: no hay suma fija \vertema{3.A.22}.
  \begin{itemize}
  \item De $d$, impuestos distorsionantes llevan a $b$ (ineficiente, mejor que $d$).
  \item Si sigue: $e$ en la bisectriz, peor que $d$ ($3{,}45^2\approx11{,}9<12{,}27$). Imposición óptima \vertema{4.B.11}.
  \end{itemize}
\end{itemize}
\graficorepaso[0.62\linewidth]{fbs-gfpu-tradeoff}{FBS sobre la GFPU: punto de felicidad $c$; con impuestos distorsionantes, $d\to b\to e$}
\subsub{c) Valoración}
\begin{itemize}
\item Resuelve la incompletitud; instrumento de la economía pública.
\item Exige comparaciones y su forma es un juicio de valor (\autor{Bergson} no propone ninguna).
\end{itemize}

\sub{II.3. Valores éticos: la forma funcional\label{rep:II.3}}
\subsub{a) Utilitarismo: \autor{Bentham} y \autor{Harsanyi}}
\begin{itemize}
\item Elige sin saber qué lugar ocupará, prob. $1/I$ (\autor{Harsanyi}, 1953) $\Rightarrow$ $\max\sum_i\frac1IU_i$.
  \begin{itemize}
  \item \emph{Teorema de agregación} (1955): vNM \vertema{3.A.10} $\Rightarrow$ suma ponderada; con anonimato, $W=\sum_iU_i$.
  \end{itemize}
\item CIS \emph{rectas de pendiente $-1$}; neutral a la \emph{utilidad}, no a la renta (\autor{Edgeworth}).
\item \emph{Críticas}: sacrifica minorías; \emph{conclusión repugnante} (\autor{Parfit}, 1984).
\end{itemize}
\subsub{b) \autor{Rawls} (1971): neocontractualismo}
\begin{itemize}
\item \emph{Posición original} tras el \concepto{velo de la ignorancia}: decisión grave, irreversible, sin probabilidades.
  \begin{itemize}
  \item \emph{Maximin} (aversión infinita a la incertidumbre): $W=\min\{U_1,\dots,U_I\}$; CIS \emph{en L}.
  \item \emph{Leximin}: si empatan, el 2.º peor; cumple \autor{Pareto} fuerte.
  \end{itemize}
\item \emph{Críticas}: \autor{Harsanyi} (1975), regla extrema; \autor{Nozick} (1974), justicia histórica.
\end{itemize}
\subsub{c) Familia isoelástica (\autor{Atkinson}, 1970)}
\begin{itemize}
\item Aversión a la desigualdad $\epsilon\geq0$:
\funcionclave{W=\sum_{i=1}^{I}\frac{U_i^{1-\epsilon}}{1-\epsilon}\ (\epsilon\neq1),\qquad W=\sum_{i=1}^{I}\ln U_i\ (\epsilon=1)}
  \begin{itemize}
  \item Pendiente de las CIS $-(U_B/U_A)^{\epsilon}$.
  \item $\epsilon=0$: utilitarista; $\epsilon=1$: \autor{Bernoulli}-\autor{Nash}; $\epsilon\to\infty$: rawlsiana.
  \item Discutir la justicia = discutir $\epsilon$.
  \end{itemize}
\end{itemize}
\graficorepaso[0.98\linewidth]{cis-formas}{CIS: utilitarista ($\epsilon=0$, rectas), intermedia ($\epsilon=1$) y rawlsiana ($\epsilon\to\infty$, en L)}

\sub{II.4. Criterios equitativos y de justicia\label{rep:II.4}}
\subsub{a) \autor{Atkinson}: la medida de la desigualdad}
\begin{itemize}
\item Todo índice de desigualdad lleva dentro una FBS (\autor{Atkinson}, 1970).
\item \concepto{Curva de \autor{Lorenz}} $L(s)$: renta de la fracción $s$ más pobre; igualdad = diagonal.
\item \concepto{Índice de \autor{Gini}}: doble del área entre diagonal y curva:
\begin{formula}
\centering $\mathrm{IG}=1-2\displaystyle\int_0^1L(s)\,ds,\qquad 0\leq\mathrm{IG}\leq1$
\end{formula}
\item \emph{Dominancia de \autor{Lorenz}}: misma media y $L_{\vect{y}}\geq L_{\vect{y}'}$ $\Rightarrow$ \emph{toda} FBS utilitarista con $U$ cóncava prefiere $\vect{y}$.
  \begin{itemize}
  \item Medias distintas: \autor{Lorenz} generalizada (\autor{Shorrocks}, 1983); si se cortan, hay que elegir FBS.
  \end{itemize}
\end{itemize}
\graficorepaso[0.45\linewidth]{lorenz}{\autor{Lorenz}: área sombreada = IG/2}
\begin{itemize}
\item \concepto{Índice de \autor{Atkinson}}, con $y_{EDE}$ = \emph{renta equivalente igualmente distribuida}:
\funcionclave{A_\epsilon=1-\frac{y_{EDE}}{\overline{y}},\quad y_{EDE}=\Big[\frac1I\sum_{i=1}^{I}y_i^{1-\epsilon}\Big]^{\frac{1}{1-\epsilon}}}
  \begin{itemize}
  \item Parte de la renta que se sacrificaría por repartir el resto por igual; ↑$\epsilon$ $\Rightarrow$ pesa más la cola baja.
  \item Ej. 10~y 30 €: $A_0=0$; $A_1\approx0{,}13$; $A_2=0{,}25$; $A_\infty=0{,}5$.
  \item \emph{Normativo}, a diferencia del \autor{Gini}; pero $\epsilon$ no sale de los datos.
  \end{itemize}
\end{itemize}
\subsub{b) Evidencia experimental}
\begin{itemize}
\item \emph{Ultimátum} (\autor{Güth} et al., 1982): ENPS = ofrecer lo mínimo \vertema{3.A.14}; oferta media 40\,\%, rechazo 16\,\%.
\item \emph{Dictador} (\autor{Forsythe} et al., 1994): 28\,\% de media; $>$1/3 no da nada.
\item Importa el reparto, pero menos (miedo al rechazo) $\Rightarrow$ FBS no neutral.
\end{itemize}
\subsub{c) Crítica de \autor{Sen}}
\begin{itemize}
\item FBS \concepto{bienestarista}: la utilidad mide mal (adaptación a la privación).
\item \concepto{Capacidades}: libertad real para lograr \emph{funcionamientos} (1985); IDH (1990).
\end{itemize}

\sub{II.5. Valoración de las FBS\label{rep:II.5}}
\begin{itemize}
\item \emph{Ventajas}: imposición óptima \vertema{4.B.11}, redistribución \vertema{4.B.3}, proyectos \vertema{4.B.6}.
\item \emph{Inconvenientes}: comparaciones; forma = juicio de valor; ¿de quién? Si son de la sociedad, volvemos a \autor{Arrow}.
\end{itemize}

\bloque[9]{III. 2.ª escapatoria: levantamiento de los axiomas\label{rep:III}}
\begin{ideaclave}
Cada renuncia tiene un coste: restringir las preferencias salva a la mayoría, pero no recoge la intensidad; recogerla invita a mentir; debilitar la transitividad crea grupos con veto, y con preferencias privadas toda regla razonable es manipulable.
\end{ideaclave}

\sub{III.1. Aceptación de la metodología: \autor{May}\label{rep:III.1}}
\begin{itemize}
\item Marco ordinal sin comparaciones; regla de votación = regla de agregación.
  \begin{itemize}
  \item \autor{Inman}: sin P, ineficiencia; sin U, IAI o T, en la decisión; sin D, dictadura: lo razonable, renunciar a U o IAI.
  \end{itemize}
\item \concepto{Teorema de \autor{May}} (1952): con 2~alternativas, mayoría simple $\iff$ decisión universal, anonimato, neutralidad, respuesta positiva.
  \begin{itemize}
  \item Resultado de \emph{posibilidad}; con $\geq3$, la mayoría tropieza con \autor{Arrow}.
  \end{itemize}
\end{itemize}

\sub{III.2. Levantamiento de los axiomas\label{rep:III.2}}
\subsub{a) Renuncia al dominio universal}
\subsubsub{i) Paradoja de \autor{Condorcet} (1785)}
\begin{itemize}
\item Individuos transitivos, mayoría intransitiva. Vecinos 1: $x\succ y\succ z$; 2: $y\succ z\succ x$; 3: $z\succ x\succ y$.
  \begin{itemize}
  \item $x$ gana a $y$ (1~y 3), $y$ a $z$ (1~y 2), $z$ a $x$ (2~y 3) $\Rightarrow$ $x\succ_Gy\succ_Gz\succ_Gx$: no hay \concepto{ganador de \autor{Condorcet}}.
  \item \emph{Agenda}: quien fija el orden fija el resultado.
  \end{itemize}
\item \emph{Causa}: el vecino 3~prefiere los extremos (\concepto{multimodales}) $\Rightarrow$ restringir a \concepto{unimodales}.
\end{itemize}
\graficorepaso[0.88\linewidth]{condorcet}{Multimodales (\autor{Condorcet}, izq.) y unimodales (pico del mediano, dcha.)}
\subsubsub{ii) Votante mediano (\autor{Black}, 1948)}
\begin{itemize}
\item \emph{Supuestos}: 1~dimensión, unimodales, nº impar, por parejas, voto sincero.
\item \concepto{Teorema del votante mediano}: la favorita del mediano gana a cualquier otra: ganador de \autor{Condorcet}, mayoría transitiva.
  \begin{itemize}
  \item T, P, IAI y D a costa de U.
  \item \autor{Hotelling}-\autor{Downs} (1957): 2~partidos convergen \vertema{3.A.18}; varias dimensiones: caos (\autor{McKelvey}, 1976).
  \end{itemize}
\item \emph{¿Eficiente?} Cuotas iguales; eficiencia = \autor{Samuelson} \vertema{3.A.23} (\autor{Bowen}, 1943):
\begin{formula}
\centering $G^m=G^*\iff\mathrm{RMS}^m_{GX}(G^*)=\dfrac1I\displaystyle\sum_{i=1}^{I}\mathrm{RMS}^i_{GX}(G^*)$
\end{formula}
  \begin{itemize}
  \item Media $>$ mediana $\Rightarrow$ se gasta de menos: la mayoría cuenta cabezas, no intensidades.
  \end{itemize}
\end{itemize}
\subsub{b) Renuncia a la IAI}
\subsubsub{i) Recuento de \autor{Borda} (1781)}
\begin{itemize}
\item $m$ puntos a la favorita, \dots, 1~a la última; ordinal (posicional). Cumple U, T, P, D; incumple IAI.
\item 5~vecinos: 3~con $x\succ y\succ z$, 2~con $y\succ z\succ x$:
\end{itemize}

\subsubsub{ii) Intercambio de votos (\emph{logrolling}, \autor{Tullock}, 1959)}
\begin{itemize}
\item «Yo voto tu carretera y tú mi polideportivo». Ascensor y rampa, coste 6 (2~por vecino):
  \begin{itemize}
  \item si valen 7: suma $+1$, se aprueban, bien;
  \item si valen 5: suma $-1$, se aprueban igual: el 3~paga 4, sobreprovisión.
  \end{itemize}
\item Exige pocos votantes y un paquete.
\end{itemize}
\subsub{c) Renuncia a T, P o D}
\begin{itemize}
\item \concepto{Cuasitransitividad} (\autor{Sen}, 1969): solo la preferencia estricta transitiva.
  \begin{itemize}
  \item Con U, P e IAI $\Rightarrow$ \concepto{oligarquía} con veto (\autor{Gibbard}, 1969; \autor{Mas-Colell}-\autor{Sonnenschein}, 1972).
  \item Aciclicidad: sigue el veto (\autor{Brown}, 1975).
  \end{itemize}
\item \textbf{Sin P}: regla nula, dictatorial o dictatorial inversa (\autor{Wilson}, 1972). \textbf{Sin D}: renunciar a la democracia.
\item \concepto{Liberal paretiano} (\autor{Sen}, 1970): U + \autor{Pareto} débil + liberalismo mínimo $\Rightarrow$ ciclos.
  \begin{itemize}
  \item Libro escandaloso: lo lee el puritano ($x$), el lascivo ($y$) o nadie ($z$); puritano $z\succ x\succ y$, lascivo $x\succ y\succ z$.
  \item $z\succ_Gx$ (puritano), $y\succ_Gz$ (lascivo), $x\succ_Gy$ (P) $\Rightarrow$ ciclo.
  \item \autor{Pareto} respeta preferencias \emph{entrometidas}; salida: derechos \emph{alienables} (\autor{Gibbard}, 1974).
  \end{itemize}
\end{itemize}
\subsub{d) Otras reglas distintas de la mayoría simple}
\begin{itemize}
\item \concepto{Unanimidad} (\autor{Wicksell}, 1896): solo mejoras de \autor{Pareto}; base de \autor{Lindahl} \vertema{3.A.23}.
  \begin{itemize}
  \item Cara, veto, \emph{statu quo} (enganche).
  \end{itemize}
\item \concepto{Mayoría óptima} (\autor{Buchanan}-\autor{Tullock}, 1962): decisión \emph{constitucional}, sin saber si estará en la mayoría.
  \begin{itemize}
  \item \concepto{Costes externos} $\mathrm{CE}(M)$ \emph{bajan} con $M$; \concepto{de decisión} $\mathrm{CD}(M)$ \emph{suben}:
  \end{itemize}
\begin{formula}
\centering $M^*=\arg\min_M\big[\mathrm{CE}(M)+\mathrm{CD}(M)\big]$
\end{formula}
\end{itemize}
\graficorepaso[0.55\linewidth]{buchanan-tullock}{$M^*$ en el mínimo de la suma, no en el corte}
\begin{itemize}
\item Más alta cuanto más graves e irreversibles las decisiones.
  \begin{itemize}
  \item CE: simple (ordinarias), absoluta (orgánicas, art.~81), 3/5 (art.~167), 2/3 + referéndum (art.~168).
  \item UE: mayoría cualificada; unanimidad en fiscalidad, exterior y MFP.
  \end{itemize}
\item \emph{Límites}: costes conocidos ex ante; olvida quién decide \vertema{4.B.2}.
\end{itemize}

\sub{III.3. Problemas de información\label{rep:III.3}}
\begin{itemize}
\item 2~problemas:
  \begin{itemize}
  \item \emph{sobre el bienestar}: con comparaciones desaparece la imposibilidad (\emph{leximin}, utilitarismo $\to$ \hyperref[rep:II]{II});
  \item \emph{sobre las preferencias}: se declaran y se puede mentir.
  \end{itemize}
\item \concepto{Función de elección social}: a cada perfil declarado, \emph{una} alternativa; \emph{no manipulable} si decir la verdad es dominante.
\item \concepto{Teorema de \autor{Gibbard}-\autor{Satterthwaite}} (1973, 1975): dominio universal y $\geq3$ alternativas elegibles $\Rightarrow$ toda función no manipulable es dictatorial.
  \begin{itemize}
  \item O dictadura, o alternativas que nunca ganan, o manipulable.
  \item No pide IAI ni T (elige, no ordena): análogo de \autor{Arrow} para incentivos; misma idea del pivote (\autor{Reny}, 2001).
  \end{itemize}
\item \emph{Salidas}:
  \begin{itemize}
  \item unimodales: mediana de los picos no manipulable (\autor{Moulin}, 1980; al vecino 3~no le sirve mentir);
  \item cuasilineal + pagos: \autor{Clarke}-\autor{Groves} (cada uno paga el daño que causa) \vertema{3.A.23}, \vertema{3.A.14}.
  \end{itemize}
\item \concepto{Diseño de mecanismos}: diseñar la regla para que el interés lleve al resultado deseado.
  \begin{itemize}
  \item \emph{Principio de revelación} (\autor{Gibbard}, 1973; \autor{Myerson}, 1979): basta buscar mecanismos directos veraces.
  \item Nobel 2007 (\autor{Hurwicz}, \autor{Maskin}, \autor{Myerson}).
  \end{itemize}
\end{itemize}

\bloque[1]{Conclusión}
\begin{itemize}
\item \textbf{Recapitulación}:
  \begin{itemize}
  \item \autor{Pareto} no ordena los óptimos; \autor{Arrow}: ninguna regla cumple U, T, P, IAI y D;
  \item las FBS escapan con juicios de valor y comparaciones ($\epsilon$, \autor{Atkinson});
  \item las reglas de votación renuncian a un axioma, con coste.
  \end{itemize}
\item \textbf{Relevancia}: las democracias combinan reglas según la decisión (mayorías reforzadas, unanimidades, vetos: Constitución, UE); el diseño institucional importa tanto como las preferencias.
\item \textbf{Extensiones}: elección pública \vertema{4.B.2}; mecanismos \vertema{3.A.14}; imposición óptima \vertema{4.B.11}; redistribución \vertema{4.B.3}; desigualdad en España \vertema{4.A.3}; ACB (\autor{Kaldor}-\autor{Hicks}) \vertema{4.B.6}.
\item \textbf{Opinión}: \autor{Arrow} no va contra la democracia, sino contra la ilusión de una regla neutral.
  \begin{itemize}
  \item Elegir qué axioma sacrificar es un juicio de valor: mejor a la vista, como en las constituciones.
  \end{itemize}
\item \textbf{Idea final}: «\emph{nunca llueve a gusto de todos}»; una sociedad decide con coherencia si dice qué valora.
\end{itemize}

\bloque{Preguntas del tribunal}
\begin{enumerate}
\item \textbf{¿Cómo casa \autor{Arrow} con los óptimos únicos de otras ramas?} Un óptimo único supone una sola función objetivo: alguien ya fijó los juicios de valor o el dominio está restringido.
\item \textbf{¿Exige la FBS utilidades cardinales?} No: comparabilidad, no cardinalidad.
\item \textbf{¿Qué axioma levanta el votante mediano? ¿Es eficiente?} U; eficiente solo si mediana = media (\autor{Bowen}).
\item \textbf{¿Aplicación en la Constitución?} Mayorías reforzadas (arts. 81, 167, 168): \autor{Buchanan}-\autor{Tullock}.
\item \textbf{¿Relación \autor{Arrow}-\autor{Gibbard}-\autor{Satterthwaite}?} Análogo para incentivos; elige, no ordena.
\item \textbf{¿Por qué \autor{Clarke}-\autor{Groves} no lo contradice?} Dominio restringido: cuasilineal y pagos.
\end{enumerate}
\begin{trampa}
\begin{itemize}
\item \autor{Arrow}: \autor{Pareto} \emph{débil}; con 2~alternativas, la mayoría funciona (\autor{May}).
\item IAI de \autor{Arrow} $\neq$ independencia vNM $\neq$ IIA de \autor{Nash}.
\item Cuasitransitividad $\Rightarrow$ \emph{oligarquía}, no dictadura.
\item \autor{Borda}: ordinal (posicional), no cardinal; incumple IAI.
\item \autor{Rawls}: aversión infinita a la \emph{incertidumbre}; CIS en L. \autor{Atkinson}: $\epsilon=1$, media geométrica.
\item \autor{Buchanan}-\autor{Tullock}: $M^*$ en el mínimo de la suma, no en el corte.
\end{itemize}
\end{trampa}
