% MathJax; en Word construir-tema.py cambia \tachado por un recuadro con tachado
\newcommand{\nuevo}[1]{{\color{red}{#1}}}
\newcommand{\tachado}[1]{\cancel{#1}}
% \tiempo{2,5} → 2′30″ (lo calcula tcee.lua, como \minutos)
\newcommand{\tiempo}[1]{\textcolor{tceetiempo}{#1}}

% \cortetitulo: punto de corte de un título largo; en la web y en Word, un espacio
